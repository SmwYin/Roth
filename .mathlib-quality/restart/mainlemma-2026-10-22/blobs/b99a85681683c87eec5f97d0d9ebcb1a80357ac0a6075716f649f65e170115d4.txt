/-- Select a nonvanishing Hasse derivative while controlling index loss and evaluation. -/
private lemma exists_hasseDeriv_index_eval_bound_aux {σ : Type*} [Fintype σ]
    {P : MvPolynomial σ ℤ} {r : σ → ℕ} {B α β γ c : ℝ} {s : ℕ} {z : σ → ℝ}
    (hP : P ≠ 0) (hd : ∀ j, degreeOf j P ≤ r j) (hc : ∀ d, |(P.coeff d : ℝ)| ≤ B)
    (hs : ∑ j, r j ≤ s) (h2 : (2 : ℝ) ^ s < β) (hα : (1 + |α|) ^ s ≤ β)
    (hγ : γ ≤ index (map (Int.castRingHom ℝ) P) (fun _ ↦ α) r)
    (hz : index (map (Int.castRingHom ℝ) P) z r < c) :
    ∃ Q : MvPolynomial σ ℤ, Q ≠ 0 ∧ (∀ j, degreeOf j Q ≤ r j) ∧
      γ - c < index (map (Int.castRingHom ℝ) Q) (fun _ ↦ α) r ∧
      eval z (map (Int.castRingHom ℝ) Q) ≠ 0 ∧
      ∀ i : σ →₀ ℕ, |eval (fun _ ↦ α) (hasseDeriv i (map (Int.castRingHom ℝ) Q))| <
        β * (β * B) * β := by
  obtain ⟨d, hdne⟩ := MvPolynomial.exists_coeff_ne_zero hP
  have hB : 0 < B := (abs_pos.mpr (Int.cast_ne_zero.mpr hdne)).trans_le (hc d)
  have hmap0 : map (Int.castRingHom ℝ) P ≠ 0 := fun h0 ↦
    hP (MvPolynomial.map_injective _ Int.cast_injective (h0.trans (map_zero _).symm))
  obtain ⟨k, hkw, hkev⟩ := exists_hasseDeriv_eval_ne_zero hmap0 z r hz
  have hmapQ : map (Int.castRingHom ℝ) (hasseDeriv k P) =
      hasseDeriv k (map (Int.castRingHom ℝ) P) := (hasseDeriv_map _ k _).symm
  have hdQ : ∀ j, degreeOf j (hasseDeriv k P) ≤ r j := fun j ↦
    (degreeOf_hasseDeriv_le j k P).trans (hd j)
  refine ⟨hasseDeriv k P, fun hQ0 ↦ hkev ?_, hdQ, ?_, by rw [hmapQ]; exact hkev,
    fun i ↦ ?_⟩
  · rw [← hmapQ, hQ0, map_zero, map_zero]
  · have hk0 : hasseDeriv k (map (Int.castRingHom ℝ) P) ≠ 0 := fun h0 ↦
      hkev (by rw [h0, map_zero])
    have hderiv := index_hasseDeriv_ge hk0 (fun _ ↦ α) r
    rw [hmapQ]
    linarith
  · rw [hasseDeriv_map]
    exact abs_eval_iterated_hasseDeriv_lt_aux hd hB hs h2 hα hc i k
