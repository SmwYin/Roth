theorem audit_no_library_match {σ : Type*} [Fintype σ]
    {P : MvPolynomial σ ℤ} {r : σ → ℕ} {B α β : ℝ} {s : ℕ}
    (hP : ∀ j, degreeOf j P ≤ r j) (hB : 0 < B) (hs : ∑ j, r j ≤ s)
    (h2 : (2 : ℝ) ^ s < β) (hα : (1 + |α|) ^ s < β)
    (hc : ∀ d, |((P.coeff d : ℤ) : ℝ)| ≤ B) (i k : σ →₀ ℕ) :
    |eval (fun _ ↦ α) (map (Int.castRingHom ℝ) (hasseDeriv i (hasseDeriv k P)))| <
      β * (β * B) * β := by
  fail_if_success (solve | grind)
  fail_if_success (solve | simp)
  fail_if_success (solve | aesop)
  fail_if_success (solve | exact?)
  fail_if_success (solve | gcongr)
  fail_if_success (solve | positivity)
  fail_if_success (solve | linarith)
  fail_if_success (solve | nlinarith)
  have hd : ∀ j, degreeOf j (hasseDeriv k P) ≤ r j := fun j ↦
    (degreeOf_hasseDeriv_le j k P).trans (hP j)
  have he := abs_eval_le (hasseDeriv i (hasseDeriv k P))
    (fun j ↦ (degreeOf_hasseDeriv_le j i _).trans (hd j))
    (abs_coeff_hasseDeriv_le i _ hd (abs_coeff_hasseDeriv_le k P hP hc)) (fun _ ↦ α)
  rw [Finset.prod_pow_eq_pow_sum] at he
  refine he.trans_lt ?_
  have ht : (2 : ℝ) ^ (∑ j, r j) < β :=
    (pow_le_pow_right₀ (by norm_num) hs).trans_lt h2
  have ha : (1 + |α|) ^ (∑ j, r j) < β :=
    (pow_le_pow_right₀ (le_add_of_nonneg_right (abs_nonneg α)) hs).trans_lt hα
  exact mul_lt_mul'' (mul_lt_mul'' ht (mul_lt_mul_of_pos_right ht hB)
    (by positivity) (by positivity)) ha (by positivity) (by positivity)

run_cmd Lean.Elab.Command.liftTermElabM do
  let env ← Lean.getEnv
  unless !(env.header.moduleNames.contains `Roth.MainLemma) do throwError "Target module present in absence probe"
  let n := ``audit_no_library_match
  let c ← Lean.getConstInfo n
  let v := c.value! true
  let opts ← Lean.getOptions
  unless Lean.maxHeartbeats.get opts == 200000 && Lean.Meta.synthInstance.maxHeartbeats.get opts == 20000 && Lean.maxRecDepth.get opts == 512 do throwError "Changed defaults"
  for e in [c.type, v] do
    if e.hasMVar || e.hasLevelMVar || e.hasFVar || e.hasLooseBVars || e.getUsedConstants.contains ``sorryAx then throwError "Open/admitted absence probe"
  unless ← Lean.Meta.isDefEq (← Lean.Meta.inferType v) c.type do throwError "Probe inferred TYPE differs"
  let axs ← Lean.collectAxioms n
  unless axs.all ([``propext, ``Classical.choice, ``Quot.sound].contains ·) do throwError "Probe nonstandard axiom"
  Lean.Core.liftIOCore <| IO.println s!"FULL COMMAND AUTOMATION NEGATIVE CERTIFICATE PASS TYPE={(repr c.type).pretty 1000000} VALUE={(repr v).pretty 1000000} AXIOMS={axs} LIMITS=200000/20000/512"
