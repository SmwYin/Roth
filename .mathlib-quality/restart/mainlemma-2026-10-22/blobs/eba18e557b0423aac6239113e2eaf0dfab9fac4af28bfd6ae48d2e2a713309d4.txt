from pathlib import Path
from snapshot_reuse_v1 import save_at_immutable
import sys,json,hashlib,time
p=Path('/private/tmp/roth-mainlemma-helper9-rem-floor');root=Path('/Users/ramin/LEAN/Roth');label,phase=sys.argv[1:3]
files=sorted(list(root.glob('Roth/*.lean'))+[root/'Roth.lean',root/'lean-toolchain',root/'lakefile.toml',root/'lake-manifest.json']);pins={};byte_counts={}
for f in files:
 b=f.read_bytes();s=hashlib.sha256(b).hexdigest();pins[str(f.relative_to(root))]=s;byte_counts[str(f.relative_to(root))]=len(b)
 d=p/'launched-inputs'/s/f.name;d.parent.mkdir(parents=True,exist_ok=True);save_at_immutable(d,b)
(p/(label+'-'+phase+'-pins.json')).write_text(json.dumps({'observed_unix':time.time(),'pins':pins,'bytes':byte_counts},indent=2))
if phase=='before':
 tool=sys.argv[3];args=json.loads(sys.argv[4]);(p/(label+'-request.json')).write_text(json.dumps({'tool':tool,'actual_args':args,'started_unix':time.time(),'before_pins':pins,'before_bytes':byte_counts},indent=2))
else:
 q=p/(label+'-request.json');j=json.loads(q.read_text());j['completed_unix']=time.time();j['after_pins']=pins;j['after_bytes']=byte_counts;j['pins_unchanged']=j['before_pins']==pins;(p/(label+'-completion.json')).write_text(json.dumps(j,indent=2))
print(label,phase)
