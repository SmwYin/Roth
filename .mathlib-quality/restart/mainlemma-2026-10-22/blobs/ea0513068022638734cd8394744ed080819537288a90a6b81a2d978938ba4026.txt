from pathlib import Path
import subprocess,json,time,hashlib
p=Path('/private/tmp/roth-mainlemma-helper8-diag-rem')
def run(label,args):
    subprocess.run(['python3',str(p/'observe.py'),label+'-prelaunch'],check=True)
    subprocess.run(['python3',str(p/'runner.py'),'--frozen-env',str(p/'parent-environment.json'),'--timeout','425',label,'--','/Users/ramin/.elan/bin/lake']+args,check=True)
    print('GATE-DRAINED '+label,flush=True)
run('v4-final-full',['build'])
run('v4-final-eval',['build','Roth.LeanEval'])
run('v4-independent-current-v1',['env','lean','-j1',str(p/'v4-independent-current-v1.lean')])
run('v4-after-inventory',['env','lean','-j1',str(p/'v4-after-inventory.lean')])
data=[json.loads(l.removeprefix('INVENTORY_JSON=')) for l in (p/'v4-after-inventory.log').read_text().splitlines() if l.startswith('INVENTORY_JSON=')]
assert len(data)==1
old=json.loads((p/'before-inventory-data.json').read_text())
assert sorted(data[0]['names'])==sorted(old['names']), 'Actual inventory changed; fresh guard generation required'
(p/'v4-after-inventory-data.json').write_text(json.dumps(data[0],indent=2)+'\n')
(p/'v4-after-pairs.lean').write_bytes((p/'before-pairs.lean').read_bytes())
run('v4-after-pairs',['env','lean','-j1',str(p/'v4-after-pairs.lean')])
run('v4-final-affected',['build','Roth.MainLemma','Roth.Main'])
print('V4-SEMANTICS-ALL-DRAINED',flush=True)
