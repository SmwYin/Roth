private def sameBinderName (a b : Lean.Name) : Bool :=
  a == b || (a.hasMacroScopes && b.hasMacroScopes && a.eraseMacroScopes == b.eraseMacroScopes)
private partial def sameTypeExceptBothGeneratedBinders (a b : Lean.Expr) : Bool :=
  if a == b then true else
  match a, b with
  | .forallE n t x bi, .forallE m u y bj => sameBinderName n m && bi == bj &&
      sameTypeExceptBothGeneratedBinders t u && sameTypeExceptBothGeneratedBinders x y
  | .lam n t x bi, .lam m u y bj => sameBinderName n m && bi == bj &&
      sameTypeExceptBothGeneratedBinders t u && sameTypeExceptBothGeneratedBinders x y
  | .letE n t v x nd, .letE m u w y md => sameBinderName n m && nd == md &&
      sameTypeExceptBothGeneratedBinders t u && sameTypeExceptBothGeneratedBinders v w &&
      sameTypeExceptBothGeneratedBinders x y
  | .app f x, .app g y => sameTypeExceptBothGeneratedBinders f g &&
      sameTypeExceptBothGeneratedBinders x y
  | .proj n i x, .proj m j y => n == m && i == j && sameTypeExceptBothGeneratedBinders x y
  | .mdata m x, .mdata n y => m == n && sameTypeExceptBothGeneratedBinders x y
  | _, _ => false
