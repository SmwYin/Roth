theorem audit_original_firstdraft {σ : Type*} [Fintype σ]
    {P : MvPolynomial σ ℤ} {r : σ → ℕ} {B α β : ℝ} {s : ℕ}
    (hP : ∀ j, degreeOf j P ≤ r j) (hB : 0 < B) (hs : ∑ j, r j ≤ s)
    (h2 : (2 : ℝ) ^ s < β) (hα : (1 + |α|) ^ s < β)
    (hc : ∀ d, |((P.coeff d : ℤ) : ℝ)| ≤ B) (i k : σ →₀ ℕ) :
    |eval (fun _ ↦ α) (map (Int.castRingHom ℝ) (hasseDeriv i (hasseDeriv k P)))| <
      β * (β * B) * β := by
  have hd : ∀ j, degreeOf j (hasseDeriv k P) ≤ r j := fun j ↦
    (degreeOf_hasseDeriv_le j k P).trans (hP j)
  have he := abs_eval_le (hasseDeriv i (hasseDeriv k P))
    (fun j ↦ (degreeOf_hasseDeriv_le j i _).trans (hd j))
    (abs_coeff_hasseDeriv_le i _ hd (abs_coeff_hasseDeriv_le k P hP hc)) (fun _ ↦ α)
  rw [Finset.prod_pow_eq_pow_sum] at he
  refine he.trans_lt ?_
  have ht : (2 : ℝ) ^ (∑ j, r j) < β :=
    (pow_le_pow_right₀ (by norm_num) hs).trans_lt h2
  have ha : (1 + |α|) ^ (∑ j, r j) < β :=
    (pow_le_pow_right₀ (le_add_of_nonneg_right (abs_nonneg α)) hs).trans_lt hα
  exact mul_lt_mul'' (mul_lt_mul'' ht (mul_lt_mul_of_pos_right ht hB)
    (by positivity) (by positivity)) ha (by positivity) (by positivity)

theorem audit_weaker_alpha {σ : Type*} [Fintype σ]
    {P : MvPolynomial σ ℤ} {r : σ → ℕ} {B α β : ℝ} {s : ℕ}
    (hP : ∀ j, degreeOf j P ≤ r j) (hB : 0 < B) (hs : ∑ j, r j ≤ s)
    (h2 : (2 : ℝ) ^ s < β) (hα : (1 + |α|) ^ s ≤ β)
    (hc : ∀ d, |((P.coeff d : ℤ) : ℝ)| ≤ B) (i k : σ →₀ ℕ) :
    |eval (fun _ ↦ α) (map (Int.castRingHom ℝ) (hasseDeriv i (hasseDeriv k P)))| <
      β * (β * B) * β := by
  have hd : ∀ j, degreeOf j (hasseDeriv k P) ≤ r j := fun j ↦
    (degreeOf_hasseDeriv_le j k P).trans (hP j)
  have he := abs_eval_le (hasseDeriv i (hasseDeriv k P))
    (fun j ↦ (degreeOf_hasseDeriv_le j i _).trans (hd j))
    (abs_coeff_hasseDeriv_le i _ hd (abs_coeff_hasseDeriv_le k P hP hc)) (fun _ ↦ α)
  rw [Finset.prod_pow_eq_pow_sum] at he
  refine he.trans_lt ?_
  have ht : (2 : ℝ) ^ (∑ j, r j) < β :=
    (pow_le_pow_right₀ (by norm_num) hs).trans_lt h2
  have ha : (1 + |α|) ^ (∑ j, r j) ≤ β :=
    (pow_le_pow_right₀ (le_add_of_nonneg_right (abs_nonneg α)) hs).trans hα
  have hβ : 0 < β := (pow_pos (by norm_num) _).trans ht
  exact mul_lt_mul (mul_lt_mul'' ht (mul_lt_mul_of_pos_right ht hB)
    (by positivity) (by positivity)) ha (by positivity) (by positivity)

theorem audit_weaker_two {σ : Type*} [Fintype σ]
    {P : MvPolynomial σ ℤ} {r : σ → ℕ} {B α β : ℝ} {s : ℕ}
    (hP : ∀ j, degreeOf j P ≤ r j) (hB : 0 < B) (hs : ∑ j, r j ≤ s)
    (h2 : (2 : ℝ) ^ s ≤ β) (hα : (1 + |α|) ^ s < β)
    (hc : ∀ d, |((P.coeff d : ℤ) : ℝ)| ≤ B) (i k : σ →₀ ℕ) :
    |eval (fun _ ↦ α) (map (Int.castRingHom ℝ) (hasseDeriv i (hasseDeriv k P)))| <
      β * (β * B) * β := by
  have hd : ∀ j, degreeOf j (hasseDeriv k P) ≤ r j := fun j ↦
    (degreeOf_hasseDeriv_le j k P).trans (hP j)
  have he := abs_eval_le (hasseDeriv i (hasseDeriv k P))
    (fun j ↦ (degreeOf_hasseDeriv_le j i _).trans (hd j))
    (abs_coeff_hasseDeriv_le i _ hd (abs_coeff_hasseDeriv_le k P hP hc)) (fun _ ↦ α)
  rw [Finset.prod_pow_eq_pow_sum] at he
  refine he.trans_lt ?_
  have ht : (2 : ℝ) ^ (∑ j, r j) ≤ β :=
    (pow_le_pow_right₀ (by norm_num) hs).trans h2
  have ha : (1 + |α|) ^ (∑ j, r j) < β :=
    (pow_le_pow_right₀ (le_add_of_nonneg_right (abs_nonneg α)) hs).trans_lt hα
  exact mul_lt_mul_of_le_of_lt_of_pos_of_nonneg (mul_le_mul ht (mul_le_mul_of_nonneg_right ht hB.le)
    (by positivity) ((by positivity : (0 : ℝ) ≤ 2 ^ (∑ j, r j)).trans ht)) ha
    (by positivity) ((pow_nonneg (by positivity) _).trans ha.le)

theorem audit_zero_B_boundary :
    (∀ j : Fin 0, degreeOf j (0 : MvPolynomial (Fin 0) ℤ) ≤ 0) ∧
    (∑ _j : Fin 0, (0 : ℕ)) ≤ 0 ∧ (2 : ℝ) ^ 0 < 2 ∧
    (1 + |(0 : ℝ)|) ^ 0 < 2 ∧
    (∀ d, |(((0 : MvPolynomial (Fin 0) ℤ).coeff d : ℤ) : ℝ)| ≤ 0) ∧
    ¬ |eval (fun _ : Fin 0 ↦ (0 : ℝ)) (map (Int.castRingHom ℝ)
      (hasseDeriv 0 (hasseDeriv 0 (0 : MvPolynomial (Fin 0) ℤ))))| < 2 * (2 * (0 : ℝ)) * 2 := by
  norm_num

theorem audit_both_weak_boundary :

    (∑ _j : Fin 0, (0 : ℕ)) ≤ 0 ∧ (0 : ℝ) < 1 ∧ (2 : ℝ) ^ 0 ≤ 1 ∧

      (hasseDeriv 0 (hasseDeriv 0 (1 : MvPolynomial (Fin 0) ℤ))))| < 1 * (1 * (1 : ℝ)) * 1 := by

open Lean Meta in
private partial def normalizeGeneratedBinders : Expr → Expr
  | .forallE n t b bi => .forallE (if n.hasMacroScopes then n.eraseMacroScopes else n)
      (normalizeGeneratedBinders t) (normalizeGeneratedBinders b) bi
  | .lam n t b bi => .lam (if n.hasMacroScopes then n.eraseMacroScopes else n)
      (normalizeGeneratedBinders t) (normalizeGeneratedBinders b) bi
  | .letE n t v b nd => .letE (if n.hasMacroScopes then n.eraseMacroScopes else n)
      (normalizeGeneratedBinders t) (normalizeGeneratedBinders v) (normalizeGeneratedBinders b) nd
  | .app f a => .app (normalizeGeneratedBinders f) (normalizeGeneratedBinders a)
  | .mdata m e => .mdata m (normalizeGeneratedBinders e)
  | .proj n i e => .proj n i (normalizeGeneratedBinders e)
  | e => e

run_cmd Lean.Elab.Command.liftTermElabM do
  let env ← Lean.getEnv
  let helpers := env.constants.fold (init := #[]) fun ns n _ ↦
    match env.getModuleIdxFor? n with
    | some idx => if env.header.moduleNames[idx]! == `Roth.MainLemma &&
        n.toString.endsWith ".Roth.abs_eval_iterated_hasseDeriv_lt_aux" then ns.push n else ns
    | none => ns
  unless helpers.size == 1 do throwError "Nonunique actual imported helper {helpers}"
  let old := helpers[0]!
  let own := ``audit_original_firstdraft
  let oc ← Lean.getConstInfo old
  let nc ← Lean.getConstInfo own
  unless oc.levelParams == nc.levelParams do throwError "Changed universes"
  unless (repr (normalizeGeneratedBinders oc.type)).pretty 1000000 ==
      (repr (normalizeGeneratedBinders nc.type)).pretty 1000000 do
    throwError "Full TYPE differs beyond generated binder macro scopes"
  let opts ← Lean.getOptions
  unless Lean.maxHeartbeats.get opts == 200000 &&
      Lean.Meta.synthInstance.maxHeartbeats.get opts == 20000 && Lean.maxRecDepth.get opts == 512 do
    throwError "Changed defaults"
  for n in [old, own, ``audit_weaker_alpha, ``audit_weaker_two, ``audit_zero_B_boundary, ``audit_both_weak_boundary, ``audit_missing_coeff_boundary, ``audit_power_coeff, ``audit_missing_degree_boundary, ``audit_missing_cap_boundary, ``audit_missing_alpha_boundary, ``audit_missing_two_boundary] do
    let c ← Lean.getConstInfo n
    let v := c.value! true
    for e in [c.type, v] do
      if e.hasMVar || e.hasLevelMVar || e.hasFVar || e.hasLooseBVars ||
          e.getUsedConstants.contains ``sorryAx then throwError "Open/admitted type or value {n}"
    unless ← Lean.Meta.isDefEq (← Lean.Meta.inferType v) c.type do
      throwError "Inferred value TYPE differs {n}"
    let axs ← Lean.collectAxioms n
    unless axs.all ([``propext, ``Classical.choice, ``Quot.sound].contains ·) do
      throwError "Nonstandard axiom {n} {axs}"
    Lean.Core.liftIOCore <| IO.println s!"FULL PAIR NAME={(repr n).pretty 1000000} LEVELS={c.levelParams} TYPE={(repr c.type).pretty 1000000} ACTUALOPAQUEVALUE={(repr v).pretty 1000000} AXIOMS={axs} LIMITS=200000/20000/512"
  let mut pending := #[own, ``audit_weaker_alpha, ``audit_weaker_two, ``audit_zero_B_boundary, ``audit_both_weak_boundary, ``audit_missing_coeff_boundary, ``audit_power_coeff, ``audit_missing_degree_boundary, ``audit_missing_cap_boundary, ``audit_missing_alpha_boundary, ``audit_missing_two_boundary]
  let mut seen : Lean.NameSet := {}
  while !pending.isEmpty do
    let n := pending.back!
    pending := pending.pop
    unless seen.contains n do
      seen := seen.insert n
      if old.isPrefixOf n || (`Roth.main_lemma).isPrefixOf n || n == ``sorryAx then
        throwError "Independent proof closure uses denied name {n}"
      let c ← Lean.getConstInfo n
      pending := pending ++ c.type.getUsedConstants
      if let some v := c.value? true then pending := pending ++ v.getUsedConstants
  Lean.Core.liftIOCore <| IO.println s!"INDEPENDENT ORIGINAL FULL CONTRACT PASS DENIED={old},Roth.main_lemma,sorryAx CLOSURE={seen.toList}"
