private lemma abs_coeff_diag_hasseDeriv_modByMonic_le_floor_aux {σ : Type*} [Fintype σ]
    {f : Polynomial ℤ} {P : MvPolynomial σ ℤ} {r : σ → ℕ} {A B β : ℝ} {s : ℕ}
    (hA : ∀ i, |(f.coeff i : ℝ)| ≤ A) (hn : 1 ≤ f.natDegree)
    (hd : ∀ k, degreeOf k P ≤ r k) (hc : ∀ d, |(P.coeff d : ℝ)| ≤ B)
    (hs : ∑ k, r k ≤ s) (h2 : (2 : ℝ) ^ s ≤ β) (hpow : (1 + A) ^ s ≤ β)
    (j : σ →₀ ℕ) (i : ℕ) :
    |(diag (hasseDeriv j P) %ₘ f).coeff i| ≤ (⌊B * β ^ 3⌋₊ : ℤ) := by
  have hB0 : 0 ≤ B := (abs_nonneg _).trans (hc 0)
  have hA0 : 0 ≤ A := (abs_nonneg _).trans (hA 0)
  have hβ0 : 0 ≤ β := (pow_nonneg (by norm_num) s).trans h2
  have hNle : (∏ k, (r k + 1) : ℕ) ≤ 2 ^ (∑ k, r k) := by
    rw [← Finset.prod_pow_eq_pow_sum]
    exact Finset.prod_le_prod fun k _ ↦ Nat.lt_two_pow_self
  have hNR : ((∏ k, (r k + 1) : ℕ) : ℝ) ≤ 2 ^ (∑ k, r k) := by exact_mod_cast hNle
  have e1 : (1 + A) ^ ((∑ k, r k) + 1 - f.natDegree) ≤ (1 + A) ^ s :=
    pow_le_pow_right₀ (le_add_of_nonneg_right hA0) (by omega)
  have e2 : (2 : ℝ) ^ (∑ k, r k) ≤ 2 ^ s :=
    pow_le_pow_right₀ (by norm_num) hs
  have hreal : |((diag (hasseDeriv j P) %ₘ f).coeff i : ℝ)| ≤ B * β ^ 3 := by
    calc _ ≤ _ := abs_coeff_diag_hasseDeriv_modByMonic_le_aux hA hd hc j i
      _ ≤ (1 + A) ^ s * (2 ^ s * (2 ^ s * B)) := by
        refine mul_le_mul e1 ?_ (by positivity) (by positivity)
        exact mul_le_mul (hNR.trans e2) (mul_le_mul_of_nonneg_right e2 hB0)
          (by positivity) (by positivity)
      _ ≤ β * (β * (β * B)) := by
        refine mul_le_mul hpow ?_ (by positivity) hβ0
        exact mul_le_mul h2 (mul_le_mul_of_nonneg_right h2 hB0) (by positivity) hβ0
      _ = B * β ^ 3 := by ring
  rw [Int.abs_eq_natAbs]
  have hcNatAbs : (((diag (hasseDeriv j P) %ₘ f).coeff i).natAbs : ℝ) =
      |((diag (hasseDeriv j P) %ₘ f).coeff i : ℝ)| := by
    rw [Nat.cast_natAbs, Int.cast_abs]
  exact_mod_cast Nat.le_floor (by rw [hcNatAbs]; exact hreal)
