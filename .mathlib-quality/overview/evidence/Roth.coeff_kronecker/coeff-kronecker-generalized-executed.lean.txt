namespace CoeffKroneckerGeneralized
open MvPolynomial
theorem positional {R σ : Type*} [CommSemiring R] {n k : ℕ} (e : σ ≃ Fin n)
    {P : MvPolynomial σ R} (hP : ∀ j, degreeOf j P < k)
    {d : σ →₀ ℕ} (hd : ∀ j, d j < k) :
    (aeval (fun j ↦ (Polynomial.X : Polynomial R) ^ k ^ (e j : ℕ)) P).coeff
      (Finsupp.weight (fun j ↦ k ^ (e j : ℕ)) d) = P.coeff d := by
  exact CoeffKroneckerProbe.abstract_coeff _ {v : σ →₀ ℕ | ∀ j, v j < k} P
    (fun v hv j ↦ (monomial_le_degreeOf j hv).trans_lt (hP j))
    (KroneckerExpInjOnNoBase.no_base e k) hd
#print axioms positional
end CoeffKroneckerGeneralized
