theorem hom_bridge {R σ : Type*} [CommSemiring R] (w : σ → ℕ) :
    (Polynomial.toFinsuppIsoAlg R).toAlgHom.comp
      (aeval (fun j ↦ (Polynomial.X : Polynomial R) ^ w j)) =
        AddMonoidAlgebra.mapDomainAlgHom R R (Finsupp.weight w) := by
  ext i
  simp [MvPolynomial.X, AddMonoidAlgebra.mapDomainAlgHom,
    AddMonoidAlgebra.mapDomainRingHom, Finsupp.weight_single,
    Polynomial.toFinsuppIsoAlg, Polynomial.toFinsupp_X_pow]
