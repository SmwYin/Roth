theorem bridge {R σ : Type*} [CommSemiring R] (w : σ → ℕ) (P : MvPolynomial σ R) :
    (aeval (fun j ↦ (Polynomial.X : Polynomial R) ^ w j) P).toFinsupp.coeff =
      P.coeff.mapDomain (Finsupp.weight w) := by
  classical
  induction P using MvPolynomial.induction_on' with
  | monomial d a =>
    simp only [aeval_monomial, Polynomial.algebraMap_eq, Finsupp.prod, Finsupp.weight_apply,
      Finsupp.sum, smul_eq_mul, ← pow_mul, mul_comm, ← Finset.prod_pow_eq_pow_sum,
      Polynomial.toFinsupp_C_mul_X_pow]
    simp [MvPolynomial.monomial, Finsupp.mapDomain_single]
  | add P Q hP hQ =>
    simp only [map_add, Polynomial.toFinsupp_add, AddMonoidAlgebra.coeff_add,
      hP, hQ, Finsupp.mapDomain_add]
theorem abstract_coeff {R σ : Type*} [CommSemiring R] (w : σ → ℕ)
    (S : Set (σ →₀ ℕ)) (P : MvPolynomial σ R) (hP : (P.coeff.support : Set _) ⊆ S)
    (hi : Set.InjOn (Finsupp.weight w) S) {d : σ →₀ ℕ} (hd : d ∈ S) :
    (aeval (fun j ↦ (Polynomial.X : Polynomial R) ^ w j) P).coeff
      (Finsupp.weight w d) = P.coeff d := by
  rw [← Polynomial.toFinsupp_apply, bridge]
  exact Finsupp.mapDomain_apply' S P.coeff hP hi hd
