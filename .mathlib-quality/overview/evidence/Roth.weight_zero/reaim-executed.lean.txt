import Mathlib.Data.Finsupp.Weight
import Mathlib.Basic.Real.Basic
#check_failure Roth.weight_zero
#check_failure Roth.weight_add
namespace RawWeightReaimChecks
variable {σ : Type*} [Fintype σ]
noncomputable def original (r : σ → ℕ) (d : σ →₀ ℕ) : ℝ :=
  ∑ j, (d j : ℝ) / r j
theorem bridge (r : σ → ℕ) (d : σ →₀ ℕ) :
    original r d = Finsupp.weight (fun j ↦ (r j : ℝ)⁻¹) d := by
  simp [original, Finsupp.weight_eq_sum, nsmul_eq_mul, div_eq_mul_inv]
theorem zero_via_existing_map (r : σ → ℕ) : original r 0 = 0 :=
  (bridge r 0).trans (map_zero _)
theorem add_via_existing_map (r : σ → ℕ) (d e : σ →₀ ℕ) :
    original r (d + e) = original r d + original r e := by
  simpa [original, Finsupp.weight_eq_sum, nsmul_eq_mul, div_eq_mul_inv] using
    (map_add (Finsupp.weight (fun j ↦ (r j : ℝ)⁻¹)) d e)
#print axioms zero_via_existing_map
#print axioms add_via_existing_map
end RawWeightReaimChecks
