#check_failure MvPolynomial.coeff_taylor
namespace MvPolynomial.CoeffTaylorComposition
variable {σ R : Type*} [CommSemiring R]
example (α : σ → R) (d : σ →₀ ℕ) (P : MvPolynomial σ R) :
    (aeval (fun j ↦ X j + C (α j)) P).coeff d =
      ∑ k ∈ P.support,
        ((d.prod (fun j n ↦ (k j).choose n) : ℕ) : R) * P.coeff k *
          (k - d).prod (fun j n ↦ α j ^ n) := by
  simp only [aeval_def, eval₂_eq, coeff_sum]
