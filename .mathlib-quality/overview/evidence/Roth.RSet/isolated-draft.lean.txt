namespace RSet179Isolated
open MvPolynomial
#check MvPolynomial.restrictSupport
#check MvPolynomial.mem_restrictSupport_iff
#check MvPolynomial.basisRestrictSupport
#check MvPolynomial.restrictSupport_add
#check MvPolynomial.mem_restrictDegree
#check MvPolynomial.degreeOf_le_iff
example {σ : Type*} {R : Type*} [CommSemiring R] (P : MvPolynomial σ R) (r : σ → ℕ) :
    (∀ j, degreeOf j P ≤ r j) ↔
      P ∈ MvPolynomial.restrictSupport R {d | ∀ j, d j ≤ r j} := by
  simp only [MvPolynomial.mem_restrictSupport_iff, Set.subset_def, Finset.mem_coe,
    MvPolynomial.degreeOf_le_iff]
  exact ⟨fun h d hd j ↦ h j d hd, fun h j d hd ↦ h d hd j⟩
end RSet179Isolated
