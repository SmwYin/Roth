/-! Isolated iterativity assessment: raw coefficient and cocycle witnesses, no project import. -/
open MvPolynomial
namespace CoeffAssessment
variable {σ R : Type*} [CommSemiring R]
noncomputable def delta (d : σ →₀ ℕ) : MvPolynomial σ R →ₗ[R] MvPolynomial σ R :=
  Finsupp.lsum R (fun k : σ →₀ ℕ ↦
    (monomial (k - d)).comp (LinearMap.lsmul R R ((d.prod (fun j ↦ (k j).choose) : ℕ) : R))) ∘ₗ
      (AddMonoidAlgebra.coeffLinearEquiv R).toLinearMap

-- Coefficient formula is independently proved below.
theorem independent (d e : σ →₀ ℕ) (P : MvPolynomial σ R) :
    (delta d P).coeff e = ((d.prod (fun j ↦ ((e + d) j).choose) : ℕ) : R) * P.coeff (e + d) := by
  classical
  have hz (k : σ →₀ ℕ) : d.prod (fun j ↦ (k j).choose) = 0 ↔ ¬d ≤ k := by
    simp [Finsupp.prod, Finset.prod_eq_zero_iff, Nat.choose_eq_zero_iff, Finsupp.le_iff,
      not_forall]
  induction P using MvPolynomial.induction_on' with
  | monomial k a =>
    rw [show delta d (monomial k a) =
      monomial (k - d) (((d.prod (fun j ↦ (k j).choose) : ℕ) : R) * a) by simp [delta],
      coeff_monomial, coeff_monomial]
    by_cases hk : k = e + d
    · simp [hk]
    by_cases hdk : d ≤ k
    · simp [hk, tsub_eq_iff_eq_add_of_le hdk]
    · simp [hk, (hz k).2 hdk]
  | add p q hp hq => simp [hp, hq, mul_add]


#print axioms independent
section
variable {S : Type*} [Semiring S]
#check_failure MvPolynomial σ S
end
end CoeffAssessment
namespace RawHasseDegree
open MvPolynomial
variable {σ R : Type*} [CommSemiring R]
theorem support_shift (j : σ) (d : σ →₀ ℕ) (P Q : MvPolynomial σ R)
    (hshift : ∀ m, Q.coeff m ≠ 0 → P.coeff (m + d) ≠ 0) :
    degreeOf j Q ≤ degreeOf j P - d j := by
  refine degreeOf_le_iff.2 fun m hm ↦ ?_
  have h := monomial_le_degreeOf j (mem_support_iff.2 (hshift m (mem_support_iff.1 hm)))
  exact Nat.le_sub_of_add_le (by simpa using h)
theorem degree_delta (j : σ) (d : σ →₀ ℕ) (P : MvPolynomial σ R) :
    degreeOf j (CoeffAssessment.delta d P) ≤ degreeOf j P - d j := by
  apply support_shift j d P
  intro m hm
  rw [CoeffAssessment.independent] at hm
  exact right_ne_zero_of_mul hm
#print axioms support_shift
#print axioms degree_delta
end RawHasseDegree
