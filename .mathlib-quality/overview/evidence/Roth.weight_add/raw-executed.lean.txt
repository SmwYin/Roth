import Mathlib.Data.Finsupp.Weight
import Mathlib.Basic.Real.Basic
#check_failure Roth.weight_add
open scoped BigOperators
namespace RawWeightAddAssessment
variable {σ : Type*}
noncomputable def original [Fintype σ] (r : σ → ℕ) (d : σ →₀ ℕ) : ℝ :=
  ∑ j, (d j : ℝ) / r j
theorem original_three_facts [Fintype σ] (r : σ → ℕ) (d e : σ →₀ ℕ) :
    original r (d + e) = original r d + original r e :=
  (congrArg (fun f : σ → ℝ ↦ ∑ j, f j)
    (funext fun j ↦ (congrArg (fun x : ℝ ↦ x / (r j : ℝ))
      (Nat.cast_add (d j) (e j))).trans (add_div _ _ _))).trans Finset.sum_add_distrib
theorem arbitrary_weight {M : Type*} [AddCommMonoid M] (w : σ → M) (d e : σ →₀ ℕ) :
    Finsupp.weight w (d + e) = Finsupp.weight w d + Finsupp.weight w e :=
  map_add (Finsupp.weight w) d e
theorem arbitrary_coefficients {M K : Type*} [Semiring K] [AddCommMonoid M] [Module K M]
    (w : σ → M) (d e : σ →₀ K) :
    Finsupp.weight w (d + e) = Finsupp.weight w d + Finsupp.weight w e :=
  map_add (Finsupp.weight w) d e
theorem finite_functions {K : Type*} [DivisionSemiring K] (s : Finset σ)
    (r d e : σ → K) :
    ∑ j ∈ s, (d j + e j) / r j = (∑ j ∈ s, d j / r j) + ∑ j ∈ s, e j / r j :=
  (congrArg (fun f : σ → K ↦ ∑ j ∈ s, f j)
    (funext fun j ↦ add_div (d j) (e j) (r j))).trans Finset.sum_add_distrib
theorem nat_coefficients [Fintype σ] {K : Type*} [DivisionSemiring K]
    (r : σ → K) (d e : σ →₀ ℕ) :
    ∑ j, ((d + e) j : K) / r j = (∑ j, (d j : K) / r j) + ∑ j, (e j : K) / r j :=
  (congrArg (fun f : σ → K ↦ ∑ j, f j)
    (funext fun j ↦ (congrArg (fun x : K ↦ x / r j)
      (Nat.cast_add (d j) (e j))).trans (add_div _ _ _))).trans Finset.sum_add_distrib
theorem bridge [Fintype σ] (r : σ → ℕ) (d : σ →₀ ℕ) :
    original r d = Finsupp.weight (fun j ↦ (r j : ℝ)⁻¹) d := by
  simp [original, Finsupp.weight_eq_sum, nsmul_eq_mul, div_eq_mul_inv]
example (d e : ℕ →₀ ℕ) :
    Finsupp.weight (fun n : ℕ ↦ (n : ℝ)) (d + e) =
      Finsupp.weight (fun n : ℕ ↦ (n : ℝ)) d + Finsupp.weight (fun n : ℕ ↦ (n : ℝ)) e :=
  map_add _ _ _
example : original (fun _ : Unit ↦ 0) (Finsupp.single () 1 + Finsupp.single () 1) = 0 := by
  simp [original]
example : ¬ original (fun _ : Unit ↦ 1) (Finsupp.single () 1 + Finsupp.single () 1) =
    original (fun _ : Unit ↦ 1) (Finsupp.single () 1) := by
  simp [original]

end RawWeightAddAssessment
