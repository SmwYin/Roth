theorem manual_reduction {α : ℝ} (hα : Irrational α) (halg : IsAlgebraic ℚ α) {κ : ℝ} (hκ : 2 < κ)
    (hinf : (approxSet α κ).Infinite) : ∃ (α' κ' : ℝ), Irrational α' ∧ IsIntegral ℤ α' ∧ 2 < κ' ∧
    (approxSet α' κ').Infinite := by
  obtain ⟨y, hy, hint⟩ :=
    ((IsFractionRing.isAlgebraic_iff ℤ ℚ ℝ).2 halg).exists_integral_multiple
  let M := y.natAbs
  have hM0 : 0 < M := Int.natAbs_pos.2 hy
  have hMint : IsIntegral ℤ ((M : ℝ) * α) := by
    simpa [M, zsmul_eq_mul, ← mul_assoc, ← Int.cast_mul, Int.sign_mul_self]
      using hint.zsmul y.sign
  obtain ⟨Q₀, hQ₀⟩ := eventually_div_rpow_lt (M := M) (show (κ + 2) / 2 < κ by linarith)
  refine ⟨M * α, (κ + 2) / 2, hα.natCast_mul hM0.ne', hMint, by linarith, ?_⟩
  refine ((hinf.sdiff (approxSet_den_le_finite α κ Q₀)).image
    ((mul_right_injective₀ (by exact_mod_cast hM0.ne' : (M : ℤ) ≠ 0)).prodMap
      Function.injective_id).injOn).mono ?_
  rintro _ ⟨⟨h, q⟩, ⟨⟨hq0, happ⟩, hnot⟩, rfl⟩
  refine ⟨hq0, ?_⟩
  dsimp only [Prod.map, id]
  push_cast
  rw [mul_div_assoc, ← mul_sub, abs_mul, Nat.abs_cast]
  calc (M : ℝ) * |α - h / q| < M / (q : ℝ) ^ κ := by
        simpa only [mul_one_div] using mul_lt_mul_of_pos_left happ (by positivity : (0 : ℝ) < M)
    _ < 1 / (q : ℝ) ^ ((κ + 2) / 2) := hQ₀ q (not_le.1 fun hc ↦ hnot ⟨⟨hq0, happ⟩, hc⟩)
theorem closure_reduction {α : ℝ} (hα : Irrational α) (halg : IsAlgebraic ℚ α) {κ : ℝ} (hκ : 2 < κ)
    (hinf : (approxSet α κ).Infinite) : ∃ (α' κ' : ℝ), Irrational α' ∧ IsIntegral ℤ α' ∧ 2 < κ' ∧
    (approxSet α' κ').Infinite := by
  obtain ⟨M, s, hM, hs⟩ :=
    ((IsFractionRing.isAlgebraic_iff ℤ ℚ ℝ).2 halg).exists_nsmul_eq (integralClosure ℤ ℝ)
  have hM0 : 0 < M := Nat.pos_of_ne_zero hM
  have hMint : IsIntegral ℤ ((M : ℝ) * α) := by
    have hi : IsIntegral ℤ (M • α) := by
      rw [hs]
      exact s.property
    simpa only [nsmul_eq_mul] using hi
  obtain ⟨Q₀, hQ₀⟩ := eventually_div_rpow_lt (M := M) (show (κ + 2) / 2 < κ by linarith)
  refine ⟨M * α, (κ + 2) / 2, hα.natCast_mul hM0.ne', hMint, by linarith, ?_⟩
  refine ((hinf.sdiff (approxSet_den_le_finite α κ Q₀)).image
    ((mul_right_injective₀ (by exact_mod_cast hM0.ne' : (M : ℤ) ≠ 0)).prodMap
      Function.injective_id).injOn).mono ?_
  rintro _ ⟨⟨h, q⟩, ⟨⟨hq0, happ⟩, hnot⟩, rfl⟩
  refine ⟨hq0, ?_⟩
  dsimp only [Prod.map, id]
  push_cast
  rw [mul_div_assoc, ← mul_sub, abs_mul, Nat.abs_cast]
  calc (M : ℝ) * |α - h / q| < M / (q : ℝ) ^ κ := by
        simpa only [mul_one_div] using mul_lt_mul_of_pos_left happ (by positivity : (0 : ℝ) < M)
    _ < 1 / (q : ℝ) ^ ((κ + 2) / 2) := hQ₀ q (not_le.1 fun hc ↦ hnot ⟨⟨hq0, happ⟩, hc⟩)
