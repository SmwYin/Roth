theorem signed {ι K : Type*} [Fintype ι] [Field K] [LinearOrder K]
    [IsStrictOrderedRing K] (Q : MvPolynomial (ι) ℤ) {r : ι → ℕ}
    (hQ : ∀ j, degreeOf j Q ≤ r j) (h : ι → ℤ) {q : ι → ℤ} (hq : ∀ j, q j ≠ 0)
    (hne : eval (fun j ↦ (h j : K) / q j) (map (Int.castRingHom K) Q) ≠ 0) :
    1 ≤ |∏ j, (q j : K) ^ r j| * |eval (fun j ↦ (h j : K) / q j) (map (Int.castRingHom K) Q)| := by
  have hqR : ∀ j, (q j : K) ≠ 0 := fun j ↦ by exact_mod_cast hq j
  have hP : (∏ j, (q j : K) ^ r j) ≠ 0 :=
    Finset.prod_ne_zero_iff.mpr fun j _ ↦ pow_ne_zero _ (hqR j)
  set N : ℤ := ∑ d ∈ Q.support, Q.coeff d * ∏ j, h j ^ d j * (q j : ℤ) ^ (r j - d j) with hN
  have hNeq : (N : K) = (∏ j, (q j : K) ^ r j) *
      eval (fun j ↦ (h j : K) / q j) (map (Int.castRingHom K) Q) := by
    rw [eval_eq', support_map_of_injective _ Int.cast_injective, Finset.mul_sum, hN]
    push_cast
    refine Finset.sum_congr rfl fun d hd ↦ ?_
    rw [coeff_map, mul_left_comm, ← Finset.prod_mul_distrib]
    congr 1
    refine Finset.prod_congr rfl fun j _ ↦ ?_
    rw [← pow_sub_mul_pow (q j : K) ((monomial_le_degreeOf j hd).trans (hQ j)), div_pow]
    rw [mul_assoc, mul_div_cancel₀ _ (pow_ne_zero _ (hqR j)), mul_comm]
  have h1 : (1 : K) ≤ |(N : K)| := by
    rw [← Int.cast_abs]
    exact_mod_cast Int.one_le_abs (Int.cast_ne_zero.mp (hNeq.symm ▸ mul_ne_zero hP hne))
  rwa [hNeq, abs_mul] at h1

theorem eval₂Form {ι K : Type*} [Fintype ι] [Field K] [LinearOrder K]
    [IsStrictOrderedRing K] (Q : MvPolynomial (ι) ℤ) {r : ι → ℕ}
    (hQ : ∀ j, degreeOf j Q ≤ r j) (h : ι → ℤ) {q : ι → ℤ} (hq : ∀ j, q j ≠ 0)
    (hne : eval₂ (Int.castRingHom K) (fun j ↦ (h j : K) / q j) Q ≠ 0) :
    1 ≤ |∏ j, (q j : K) ^ r j| * |eval₂ (Int.castRingHom K) (fun j ↦ (h j : K) / q j) Q| := by
  have hn : eval (fun j ↦ (h j : K) / q j) (map (Int.castRingHom K) Q) ≠ 0 := by
    simpa only [eval_map] using hne
  simpa only [eval_map] using signed (K := K) Q hQ h hq hn
