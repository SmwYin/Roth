universe u v
theorem modern {σ : Type u} {R : Type v} [CommSemiring R] [NoZeroDivisors R]
    [Nontrivial R] {ι : Type*} (w : σ → ℕ) (f : ι → MvPowerSeries σ R) (s : Finset ι) :
    (∏ i ∈ s, f i).weightedOrder w = ∑ i ∈ s, (f i).weightedOrder w :=
  MvPowerSeries.weightedOrder_prod w f s
theorem polynomial {σ : Type u} {R : Type v} [CommSemiring R] [NoZeroDivisors R]
    [Nontrivial R] {ι : Type*} (w : σ → ℕ) (f : ι → MvPolynomial σ R) (s : Finset ι) :
    ((∏ i ∈ s, f i : MvPolynomial σ R) : MvPowerSeries σ R).weightedOrder w =
      ∑ i ∈ s, (f i : MvPowerSeries σ R).weightedOrder w := by
  rw [MvPolynomial.coe_prod, MvPowerSeries.weightedOrder_prod]
