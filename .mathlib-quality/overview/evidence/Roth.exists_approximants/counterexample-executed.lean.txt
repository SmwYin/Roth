namespace Approximants277Counterexample
def S (α κ : ℝ) : Set (ℤ × ℕ) :=
  {hq | 0 < hq.2 ∧ |α - (hq.1 : ℝ) / hq.2| < 1 / (hq.2 : ℝ) ^ κ}
theorem infinite_zero_pairs : (S 0 3).Infinite := by
  refine (Set.infinite_range_of_injective (f := fun n : ℕ ↦ ((0 : ℤ), n + 1))
    (fun _ _ h ↦ Nat.add_right_cancel (congrArg Prod.snd h))).mono ?_
  rintro _ ⟨n, rfl⟩
  refine ⟨by simp, ?_⟩
  simp only [Int.cast_zero, zero_div, sub_zero, abs_zero]
  positivity

theorem no_large_reduced : ¬ ∃ hq ∈ S 0 3, 1 < hq.2 ∧
    IsCoprime hq.1 (hq.2 : ℤ) := by
  rintro ⟨⟨h,q⟩, ⟨hq, happ⟩, hq1, hcop⟩
  have hqR : (1 : ℝ) < q := by exact_mod_cast hq1
  have hzero : h = 0 := by
    by_contra hn
    have h1 : (1 : ℝ) ≤ |(h : ℝ)| := by
      exact_mod_cast Int.one_le_abs hn
    have hqpos : (0 : ℝ) < q := by positivity
    have hbound : |(h : ℝ)| < 1 / (q : ℝ) ^ (2 : ℕ) := by
      have ht : |(h : ℝ)| / q < 1 / (q : ℝ) ^ (3 : ℕ) := by
        simpa [abs_div, abs_of_pos hqpos, Real.rpow_ofNat] using happ
      rw [div_lt_iff₀ hqpos] at ht
      convert ht using 1
      field_simp
    have : 1 / (q : ℝ) ^ (2 : ℕ) < 1 := by
      rw [div_lt_one (by positivity)]
      nlinarith
    linarith
  subst h
  have : q = 1 := by simpa [Int.isCoprime_iff_gcd_eq_one] using hcop
  omega
#print axioms Approximants277Counterexample.infinite_zero_pairs
#print axioms Approximants277Counterexample.no_large_reduced
theorem no_chain : ¬ ∃ (h : Fin 1 → ℤ) (q : Fin 1 → ℕ),
    (∀ j, (h j, q j) ∈ S 0 3) ∧ (∀ j, IsCoprime (h j) (q j : ℤ)) ∧
      (∀ j, 1 < q j) ∧ (0 : ℝ) < Real.log (q 0) ∧
      ∀ i j : Fin 1, i.val + 1 = j.val →
        (2 / (1 : ℝ)) * Real.log (q i) < Real.log (q j) := by
  rintro ⟨h, q, hm, hc, hq, -⟩
  exact no_large_reduced ⟨(h 0, q 0), hm 0, hq 0, hc 0⟩
#print axioms Approximants277Counterexample.no_chain
end Approximants277Counterexample
