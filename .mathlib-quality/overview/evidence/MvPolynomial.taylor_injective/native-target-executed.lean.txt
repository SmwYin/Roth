#check @Function.LeftInverse.injective
namespace MvPolynomial.TaylorInjectiveAssessment
variable {σ A : Type*} [CommRing A]
theorem replay (α : σ → A) : Function.Injective (taylor α : MvPolynomial σ A → _) :=
  Function.LeftInverse.injective (taylor_neg_taylor α)
theorem independent (α : σ → A) : Function.Injective (taylor α : MvPolynomial σ A → _) :=
  Function.LeftInverse.injective (TaylorInverseAssessment.replay α)
theorem bundled (α : σ → A) : Function.Injective (taylor α : MvPolynomial σ A → _) :=
  (TaylorInverseExtra.equivalence α).injective
theorem zeroCaller (α : σ → A) (P : MvPolynomial σ A) : taylor α P = 0 ↔ P = 0 :=
  (replay α).eq_iff' (map_zero _)

end MvPolynomial.TaylorInjectiveAssessment
