namespace MvPolynomial.DegreeTaylorRaw
variable {σ R : Type*} [CommSemiring R]
def shift (α : σ → R) : MvPolynomial σ R →ₐ[R] MvPolynomial σ R :=
  aeval fun j ↦ X j + C (α j)
theorem coefficientCriterion (Q : MvPolynomial σ R) (j : σ) (n : ℕ)
    (h : ∀ d : σ →₀ ℕ, n < d j → Q.coeff d = 0) : degreeOf j Q ≤ n :=
  degreeOf_le_iff.mpr fun d hd ↦ not_lt.mp fun hlt ↦
    (mem_support_iff.mp hd) (h d hlt)
theorem shiftCriterion (α : σ → R) (j : σ) (P : MvPolynomial σ R)
    (h : ∀ d : σ →₀ ℕ, degreeOf j P < d j → (shift α P).coeff d = 0) :
    degreeOf j (shift α P) ≤ degreeOf j P :=
  coefficientCriterion (shift α P) j (degreeOf j P) h
example (Q : Polynomial R) (r : R) : (Polynomial.taylor r Q).natDegree = Q.natDegree :=
  Polynomial.natDegree_taylor Q r
