namespace Witness254
universe u v w
theorem abstract_cutoff {I : Type u} {A : Type v} {L : Type w} [Semiring A]
    [LinearOrder L] [OrderTop L] (D : I → L) (P : AddMonoidAlgebra A I) (c : L) :
    AddMonoidAlgebra.infDegree D P < c ↔ ∃ i ∈ P.coeff.support, D i < c :=
  Finset.inf_lt_iff
theorem coefficient_cutoff {I A L : Type*} [Semiring A] [LinearOrder L] [OrderTop L]
    (D : I → L) (P : AddMonoidAlgebra A I) {c : L}
    (hc : AddMonoidAlgebra.infDegree D P < c) : ∃ i, D i < c ∧ P.coeff i ≠ 0 :=
  (Finset.inf_lt_iff.mp hc).imp fun _ ⟨hi, hw⟩ ↦ ⟨hw, Finsupp.mem_support_iff.mp hi⟩
theorem projected_cutoff {I A : Type*} [Semiring A] (D : I → ℝ)
    (P : AddMonoidAlgebra A I) (hP : P.coeff.support.Nonempty) {c : ℝ}
    (hc : (AddMonoidAlgebra.infDegree (fun i ↦ (D i : WithTop ℝ)) P).untopD 0 < c) :
    ∃ i, D i < c ∧ P.coeff i ≠ 0 := by
  change (P.coeff.support.inf (WithTop.some ∘ D)).untopD 0 < c at hc
  rw [← Finset.coe_inf' hP D, WithTop.untopD_coe] at hc
  exact (Finset.inf'_lt_iff hP |>.mp hc).imp fun _ ⟨hi, hw⟩ ↦
    ⟨hw, Finsupp.mem_support_iff.mp hi⟩
theorem finite_real_cutoff {I : Type*} (s : Finset I) (hs : s.Nonempty) (D : I → ℝ)
    {c : ℝ} (hc : sInf (D '' (↑s : Set I)) < c) : ∃ i ∈ s, D i < c := by
  rw [← Finset.inf'_eq_csInf_image] at hc
  exact (Finset.inf'_lt_iff hs).mp hc
theorem translated_cutoff {σ : Type*} (P : MvPolynomial σ ℝ) (α : σ → ℝ)
    (D : (σ →₀ ℕ) → ℝ)
    (hP : (MvPolynomial.aeval (fun j ↦ MvPolynomial.X j + MvPolynomial.C (α j)) P)
      .coeff.support.Nonempty) {c : ℝ}
    (hc : (AddMonoidAlgebra.infDegree (fun d ↦ (D d : WithTop ℝ))
      (MvPolynomial.aeval (fun j ↦ MvPolynomial.X j + MvPolynomial.C (α j)) P))
      .untopD 0 < c) : ∃ d, D d < c ∧
      (MvPolynomial.aeval (fun j ↦ MvPolynomial.X j + MvPolynomial.C (α j)) P).coeff d ≠ 0 :=
  projected_cutoff D _ hP hc
end Witness254
