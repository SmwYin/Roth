-- Complete arithmetic predicate, without importing any Roth module.
theorem arithmetic {σ : Type u} {B : ℝ} (hB : 1 ≤ B) (r : σ → ℕ) :
    (1 : MvPolynomial σ ℤ) ≠ 0 ∧
      (∀ j, degreeOf j (1 : MvPolynomial σ ℤ) ≤ r j) ∧
      ∀ d, |(((1 : MvPolynomial σ ℤ).coeff d : ℤ) : ℝ)| ≤ B := by
  classical
  simp [coeff_one, apply_ite, ite_apply, hB, zero_le_one.trans hB]
#print axioms arithmetic
-- The fixed coefficient and bound representation can be expanded to normed fields.
theorem normed_comm_ring {σ : Type u} {K : Type v} [NormedCommRing K] [NormOneClass K]
    [Nontrivial K] {B : ℝ}
    (hB : 1 ≤ B) (r : σ → ℕ) :
    (1 : MvPolynomial σ K) ≠ 0 ∧
      (∀ j, degreeOf j (1 : MvPolynomial σ K) ≤ r j) ∧
      ∀ d, ‖(1 : MvPolynomial σ K).coeff d‖ ≤ B := by
  classical
  simp [coeff_one, apply_ite, ite_apply, hB, zero_le_one.trans hB]
#print axioms normed_comm_ring
#check @normed_comm_ring
-- Necessity of the lower bound, witnessed at the constant coefficient.
theorem iff_bound {σ : Type u} {B : ℝ} (r : σ → ℕ) :
    ((1 : MvPolynomial σ ℤ) ≠ 0 ∧
      (∀ j, degreeOf j (1 : MvPolynomial σ ℤ) ≤ r j) ∧
      ∀ d, |(((1 : MvPolynomial σ ℤ).coeff d : ℤ) : ℝ)| ≤ B) ↔ 1 ≤ B := by
  constructor
  · intro h
    simpa using h.2.2 0
  · intro h
    exact arithmetic h r
#print axioms iff_bound
