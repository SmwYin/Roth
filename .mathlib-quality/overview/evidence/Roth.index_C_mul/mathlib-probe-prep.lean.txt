universe u v w z
theorem modern {A : Type u} {R : Type v} {S : Type w} {T : Type z}
    [Semiring R] [Semiring S] [IsDomain S] [Module S R] [Module.IsTorsionFree S R]
    [SemilatticeInf T] [OrderTop T] (D : A → T) {c : S} (hc : c ≠ 0)
    (f : AddMonoidAlgebra R A) :
    AddMonoidAlgebra.infDegree D (c • f) = AddMonoidAlgebra.infDegree D f := by
  classical
  simp only [AddMonoidAlgebra.infDegree, AddMonoidAlgebra.coeff_smul,
    Finsupp.support_smul_eq hc]
theorem supportGeneral {A : Type u} {M : Type v} {S : Type w}
    [Semiring S] [IsDomain S] [AddCommMonoid M] [Module S M]
    [Module.IsTorsionFree S M] {c : S} (hc : c ≠ 0) (f : A →₀ M) :
    (c • f).support = f.support := Finsupp.support_smul_eq hc
