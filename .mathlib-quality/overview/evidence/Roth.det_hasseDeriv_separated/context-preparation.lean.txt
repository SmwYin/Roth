namespace Det169Context
open MvPolynomial Roth
variable {τ : Type*} {K : Type*} [Field K]
theorem originalClone169 {l : ℕ} (D : Fin l → (τ →₀ ℕ)) (φ : Fin l → Polynomial K)
    (ψ : Fin l → MvPolynomial τ K) :
    (Matrix.of fun μ ν : Fin l ↦ hasseDeriv ((D μ).mapDomain some + Finsupp.single none (ν : ℕ))
        (∑ ρ, Polynomial.aeval (X none : MvPolynomial (Option τ) K) (φ ρ) *
          rename some (ψ ρ))).det = rename some (genWronskian D ψ) *
      Polynomial.aeval (X none : MvPolynomial (Option τ) K) (wronskian φ) := by
  rw [genWronskian, wronskian, AlgHom.map_det, AlgHom.map_det,
    ← Matrix.det_transpose ((Polynomial.aeval (X none)).mapMatrix (wronskianMatrix φ)),
    ← Matrix.det_mul]
  congr 1
  refine Matrix.ext fun μ ν ↦ ?_
  rw [Matrix.of_apply, map_sum, Matrix.mul_apply]
  refine Finset.sum_congr rfl fun ρ _ ↦ ?_
  rw [mul_comm (Polynomial.aeval (X none : MvPolynomial (Option τ) K) (φ ρ)),
    hasseDeriv_rename_mul_aeval]
  rfl

#print axioms originalClone169
section Ring
variable {R : Type*} [CommRing R]
theorem expandedRing169 {l : ℕ} (D : Fin l → (τ →₀ ℕ)) (φ : Fin l → Polynomial R)
    (ψ : Fin l → MvPolynomial τ R) :
    (Matrix.of fun μ ν : Fin l ↦ hasseDeriv ((D μ).mapDomain some + Finsupp.single none (ν : ℕ))
        (∑ ρ, Polynomial.aeval (X none : MvPolynomial (Option τ) R) (φ ρ) *
          rename some (ψ ρ))).det = rename some ((Matrix.of fun μ ρ ↦ hasseDeriv (D μ) (ψ ρ)).det) *
      Polynomial.aeval (X none : MvPolynomial (Option τ) R)
        ((Matrix.of fun ν ρ : Fin l ↦ Polynomial.hasseDeriv (ν : ℕ) (φ ρ)).det) := by
  rw [AlgHom.map_det, AlgHom.map_det,
    ← Matrix.det_transpose ((Polynomial.aeval (X none)).mapMatrix
      (Matrix.of fun ν ρ : Fin l ↦ Polynomial.hasseDeriv (ν : ℕ) (φ ρ))),
    ← Matrix.det_mul]
  congr 1
  refine Matrix.ext fun μ ν ↦ ?_
  rw [Matrix.of_apply, map_sum, Matrix.mul_apply]
  refine Finset.sum_congr rfl fun ρ _ ↦ ?_
  rw [mul_comm (Polynomial.aeval (X none : MvPolynomial (Option τ) R) (φ ρ)),
    hasseDeriv_rename_mul_aeval]
  rfl
