namespace Det169Guards
open MvPolynomial Roth
variable {τ : Type*} {R : Type*} [CommRing R] {l : ℕ}
theorem parentFloor (D : Fin l → τ →₀ ℕ) (φ : Fin l → Polynomial R)
    (ψ : Fin l → MvPolynomial τ R) : True := by
  fail_if_success
    have _ : Field R := inferInstance
  fail_if_success
    have _ := genWronskian D ψ
  fail_if_success
    have _ := wronskian φ
  trivial
#print axioms parentFloor
end Det169Guards
