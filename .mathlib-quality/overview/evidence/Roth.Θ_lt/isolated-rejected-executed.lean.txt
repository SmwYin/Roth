theorem uniform {α : Type*} [ConditionallyCompleteLinearOrder α] {S : Set α}
    (hS : S.Nonempty) {b c : α} (hb : ∀ x ∈ S, x ≤ b) (hbc : b < c) : sSup S < c :=
  (csSup_le hS hb).trans_lt hbc
theorem uniformReal {S : Set ℝ} {b c : ℝ} (hb : ∀ x ∈ S, x ≤ b)
    (hb0 : 0 ≤ b) (hbc : b < c) : sSup S < c :=
  (Real.sSup_le hb hb0).trans_lt hbc
theorem strictPointwiseNotEnough :
    (∀ x ∈ Set.Iio (1 : ℝ), x < 1) ∧ ¬sSup (Set.Iio (1 : ℝ)) < 1 := by
  simp only [Set.mem_Iio, csSup_Iio, lt_self_iff_false, not_false_eq_true, and_true]
  exact fun _ h ↦ h
theorem emptyGuard : ¬sSup (∅ : Set ℝ) ≤ -1 := by simp
open MvPolynomial
noncomputable def weight {σ : Type*} [Fintype σ] (r : σ → ℕ) (d : σ →₀ ℕ) : ℝ :=
  ∑ j, (d j : ℝ) / r j
noncomputable def index {σ : Type*} [Fintype σ] (P : MvPolynomial σ ℝ)
    (α : σ → ℝ) (r : σ → ℕ) : ℝ :=
  sInf (weight r '' ((taylor α P).support : Set (σ →₀ ℕ)))
def RSet {σ : Type*} (B : ℝ) (r : σ → ℕ) : Set (MvPolynomial σ ℤ) :=
  {R | R ≠ 0 ∧ (∀ j, degreeOf j R ≤ r j) ∧ ∀ d, |((R.coeff d : ℤ) : ℝ)| ≤ B}
def indexSet {σ : Type*} [Fintype σ] (B : ℝ) (q r : σ → ℕ) : Set ℝ :=
  {x | ∃ R ∈ RSet B r, ∃ h : σ → ℤ, (∀ j, IsCoprime (h j) (q j : ℤ)) ∧
    index (map (Int.castRingHom ℝ) R) (fun j ↦ (h j : ℝ) / q j) r = x}
noncomputable def theta {σ : Type*} [Fintype σ] (B : ℝ) (q r : σ → ℕ) : ℝ :=
  sSup (indexSet B q r)
-- The named general supremum laws cannot discharge the arithmetic uniform-bound premise.
example (n : ℕ) {δ : ℝ} (hδ0 : 0 < δ) {r q : Fin (n + 1) → ℕ}
    (hrlast : 3 ≤ (r (Fin.last n) : ℝ))
    (hratio : ∀ i : Fin n, (r i.succ : ℝ) ≤ δ * r i.castSucc)
    (hq1 : ((n : ℝ) + 1) * (2 * ((n : ℝ) + 1) + 1) / δ ≤ Real.log (q 0))
    (hqr : ∀ j, (r 0 : ℝ) * Real.log (q 0) ≤ r j * Real.log (q j)) : True := by
  fail_if_success
    have whole : theta ((q 0 : ℝ) ^ (δ * r 0)) q r <
        10 ^ (n + 1) * δ ^ ((1 / 2 : ℝ) ^ (n + 1)) := by
      exact?
  trivial
