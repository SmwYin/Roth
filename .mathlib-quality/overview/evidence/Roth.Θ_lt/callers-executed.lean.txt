variable {K : Type*} [Field K] [LinearOrder K] [IsStrictOrderedRing K]

theorem mul_div_two_le_sum_max_aux {θ r : K} {l : ℕ} (hl : (l : K) ≤ θ * r) :
    l * θ / 2 ≤ ∑ ν ∈ Finset.range l, max 0 (θ - ν / r) := by
  rcases Nat.eq_zero_or_pos l with rfl | hl0
  · simp
  by_cases hθ : θ ≤ 0
  · exact (div_nonpos_of_nonpos_of_nonneg
      (mul_nonpos_of_nonneg_of_nonpos (Nat.cast_nonneg l) hθ) (by norm_num)).trans
        (Finset.sum_nonneg fun _ _ ↦ le_max_left _ _)
  have hr := pos_of_mul_pos_right ((Nat.cast_pos.mpr hl0).trans_le hl) (le_of_not_ge hθ)
  have hsum := congrArg (Nat.cast (R := K)) (Finset.sum_range_id_mul_two l)
  push_cast [Nat.cast_sub hl0] at hsum
  refine le_trans ?_ (Finset.sum_le_sum fun ν _ ↦ le_max_right 0 (θ - ν / r))
  rw [Finset.sum_sub_distrib, Finset.sum_const, Finset.card_range, nsmul_eq_mul,
    ← Finset.sum_div, le_sub_iff_add_le, div_add_div _ _ (by norm_num) hr.ne',
    div_le_iff₀ (by positivity)]
  linarith only [hsum, (Nat.cast_nonneg l : (0 : K) ≤ l),
    mul_nonneg (Nat.cast_nonneg l) (sub_nonneg.mpr hl)]

theorem mul_sq_div_three_le_sum_max_of_pos_aux {θ r : K} (hr : 0 < r) (hθ : 0 ≤ θ) {l : ℕ}
    (hl : θ * r < l) : r * θ ^ 2 / 2 ≤ ∑ ν ∈ Finset.range l, max 0 (θ - ν / r) := by
  let n := Nat.findGreatest (fun k : ℕ ↦ (k : K) ≤ θ * r) l
  have hnle : (n : K) ≤ θ * r := Nat.findGreatest_spec (P := fun k : ℕ ↦ (k : K) ≤ θ * r)
    (Nat.zero_le l) (by simpa only [Nat.cast_zero] using mul_nonneg hθ hr.le)
  have hnl : n < l := by exact_mod_cast hnle.trans_lt hl
  have hnlt : θ * r < (n : K) + 1 := by
    have h := Nat.findGreatest_is_greatest (P := fun k : ℕ ↦ (k : K) ≤ θ * r)
      (Nat.lt_succ_self n) (Nat.succ_le_of_lt hnl)
    simpa only [Nat.cast_succ] using lt_of_not_ge h
  have hsum := congrArg (Nat.cast (R := K)) (Finset.sum_range_id_mul_two (n + 1))
  push_cast [Nat.add_sub_cancel] at hsum
  have hbound : (θ * r) ^ 2 / 2 ≤ ∑ ν ∈ Finset.range (n + 1), (θ * r - ν) := by
    rw [Finset.sum_sub_distrib, Finset.sum_const, Finset.card_range, nsmul_eq_mul]
    push_cast
    generalize θ * r = a at hnle hnlt ⊢
    linear_combination hsum / 2 + (le_trans (Nat.cast_nonneg n) hnle) / 2 +
      mul_nonneg (sub_nonneg.mpr hnle) (sub_nonneg.mpr hnlt.le) / 2
  calc
    r * θ ^ 2 / 2 = (θ * r) ^ 2 / 2 / r := (eq_div_iff hr.ne').2 (by ring)
    _ ≤ (∑ ν ∈ Finset.range (n + 1), (θ * r - ν)) / r :=
      (div_le_div_iff_of_pos_right hr).2 hbound
    _ ≤ ∑ ν ∈ Finset.range (n + 1), max 0 (θ - ν / r) := by
      simp only [Finset.sum_div, sub_div, mul_div_cancel_right₀ _ hr.ne']
      exact Finset.sum_le_sum fun ν _ ↦ le_max_right _ _
    _ ≤ ∑ ν ∈ Finset.range l, max 0 (θ - ν / r) :=
      Finset.sum_le_sum_of_subset_of_nonneg (Finset.range_subset_range.2
        (Nat.succ_le_of_lt hnl))
        fun _ _ _ ↦ le_max_left _ _

theorem mul_sq_div_three_le_sum_max_aux {θ r : K} (hθ : 0 ≤ θ) {l : ℕ} (hl : θ * r ≤ l) :
    r * θ ^ 2 / 2 ≤ ∑ ν ∈ Finset.range l, max 0 (θ - ν / r) := by
  rcases le_or_gt r 0 with hr | hr
  · exact (div_nonpos_of_nonpos_of_nonneg (mul_nonpos_of_nonpos_of_nonneg hr (sq_nonneg θ))
      (by norm_num)).trans (Finset.sum_nonneg fun _ _ ↦ le_max_left _ _)
  rcases lt_or_eq_of_le hl with hl | hl
  · exact mul_sq_div_three_le_sum_max_of_pos_aux hr hθ hl
  · linarith only [mul_div_two_le_sum_max_aux hl.ge, congrArg (· * θ) hl,
      mul_nonneg hr.le (sq_nonneg θ)]

theorem generic {θ rp : K} (l : ℕ) :
    min (l * θ / 2) (rp * θ ^ 2 / 2) ≤ ∑ ν ∈ Finset.range l, max 0 (θ - ν / rp) := by
  fail_if_success solve | exact?
  fail_if_success solve | simp
  fail_if_success solve | positivity
  by_cases hθ : θ ≤ 0
  · exact (min_le_left _ _).trans ((div_nonpos_of_nonpos_of_nonneg
      (mul_nonpos_of_nonneg_of_nonpos (Nat.cast_nonneg l) hθ) (by norm_num)).trans
        (Finset.sum_nonneg fun _ _ ↦ le_max_left _ _))
  rcases le_or_gt (l : K) (θ * rp) with hl | hl
  · exact (min_le_left _ _).trans (mul_div_two_le_sum_max_aux hl)
  · exact (min_le_right _ _).trans (mul_sq_div_three_le_sum_max_aux (le_of_not_ge hθ) hl.le)


end LemSix213Finite

variable {σ : Type*} [Fintype σ] [DecidableEq σ]

/-- The set `ℛ_m(B; r)` of nonzero integer polynomials whose degree in each variable `X j` is at
most `r j` and whose coefficients have absolute value at most `B`.
Source: §5 (items (1)–(3) of the definition of `ℛ_m`). -/
def RSet (B : ℝ) (r : σ → ℕ) : Set (MvPolynomial σ ℤ) :=
  {R | R ≠ 0 ∧ (∀ j, degreeOf j R ≤ r j) ∧ ∀ d, |((R.coeff d : ℤ) : ℝ)| ≤ B}

/-- The set of indices `ind(R; h/q; r)` of polynomials `R ∈ ℛ_m(B; r)` at the points
`(h_j / q_j)_j` with each `h_j` coprime to `q_j`; its supremum is `Θ_m(B; q; r)`.
Source: §5, definition of `Θ_m`. -/
def indexSet (B : ℝ) (q r : σ → ℕ) : Set ℝ :=
  {x | ∃ R ∈ RSet B r, ∃ h : σ → ℤ, (∀ j, IsCoprime (h j) (q j : ℤ)) ∧
    index (map (Int.castRingHom ℝ) R) (fun j ↦ (h j : ℝ) / q j) r = x}

/-- `Θ_m(B; q_1, …, q_m; r_1, …, r_m)` is the supremum of the indices `ind(R; h/q; r)` over
`R ∈ ℛ_m(B; r)` and `h` with each `h_j` coprime to `q_j`. Source: §5, definition of `Θ_m`. -/
noncomputable def Θ (B : ℝ) (q r : σ → ℕ) : ℝ := sSup (indexSet B q r)

omit [Fintype σ] [DecidableEq σ] in
/-- If `1 ≤ B`, the constant polynomial `1` lies in `ℛ_m(B; r)`. Source: §5 ("the constant
polynomial `1` with every `h_i = 1` is allowed"). -/
theorem one_mem_RSet {B : ℝ} (hB : 1 ≤ B) (r : σ → ℕ) : (1 : MvPolynomial σ ℤ) ∈ RSet B r := by
  classical
  simp [RSet, coeff_one, apply_ite, ite_apply, hB, zero_le_one.trans hB]

omit [DecidableEq σ] in
/-- If `1 ≤ B`, the set of indices defining `Θ_m(B; q; r)` is nonempty. Source: §5 ("This
supremum is well-defined"). -/
theorem indexSet_nonempty {B : ℝ} (hB : 1 ≤ B) (q r : σ → ℕ) : (indexSet B q r).Nonempty :=
  ⟨_, 1, one_mem_RSet hB r, fun _ ↦ 1, fun _ ↦ isCoprime_one_left, rfl⟩

omit [DecidableEq σ] in
/-- If every `r j` is positive, the indices defining `Θ_m(B; q; r)` are bounded above (by the
number `m` of variables). Source: §5 ("`0 ≤ Θ_m ≤ m`": Taylor's theorem gives a nonzero
derivative of weight at most `m`). -/
theorem indexSet_bddAbove (B : ℝ) (q : σ → ℕ) {r : σ → ℕ} : BddAbove (indexSet B q r) := by
  classical
  refine ⟨Fintype.card σ, ?_⟩
  rintro _ ⟨R, ⟨hR0, hRd, -⟩, h, -, rfl⟩
  exact index_le_card ((map_ne_zero_iff _ (map_injective _ Int.cast_injective)).2 hR0)
    (by simpa [degreeOf, degrees_map_of_injective R (f := Int.castRingHom ℝ)
          Int.cast_injective] using hRd) _

omit [DecidableEq σ] in
/-- The index of any `R ∈ ℛ_m(B; r)` at a point `(h_j / q_j)_j` with each `h_j` coprime to `q_j`
is at most `Θ_m(B; q; r)`. Source: §5, definition of `Θ_m` as a supremum, and the Lean note
("lemmas for bounding an individual index"). -/
theorem index_le_Θ {B : ℝ} {q r : σ → ℕ} {R : MvPolynomial σ ℤ} (hR : R ∈ RSet B r) {h : σ → ℤ}
    (hh : ∀ j, IsCoprime (h j) (q j : ℤ)) :
    index (map (Int.castRingHom ℝ) R) (fun j ↦ (h j : ℝ) / q j) r ≤ Θ B q r :=
  le_csSup (indexSet_bddAbove B q) ⟨R, hR, h, hh, rfl⟩

omit [DecidableEq σ] in
/-- `Θ_m(B; q; r)` is nonnegative for every coefficient bound and weight vector. Source: §5
("`0 ≤ Θ_m`"). -/
theorem Θ_nonneg {B : ℝ} (q : σ → ℕ) {r : σ → ℕ} : 0 ≤ Θ B q r := by
  apply Real.sSup_nonneg
  rintro _ ⟨R, _, h, _, rfl⟩
  exact index_nonneg _ _ _

omit [DecidableEq σ] in
/-- `Θ_m(B; q; r)` is at most the number `m` of variables. Source: §5 ("`Θ_m ≤ m`"). -/
theorem Θ_le_card {B : ℝ} (q : σ → ℕ) {r : σ → ℕ} : Θ B q r ≤ Fintype.card σ := by
  refine Real.sSup_le ?_ (Nat.cast_nonneg _)
  rintro _ ⟨R, ⟨hR0, hRd, -⟩, h, -, rfl⟩
  exact index_le_card ((map_ne_zero_iff _ (map_injective _ Int.cast_injective)).2 hR0)
    (by simpa [degreeOf, degrees_map_of_injective R (f := Int.castRingHom ℝ)
          Int.cast_injective] using hRd) _

omit [DecidableEq σ] in
/-- A bound on every index is a bound on `Θ`. Source: §5, Lemma 5 ("Taking the supremum proves
the claim"). -/
theorem Θ_le_of_forall_index_le {B : ℝ} (hB : 1 ≤ B) {q r : σ → ℕ} {c : ℝ}
    (hc : ∀ R ∈ RSet B r, ∀ h : σ → ℤ, (∀ j, IsCoprime (h j) (q j : ℤ)) →
      index (map (Int.castRingHom ℝ) R) (fun j ↦ (h j : ℝ) / q j) r ≤ c) : Θ B q r ≤ c := by
  refine csSup_le (indexSet_nonempty hB q r) ?_
  rintro _ ⟨R, hR, h, hh, rfl⟩
  exact hc R hR h hh

omit [DecidableEq σ] in
/-- `Θ_m(B; q; r)` is monotone in `B`. Source: §5 ("`Θ_m` is nondecreasing in `B`, since
increasing `B` enlarges the set of allowed polynomials"). -/
theorem Θ_mono {B B' : ℝ} (hB : 1 ≤ B) (hBB' : B ≤ B') (q : σ → ℕ) {r : σ → ℕ} :
    Θ B q r ≤ Θ B' q r := by
  refine csSup_le_csSup (indexSet_bddAbove B' q) (indexSet_nonempty hB q r) ?_
  rintro _ ⟨R, ⟨hR0, hRd, hRc⟩, h, hh, rfl⟩
  exact ⟨R, ⟨hR0, hRd, fun d ↦ (hRc d).trans hBB'⟩, h, hh, rfl⟩

omit [DecidableEq σ] in
private lemma indexSet_comp_subset {τ : Type*} [Fintype τ] (e : σ ≃ τ) (B : ℝ) (q r : τ → ℕ) :
    indexSet B (q ∘ e) (r ∘ e) ⊆ indexSet B q r := by
  rintro _ ⟨R, ⟨hR0, hRd, hRc⟩, h, hh, rfl⟩
  refine ⟨rename e R, ⟨?_, ?_, ?_⟩, h ∘ e.symm, ?_, ?_⟩
  · simpa [rename_eq_zero_iff_of_injective R e.injective] using hR0
  · intro j
    obtain ⟨i, rfl⟩ := e.surjective j
    simpa [degreeOf_rename_of_injective e.injective] using hRd i
  · intro d
    obtain ⟨d, rfl⟩ := Finsupp.mapDomain_surjective e.surjective d
    simpa [coeff_rename_mapDomain _ e.injective] using hRc d
  · intro j
    simpa using hh (e.symm j)
  · simp [map_rename, index_rename e.injective, Function.comp_def]

omit [DecidableEq σ] in
/-- `Θ` is invariant under renaming the variables. Source: §4, Lean note after Lemma 4-2
("invariance under ... variable renaming"). -/
theorem Θ_equiv {τ : Type*} [Fintype τ] (e : σ ≃ τ) (B : ℝ) (q r : τ → ℕ) :
    Θ B (q ∘ e) (r ∘ e) = Θ B q r := by
  apply congrArg sSup
  refine subset_antisymm (indexSet_comp_subset e B q r) ?_
  simpa [Function.comp_def] using indexSet_comp_subset e.symm B (q ∘ e) (r ∘ e)

section Lemma5

private lemma isPrimitive_linear {h : ℤ} {q : ℕ} (hcop : IsCoprime h (q : ℤ)) :
    (Polynomial.C (q : ℤ) * Polynomial.X - Polynomial.C h).IsPrimitive := by
  intro c hc
  rw [Polynomial.C_dvd_iff_dvd_coeff] at hc
  exact hcop.isUnit_of_dvd' (by simpa using hc 0)
    (by simpa only [Polynomial.coeff_sub, Polynomial.coeff_C_mul, Polynomial.coeff_X_one,
      Polynomial.coeff_C, mul_one, one_ne_zero, ↓reduceIte, sub_zero] using hc 1)

private lemma isPrimitive_pow {p : Polynomial ℤ} (hp : p.IsPrimitive) (t : ℕ) :
    (p ^ t).IsPrimitive :=
  Nat.rec Polynomial.isPrimitive_one (fun t ih ↦ by simpa [pow_succ] using ih.mul hp) t

/-- If `h` is coprime to `q > 0` and `t` is the multiplicity of `h / q` as a root of a nonzero
`R ∈ ℤ[X]`, then `R = (q X - h)^t Q` for some `Q ∈ ℤ[X]`.
Source: §5, proof of Lemma 5 ("The polynomial `(q_1 x_1 - h_1)^t` is primitive and divides
`R` over `ℚ`. Gauss's lemma therefore gives a nonzero `Q ∈ ℤ[x_1]` with
`R = (q_1 x_1 - h_1)^t Q`"). -/
theorem exists_eq_pow_mul_of_rootMultiplicity {R : Polynomial ℤ} {h : ℤ} {q : ℕ} (hq : 0 < q)
    (hcop : IsCoprime h (q : ℤ)) : ∃ Q : Polynomial ℤ, R =
      (Polynomial.C (q : ℤ) * Polynomial.X - Polynomial.C h) ^
      Polynomial.rootMultiplicity ((h : ℚ) / q) (R.map (Int.castRingHom ℚ)) * Q := by
  set t := Polynomial.rootMultiplicity ((h : ℚ) / q) (R.map (Int.castRingHom ℚ))
  have hq' : (q : ℚ) ≠ 0 := by exact_mod_cast hq.ne'
  have hmap : (Polynomial.C (q : ℤ) * Polynomial.X - Polynomial.C h).map (Int.castRingHom ℚ) =
      Polynomial.C (q : ℚ) * (Polynomial.X - Polynomial.C ((h : ℚ) / q)) := by
    rw [Polynomial.map_sub, Polynomial.map_mul, Polynomial.map_C, Polynomial.map_X,
      Polynomial.map_C, mul_sub, ← Polynomial.C_mul, mul_div_cancel₀ _ hq']
    simp
  apply (Polynomial.IsPrimitive.Int.dvd_iff_map_cast_dvd_map_cast _ _
    (isPrimitive_pow (isPrimitive_linear hcop) t)).2
  rw [Polynomial.map_pow, hmap, mul_pow, ← Polynomial.C_pow,
    Polynomial.C_mul_dvd (pow_ne_zero _ hq')]
  exact Polynomial.pow_rootMultiplicity_dvd _ _

/-- If the coefficients of a nonzero `R ∈ ℤ[X]` are bounded by `B` and `h / q`, with `h` coprime
to `q > 0`, is a root of `R` of multiplicity `t`, then `q^t ≤ B`.
It suffices to bound the absolute value of the leading coefficient.
Source: §5, proof of Lemma 5 ("Comparing nonzero leading coefficients yields
`B ≥ |lc(R)| = q_1^t |lc(Q)| ≥ q_1^t`"). -/
theorem pow_rootMultiplicity_le {R : Polynomial ℤ} (hR : R ≠ 0) {h : ℤ} {q : ℕ} (hq : 0 < q)
    (hcop : IsCoprime h (q : ℤ)) {B : ℝ} (hB : |(R.leadingCoeff : ℝ)| ≤ B) :
    (q : ℝ) ^ Polynomial.rootMultiplicity ((h : ℚ) / q) (R.map (Int.castRingHom ℚ)) ≤ B := by
  obtain ⟨Q, hQ⟩ := exists_eq_pow_mul_of_rootMultiplicity hq hcop
  set t := Polynomial.rootMultiplicity ((h : ℚ) / q) (R.map (Int.castRingHom ℚ))
  have hlc : R.leadingCoeff = (q : ℤ) ^ t * Q.leadingCoeff := by
    rw [hQ, Polynomial.leadingCoeff_mul, Polynomial.leadingCoeff_pow, sub_eq_add_neg,
      ← Polynomial.C_neg, Polynomial.leadingCoeff_linear (by exact_mod_cast hq.ne')]
  have h1 : (1 : ℝ) ≤ |(Q.leadingCoeff : ℝ)| := by
    exact_mod_cast Int.one_le_abs (Polynomial.leadingCoeff_ne_zero.2
      (right_ne_zero_of_mul (hQ ▸ hR)))
  rw [hlc] at hB
  push_cast at hB
  rw [abs_mul, abs_pow, Nat.abs_cast] at hB
  exact (le_mul_of_one_le_right (by positivity) h1).trans hB

private lemma index_unique_le_log_div [Unique σ] {B : ℝ} {q r : σ → ℕ} (hq : 1 < q default)
    {R : MvPolynomial σ ℤ} (hR : R ≠ 0) (hRB : |((uniqueAlgEquiv ℤ σ R).leadingCoeff : ℝ)| ≤ B)
    {h : σ → ℤ} (hh : IsCoprime (h default) (q default : ℤ)) :
    index (map (Int.castRingHom ℝ) R) (fun j ↦ (h j : ℝ) / q j) r ≤
      Real.log B / (r default * Real.log (q default)) := by
  classical
  let R1 : Polynomial ℤ := uniqueAlgEquiv ℤ σ R
  have hmapeq : uniqueAlgEquiv ℝ σ (map (Int.castRingHom ℝ) R) =
      R1.map (Int.castRingHom ℝ) := by
    dsimp only [R1]
    ext n
    rw [coeff_uniqueAlgEquiv, Polynomial.coeff_map, coeff_uniqueAlgEquiv, coeff_map]
  have hmult : Polynomial.rootMultiplicity ((h default : ℝ) / q default)
      (R1.map (Int.castRingHom ℝ)) =
      Polynomial.rootMultiplicity ((h default : ℚ) / q default) (R1.map (Int.castRingHom ℚ)) := by
    rw [Polynomial.eq_rootMultiplicity_map (f := Rat.castHom ℝ) Rat.cast_injective,
      Polynomial.map_map]
    congr 1
    simp
  rw [index_unique ((map_ne_zero_iff _ (map_injective _ Int.cast_injective)).2 hR), hmapeq, hmult]
  have hlog := Real.log_le_log (by positivity : (0 : ℝ) < (q default : ℝ) ^
    Polynomial.rootMultiplicity ((h default : ℚ) / q default) (R1.map (Int.castRingHom ℚ)))
    (pow_rootMultiplicity_le ((map_ne_zero_iff _ (uniqueAlgEquiv ℤ σ).injective).2 hR)
      (Nat.zero_lt_of_lt hq) hh hRB)
  rw [Real.log_pow] at hlog
  simpa [div_div, mul_comm] using
    div_le_div_of_nonneg_right
      ((le_div_iff₀ (Real.log_pos (by exact_mod_cast hq))).2 hlog) (Nat.cast_nonneg _)

omit [DecidableEq σ] in
/-- **Lemma 5**: in one variable, if `q > 1` then `Θ_1(B; q; r) ≤ log B / (r log q)`.
Source: §5, Lemma `roth:lem5` ("Suppose additionally that `q_1 > 1`. Then
`Θ_1(B; q_1; r_1) ≤ log B / (r_1 log q_1)`"), for a single variable `σ`. -/

section Lemma6

variable {τ : Type*} [Fintype τ] [DecidableEq τ]

/-- The constant `M = (r_1 + 1)^{p l} l! B^l 2^{p l r_1}` of Lemma 6. Source: §5, proof of
Lemma 6, eq. (16). -/
noncomputable def lem6M (p r₁ : ℕ) (B : ℝ) (l : ℕ) : ℝ :=
  ((r₁ : ℝ) + 1) ^ (p * l) * l.factorial * B ^ l * 2 ^ (p * l * r₁)

/-- The quantity `Φ = Θ_1(M; q_p; l r_p) + Θ_{p-1}(M; q_1, …, q_{p-1}; l r_1, …, l r_{p-1})` of
Lemma 6. Source: §5, proof of Lemma 6, eq. (15). -/
noncomputable def lem6Φ (q r : Option τ → ℕ) (M : ℝ) (l : ℕ) : ℝ :=
  Θ M (fun _ : Unit ↦ q none) (fun _ ↦ l * r none) + Θ M (q ∘ some) fun j ↦ l * r (some j)

/-- The constant `M` of Lemma 6 is at least `1` when `B` is. Source: §5, proof of Lemma 6. -/
theorem one_le_lem6M (p r₁ : ℕ) {B : ℝ} (hB : 1 ≤ B) (l : ℕ) : 1 ≤ lem6M p r₁ B l := by
  rw [lem6M]
  refine one_le_mul_of_one_le_of_one_le
    (one_le_mul_of_one_le_of_one_le (one_le_mul_of_one_le_of_one_le ?_ ?_)
      (one_le_pow₀ hB)) (one_le_pow₀ (by norm_num))
  · exact one_le_pow₀ (le_add_of_nonneg_left (Nat.cast_nonneg r₁))
  · exact_mod_cast Nat.factorial_pos l

/-- For `a ≥ 0`, `⌊a⌋² / 2 ≤ ∑_{0 ≤ ν ≤ ⌊a⌋} (a - ν)`. Source: §5, proof of Lemma 6, first

private lemma mul_div_two_le_sum_max_aux {θ r : ℝ} {l : ℕ} (hl : (l : ℝ) ≤ θ * r) :
    l * θ / 2 ≤ ∑ ν ∈ Finset.range l, max 0 (θ - ν / r) := by
  rcases Nat.eq_zero_or_pos l with rfl | hl0
  · simp
  by_cases hθ : θ ≤ 0
  · exact (div_nonpos_of_nonpos_of_nonneg
      (mul_nonpos_of_nonneg_of_nonpos (Nat.cast_nonneg l) hθ) (by norm_num)).trans
        (Finset.sum_nonneg fun _ _ ↦ le_max_left _ _)
  have hr := pos_of_mul_pos_right ((Nat.cast_pos.mpr hl0).trans_le hl) (le_of_not_ge hθ)
  have hsum := congrArg (Nat.cast (R := ℝ)) (Finset.sum_range_id_mul_two l)
  push_cast [Nat.cast_sub hl0] at hsum
  refine le_trans ?_ (Finset.sum_le_sum fun ν _ ↦ le_max_right 0 (θ - ν / r))
  rw [Finset.sum_sub_distrib, Finset.sum_const, Finset.card_range, nsmul_eq_mul,
    ← Finset.sum_div, le_sub_iff_add_le, div_add_div _ _ (by norm_num) hr.ne',
    div_le_iff₀ (by positivity)]
  linarith only [hsum, (Nat.cast_nonneg l : (0 : ℝ) ≤ l),
    mul_nonneg (Nat.cast_nonneg l) (sub_nonneg.mpr hl)]

private lemma mul_sq_div_three_le_sum_max_of_pos_aux {θ r : ℝ} (hr : 0 < r) (hθ : 0 ≤ θ) {l : ℕ}
    (hl : θ * r < l) : r * θ ^ 2 / 3 ≤ ∑ ν ∈ Finset.range l, max 0 (θ - ν / r) := by
  set n := ⌊θ * r⌋₊
  have hnle : (n : ℝ) ≤ θ * r := Nat.floor_le (mul_nonneg hθ hr.le)
  have hnlt : θ * r < n + 1 := Nat.lt_floor_add_one _
  have hsum := congrArg (Nat.cast (R := ℝ)) (Finset.sum_range_id_mul_two (n + 1))
  push_cast [Nat.add_sub_cancel] at hsum
  have hbound : (θ * r) ^ 2 / 3 ≤ ∑ ν ∈ Finset.range (n + 1), (θ * r - ν) := by
    rw [Finset.sum_sub_distrib, Finset.sum_const, Finset.card_range, nsmul_eq_mul]
    push_cast
    generalize θ * r = a at hnle hnlt ⊢
    linear_combination hsum / 2 + sq_nonneg a / 6 + (le_trans (Nat.cast_nonneg n) hnle) / 2 +
      mul_nonneg (sub_nonneg.mpr hnle) (sub_nonneg.mpr hnlt.le) / 2
  calc
    r * θ ^ 2 / 3 = (θ * r) ^ 2 / 3 / r := (eq_div_iff hr.ne').2 (by ring)
    _ ≤ (∑ ν ∈ Finset.range (n + 1), (θ * r - ν)) / r :=
      (div_le_div_iff_of_pos_right hr).2 hbound
    _ ≤ ∑ ν ∈ Finset.range (n + 1), max 0 (θ - ν / r) := by
      simp only [Finset.sum_div, sub_div, mul_div_cancel_right₀ _ hr.ne']
      exact Finset.sum_le_sum fun ν _ ↦ le_max_right _ _
    _ ≤ ∑ ν ∈ Finset.range l, max 0 (θ - ν / r) :=
      Finset.sum_le_sum_of_subset_of_nonneg (Finset.range_subset_range.2
        ((Nat.floor_lt (mul_nonneg hθ hr.le)).2 hl))
        fun _ _ _ ↦ le_max_left _ _

private lemma mul_sq_div_three_le_sum_max_aux {θ r : ℝ} (hθ : 0 ≤ θ) {l : ℕ} (hl : θ * r ≤ l) :
    r * θ ^ 2 / 3 ≤ ∑ ν ∈ Finset.range l, max 0 (θ - ν / r) := by

  rcases lt_or_eq_of_le hl with hl | hl
  · exact mul_sq_div_three_le_sum_max_of_pos_aux hr hθ hl
  · linarith only [mul_div_two_le_sum_max_aux hl.ge, congrArg (· * θ) hl,
      mul_nonneg hr.le (sq_nonneg θ)]

private lemma factorial_coefficient_le_two_pow_aux (p r₁ l : ℕ) (hl : l ≤ 2 ^ r₁) :
    (r₁ + 1) ^ (p * l) * l.factorial * 2 ^ (p * l * r₁) ≤ 2 ^ ((2 * p + 1) * r₁ * l) := by
  calc
    (r₁ + 1) ^ (p * l) * l.factorial * 2 ^ (p * l * r₁)
        ≤ (2 ^ r₁) ^ (p * l) * (2 ^ r₁) ^ l * 2 ^ (p * l * r₁) := by
          gcongr
          · exact Nat.lt_two_pow_self
          · exact (Nat.factorial_le_pow l).trans (Nat.pow_le_pow_left hl _)
    _ = 2 ^ ((2 * p + 1) * r₁ * l) := by
      rw [← pow_mul, ← pow_mul, ← pow_add, ← pow_add]
      ring_nf

private lemma two_pow_lt_rpow_aux {p r₁ l : ℕ} (hp : 0 < p) {δ q : ℝ} (hδ : 0 < δ) (hq : 0 ≤ q)
    (hlog : (p : ℝ) * (2 * p + 1) / δ ≤ Real.log q) (hlr : 0 < l * r₁) :
    (2 : ℝ) ^ ((2 * p + 1) * r₁ * l) < q ^ (δ / p * (l * r₁)) := by
  have hp0 : (0 : ℝ) < p := by exact_mod_cast hp
  replace hq : 0 < q := zero_lt_one.trans <|
    (Real.log_pos_iff hq).1 <| lt_of_lt_of_le (by positivity) hlog
  have hlog2 : Real.log 2 < 1 := by
    exact (Real.log_lt_sub_one_of_pos (by norm_num : (0 : ℝ) < 2) (by norm_num)).trans_eq
      (by norm_num)
  have h3 : (2 * p + 1 : ℝ) ≤ Real.log q * (δ / p) := by
    rw [mul_div_assoc', le_div_iff₀ hp0]
    rw [div_le_iff₀ hδ] at hlog
    linarith

  calc

/-- For `log q_1 ≥ p (2p + 1) / δ` and `p > 0`, the constant `M` of Lemma 6 with

Source: §5, proof of Lemma 7 ("`M < q_1^{δ_1 l r_1}` with `δ_1 = δ(1 + p⁻¹)`"). -/
theorem lem6M_lt {p : ℕ} (hp : 0 < p) {δ : ℝ} (hδ0 : 0 < δ) {r₁ q₁ : ℕ}
    (hq₁ : (p : ℝ) * (2 * p + 1) / δ ≤ Real.log q₁) {l : ℕ} (hl : 1 ≤ l) (hl' : l ≤ r₁) :
    lem6M p r₁ ((q₁ : ℝ) ^ (δ * r₁)) l < (q₁ : ℝ) ^ (δ * (1 + 1 / p) * (l * r₁)) := by
  have hp0 : (0 : ℝ) < p := by exact_mod_cast hp
  have hq1pos : (0 : ℝ) < q₁ := zero_lt_one.trans <|
    (Real.log_pos_iff (Nat.cast_nonneg q₁)).1 <| lt_of_lt_of_le (by positivity) hq₁
  -- the factors other than `B^l` are at most `2^{(2p+1) r₁ l}`
  have hnatR : ((r₁ : ℝ) + 1) ^ (p * l) * l.factorial * 2 ^ (p * l * r₁) ≤
      2 ^ ((2 * p + 1) * r₁ * l) := by

  calc
    lem6M p r₁ ((q₁ : ℝ) ^ (δ * r₁)) l = ((r₁ : ℝ) + 1) ^ (p * l) * l.factorial *
        2 ^ (p * l * r₁) * ((q₁ : ℝ) ^ (δ * r₁)) ^ l := by
          rw [lem6M]
          ring

      mul_le_mul_of_nonneg_right hnatR (by positivity)

      mul_lt_mul_of_pos_right
        (two_pow_lt_rpow_aux hp hδ0 hq1pos.le hq₁ (Nat.mul_pos (by lia) (by lia))) (by positivity)
    _ = (q₁ : ℝ) ^ (δ * (1 + 1 / p) * (l * r₁)) := by
      rw [← Real.rpow_mul_natCast hq1pos.le, ← Real.rpow_add hq1pos]

/-- The parameter `λ = 4(1 + 3δ) n m^{1/2}`. Source: §6, eq. (34). -/
noncomputable def lam (δ : ℝ) (n m : ℕ) : ℝ := 4 * (1 + 3 * δ) * n * √(m : ℝ)

/-- The parameter `γ = (m - λ)/2`. Source: §6, eq. (35). -/
noncomputable def gam (δ : ℝ) (n m : ℕ) : ℝ := ((m : ℝ) - lam δ n m) / 2

/-- The parameter `η = 10^m δ^{(1/2)^m}`. Source: §6, eq. (36). -/
noncomputable def eta (δ : ℝ) (m : ℕ) : ℝ := 10 ^ m * δ ^ ((1 / 2 : ℝ) ^ m)

/-- The coefficient bound `B_1 = ⌊q_1^{δ r_1}⌋`. Source: §6, eq. (37). -/
noncomputable def mainCoeffBound (δ : ℝ) (q₁ r₁ : ℕ) : ℕ := ⌊(q₁ : ℝ) ^ (δ * r₁)⌋₊

/-- For fixed `m > 16 n²` there is `δ₀ > 0` such that every `0 ≤ δ ≤ δ₀` satisfies
`δ < 1/m` (29) and `η + 2(1 + 3δ) n √m < m/2` (30).

Lemma `m-δ`, fixed-`m` form. Source: §6, Lemma `lem:m-delta` and its Lean note ("For each
fixed `m > 16n²`, prove the stronger statement that some `δ_0 > 0` makes (29) and (30) hold for
every `0 < δ < δ_0`"). -/
theorem exists_delta_bound {n m : ℕ} (hm : 16 * n ^ 2 < m) : ∃ δ₀ : ℝ, 0 < δ₀ ∧ ∀ δ : ℝ, 0 ≤ δ →
    δ ≤ δ₀ → δ < 1 / m ∧ eta δ m + 2 * (1 + 3 * δ) * n * √(m : ℝ) < m / 2 := by
  have hm' : (16 * n ^ 2 : ℝ) < m := by exact_mod_cast hm
  have hm0 : (0 : ℝ) < m := by exact_mod_cast (Nat.zero_le _).trans_lt hm
  have h4n : 4 * (n : ℝ) < √(m : ℝ) := Real.lt_sqrt_of_sq_lt (by nlinarith only [hm'])
  have hgap : 2 * (n : ℝ) * √(m : ℝ) < m / 2 := by
    nlinarith only [Real.sq_sqrt hm0.le,
      mul_lt_mul_of_pos_right h4n (Real.sqrt_pos.mpr hm0)]
  have hc : ContinuousAt (fun δ ↦ eta δ m + 2 * (1 + 3 * δ) * n * √(m : ℝ)) 0 :=
    (Real.continuousAt_rpow_const 0 _ (.inr (by positivity))).const_mul _ |>.add (by fun_prop)
  obtain ⟨ε, hε, hbound⟩ := Metric.eventually_nhds_iff.mp
    (hc.eventually_lt continuousAt_const (by
      simpa [eta, Real.zero_rpow (by positivity : (1 / 2 : ℝ) ^ m ≠ 0)] using hgap))
  refine ⟨min (1 / (m : ℝ)) ε / 2, by positivity, fun δ hδ hle ↦ ?_⟩
  have hlt := hle.trans_lt (half_lt_self (by positivity : 0 < min (1 / (m : ℝ)) ε))
  exact ⟨(lt_min_iff.mp hlt).1,
    hbound (by simpa [Real.dist_eq, abs_of_nonneg hδ] using (lt_min_iff.mp hlt).2)⟩

/-- Inequality (30) is equivalent to `η < γ`.

Apply `.mp` to (30), then `sub_pos.mpr` to obtain positivity of the gap `γ - η`.

theorem eta_lt_gam_iff (δ : ℝ) (n m : ℕ) : eta δ m + 2 * (1 + 3 * δ) * n * √(m : ℝ) < m / 2 ↔
    eta δ m < gam δ n m := by
  grind [gam, lam]

private lemma sqrt_lt_nat_floor_aux {X : ℝ} (hX : 2 ≤ X) : √X < ⌊X⌋₊ :=

/-- If `X ≥ 2` then `√X < ⌊X⌋ ≤ X`.

The natural floor gives an integer coefficient bound; use `.1` for its square-root lower
bound and `.2` for its non-strict upper bound.

Source: §6 ("writing `X = q_1^{δ r_1}`, we have `X > 4`, whence `B_1 = ⌊X⌋ > X - 1 > √X`"
and the Lean note "Retain the non-strict upper bound `B_1 ≤ X`"). -/
theorem sqrt_lt_floor {X : ℝ} (hX : 2 ≤ X) : √X < ⌊X⌋₊ ∧ (⌊X⌋₊ : ℝ) ≤ X :=
  ⟨sqrt_lt_nat_floor_aux hX, Nat.floor_le (by positivity)⟩

/-- The bound `3 ≤ δ² log q_1`, with `δ ≥ 0` and positive natural `r_1`, gives
`q_1^{δ r_1} > 4`. In the original setting, `1 ≤ m` and (32) imply this bound.

Source: §6 ("`X > 4`", from (32) and (29): `δ r_1 log q_1 > m(2m+1) r_1`). -/
theorem four_lt_rpow {δ : ℝ} (hδ0 : 0 ≤ δ) {q₁ r₁ : ℕ} (hr₁ : 0 < r₁)
    (hq₁ : 3 ≤ δ ^ 2 * Real.log q₁) : 4 < (q₁ : ℝ) ^ (δ * r₁) := by

    pos_of_mul_pos_right (lt_of_lt_of_le (by norm_num) hq₁) (sq_nonneg δ)
  have hq : 2 ≤ q₁ := by exact_mod_cast (Real.log_pos_iff (by positivity)).1 hL
  have hlog2 : Real.log 2 < 3 / 4 := by
    apply (Real.log_lt_iff_lt_exp (by norm_num)).2
    nlinarith only [Real.quadratic_le_exp_of_nonneg (by norm_num : (0 : ℝ) ≤ 3 / 4)]
  have hr : (1 : ℝ) ≤ r₁ := by exact_mod_cast hr₁
  apply Real.lt_rpow_of_log_lt <| lt_trans zero_lt_one ((Real.log_pos_iff (by positivity)).1 hL)
  have hkey : Real.log 4 < δ * Real.log q₁ := by
    rw [show (4 : ℝ) = 2 ^ 2 by norm_num, Real.log_pow]
    by_contra! h
    norm_num at h

/-- If `2 ≤ q_1^{δ r_1}`, then `½ δ r_1 log q_1 < log B_1`.

private lemma two_lt_rpow_aux {δ : ℝ} (hδ0 : 0 ≤ δ) {q₁ r₁ : ℕ} (hr₁ : 0 < r₁)

/-- For positive `r_1`, the non-strict form of (32), with `δ ≥ 0` and `A ≥ 0`, makes each of
`2^{m r_1}`, `(1 + A)^{m r_1}` and `(1 + |α|)^{m r_1}` less than `B_1^δ`.

Destructure the result to obtain all three bounds; the conclusions remain strict on the
hypothesis boundary and also hold with zero variables.

Source: §6, eq. `three-small-factors` ("`2^{m r_1} < B_1^δ`, `(1+A)^{m r_1} < B_1^δ`,
`(1+|α|)^{m r_1} < B_1^δ`"). -/
theorem three_small_factors {δ : ℝ} (hδ0 : 0 ≤ δ) {m q₁ r₁ : ℕ} (hr₁ : 0 < r₁) {A α : ℝ}
    (hA : 0 ≤ A) (h32 : 2 * m + 1 + 2 * m * Real.log (1 + A) + 2 * m * Real.log (1 + |α|) ≤
      δ ^ 2 * Real.log q₁) : (2 : ℝ) ^ (m * r₁) < (mainCoeffBound δ q₁ r₁ : ℝ) ^ δ ∧

    (1 + |α|) ^ (m * r₁) < (mainCoeffBound δ q₁ r₁ : ℝ) ^ δ := by
  have hm0 : (0 : ℝ) ≤ m := Nat.cast_nonneg m
  have hlogA : 0 ≤ Real.log (1 + A) := Real.log_nonneg (le_add_of_nonneg_right hA)
  have hlogα : 0 ≤ Real.log (1 + |α|) :=
    Real.log_nonneg (le_add_of_nonneg_right (abs_nonneg α))
  have h1 : 1 ≤ δ ^ 2 * Real.log q₁ := by
    nlinarith only [h32, hm0, mul_nonneg hm0 hlogA, mul_nonneg hm0 hlogα]
  have hlog2 : Real.log 2 < 1 := by
    linarith only [Real.log_lt_sub_one_of_pos (by norm_num : (0 : ℝ) < 2) (by norm_num)]

    rintro rfl
    norm_num at h1)

      t ^ (m * r₁) < (mainCoeffBound δ q₁ r₁ : ℝ) ^ δ := by

      (Real.sqrt_nonneg _).trans_lt (sqrt_lt_floor hX).1
    rw [Real.log_pow]
    push_cast
    nlinarith only [mul_le_mul_of_nonneg_left h (Nat.cast_nonneg r₁),
      mul_lt_mul_of_pos_left (mul_log_div_two_lt_log_mainCoeffBound hX) hδ]
  refine ⟨key 2 ?_, key _ ?_, key _ ?_⟩
  · nlinarith only [h32, hm0, mul_nonneg hm0 hlogA, mul_nonneg hm0 hlogα,
      mul_nonneg hm0 (sub_nonneg.mpr hlog2.le)]
  · nlinarith only [h32, hm0, mul_nonneg hm0 hlogα]
  · nlinarith only [h32, hm0, mul_nonneg hm0 hlogA]

private lemma exists_norm_coeff_reduction_aux {R : Type*} [SeminormedRing R] {f P : Polynomial R}
    {A C : ℝ} (hfA : ∀ i, ‖f.coeff i‖ ≤ A) (hP : ∀ i, ‖P.coeff i‖ ≤ C) {s : ℕ}
    (hs : P.natDegree ≤ s) (hf1 : f ≠ 1) (hself : P %ₘ f ≠ P) : 0 < f.natDegree ∧ f.natDegree ≤ s ∧
    ∃ Q : Polynomial R, Q.natDegree ≤ s - 1 ∧ (∀ i, ‖Q.coeff i‖ ≤ (1 + A) * C) ∧
    P %ₘ f = Q %ₘ f := by
  have : Nontrivial R := Polynomial.Nontrivial.of_polynomial_ne hself
  have hf : f.Monic := by
    by_contra hf
    exact hself (Polynomial.modByMonic_eq_of_not_monic P hf)
  have hP0 : P ≠ 0 := fun h ↦ hself (by simp [h])
  have hdeg : f.degree ≤ P.degree := le_of_not_gt fun h ↦
    hself ((Polynomial.modByMonic_eq_self_iff hf).2 h)
  let Q := P - f * (Polynomial.C P.leadingCoeff * Polynomial.X ^ (P.natDegree - f.natDegree))
  have hmod : P %ₘ f = Q %ₘ f := by
    conv_lhs =>
      rw [Polynomial.modByMonic, dite_eq_left hf, Polynomial.divModByMonicAux,
        dite_eq_left ⟨hdeg, hP0⟩]
    rw [Polynomial.modByMonic, dite_eq_left hf]
  refine ⟨Nat.pos_of_ne_zero (fun hn ↦ hf1 (Polynomial.eq_one_of_monic_natDegree_zero hf hn)),
    (Polynomial.natDegree_le_natDegree hdeg).trans hs, Q, ?_, ?_, hmod⟩
  · by_cases hQ0 : Q = 0
    · simp [hQ0]
    · exact Nat.le_sub_one_of_lt ((Polynomial.natDegree_lt_natDegree hQ0
        (Polynomial.div_wf_lemma ⟨hdeg, hP0⟩ hf)).trans_le hs)
  · intro i
    rw [Polynomial.coeff_sub, ← mul_assoc, Polynomial.coeff_mul_X_pow', add_mul, one_mul]
    refine (norm_sub_le _ _).trans (add_le_add (hP i) ?_)
    split_ifs
    · rw [Polynomial.coeff_mul_C]
      exact (norm_mul_le _ _).trans (mul_le_mul (hfA _) (hP _) (norm_nonneg _)
        ((norm_nonneg _).trans (hfA 0)))
    · simpa using mul_nonneg ((norm_nonneg _).trans (hfA 0)) ((norm_nonneg _).trans (hP 0))

/-- For polynomials over a seminormed ring with coefficient norm bounds `A` and `C` and
`P.natDegree ≤ s`, the coefficients of `P %ₘ f` are bounded by `(1 + A)^{s + 1 - deg f} C`.

No sign or monicity premises are needed; for a nonmonic divisor, `modByMonic` returns `P`.

Source: §6, proof of Lemma 9 ("each leading-term elimination multiplies the bound by at most
`1 + A` and lowers the degree ... `abc(T) ≤ (1+A)^{max(0, s-n+1)} B_1^{1+2δ}`"). -/
theorem norm_coeff_modByMonic_le {R : Type*} [SeminormedRing R] {f : Polynomial R} {A : ℝ}
    (hfA : ∀ i, ‖f.coeff i‖ ≤ A) (P : Polynomial R) {C : ℝ} (hP : ∀ i, ‖P.coeff i‖ ≤ C) {s : ℕ}
    (hs : P.natDegree ≤ s) (i : ℕ) : ‖(P %ₘ f).coeff i‖ ≤ (1 + A) ^ (s + 1 - f.natDegree) * C := by
  have hA : 1 ≤ 1 + A := le_add_of_nonneg_right ((norm_nonneg _).trans (hfA 0))
  by_cases hf1 : f = 1
  · simpa [hf1] using mul_nonneg (pow_nonneg (zero_le_one.trans hA) (s + 1))
      ((norm_nonneg _).trans (hP 0))
  induction s using Nat.strong_induction_on generalizing P C with
  | _ s ih =>
  by_cases hself : P %ₘ f = P
  · simpa [hself] using (hP i).trans
      (le_mul_of_one_le_left ((norm_nonneg _).trans (hP 0)) (one_le_pow₀ hA))
  obtain ⟨hn, hfs, Q, hQdeg, hQc, hmod⟩ := exists_norm_coeff_reduction_aux hfA hP hs hf1 hself
  rw [hmod]
  refine (ih (s - 1) (by lia) Q hQc hQdeg).trans_eq ?_
  rw [← mul_assoc, ← pow_succ]
  congr 2
  lia

/-- Integer polynomials with coordinate degrees bounded by `r` and coefficients in `[0, B]`.
Finiteness and counting below need only finitely supported bounds.
Source: §6, proof of Lemma 9 ("Consider the set `𝒲` of all polynomials
`W = ∑ c(s_1, …, s_m) x^s` where ... `0 ≤ c(s_1, …, s_m) ≤ B_1`"). -/
def boxFamily {σ : Type*} (r : σ → ℕ) (B : ℕ) : Set (MvPolynomial σ ℤ) :=
  {W | (∀ j, degreeOf j W ≤ r j) ∧ ∀ d, 0 ≤ W.coeff d ∧ W.coeff d ≤ B}

private lemma floor_rpow_count_lt_aux {x c : ℝ} (hx : 0 < x) (hc : 1 ≤ c) :
    ((2 * ⌊x ^ c⌋₊ + 1 : ℕ) : ℝ) < (x + 1) ^ (2 * c) := by
  have hu : 0 < x ^ c := Real.rpow_pos_of_pos hx c
  have hsum : x ^ c + 1 ≤ (x + 1) ^ c := by
    simpa using Real.add_rpow_le_rpow_add hx.le zero_le_one hc
  rw [mul_comm 2 c, Real.rpow_mul (by positivity), Real.rpow_two]
  push_cast
  nlinarith only [pow_le_pow_left₀ (by positivity) hsum 2, sq_pos_of_pos hu, Nat.floor_le hu.le]

/-- For positive real `B`, positive `N`, `1 ≤ c` and `c n D ≤ N / 2`, the remainder count is
below `(B + 1)^N`. Natural coefficient bounds specialize by coercion.

Source: §6, proof of Lemma 9 ("`1 + 2B_1^c < [2(1+B_1)]^c` ... `(1 + 2B_1^c)^{nD} <
(1+B_1)^r`"). -/
theorem pigeonhole_count {B : ℝ} (hB : 0 < B) {c : ℝ} (hc : 1 ≤ c) {n D N : ℕ} (hN : 0 < N)
    (hD : c * n * D ≤ N / 2) : ((2 * ⌊B ^ c⌋₊ + 1 : ℕ) : ℝ) ^ (n * D) < (B + 1) ^ N := by
  -- The source sets Y = 2(1+B) and uses the intermediate estimate
  -- `2⌊B^c⌋ + 1 < Y^c`
  -- The helper instead bounds the count directly by `(B + 1)^(2c)`.
  by_cases hk : n * D = 0
  · simpa [hk] using one_lt_pow₀ (by linarith) hN.ne'
  calc
    _ < ((B + 1) ^ (2 * c)) ^ (n * D) :=
      pow_lt_pow_left₀ (floor_rpow_count_lt_aux hB hc) (Nat.cast_nonneg _) hk
    _ = (B + 1) ^ (2 * c * (n * D : ℕ)) := (Real.rpow_mul_natCast (by positivity) _ _).symm
    _ ≤ (B + 1) ^ N := by
      rw [← Real.rpow_natCast]
      apply Real.rpow_le_rpow_of_exponent_le (by linarith)
      push_cast
      linarith only [hD]

/-- Finitely supported degree bounds and a strict remainder count give one common pair of
distinct box polynomials for all selected diagonal Hasse remainders.
Only coefficients below `f.natDegree` need bounding; finite index types supply the support
instance, and `finprod_eq_prod_of_fintype` recovers the usual product count.

Source: §6, proof of Lemma 9 ("By the pigeonhole principle, there are two distinct
polynomials `W', W'' ∈ 𝒲` such that `Δ_j W'(x, …, x) - Δ_j W''(x, …, x)` is divisible by `f(x)`
for all `j` satisfying (44)"). -/
theorem exists_ne_modByMonic_eq {σ : Type*} (r : σ → ℕ) [Finite (Function.support r)] (B : ℕ)
    {f : Polynomial ℤ} (hf : f.Monic) (J : Finset (σ →₀ ℕ)) (C : ℕ)
    (hC : ∀ W ∈ boxFamily r B, ∀ j ∈ J, ∀ i < f.natDegree,
      |(diag (hasseDeriv j W) %ₘ f).coeff i| ≤ C)
    (hcount : ((2 * C + 1 : ℕ) : ℝ) ^ (f.natDegree * J.card) < ((B : ℝ) + 1) ^ ∏ᶠ j, (r j + 1)) :
    ∃ W' ∈ boxFamily r B, ∃ W'' ∈ boxFamily r B, W' ≠ W'' ∧
      ∀ j ∈ J, diag (hasseDeriv j W') %ₘ f = diag (hasseDeriv j W'') %ₘ f := by
  classical
  let : Fintype (boxFamily r B) := (boxFamily_finite r B).fintype
  let F (W : boxFamily r B) (p : J × Fin f.natDegree) : Set.Icc (-(C : ℤ)) C :=
    ⟨(diag (hasseDeriv p.1.1 W.1) %ₘ f).coeff p.2, abs_le.1 (hC W.1 W.2 p.1.1 p.1.2 p.2.1 p.2.2)⟩
  obtain ⟨W', W'', hne, h⟩ := Fintype.exists_ne_map_eq_of_card_lt F <| by
    simpa only [Fintype.card_fun, Fintype.card_ofFinset, Int.card_Icc, sub_neg_eq_add,
      Fintype.card_prod, Fintype.card_coe, Fintype.card_fin, mul_comm, Set.fintypeCard_eq_ncard,
      ncard_boxFamily, show ((C : ℤ) + 1 + C).toNat = 2 * C + 1 by lia] using
      (show (2 * C + 1) ^ (f.natDegree * J.card) < (B + 1) ^ ∏ᶠ j, (r j + 1) by
        exact_mod_cast hcount)
  refine ⟨W', W'.2, W'', W''.2, Subtype.coe_ne_coe.2 hne, fun j hj ↦ ?_⟩
  simpa [F, Polynomial.sum_modByMonic_coeff hf Polynomial.degree_le_natDegree] using
    congrArg (fun g ↦ ∑ i : Fin f.natDegree, Polynomial.monomial (i : ℕ)
      (g (⟨j, hj⟩, i) : ℤ)) h

/-- The difference of two distinct members of `𝒲` is a nonzero box polynomial with

No finiteness assumption on the variables or the support of the degree bounds is needed.

  simpa only [Int.norm_eq_abs, Nat.cast_prod, Nat.cast_add, Nat.cast_one] using
    norm_coeff_modByMonic_le (by simpa [Int.norm_eq_abs] using hA) _
      (by simpa [Int.norm_eq_abs] using hcD) (natDegree_diag_le _ hdP) i

private lemma abs_coeff_diag_hasseDeriv_modByMonic_le_floor_aux {σ : Type*} [Fintype σ]
    {f : Polynomial ℤ} {P : MvPolynomial σ ℤ} {r : σ → ℕ} {A B β : ℝ} {s : ℕ}
    (hA : ∀ i, |(f.coeff i : ℝ)| ≤ A) (hd : ∀ k, degreeOf k P ≤ r k)
    (hc : ∀ d, |(P.coeff d : ℝ)| ≤ B) (hs : ∑ k, r k ≤ s) (h2 : 2 ^ s ≤ β) (hpow : (1 + A) ^ s ≤ β)
    (j : σ →₀ ℕ) (i : ℕ) : |(diag (hasseDeriv j P) %ₘ f).coeff i| ≤ (⌊B * β ^ 3⌋₊ : ℤ) := by
  have hB0 : 0 ≤ B := (abs_nonneg _).trans (hc 0)
  have hβ1 : 1 ≤ β := (one_le_pow₀ (by norm_num)).trans h2
  have hβ0 : 0 ≤ β := zero_le_one.trans hβ1
  have h2' : (2 : ℝ) ^ (∑ k, r k) ≤ β :=
    (pow_le_pow_right₀ (by norm_num) hs).trans h2
  have hNR : ((∏ k, (r k + 1) : ℕ) : ℝ) ≤ β := by
    refine le_trans ?_ h2'
    exact_mod_cast (show (∏ k, (r k + 1) : ℕ) ≤ 2 ^ (∑ k, r k) by
      rw [← Finset.prod_pow_eq_pow_sum]
      exact Finset.prod_le_prod fun k _ ↦ Nat.lt_two_pow_self)
  have bound {g : Polynomial ℤ} {C : ℝ} (hC : ∀ i, |(g.coeff i : ℝ)| ≤ C)
      (he : (1 + C) ^ ((∑ k, r k) + 1 - g.natDegree) ≤ β) :
      |((diag (hasseDeriv j P) %ₘ g).coeff i : ℝ)| ≤ B * β ^ 3 := by
    calc _ ≤ _ := abs_coeff_diag_hasseDeriv_modByMonic_le_aux hC hd hc j i
      _ ≤ β * (β * (β * B)) :=
        (mul_le_mul he (mul_le_mul hNR (mul_le_mul_of_nonneg_right h2' hB0)
          (by positivity) hβ0) (by positivity) hβ0)
      _ = B * β ^ 3 := by ring
  have hreal : |((diag (hasseDeriv j P) %ₘ f).coeff i : ℝ)| ≤ B * β ^ 3 := by
    by_cases hn : f.natDegree = 0
    · by_cases hf : f.Monic
      · simpa [Polynomial.eq_one_of_monic_natDegree_zero hf hn] using
          mul_nonneg hB0 (pow_nonneg hβ0 3)
      · simpa [Polynomial.modByMonic_eq_of_not_monic _ hf] using
          bound (g := 0) (C := 0) (by simp) (by simpa using hβ1)
    · exact bound hA ((pow_le_pow_right₀ (le_add_of_nonneg_right
        ((abs_nonneg _).trans (hA 0))) (by lia)).trans hpow)
  simpa [Int.natCast_floor_eq_floor ((abs_nonneg _).trans hreal), Int.le_floor,
    Int.cast_abs] using hreal

private lemma exists_mem_RSet_le_index_aux {σ : Type*} [Fintype σ] {f : Polynomial ℤ} {α : ℝ}
    (hf : f.Monic) (hfα : Polynomial.aeval α f = 0) (r : σ → ℕ) (B C : ℕ) (J : Finset (σ →₀ ℕ))
    {γ : ℝ} (hC : ∀ W ∈ boxFamily r B, ∀ j ∈ J, ∀ i < f.natDegree,
      |(diag (hasseDeriv j W) %ₘ f).coeff i| ≤ (C : ℤ))
    (hcount : ((2 * C + 1 : ℕ) : ℝ) ^ (f.natDegree * J.card) < ((B : ℝ) + 1) ^ (∏ k, (r k + 1) : ℕ))
    (hJ : ∀ d, (∀ k, d k ≤ r k) → weight r d < γ → d ∈ J) : ∃ P : MvPolynomial σ ℤ, P ∈ RSet B r ∧
      γ ≤ index (map (Int.castRingHom ℝ) P) (fun _ ↦ α) r := by
  obtain ⟨W', hW', W'', hW'', hne, hWeq⟩ := exists_ne_modByMonic_eq r B hf J C hC
    (by simpa [finprod_eq_prod_of_fintype] using hcount)
  -- `W* = W' - W''`
  have hP := sub_mem_RSet hW' hW'' hne
  refine ⟨W' - W'', hP, ?_⟩
  exact le_index_of_forall_eval_eq_zero
    ((map_ne_zero_iff _ (MvPolynomial.map_injective _ Int.cast_injective)).2 hP.1)
    (by simpa only [degreeOf,
          degrees_map_of_injective _ (f := Int.castRingHom ℝ) Int.cast_injective]
        using hP.2.1) (fun _ ↦ α)
    fun d hd hw ↦ eval_hasseDeriv_sub_eq_zero hfα (hWeq d (hJ d hd hw))

private lemma mul_card_filter_weight_le_aux {m : ℕ} (hm : 0 < m) (r : Fin m → ℕ) (hr : ∀ j, 0 < r j)
    (a : ℝ) : a * (((box r).filter fun d ↦ weight r d ≤ ((m : ℝ) - 4 * a * √(m : ℝ)) / 2).card : ℝ)
      ≤ ((∏ j, (r j + 1) : ℕ) : ℝ) / 2 := by
  by_cases ha : 0 < a
  · refine (mul_le_mul_of_nonneg_left
      (card_filter_weight_le m r hr (by positivity)) ha.le).trans_eq ?_
    push_cast
    field_simp [ha.ne', (Real.sqrt_pos.2 (Nat.cast_pos.2 hm)).ne']
    ring
  · exact (mul_nonpos_of_nonpos_of_nonneg (le_of_not_gt ha) (Nat.cast_nonneg _)).trans
      (by positivity)

private lemma exists_mem_RSet_le_index_of_rpow_bound_aux {f : Polynomial ℤ} {α A : ℝ} (hf : f.Monic)
    (hfα : Polynomial.aeval α f = 0) (hA : ∀ i, |(f.coeff i : ℝ)| ≤ A) {m : ℕ} (hm : 0 < m)
    (r : Fin m → ℕ) (hr : ∀ j, 0 < r j) {c β : ℝ} {B s : ℕ} (hB : 0 < B) (hs : ∑ j, r j ≤ s)
    (hAβ : (1 + A) ^ s ≤ β) (hpow : (B : ℝ) * β ^ 3 ≤ (B : ℝ) ^ c) : ∃ P : MvPolynomial (Fin m) ℤ,
      P ∈ RSet B r ∧ ((m : ℝ) - 4 * c * f.natDegree * √(m : ℝ)) / 2 ≤
        index (map (Int.castRingHom ℝ) P) (fun _ ↦ α) r := by
  have hA1 : 1 ≤ A := by simpa [hf.coeff_natDegree] using hA f.natDegree
  have h2 : (2 : ℝ) ^ s ≤ β :=
    (pow_le_pow_left₀ (by norm_num) (by linarith : (2 : ℝ) ≤ 1 + A) s).trans hAβ
  have hc : 1 ≤ c := by
    by_contra! hc
    have hBc : (B : ℝ) ^ c ≤ B := by
      simpa using Real.rpow_le_rpow_of_exponent_le (by exact_mod_cast hB) hc.le
    have hs0 : 0 < s := (hr ⟨0, hm⟩).trans_le
      ((Finset.single_le_sum (fun j _ ↦ Nat.zero_le (r j)) (Finset.mem_univ _)).trans hs)
    have hβ : 1 < β := (one_lt_pow₀ (by norm_num : (1 : ℝ) < 2) hs0.ne').trans_le h2
    exact (not_lt_of_ge hBc) ((lt_mul_of_one_lt_right (Nat.cast_pos.2 hB)
      (one_lt_pow₀ hβ (by decide : 3 ≠ 0))).trans_le hpow)
  -- the bound `C` for the remainders
  let Cn := ⌊(B : ℝ) ^ c⌋₊
  have hC : ∀ W ∈ boxFamily r B, ∀ j i,
      |(diag (hasseDeriv j W) %ₘ f).coeff i| ≤ (Cn : ℤ) := by
    intro W hW j i
    have hcW : ∀ e, |(W.coeff e : ℝ)| ≤ B := fun e ↦ by
      exact_mod_cast (abs_of_nonneg (hW.2 e).1).trans_le (hW.2 e).2
    exact (abs_coeff_diag_hasseDeriv_modByMonic_le_floor_aux
      hA hW.1 hcW hs h2 hAβ j i).trans (by exact_mod_cast Nat.floor_mono hpow)
  let J := (box r).filter fun d ↦
    weight r d ≤ ((m : ℝ) - 4 * c * f.natDegree * √(m : ℝ)) / 2
  -- the counting inequality
  have hD : c * f.natDegree * J.card ≤ ((∏ j, (r j + 1) : ℕ) : ℝ) / 2 := by
    simpa [J, mul_assoc] using mul_card_filter_weight_le_aux hm r hr (c * f.natDegree)
  exact exists_mem_RSet_le_index_aux hf hfα r B Cn J
    (fun W hW j _ i _ ↦ hC W hW j i)
    (pigeonhole_count (Nat.cast_pos.2 hB) hc (Finset.prod_pos fun j _ ↦ Nat.succ_pos _) hD)
    (fun d hd hw ↦ Finset.mem_filter.2 ⟨mem_box.2 hd, hw.le⟩)

private lemma exists_mem_RSet_le_index_of_log_bound_aux {f : Polynomial ℤ} {α A : ℝ} (hf : f.Monic)
    (hfα : Polynomial.aeval α f = 0) (hA : ∀ i, |(f.coeff i : ℝ)| ≤ A) {m q₁ r₁ : ℕ} (hm : 0 < m)
    (r : Fin m → ℕ) (hr : ∀ j, 0 < r j) (hs : ∑ j, r j ≤ m * r₁) {δ : ℝ} (hδ0 : 0 ≤ δ)
    (hL : 2 * m + 1 + 2 * m * Real.log (1 + A) ≤ δ ^ 2 * Real.log q₁) :
    ∃ P : MvPolynomial (Fin m) ℤ, P ∈ RSet (mainCoeffBound δ q₁ r₁) r ∧
      gam δ f.natDegree m ≤ index (map (Int.castRingHom ℝ) P) (fun _ ↦ α) r := by
  have hA1 : 1 ≤ A := by simpa [hf.coeff_natDegree] using hA f.natDegree
  have hr₁ : 0 < r₁ := Nat.pos_of_mul_pos_left <| (hr ⟨0, hm⟩).trans_le
    ((Finset.single_le_sum (fun j _ ↦ Nat.zero_le (r j)) (Finset.mem_univ _)).trans hs)
  have hq : 3 ≤ δ ^ 2 * Real.log q₁ := by
    have hmR : (1 : ℝ) ≤ m := by exact_mod_cast hm
    nlinarith only [hL, hmR, mul_nonneg (by positivity : (0 : ℝ) ≤ 2 * m)
      (Real.log_nonneg (by linarith : 1 ≤ 1 + A))]
  obtain ⟨_, hTA, _⟩ := three_small_factors (m := m) (q₁ := q₁) (α := 0)
    hδ0 hr₁ (zero_le_one.trans hA1) (by simpa using hL)
  have hpow : (mainCoeffBound δ q₁ r₁ : ℝ) ^ (1 + 3 * δ) =
      mainCoeffBound δ q₁ r₁ * ((mainCoeffBound δ q₁ r₁ : ℝ) ^ δ) ^ 3 := by
    rw [Real.rpow_add_of_nonneg (by positivity) zero_le_one (by positivity), Real.rpow_one,
      ← Real.rpow_mul_natCast (by positivity)]
    simp [mul_comm]
  simpa only [gam, lam] using
    exists_mem_RSet_le_index_of_rpow_bound_aux (B := mainCoeffBound δ q₁ r₁) hf hfα hA
    hm r hr (Nat.floor_pos.2 <| (by norm_num : (1 : ℝ) ≤ 4).trans
      (four_lt_rpow hδ0 hr₁ hq).le) hs hTA.le hpow.ge

private lemma exists_auxiliary_polynomial_of_log_bound_aux {f : Polynomial ℤ} {α A : ℝ}
    (hf : f.Monic) (hfα : Polynomial.aeval α f = 0) (hA : ∀ i, |(f.coeff i : ℝ)| ≤ A) {m q₁ r₁ : ℕ}
    (hm : 0 < m) (r : Fin m → ℕ) (hr : ∀ j, 0 < r j) (hs : ∑ j, r j ≤ m * r₁) {δ : ℝ} (hδ0 : 0 ≤ δ)
    (hL : 2 * m + 1 + 2 * m * Real.log (1 + A) + 2 * m * Real.log (1 + |α|) ≤ δ ^ 2 * Real.log q₁) :
    ∃ P : MvPolynomial (Fin m) ℤ, P ∈ RSet ((q₁ : ℝ) ^ (δ * r₁)) r ∧
      gam δ f.natDegree m ≤ index (map (Int.castRingHom ℝ) P) (fun _ ↦ α) r ∧ ∀ i k : Fin m →₀ ℕ,
      |eval (fun _ ↦ α) (map (Int.castRingHom ℝ) (hasseDeriv i (hasseDeriv k P)))| <
        (mainCoeffBound δ q₁ r₁ : ℝ) ^ (1 + 3 * δ) := by
  let B := mainCoeffBound δ q₁ r₁
  let β : ℝ := (B : ℝ) ^ δ
  obtain ⟨P, hPR, hγ⟩ := exists_mem_RSet_le_index_of_log_bound_aux hf hfα hA hm r hr hs hδ0
    ((le_add_of_nonneg_right (mul_nonneg (by positivity)
      (Real.log_nonneg (le_add_of_nonneg_right (abs_nonneg α))))).trans hL)
  obtain ⟨h2, _, hα⟩ := three_small_factors hδ0 (Nat.pos_of_mul_pos_left <|
    (hr ⟨0, hm⟩).trans_le ((Finset.single_le_sum (fun j _ ↦ Nat.zero_le (r j))
      (Finset.mem_univ _)).trans hs)) ((abs_nonneg _).trans (hA 0)) hL
  obtain ⟨d, hdne⟩ := MvPolynomial.exists_coeff_ne_zero hPR.1
  have hBpos : 0 < (B : ℝ) :=
    (abs_pos.mpr (Int.cast_ne_zero.2 hdne)).trans_le (hPR.2.2 d)
  have hpow : (B : ℝ) ^ (1 + 3 * δ) = B * β ^ 3 := by
    rw [Real.rpow_add_of_nonneg hBpos.le zero_le_one (by positivity), Real.rpow_one,
      ← Real.rpow_mul_natCast hBpos.le]
    simp [mul_comm]
  refine ⟨P, ⟨hPR.1, hPR.2.1, fun d ↦ (hPR.2.2 d).trans (Nat.floor_le (by positivity))⟩,
    hγ, fun i k ↦
      (abs_eval_iterated_hasseDeriv_lt_aux hPR.2.1 hBpos hs h2 hα.le hPR.2.2 i k).trans_eq ?_⟩
  linear_combination -hpow

private lemma theta_lt_eta_of_sq_log_bound_aux {m : ℕ} [NeZero m] {δ : ℝ} (hδ0 : 0 ≤ δ)
    (hδm : δ * (m : ℝ) ≤ 1) {r q : Fin m → ℕ} (hrlast : 3 ≤ r (Fin.rev 0))
    (hratio : ∀ i j : Fin m, i.val + 1 = j.val → (r j : ℝ) / δ ≤ r i)
    (hL : 2 * (m : ℝ) + 1 ≤ δ ^ 2 * Real.log (q 0))
    (hqr : ∀ j, (r 0 : ℝ) * Real.log (q 0) ≤ r j * Real.log (q j)) :
    Θ ((q 0 : ℝ) ^ (δ * r 0)) q r < eta δ m := by
  obtain ⟨n, rfl⟩ := Nat.exists_eq_succ_of_ne_zero (NeZero.ne m)
  let M : ℝ := (n + 1 : ℕ)
  have hδpos : 0 < δ := lt_of_le_of_ne hδ0 fun he ↦ by
    simpa [← he] using ((by positivity : 0 < 2 * M + 1).trans_le hL)
  have hq : M * (2 * M + 1) / δ ≤ Real.log (q 0) := by
    apply (mul_le_mul_iff_of_pos_right (sq_pos_of_pos hδpos)).1
    convert ((mul_le_mul_of_nonneg_right hδm (by positivity : 0 ≤ 2 * M + 1)).trans_eq
      (one_mul _)).trans hL using 1
    all_goals field_simp
    all_goals ring
  simpa [eta, Theta228Broad.bridge] using Theta228Broad.replay n hδpos
    (by have h : 1 ≤ r (Fin.rev 0) := (by decide : 1 ≤ 3).trans hrlast
        simpa only [Fin.rev_zero, Nat.cast_one] using (Nat.cast_le (α := ℝ)).2 h)
    (fun i ↦ by simpa [mul_comm] using ((div_le_iff₀ hδpos).1 (hratio _ _ (by simp))))
    (by simpa [M] using hq) hqr

private lemma exists_auxiliary_polynomial_with_index_gap_aux {α : ℝ} {f : Polynomial ℤ}
    (hf : f.Monic) (hfα : Polynomial.aeval α f = 0) {A : ℝ} (hA : ∀ i, |(f.coeff i : ℝ)| ≤ A)
    {m : ℕ} [NeZero m] {δ : ℝ} (hδ0 : 0 ≤ δ) (hδm : δ ≤ 1 / m) {r : Fin m → ℕ}
    (hrlast : 3 ≤ r (Fin.rev 0)) (h31b : ∀ i j : Fin m, i.val + 1 = j.val → (r j : ℝ) / δ ≤ r i)
    {q : Fin m → ℕ} (h32 : 2 * m + 1 + 2 * m * Real.log (1 + A) + 2 * m * Real.log (1 + |α|) ≤
      δ ^ 2 * Real.log (q 0)) (h33 : ∀ j, (r 0 : ℝ) * Real.log (q 0) ≤ r j * Real.log (q j)) :
    ∃ W : MvPolynomial (Fin m) ℤ, W ∈ RSet ((q 0 : ℝ) ^ (δ * r 0)) r ∧
      gam δ f.natDegree m - eta δ m + Θ ((q 0 : ℝ) ^ (δ * r 0)) q r <
        index (map (Int.castRingHom ℝ) W) (fun _ ↦ α) r ∧ ∀ i k : Fin m →₀ ℕ,
        |eval (fun _ ↦ α) (map (Int.castRingHom ℝ) (hasseDeriv i (hasseDeriv k W)))| <
          (mainCoeffBound δ (q 0) (r 0) : ℝ) ^ (1 + 3 * δ) := by
  obtain ⟨m', rfl⟩ := Nat.exists_eq_succ_of_ne_zero (NeZero.ne m)
  have h32' : 2 * ((m' + 1 : ℕ) : ℝ) + 1 ≤ δ ^ 2 * Real.log (q 0) :=
    (le_add_of_nonneg_right (mul_nonneg (by positivity)
      (Real.log_nonneg (le_add_of_nonneg_right ((abs_nonneg _).trans (hA 0)))))).trans
      ((le_add_of_nonneg_right (mul_nonneg (by positivity)
        (Real.log_nonneg (le_add_of_nonneg_right (abs_nonneg α))))).trans h32)
  have hδ0 : 0 < δ := lt_of_le_of_ne hδ0 fun he ↦ by
    simpa [← he] using ((by positivity : 0 < 2 * ((m' + 1 : ℕ) : ℝ) + 1).trans_le h32')
  have hmR : (1 : ℝ) ≤ ((m' + 1 : ℕ) : ℝ) := by
    exact_mod_cast (Nat.succ_le_succ (Nat.zero_le m'))
  -- `r` is decreasing, so `r j ≤ r 0`
  have hanti : Antitone r := Fin.antitone_iff_succ_le.2 fun i ↦ by
    exact_mod_cast (le_div_self (Nat.cast_nonneg _) hδ0
      (hδm.trans ((div_le_one (by positivity)).2 hmR))).trans (h31b i.castSucc i.succ rfl)
  have hsumr : ∑ j, r j ≤ (m' + 1) * r 0 := by
    simpa using Finset.sum_le_card_nsmul Finset.univ r (r 0) (fun j _ ↦ hanti (Fin.zero_le j))
  obtain ⟨W, hWR, hγ, hWbound⟩ := exists_auxiliary_polynomial_of_log_bound_aux hf hfα hA
    (Nat.succ_pos m') r (fun j ↦ (by decide : 0 < 3).trans_le
      (hrlast.trans (hanti (by simpa using Fin.le_last j)))) hsumr hδ0.le h32
  -- Lemma 7
  refine ⟨W, hWR, ?_, hWbound⟩
  linarith only [hγ, theta_lt_eta_of_sq_log_bound_aux hδ0.le
    ((le_div_iff₀ (by positivity)).1 hδm) hrlast h31b h32' h33]

/-- **Roth's main lemma** (Lemma 9), with weaker numerical hypotheses and a strict index bound.
For `m ≥ 1`, a monic integer equation for `α`, `0 ≤ δ ≤ 1/m`, last degree at least `3`, and the
displayed weak ratio and logarithmic bounds give a nonzero integer polynomial `Q` of degree
`≤ r_j` in `X j`, with index greater than `γ - η` at `(α, …, α)`, not vanishing at `(h_j/q_j)_j`.
Every Hasse derivative at `(α, …, α)` is bounded by `B_1^{1 + 3δ}`.

Source: §6, Lemma `roth:lem9` (for `m ≥ 1` variables indexed by `Fin m`; the chain conditions
(31) are stated for consecutive indices). -/
theorem main_lemma {α : ℝ} {f : Polynomial ℤ} (hf : f.Monic) (hfα : Polynomial.aeval α f = 0)
    {A : ℝ} (hA : ∀ i, |(f.coeff i : ℝ)| ≤ A) {m : ℕ} [NeZero m] {δ : ℝ} (hδ0 : 0 ≤ δ)
    (hδm : δ ≤ 1 / m) {r : Fin m → ℕ} (hrlast : 3 ≤ r (Fin.rev 0))
    (h31b : ∀ i j : Fin m, i.val + 1 = j.val → (r j : ℝ) / δ ≤ r i) {q : Fin m → ℕ}
    (h32 : 2 * m + 1 + 2 * m * Real.log (1 + A) + 2 * m * Real.log (1 + |α|) ≤
      δ ^ 2 * Real.log (q 0)) (h33 : ∀ j, (r 0 : ℝ) * Real.log (q 0) ≤ r j * Real.log (q j))
    {h : Fin m → ℤ} (hh : ∀ j, IsCoprime (h j) (q j : ℤ)) : ∃ Q : MvPolynomial (Fin m) ℤ, Q ≠ 0 ∧
      (∀ j, degreeOf j Q ≤ r j) ∧
      gam δ f.natDegree m - eta δ m < index (map (Int.castRingHom ℝ) Q) (fun _ ↦ α) r ∧
      eval (fun j ↦ (h j : ℝ) / q j) (map (Int.castRingHom ℝ) Q) ≠ 0 ∧
      ∀ i : Fin m →₀ ℕ, |eval (fun _ ↦ α) (hasseDeriv i (map (Int.castRingHom ℝ) Q))| <
        (mainCoeffBound δ (q 0) (r 0) : ℝ) ^ (1 + 3 * δ) := by
  obtain ⟨W, hW, hgap, hbound⟩ := exists_auxiliary_polynomial_with_index_gap_aux hf hfα hA hδ0 hδm
    hrlast h31b h32 h33
  let c := (index (map (Int.castRingHom ℝ) W) (fun _ ↦ α) r -
    (gam δ f.natDegree m - eta δ m) + Θ ((q 0 : ℝ) ^ (δ * r 0)) q r) / 2
  simpa only [add_sub_cancel_right] using
    exists_hasseDeriv_index_eval_bound_aux hW.1 hW.2.1 hbound
      (γ := gam δ f.natDegree m - eta δ m + c) (c := c)
      (z := fun j ↦ (h j : ℝ) / q j)
      (by
        dsimp only [c]
        linarith only [hgap])
      (by
        dsimp only [c]
        linarith only [hgap, index_le_Θ hW hh])
