theorem directReal (p r : ℕ) {B : ℝ} (hB : 1 ≤ B) (l : ℕ) :
    1 ≤ ((r : ℝ) + 1) ^ (p * l) * l.factorial * B ^ l * 2 ^ (p * l * r) := by
  refine one_le_mul_of_one_le_of_one_le
    (one_le_mul_of_one_le_of_one_le (one_le_mul_of_one_le_of_one_le ?_ ?_)
      (one_le_pow₀ hB)) (one_le_pow₀ (by norm_num))
  · exact one_le_pow₀ (le_add_of_nonneg_left (Nat.cast_nonneg r))
  · exact_mod_cast Nat.factorial_pos l
-- A true scalar generalization of the expanded expression; parent remains real-valued.

    1 ≤ ((r : K) + 1) ^ (p * l) * l.factorial * B ^ l * 2 ^ (p * l * r) := by
  refine one_le_mul_of_one_le_of_one_le

      (one_le_pow₀ hB)) (one_le_pow₀ (by norm_num))

-- Existing abstract list theorem with all actual four factor obligations discharged.
theorem listReal (p r : ℕ) {B : ℝ} (hB : 1 ≤ B) (l : ℕ) :
    1 ≤ ((r : ℝ) + 1) ^ (p * l) * l.factorial * B ^ l * 2 ^ (p * l * r) := by
  have hf : ∀ x ∈ [((r : ℝ) + 1) ^ (p * l), (l.factorial : ℝ),
      B ^ l, (2 : ℝ) ^ (p * l * r)], 1 ≤ x := by
    simp only [List.mem_cons, forall_eq_or_imp]
    exact ⟨one_le_pow₀ (le_add_of_nonneg_left (Nat.cast_nonneg r)),
      by exact_mod_cast Nat.factorial_pos l, one_le_pow₀ hB,
      one_le_pow₀ (by norm_num : (1 : ℝ) ≤ 2), fun _ h ↦ nomatch h⟩
  simpa only [List.prod_cons, List.prod_nil, mul_one, mul_assoc] using List.one_le_prod hf
-- Generic order form already in mathlib; this is a specialization, not new API.
theorem abstractProduct {K : Type*} [CommMonoidWithZero K] [PartialOrder K]
    [ZeroLEOneClass K] [PosMulMono K] {s : List K} (hs : ∀ x ∈ s, 1 ≤ x) :
    1 ≤ s.prod := List.one_le_prod hs
-- Hypothesis can be weakened for a particular exponent, but does not disappear uniformly.

  refine one_le_mul_of_one_le_of_one_le

-- Weak order, no strictness/linearity/characteristic-zero assumption required.
theorem genericWeak {K : Type*} [Semiring K] [PartialOrder K] [IsOrderedRing K]
    (p r : ℕ) {B : K} (hB : 1 ≤ B) (l : ℕ) :
    1 ≤ ((r : K) + 1) ^ (p * l) * l.factorial * B ^ l * 2 ^ (p * l * r) := by
  refine one_le_mul_of_one_le_of_one_le
    (one_le_mul_of_one_le_of_one_le (one_le_mul_of_one_le_of_one_le ?_ ?_)
      (one_le_pow₀ hB)) (one_le_pow₀ ?_)
  · exact one_le_pow₀ (le_add_of_nonneg_left (Nat.cast_nonneg r))
  · simpa only [Nat.cast_one] using (Nat.mono_cast (α := K) (Nat.factorial_pos l))
  · exact le_add_of_nonneg_right (zero_le_one : (0 : K) ≤ 1)
