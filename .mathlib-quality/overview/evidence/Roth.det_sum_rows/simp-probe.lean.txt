import Mathlib.LinearAlgebra.Matrix.Determinant.Basic
#check_failure Roth.det_sum_rows
namespace DetSumRowsSimp
variable {A J : Type*} [CommRing A] [Fintype J] [DecidableEq J] {I : J → Type*}
theorem weighted (s : ∀ μ, Finset (I μ)) (c : ∀ μ, I μ → A)
    (N : ∀ μ, I μ → J → A) :
    (Matrix.of fun μ ν ↦ ∑ i ∈ s μ, c μ i * N μ i ν).det =
      ∑ f ∈ Fintype.piFinset s, (∏ μ, c μ (f μ)) * (Matrix.of fun μ ν ↦ N μ (f μ) ν).det := by
  simpa only [MultilinearMap.map_smul_univ, smul_eq_mul, Matrix.det, Finset.sum_fn,
    Pi.smul_def] using
    (Matrix.detRowAlternating (R := A) (n := J)).toMultilinearMap.map_sum_finset
      (fun μ i ↦ c μ i • N μ i) s
#print axioms weighted
end DetSumRowsSimp
