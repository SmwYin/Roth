theorem explicitComposition {R ι : Type*} [CommRing R] [IsBezout R]
    [NormalizedGCDMonoid R] (s : Finset ι) (u : ι → R) :
    ∃ z : ι → R, ∑ i ∈ s, z i * u i = s.gcd u := by
  obtain ⟨z, hz⟩ := Finset.gcd_eq_sum_mul s u
  exact ⟨z, (Finset.sum_congr rfl (fun i _ ↦ mul_comm (z i) (u i))).trans hz.symm⟩
#print axioms explicitComposition
end Bezout170Direct
