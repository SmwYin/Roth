namespace WronskianIndependentGuards
noncomputable abbrev family : Fin 2 → (ZMod 2)[X] := ![1, X ^ 2]
theorem independent : LinearIndependent (ZMod 2) family := by
  apply linearIndependent_fin2.mpr
  constructor
  · simp [family]
  · intro a ha
    have h0 := congrArg (fun p : (ZMod 2)[X] ↦ p.coeff 0) ha
    simp [family, Polynomial.coeff_smul] at h0

theorem vanishes :
    (Matrix.of fun μ ν : Fin 2 ↦ hasseDeriv (μ : ℕ) (family ν)).det = 0 := by
  norm_num [family, Matrix.det_fin_two]
  exact (show (2 : ZMod 2) = 0 from by decide)
