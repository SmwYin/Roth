/-- Natural multiples of finitely many positive weights can be balanced with arbitrary error. -/
theorem balanced {ι : Type*} [Finite ι] (b : ι) {ℓ : ι → ℝ} (hℓ : ∀ j, 0 < ℓ j)
    {ε B : ℝ} (hε : 0 < ε) (hB : 0 < B) : ∃ r : ι → ℕ, (∀ j, 0 < r j) ∧
      (∀ j, B < r j) ∧ (∀ j, (r b : ℝ) * ℓ b ≤ r j * ℓ j) ∧
      ∀ j, (r j : ℝ) * ℓ j < (1 + ε) * (r b * ℓ b) := by
  classical
  let : Fintype ι := Fintype.ofFinite ι
  have : Nonempty ι := ⟨b⟩
  let K := max B (1 / ε)
  obtain ⟨s, hs⟩ := exists_nat_gt
    (Finset.univ.sup' Finset.univ_nonempty (fun j ↦ K * ℓ j / ℓ b))
  have hS (j) : K * ℓ j < (s : ℝ) * ℓ b :=
    (div_lt_iff₀ (hℓ b)).1 ((Finset.le_sup' _ (Finset.mem_univ j)).trans_lt hs)
  let r : ι → ℕ := fun j ↦ ⌈s * ℓ b / ℓ j⌉₊
  have hrb : r b = s := by
    simp only [r, mul_div_cancel_right₀ _ (hℓ b).ne', Nat.ceil_natCast]
  have hlow (j) := (div_le_iff₀ (hℓ j)).1 (Nat.le_ceil (s * ℓ b / ℓ j))
  have hlarge (j) : B < r j := by
    apply (mul_lt_mul_iff_left₀ (hℓ j)).1
    exact ((mul_le_mul_of_nonneg_right (le_max_left _ _) (hℓ j).le).trans_lt
      (hS j)).trans_le (hlow j)
  have hsmall (j) : ℓ j < ε * (s * ℓ b) := by
    have h := ((mul_le_mul_of_nonneg_right (le_max_right B (1 / ε)) (hℓ j).le).trans_lt
      (hS j))
    have h' := mul_lt_mul_of_pos_left h hε
    have he : ε * (1 / ε * ℓ j) = ℓ j := by field_simp
    rwa [he] at h'
  refine ⟨r, fun j ↦ ?_, hlarge, ?_, ?_⟩
  · exact_mod_cast hB.trans (hlarge j)
  · simpa only [hrb] using hlow
  · intro j
    rw [hrb]
    refine (mul_lt_mul_of_pos_right (Nat.ceil_lt_add_one
      (div_nonneg (mul_nonneg (Nat.cast_nonneg _) (hℓ b).le) (hℓ j).le)) (hℓ j)).trans ?_
    rw [add_mul, one_mul, div_mul_cancel₀ _ (hℓ j).ne', add_mul, one_mul]
    linarith only [hsmall j]

/-- A strict growth margin survives simultaneous natural rounding. -/
theorem separated {ι : Type*} [Finite ι] (b : ι) {ℓ : ι → ℝ} (hℓ : ∀ j, 0 < ℓ j)
    {ε B δ c : ℝ} (hε : 0 < ε) (hB : 0 < B) (hδ : 0 < δ) (hc : 1 < c)
    (R : ι → ι → Prop) (hg : ∀ i j, R i j → c / δ * ℓ i ≤ ℓ j) :
    ∃ r : ι → ℕ, (∀ j, 0 < r j) ∧ (∀ j, B < r j) ∧
      (∀ i j, R i j → (r j : ℝ) / δ < r i) ∧
      (∀ j, (r b : ℝ) * ℓ b ≤ r j * ℓ j) ∧
      ∀ j, (r j : ℝ) * ℓ j < (1 + ε) * (r b * ℓ b) := by
  let τ := min ε ((c - 1) / 2)
  have hτ : 0 < τ := lt_min hε (by linarith)
  obtain ⟨r, hr, hlarge, hlow, hup⟩ := balanced b hℓ hτ hB
  have hτc : 1 + τ ≤ c := by
    have := min_le_right ε ((c - 1) / 2)
    dsimp [τ]
    linarith
  refine ⟨r, hr, hlarge, ?_, hlow, fun j ↦ (hup j).trans_le ?_⟩
  · intro i j hij
    have h : (r j : ℝ) * (c / δ) < (1 + τ) * r i := by
      apply (mul_lt_mul_iff_left₀ (hℓ i)).1
      simpa only [mul_assoc] using
        (mul_le_mul_of_nonneg_left (hg i j hij) (Nat.cast_nonneg (r j))).trans_lt
          ((hup j).trans_le (mul_le_mul_of_nonneg_left (hlow i) (by positivity)))
    have h' := h.trans_le (mul_le_mul_of_nonneg_right hτc (Nat.cast_nonneg (r i)))
    have hm : (r j : ℝ) * (c / δ) = c * (r j / δ) := by ring
    rw [hm] at h'
    exact (mul_lt_mul_iff_right₀ (zero_lt_one.trans hc)).1 h'
  · exact mul_le_mul_of_nonneg_right (add_le_add le_rfl (min_le_left _ _))
      (mul_nonneg (Nat.cast_nonneg _) (hℓ b).le)

end Degrees278Generic

namespace Degrees278Recovery
theorem all_positive_delta {m : ℕ} [NeZero m] {δ : ℝ} (hδ0 : 0 < δ) {q : Fin m → ℕ}
    (hq : ∀ j, 1 < q j) (h50 : ∀ i j : Fin m, i.val + 1 = j.val →
      2 / δ * Real.log (q i) ≤ Real.log (q j)) : ∃ r : Fin m → ℕ, (∀ j, 0 < r j) ∧
      (∀ j, 10 / δ < r j) ∧ (∀ i j : Fin m, i.val + 1 = j.val → (r j : ℝ) / δ < r i) ∧
      (∀ j, (r 0 : ℝ) * Real.log (q 0) ≤ r j * Real.log (q j)) ∧ ∀ j,
      (r j : ℝ) * Real.log (q j) < (1 + δ / 10) * (r 0 * Real.log (q 0))  := by
  exact Degrees278Generic.separated (0 : Fin m)
    (fun j ↦ Real.log_pos (by exact_mod_cast hq j))
    (div_pos hδ0 (by norm_num)) (div_pos (by norm_num) hδ0) hδ0 (by norm_num)
    (fun i j ↦ i.val + 1 = j.val) h50

theorem recovered {m : ℕ} [NeZero m] {δ : ℝ} (hδ0 : 0 < δ) (_hδ1 : δ ≤ 1) {q : Fin m → ℕ}
    (hq : ∀ j, 1 < q j) (h50 : ∀ i j : Fin m, i.val + 1 = j.val →
      2 / δ * Real.log (q i) ≤ Real.log (q j)) : ∃ r : Fin m → ℕ, (∀ j, 0 < r j) ∧
      (∀ j, 10 / δ < r j) ∧ (∀ i j : Fin m, i.val + 1 = j.val → (r j : ℝ) / δ < r i) ∧
      (∀ j, (r 0 : ℝ) * Real.log (q 0) ≤ r j * Real.log (q j)) ∧ ∀ j,
      (r j : ℝ) * Real.log (q j) < (1 + δ / 10) * (r 0 * Real.log (q 0))  := by
  exact all_positive_delta hδ0 hq h50

end Degrees278Recovery

theorem inlined {α κ : ℝ} (hα : Irrational α) (hκ : 0 < κ)
    (hinf : (approxSet α κ).Infinite) (m : ℕ) [NeZero m] {δ : ℝ} (hδ0 : 0 < δ) (hδ1 : δ ≤ 1)
    (A : ℝ) : ∃ (h : Fin m → ℤ) (q r : Fin m → ℕ), (∀ j, (h j, q j) ∈ approxSet α κ) ∧
      (∀ j, IsCoprime (h j) (q j : ℤ)) ∧ (∀ j, 1 < q j) ∧ (∀ j, 0 < r j) ∧ 3 ≤ r (Fin.rev 0) ∧
      (∀ i j : Fin m, i.val + 1 = j.val → (r j : ℝ) / δ ≤ r i) ∧ (∀ j, r j ≤ r 0) ∧
      (2 * m + 1 + 2 * m * Real.log (1 + A) + 2 * m * Real.log (1 + |α|) < δ ^ 2 * Real.log (q 0)) ∧
      (∀ j, (r 0 : ℝ) * Real.log (q 0) ≤ r j * Real.log (q j)) ∧
      ∀ j, (r j : ℝ) * Real.log (q j) < (1 + δ / 10) * (r 0 * Real.log (q 0)) := by
  -- the approximants and the degrees
  obtain ⟨h, q, happ, hcop, hq1, hL, h50⟩ := exists_approximants hα hκ hinf m
    ((2 * m + 1 + 2 * m * Real.log (1 + A) + 2 * m * Real.log (1 + |α|)) / δ ^ 2)
  obtain ⟨r, hr0, h31a, h31b, h33, h53⟩ :=
    (show ∃ r: Fin m → ℕ, (∀ j, 0 < r j) ∧
      (∀ j, 10 / δ < r j) ∧ (∀ i j : Fin m, i.val + 1 = j.val → (r j : ℝ) / δ < r i) ∧
      (∀ j, (r 0 : ℝ) * Real.log (q 0) ≤ r j * Real.log (q j)) ∧ ∀ j,
      (r j : ℝ) * Real.log (q j) < (1 + δ / 10) * (r 0 * Real.log (q 0)) from by
    have hq := hq1
    have h50 := fun i j hij ↦ (h50 i j hij).le
    let ℓ (j) := Real.log (q j)
    have hℓ : ∀ j, 0 < ℓ j := fun j ↦ Real.log_pos (by exact_mod_cast hq j)
    obtain ⟨s, hs⟩ := exists_nat_gt (Finset.univ.sup' Finset.univ_nonempty (10 * ℓ · / (δ * ℓ 0)))
    have hS (j) : 10 * ℓ j < ((s : ℝ) * ℓ 0) * δ := by
      simpa only [mul_assoc, mul_comm δ] using (div_lt_iff₀ (mul_pos hδ0 (hℓ 0))).1
        ((Finset.le_sup' _ (Finset.mem_univ j)).trans_lt hs)
    let r : Fin m → ℕ := fun j ↦ ⌈s * ℓ 0 / ℓ j⌉₊
    have hr0 : r 0 = s := by simp only [r, mul_div_cancel_right₀ _ (hℓ 0).ne', Nat.ceil_natCast]
    have hlow (j) := (div_le_iff₀ (hℓ j)).1 (Nat.le_ceil (s * ℓ 0 / ℓ j))
    have hup (j) : (r j : ℝ) * ℓ j < (1 + δ / 10) * (s * ℓ 0) := by
      refine (mul_lt_mul_of_pos_right (Nat.ceil_lt_add_one (by positivity)) (hℓ j)).trans_le ?_
      rw [add_mul, one_mul, div_mul_cancel₀ _ (hℓ j).ne', add_mul, one_mul]
      refine add_le_add le_rfl ?_
      rw [div_mul_eq_mul_div, le_div_iff₀ (by norm_num : (0 : ℝ) < 10)]
      simpa only [mul_comm] using (hS j).le
    have hlarge (j) : 10 / δ < r j := by
      refine (div_lt_iff₀ hδ0).2 ((mul_lt_mul_iff_left₀ (hℓ j)).1 ?_)
      simpa only [mul_right_comm] using (hS j).trans_le (mul_le_mul_of_nonneg_right (hlow j) hδ0.le)
    refine ⟨r, fun j ↦ ?_, hlarge, fun i j hij ↦ ?_, ?_⟩
    · exact_mod_cast (div_pos (by norm_num : (0 : ℝ) < 10) hδ0).trans (hlarge j)
    · have h : (r j : ℝ) * (2 / δ) < (1 + δ / 10) * r i := by
        apply (mul_lt_mul_iff_left₀ (hℓ i)).1
        simpa only [mul_assoc] using
          (mul_le_mul_of_nonneg_left (h50 i j hij) (Nat.cast_nonneg (r j))).trans_lt
            ((hup j).trans_le (mul_le_mul_of_nonneg_left (hlow i) (by positivity)))
      rw [← mul_div_assoc, mul_comm (r j : ℝ) 2, mul_div_assoc] at h
      exact (mul_lt_mul_iff_right₀ (by norm_num : (0 : ℝ) < 2)).1 <| h.trans_le
        (mul_le_mul_of_nonneg_right ((add_le_add le_rfl
          (div_le_div_of_nonneg_right hδ1 (by norm_num))).trans (by norm_num)) (Nat.cast_nonneg _))
    · simpa only [hr0] using And.intro hlow hup)
  have hranti : Antitone r := by
    obtain ⟨m, rfl⟩ := Nat.exists_eq_succ_of_ne_zero (NeZero.ne m)
    refine Fin.antitone_iff_succ_le.2 fun i ↦ ?_
    exact_mod_cast (le_div_self (Nat.cast_nonneg _) hδ0 hδ1).trans
      (h31b i.castSucc i.succ rfl).le
  refine ⟨h, q, r, happ, hcop, hq1, hr0, ?_, fun i j hij ↦ (h31b i j hij).le,
    fun j ↦ hranti (Fin.zero_le j), (div_lt_iff₀' (by positivity)).1 hL, h33, h53⟩
  exact_mod_cast (by norm_num : (3 : ℝ) ≤ 10).trans
    ((le_div_self (by norm_num) hδ0 hδ1).trans (h31a (Fin.rev 0)).le)

theorem replaced {α κ : ℝ} (hα : Irrational α) (hκ : 0 < κ)
    (hinf : (approxSet α κ).Infinite) (m : ℕ) [NeZero m] {δ : ℝ} (hδ0 : 0 < δ) (hδ1 : δ ≤ 1)
    (A : ℝ) : ∃ (h : Fin m → ℤ) (q r : Fin m → ℕ), (∀ j, (h j, q j) ∈ approxSet α κ) ∧
      (∀ j, IsCoprime (h j) (q j : ℤ)) ∧ (∀ j, 1 < q j) ∧ (∀ j, 0 < r j) ∧ 3 ≤ r (Fin.rev 0) ∧
      (∀ i j : Fin m, i.val + 1 = j.val → (r j : ℝ) / δ ≤ r i) ∧ (∀ j, r j ≤ r 0) ∧

      (∀ j, (r 0 : ℝ) * Real.log (q 0) ≤ r j * Real.log (q j)) ∧
      ∀ j, (r j : ℝ) * Real.log (q j) < (1 + δ / 10) * (r 0 * Real.log (q 0)) := by
