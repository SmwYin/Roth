namespace IndexSet180Isolated
universe u v w
variable {A : Type u} {H : Type v} {V : Type w}
theorem exact_image2 (f : A → H → V) (s : Set A) (t : Set H) :
    {x | ∃ a ∈ s, ∃ h ∈ t, f a h = x} = Set.image2 f s t := rfl
theorem nonempty (f : A → H → V) {s : Set A} {t : Set H}
    (hs : s.Nonempty) (ht : t.Nonempty) : (Set.image2 f s t).Nonempty := hs.image2 ht
theorem mono (f : A → H → V) {s s' : Set A} {t : Set H} (hs : s ⊆ s') :
    Set.image2 f s t ⊆ Set.image2 f s' t := Set.image2_subset hs Set.Subset.rfl
theorem point_bound (f : A → H → ℝ) (s : Set A) (t : Set H) (c : ℝ) :
    (∀ z ∈ Set.image2 f s t, z ≤ c) ↔ ∀ a ∈ s, ∀ h ∈ t, f a h ≤ c :=
  Set.forall_mem_image2
