namespace MvPolynomial.TaylorMapRaw
variable {σ τ R S : Type*} [CommSemiring R] [CommSemiring S]
def shift (α : σ → R) : MvPolynomial σ R →ₐ[R] MvPolynomial σ R :=
  aeval fun j ↦ X j + C (α j)
theorem substitution (f : R →+* S) (g : σ → MvPolynomial τ R)
    (P : MvPolynomial σ R) :
    map f (aeval g P) = aeval (map f ∘ g) (map f P) :=
  map_eval₂ f g P
theorem oldForm (f : R →+* S) (α : σ → R) (P : MvPolynomial σ R) :
    shift (fun j ↦ f (α j)) (map f P) = map f (shift α P) := by
  change eval₂ C (fun j ↦ X j + C (f (α j))) (map f P) =
    map f (eval₂ C (fun j ↦ X j + C (α j)) P)
  rw [map_eval₂]
  congr 1
  funext j
  simp only [Function.comp_apply, map_add, map_X, map_C]

#print oldForm
end MvPolynomial.TaylorMapRaw
