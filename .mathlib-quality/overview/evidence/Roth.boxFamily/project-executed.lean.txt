def family {σ R : Type*} [CommSemiring R] (r : σ → ℕ) (t : (σ →₀ ℕ) → Set R) :
    Set (MvPolynomial σ R) := {P | (∀ j, degreeOf j P ≤ r j) ∧ ∀ d, P.coeff d ∈ t d}
open Classical in
theorem family_eq_image {σ R : Type*} [CommSemiring R] (r : σ →₀ ℕ)
    (t : (σ →₀ ℕ) → Finset R) (h0 : ∀ d, 0 ∈ t d) :
    family r (fun d ↦ (t d : Set R)) = AddMonoidAlgebra.ofCoeff ''
      (↑((Finset.Iic r).finsupp t) : Set ((σ →₀ ℕ) →₀ R)) := by
  classical
  fail_if_success rfl
  ext P
  constructor
  · intro hP
    refine ⟨P.coeff, ?_, rfl⟩
    rw [Finset.mem_coe, Finset.mem_finsupp_iff]
    exact ⟨fun d hd ↦ Finset.mem_Iic.2 fun j ↦
      (monomial_le_degreeOf j hd).trans (hP.1 j), fun d _ ↦ hP.2 d⟩
  · rintro ⟨c, hc, rfl⟩
    obtain ⟨hs, hc⟩ := Finset.mem_finsupp_iff.1 hc
    refine ⟨fun j ↦ degreeOf_le_iff.2 fun d hd ↦ (Finset.mem_Iic.1 (hs hd)) j, fun d ↦ ?_⟩
    by_cases hd : d ∈ Finset.Iic r
    · exact hc d hd
    · simpa [Finsupp.notMem_support_iff.1 (fun h ↦ hd (hs h))] using h0 d
open Classical in
theorem family_finite {σ R : Type*} [CommSemiring R] (r : σ →₀ ℕ)
    (t : (σ →₀ ℕ) → Finset R) (h0 : ∀ d, 0 ∈ t d) :
    (family r (fun d ↦ (t d : Set R))).Finite := by
  rw [family_eq_image r t h0]
  exact (Finset.finite_toSet _).image _
open Classical in
theorem family_ncard {σ R : Type*} [CommSemiring R] (r : σ →₀ ℕ)
    (t : (σ →₀ ℕ) → Finset R) (h0 : ∀ d, 0 ∈ t d) :
    (family r (fun d ↦ (t d : Set R))).ncard = ∏ d ∈ Finset.Iic r, (t d).card := by
  rw [family_eq_image r t h0, Set.ncard_image_of_injective _
    AddMonoidAlgebra.ofCoeff_injective, Set.ncard_coe_finset, Finset.card_finsupp]
theorem zero_mem {σ : Type*} (r : σ → ℕ) (B : ℕ) :
    (0 : MvPolynomial σ ℤ) ∈ family r (fun _ ↦ Set.Icc (0 : ℤ) B) := by
  simp [family]
theorem zero_bound {σ : Type*} (r : σ → ℕ) :
    family r (fun _ ↦ Set.Icc (0 : ℤ) 0) = {0} := by
  ext P
  constructor
  · intro h
    apply Set.mem_singleton_iff.2
    apply MvPolynomial.ext
    intro d
    simpa using le_antisymm (h.2 d).2 (h.2 d).1
  · rintro rfl
    exact zero_mem r 0
theorem degree_support {σ R : Type*} [CommSemiring R] (P : MvPolynomial σ R) (r : σ → ℕ) :
    (∀ j, degreeOf j P ≤ r j) ↔ ∀ d ∈ P.support, ∀ j, d j ≤ r j := by
  constructor
  · exact fun h d hd j ↦ (monomial_le_degreeOf j hd).trans (h j)
  · exact fun h j ↦ degreeOf_le_iff.2 fun d hd ↦ h d hd j
theorem member_X (i : ℕ) :
    X i ∈ family (fun _ : ℕ ↦ 1) (fun _ ↦ Set.Icc (0 : ℤ) 1) := by
  constructor
  · intro j
    by_cases h : i = j
    · subst j
      simp
    · simp [degreeOf_X_of_ne (Ne.symm h)]
  · intro d
    simp only [MvPolynomial.coeff_X]
    split_ifs <;> norm_num
theorem X_injective : Function.Injective (fun i : ℕ ↦ (X i : MvPolynomial ℕ ℤ)) := by
  intro i j h
  exact MvPolynomial.X_injective h
theorem infinite_family : (family (fun _ : ℕ ↦ 1) (fun _ ↦ Set.Icc (0 : ℤ) 1)).Infinite :=
  Set.infinite_range_of_injective X_injective |>.mono (Set.range_subset_iff.2 member_X)
theorem restrict_support {σ R : Type*} [CommSemiring R] (P : MvPolynomial σ R)
    (r : σ → ℕ) :
    (∀ j, degreeOf j P ≤ r j) ↔ P ∈ restrictSupport R {d | ∀ j, d j ≤ r j} := by
  rw [degree_support, mem_restrictSupport_iff]
  rfl
theorem family_specialization {σ : Type*} (r : σ → ℕ) (B : ℕ) :
    family r (fun _ ↦ Set.Icc (0 : ℤ) B) =
      {P | (∀ j, degreeOf j P ≤ r j) ∧ ∀ d, 0 ≤ P.coeff d ∧ P.coeff d ≤ (B : ℤ)} := rfl

theorem specialization {σ : Type*} (r : σ → ℕ) (B : ℕ) :
    Roth.boxFamily r B = Box233Generic.family r (fun _ ↦ Set.Icc (0 : ℤ) B) := rfl
theorem zero_bound {σ : Type*} (r : σ → ℕ) : Roth.boxFamily r 0 = {0} :=
  Box233Generic.zero_bound r
theorem infinite_family : (Roth.boxFamily (fun _ : ℕ ↦ 1) 1).Infinite :=
  Box233Generic.infinite_family
open Classical in
theorem generic_bridge {σ : Type*} (r : σ → ℕ) [Finite (Function.support r)] (B : ℕ) :
    Roth.boxFamily r B = AddMonoidAlgebra.ofCoeff ''
      (↑((Finset.Iic (Finsupp.ofSupportFinite r (Set.toFinite _))).finsupp
        fun _ ↦ Finset.Icc (0 : ℤ) B) : Set ((σ →₀ ℕ) →₀ ℤ)) := by
  rw [specialization]
  simpa only [Finset.coe_Icc, Finsupp.ofSupportFinite_coe] using
    Box233Generic.family_eq_image (Finsupp.ofSupportFinite r (Set.toFinite _))
    (fun _ ↦ Finset.Icc (0 : ℤ) B) (fun _ ↦ Finset.mem_Icc.2 ⟨le_rfl, by positivity⟩)
theorem interval_intersection {σ : Type*} (r : σ → ℕ) (B : ℕ) :
    Roth.boxFamily r B = {P | ∀ j, degreeOf j P ≤ r j} ∩
      ⋂ d, (fun P : MvPolynomial σ ℤ ↦ P.coeff d) ⁻¹' Set.Icc 0 (B : ℤ) := by
  ext P
  simp only [Roth.boxFamily, Set.mem_ofPred_eq, Set.mem_inter_iff, Set.mem_iInter,
    Set.mem_preimage, Set.mem_Icc]
theorem not_add_closed : ¬ ∀ P ∈ Roth.boxFamily (fun _ : Unit ↦ 0) 1,
    ∀ Q ∈ Roth.boxFamily (fun _ : Unit ↦ 0) 1, P + Q ∈ Roth.boxFamily (fun _ : Unit ↦ 0) 1 := by
  intro h
  have h1 : (1 : MvPolynomial Unit ℤ) ∈ Roth.boxFamily (fun _ ↦ 0) 1 := by
    constructor
    · simp
    · intro d
      simp only [MvPolynomial.coeff_one]
      split_ifs <;> norm_num
  have h2 := (h 1 h1 1 h1).2 0
  norm_num at h2
