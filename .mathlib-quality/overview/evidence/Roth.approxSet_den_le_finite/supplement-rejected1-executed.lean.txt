theorem finite_radius {K ι : Type*} [Ring K] [LinearOrder K]
    [IsStrictOrderedRing K] [FloorRing K] (s : Set ι) (hs : s.Finite) (c r : ι → K) :
    {p : ℤ × ι | p.2 ∈ s ∧ |(p.1 : K) - c p.2| ≤ r p.2}.Finite := by
  refine (hs.biUnion fun i _ ↦
    (Set.finite_Icc ⌈c i - r i⌉ ⌊c i + r i⌋).prod (Set.finite_singleton i)).subset ?_
  rintro ⟨h, i⟩ ⟨hi, hb⟩
  have hb' := abs_le.mp hb
  have hlo : c i - r i ≤ (h : K) := by linarith only [hb'.1]
  have hhi : (h : K) ≤ c i + r i := by linarith only [hb'.2]
  simp only [Set.mem_iUnion, Set.mem_prod, Set.mem_Icc, Set.mem_singleton_iff]
  exact ⟨i, hi, ⟨Int.ceil_le.mpr hlo, Int.le_floor.mpr hhi⟩, rfl⟩

end Bounded269Supplement
