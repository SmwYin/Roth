namespace Factor196RelPrime
open Polynomial
variable {A K : Type*} [CommRing A] [IsDomain A] [NormalizedGCDMonoid A]
  [Field K] [Algebra A K] [IsFractionRing A K]
omit [IsDomain A] [NormalizedGCDMonoid A] in
theorem primitive_linear {h q : A} (hcop : IsRelPrime h q) :
    (C q * X - C h).IsPrimitive := by
  intro c hc
  rw [C_dvd_iff_dvd_coeff] at hc
  exact hcop (by simpa using hc 0)
    (by simpa only [coeff_sub, coeff_C_mul, coeff_X_one,
      coeff_C, mul_one, one_ne_zero, ↓reduceIte, sub_zero] using hc 1)
omit [IsDomain A] in
theorem primitive_pow {p : A[X]} (hp : p.IsPrimitive) (t : ℕ) :
    (p ^ t).IsPrimitive :=
  Nat.rec isPrimitive_one (fun t ih ↦ by simpa [pow_succ] using ih.mul hp) t
theorem generic_nonzero {R : A[X]} {h q : A} (hq : q ≠ 0)
    (hcop : IsRelPrime h q) :
    (C q * X - C h) ^ rootMultiplicity (algebraMap A K h / algebraMap A K q)
      (R.map (algebraMap A K)) ∣ R := by
  set t := rootMultiplicity (algebraMap A K h / algebraMap A K q)
    (R.map (algebraMap A K))
  have hq' : algebraMap A K q ≠ 0 :=
    (map_ne_zero_iff _ (IsFractionRing.injective A K)).2 hq
  have hmap : (C q * X - C h).map (algebraMap A K) =
      C (algebraMap A K q) * (X - C (algebraMap A K h / algebraMap A K q)) := by
    rw [Polynomial.map_sub, Polynomial.map_mul, Polynomial.map_C, Polynomial.map_X,
      Polynomial.map_C, mul_sub, ← C_mul, mul_div_cancel₀ _ hq']
  apply (primitive_pow (primitive_linear hcop) t).dvd_of_fraction_map_dvd_fraction_map
    (K := K)
  rw [Polynomial.map_pow, hmap, mul_pow, ← C_pow, C_mul_dvd (pow_ne_zero _ hq')]
  exact pow_rootMultiplicity_dvd _ _
theorem generic {R : A[X]} {h q : A} (hcop : IsRelPrime h q) :
    (C q * X - C h) ^ rootMultiplicity (algebraMap A K h / algebraMap A K q)
      (R.map (algebraMap A K)) ∣ R := by
  by_cases hq : q = 0
  · subst q
    have hh : IsUnit h := isRelPrime_zero_right.mp hcop
    simpa only [C_0, zero_mul, zero_sub] using
      (((hh.map C).neg).pow
        (rootMultiplicity (algebraMap A K h / algebraMap A K 0)
          (R.map (algebraMap A K)))).dvd
  · exact generic_nonzero hq hcop
theorem original_without_hq {R : Polynomial ℤ} {h : ℤ} {q : ℕ}
    (hcop : IsCoprime h (q : ℤ)) : ∃ Q : Polynomial ℤ, R =
      (Polynomial.C (q : ℤ) * Polynomial.X - Polynomial.C h) ^
      Polynomial.rootMultiplicity ((h : ℚ) / q) (R.map (Int.castRingHom ℚ)) * Q := by
  exact generic (K := ℚ) hcop.isRelPrime

end Factor196RelPrime

namespace Bound197Generic
open Polynomial
theorem denominator_dvd {A K : Type*} [CommRing A] [IsDomain A] [IsGCDMonoid A]
    [Field K] [Algebra A K] [IsFractionRing A K] {R : A[X]} {h q : A}
    (hq : q ≠ 0) (hcop : IsRelPrime h q) :
    q ^ rootMultiplicity (algebraMap A K h / algebraMap A K q)
      (R.map (algebraMap A K)) ∣ R.leadingCoeff := by
  have hd := leadingCoeff_dvd_leadingCoeff (Factor196GCD.generic (R := R) (K := K) hcop)
  simpa only [leadingCoeff_pow, sub_eq_add_neg, ← C_neg, leadingCoeff_linear hq] using hd
theorem signed_bound {L : Type*} [Ring L] [LinearOrder L] [IsStrictOrderedRing L]
    {R : Polynomial ℤ} (hR : R ≠ 0) {h q : ℤ} (hcop : IsRelPrime h q) :
    |(q : L)| ^ rootMultiplicity ((h : ℚ) / q) (R.map (Int.castRingHom ℚ)) ≤
      |(R.leadingCoeff : L)| := by
  by_cases hq : q = 0
  · subst q
    have h1 : (1 : L) ≤ |(R.leadingCoeff : L)| := by
      exact_mod_cast Int.one_le_abs (leadingCoeff_ne_zero.2 hR)
    simpa only [Int.cast_zero, abs_zero] using
      (pow_le_one₀ (by norm_num) (by norm_num)).trans h1
  · have hd0 : q ^ rootMultiplicity ((h : ℚ) / q) (R.map (Int.castRingHom ℚ)) ∣
        R.leadingCoeff := denominator_dvd (R := R) (K := ℚ) hq hcop
    have hd := Int.natAbs_le_of_dvd_ne_zero hd0 (leadingCoeff_ne_zero.2 hR)
    rw [Int.natAbs_pow] at hd
    have hc := Nat.cast_le (α := L).2 hd
    simpa only [Nat.cast_pow, Nat.cast_natAbs, Int.cast_abs] using hc
theorem original_no_pos {R : Polynomial ℤ} (hR : R ≠ 0) {h : ℤ} {q : ℕ}
    (hcop : IsCoprime h (q : ℤ)) {B : ℝ} (hB : |(R.leadingCoeff : ℝ)| ≤ B) :
    (q : ℝ) ^ rootMultiplicity ((h : ℚ) / q) (R.map (Int.castRingHom ℚ)) ≤ B := by
  have hb := signed_bound (L := ℝ) hR hcop.isRelPrime
  simpa only [Int.cast_natCast, Nat.abs_cast] using hb.trans hB
#check @denominator_dvd
#check @signed_bound
#print axioms denominator_dvd
#print axioms signed_bound
#print axioms original_no_pos
example {R : Polynomial ℤ} (hR : R ≠ 0) {h : ℤ} {q : ℕ}
    (hcop : IsCoprime h (q : ℤ)) :
    (q : ℝ) ^ rootMultiplicity ((h : ℚ) / q) (R.map (Int.castRingHom ℚ)) ≤
      |(R.leadingCoeff : ℝ)| := by
  fail_if_success
    exact Polynomial.pow_rootMultiplicity_dvd _ _
  fail_if_success
    exact Polynomial.leadingCoeff_dvd_leadingCoeff
      (Polynomial.pow_rootMultiplicity_dvd _ _)
  exact original_no_pos hR hcop le_rfl
end Bound197Generic

namespace Bound197Guards
open Polynomial
theorem linear_multiplicity :
    rootMultiplicity (1 : ℚ) ((X - C 1 : Polynomial ℤ).map (Int.castRingHom ℚ)) = 1 := by
  simpa only [Polynomial.map_sub, Polynomial.map_X, Polynomial.map_C,
    Polynomial.map_one, map_one] using
    (rootMultiplicity_X_sub_C_self (x := (1 : ℚ)))
theorem coprime_necessary : ¬ ((2 : ℝ) ^
    rootMultiplicity ((2 : ℚ) / 2) ((X - C 1 : Polynomial ℤ).map (Int.castRingHom ℚ)) ≤
      |((X - C 1 : Polynomial ℤ).leadingCoeff : ℝ)|) := by
  rw [div_self (by norm_num : (2 : ℚ) ≠ 0), linear_multiplicity, leadingCoeff_X_sub_C]
  norm_num
theorem arbitrary_coefficients_fail : ¬ ((1 : ℝ) ^
    rootMultiplicity (1 : ℚ) (C (1 / 100 : ℚ) * (X - C 1)) ≤
      |((C (1 / 100 : ℚ) * (X - C 1)).leadingCoeff : ℝ)|) := by
  rw [one_pow, leadingCoeff_mul, leadingCoeff_C, leadingCoeff_X_sub_C]
  norm_num
theorem zero_denominator_divisibility_fails : ¬ ((0 : ℤ) ^
    rootMultiplicity ((1 : ℚ) / 0) ((X : Polynomial ℤ).map (Int.castRingHom ℚ)) ∣
      (X : Polynomial ℤ).leadingCoeff) := by
  have hm : rootMultiplicity (0 : ℚ) (X : Polynomial ℚ) = 1 := by
    simpa only [C_0, sub_zero] using (rootMultiplicity_X_sub_C_self (x := (0 : ℚ)))
  norm_num [hm]
#print axioms linear_multiplicity
#print axioms coprime_necessary
#print axioms arbitrary_coefficients_fail
#print axioms zero_denominator_divisibility_fails
end Bound197Guards
