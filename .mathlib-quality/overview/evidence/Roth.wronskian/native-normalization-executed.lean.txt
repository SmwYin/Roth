def W {R : Type*} [CommRing R] {l : ℕ} (φ : Fin l → R[X]) : R[X] :=
  (Matrix.of fun i j : Fin l ↦ Polynomial.hasseDeriv i (φ j)).det
theorem direct {R : Type*} [CommRing R] {l : ℕ} (φ : Fin l → R[X]) :
    W φ = (Matrix.of fun i j : Fin l ↦ Polynomial.hasseDeriv i (φ j)).det := rfl
theorem fin2 {R : Type*} [CommRing R] (a b : R[X]) :
    W ![a, b] = Polynomial.wronskian a b := by
  simp [W, Matrix.det_fin_two, Polynomial.hasseDeriv_zero,
    Polynomial.hasseDeriv_one, Polynomial.wronskian, mul_comm]
theorem fin0 {R : Type*} [CommRing R] (φ : Fin 0 → R[X]) : W φ = 1 := by
  simp [W]
theorem fin1 {R : Type*} [CommRing R] (φ : Fin 1 → R[X]) : W φ = φ 0 := by
  simp [W]

def ordinary {R : Type*} [CommRing R] {l : ℕ} (φ : Fin l → R[X]) : R[X] :=
  (Matrix.of fun i j : Fin l ↦ (Polynomial.derivative^[i.val]) (φ j)).det
theorem normalized_three :
    W (![1, X, X ^ 2] : Fin 3 → ℤ[X]) = 1 := by
  have ht : hasseDeriv 2 (X ^ 2 : ℤ[X]) = 1 := by
    rw [X_pow_eq_monomial, hasseDeriv_monomial]
    norm_num
  norm_num [W, Matrix.det_fin_three, ht, hasseDeriv_X, hasseDeriv_apply_one]
theorem ordinary_three :
    ordinary (![1, X, X ^ 2] : Fin 3 → ℤ[X]) = 2 := by
  norm_num [ordinary, Matrix.det_fin_three, Function.iterate_succ_apply']
