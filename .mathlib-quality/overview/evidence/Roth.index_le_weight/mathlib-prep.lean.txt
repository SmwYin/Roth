import Mathlib.Tactic
#check_failure Roth.index_le_weight
#check @Finset.inf_le
#check @WithTop.untopD_le
#check @Finsupp.mem_support_iff
namespace IndexLeWeight135
open MvPolynomial
universe u v w
 theorem abstractLaw {A : Type u} {R : Type v} [Semiring R]
    {T : Type w} [SemilatticeInf T] [OrderTop T] (D : A → T)
    (P : AddMonoidAlgebra R A) {d : A} (h : P.coeff d ≠ 0) :
    AddMonoidAlgebra.infDegree D P ≤ D d :=
  Finset.inf_le (Finsupp.mem_support_iff.mpr h)
theorem projectedLaw {A : Type u} {R : Type v} [Semiring R]
    (D : A → ℝ) (P : AddMonoidAlgebra R A) {d : A} (h : P.coeff d ≠ 0) :
    (AddMonoidAlgebra.infDegree (fun a ↦ (D a : WithTop ℝ)) P).untopD 0 ≤ D d :=
  WithTop.untopD_le (Finset.inf_le (Finsupp.mem_support_iff.mpr h))
theorem expandedLaw {σ : Type u} {R : Type v} [CommSemiring R]
    (P : MvPolynomial σ R) (α : σ → R) (D : (σ →₀ ℕ) → ℝ) {d : σ →₀ ℕ}
    (h : (aeval (fun j ↦ X j + C (α j)) P).coeff d ≠ 0) :
    (AddMonoidAlgebra.infDegree (fun a ↦ (D a : WithTop ℝ))
      (aeval (fun j ↦ X j + C (α j)) P)).untopD 0 ≤ D d :=
  WithTop.untopD_le (Finset.inf_le (Finsupp.mem_support_iff.mpr h))
#print axioms abstractLaw
#print axioms projectedLaw
#print axioms expandedLaw
end IndexLeWeight135
