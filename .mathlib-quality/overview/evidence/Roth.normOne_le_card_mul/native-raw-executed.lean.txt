namespace NormCardRaw
variable {σ : Type*}
theorem polynomialSupport (P : MvPolynomial σ ℤ) {B : ℝ}
    (hB : ∀ d ∈ P.support, |((P.coeff d : ℤ) : ℝ)| ≤ B) :
    (∑ d ∈ P.support, |((P.coeff d : ℤ) : ℝ)|) ≤ P.support.card * B := by
  simpa only [nsmul_eq_mul] using
    P.support.sum_le_card_nsmul (fun d ↦ |((P.coeff d : ℤ) : ℝ)|) B hB
theorem abstractBound {ι A : Type*} [AddCommMonoid A] [Preorder A] [AddLeftMono A]
    (s : Finset ι) (f : ι → A) (B : A) (hB : ∀ d ∈ s, f d ≤ B) :
    (∑ d ∈ s, f d) ≤ s.card • B :=
  Finset.sum_le_card_nsmul s f B hB
