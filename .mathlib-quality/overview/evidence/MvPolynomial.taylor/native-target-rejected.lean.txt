namespace PrivateScalarAssessment
open MvPolynomial
variable {σ : Type*}
theorem replay (f d e : σ →₀ ℕ) :
    hasseCoeff d (f + d) * hasseCoeff e (f + d + e) =
      hasseCoeff d (d + e) * hasseCoeff (d + e) (f + (d + e)) := by
  classical
  simp only [hasseCoeff]
  have hd : d.support ⊆ d.support ∪ e.support := Finset.subset_union_left
  rw [Finsupp.prod_of_support_subset _ hd _ fun _ _ ↦ Nat.choose_zero_right _,
    Finsupp.prod_of_support_subset _ Finset.subset_union_right _
      fun _ _ ↦ Nat.choose_zero_right _,
    Finsupp.prod_of_support_subset _ hd _ fun _ _ ↦ Nat.choose_zero_right _,
    Finsupp.prod_of_support_subset _ Finsupp.support_add _ fun _ _ ↦ Nat.choose_zero_right _,
    ← Finset.prod_mul_distrib, ← Finset.prod_mul_distrib]
  exact Finset.prod_congr rfl fun i _ ↦ by
    simpa [← add_assoc, mul_comm, ← Nat.choose_symm_add (a := d i) (b := e i)] using
      (Nat.choose_mul (n := f i + d i + e i) (k := d i + e i) (s := e i)
        (Nat.le_add_left (e i) (d i))).symm
#check @replay
#print axioms replay
end PrivateScalarAssessment

  induction hN : ∑ ρ, (φ ρ).natDegree using Nat.strong_induction_on generalizing φ with | _ N ih =>
    by_cases hinj : Function.Injective fun ν ↦ (φ ν).natDegree
    · exact wronskian_ne_zero_of_natDegree_injective h.ne_zero hinj
    obtain ⟨ν, ν', hdeg, hne⟩ := Function.not_injective_iff.1 hinj
    rw [← wronskian_update_sub_smul hne ((φ ν').leadingCoeff / (φ ν).leadingCoeff)]
    exact ih _ (hN ▸ sum_natDegree_update_lt h hne hdeg)
      (linearIndependent_update_sub_smul hne _ h) rfl


#print axioms replay
end WronskianIndependentAssessment

namespace WronskianDomainProbe
open Polynomial
theorem expanded {R : Type*} [CommRing R] [IsDomain R] [CharZero R]
    {l : ℕ} {φ : Fin l → R[X]} (hφ : LinearIndependent R φ) :
    (Matrix.of fun μ ν : Fin l ↦ hasseDeriv (μ : ℕ) (φ ν)).det ≠ 0 := by
  let K := FractionRing R
  have hf : Function.Injective (algebraMap R K) := IsFractionRing.injective R K
  have : CharZero K := (RingHom.charZero_iff hf).mp inferInstance
  let f := Polynomial.mapAlgHom (Algebra.ofId R K)
  have hfpoly : Function.Injective f := Polynomial.map_injective (algebraMap R K) hf
  have hiR : LinearIndependent R (f ∘ φ) :=
    hφ.map' f.toLinearMap (LinearMap.ker_eq_bot.mpr hfpoly)
  have hiK : LinearIndependent K (f ∘ φ) :=
    (LinearIndependent.iff_fractionRing R K).mp hiR
  have hw := WronskianIndependentAssessment.replay hiK
  have hm : Roth.wronskian (f ∘ φ) =
      (Polynomial.mapRingHom (algebraMap R K))
        (Matrix.of fun μ ν : Fin l ↦ hasseDeriv (μ : ℕ) (φ ν)).det := by
    rw [Roth.wronskian, Roth.wronskianMatrix, RingHom.map_det]
    congr 1
    apply Matrix.ext
    intro μ ν
    exact hasseDeriv_map (algebraMap R K) μ (φ ν)
  rw [hm] at hw
  intro hz
  exact hw (by rw [hz, map_zero])
#print axioms expanded
end WronskianDomainProbe
