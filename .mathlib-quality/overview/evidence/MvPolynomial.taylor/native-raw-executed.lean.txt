noncomputable def translate {σ : Type u} {R : Type v} [CommSemiring R] (α : σ → R) :
    MvPolynomial σ R →ₐ[R] MvPolynomial σ R :=
  MvPolynomial.aeval fun j ↦ MvPolynomial.X j + MvPolynomial.C (α j)
example {σ : Type u} {R : Type v} [CommSemiring R] (α : σ → R) :
    translate α = MvPolynomial.aeval (fun j ↦ MvPolynomial.X j + MvPolynomial.C (α j)) := rfl

example {σ R : Type*} [CommSemiring R] (α : σ → R) (P Q : MvPolynomial σ R) :
    translate α (P * Q) = translate α P * translate α Q := map_mul _ _ _
noncomputable def translateEquiv {σ R : Type*} [CommRing R] (α : σ → R) :
    MvPolynomial σ R ≃ₐ[R] MvPolynomial σ R where
  __ := translate α
  invFun := translate (fun j ↦ -α j)
  left_inv P := by
    simp [translate, MvPolynomial.comp_aeval_apply, MvPolynomial.aeval_X_left_apply]
  right_inv P := by
    simp [translate, MvPolynomial.comp_aeval_apply, MvPolynomial.aeval_X_left_apply]
example {σ R : Type*} [CommRing R] (α : σ → R) :
    Function.Injective (translate α : MvPolynomial σ R → _) := (translateEquiv α).injective
