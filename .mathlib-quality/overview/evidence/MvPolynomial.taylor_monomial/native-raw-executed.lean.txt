theorem universal {σ R A : Type*} [CommSemiring R] [CommSemiring A] [Algebra R A]
    (f : σ → A) (k : σ →₀ ℕ) (a : R) :
    aeval (R := R) f (monomial k a) = algebraMap R A a * k.prod (fun j n ↦ f j ^ n) :=
  aeval_monomial _ _ _
