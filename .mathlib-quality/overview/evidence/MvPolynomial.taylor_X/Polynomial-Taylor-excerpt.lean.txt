37: /-- The Taylor expansion of a polynomial `f` at `r`. -/
38: def taylor (r : R) : R[X] →ₗ[R] R[X] where
39:   toFun f := f.comp (X + C r)
40:   map_add' _ _ := add_comp
41:   map_smul' c f := by simp only [smul_eq_C_mul, C_mul_comp, RingHom.id_apply]
42: 
43: theorem taylor_apply : taylor r f = f.comp (X + C r) :=
44:   rfl
45:

47: theorem taylor_X : taylor r X = X + C r := X_comp
48: 
49: @[simp]
50: theorem taylor_X_pow (n : ℕ) : taylor r (X ^ n) = (X + C r) ^ n := X_pow_comp
51:

53: theorem taylor_C (x : R) : taylor r (C x) = C x := C_comp

55: theorem taylor_zero (f : R[X]) : taylor 0 f = f := by rw [taylor_apply, C_0, add_zero, comp_X]
