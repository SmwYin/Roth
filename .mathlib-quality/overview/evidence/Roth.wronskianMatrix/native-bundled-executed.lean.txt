noncomputable def matrix {R : Type*} [Semiring R] {l : ℕ}
    (φ : Fin l → R[X]) : Matrix (Fin l) (Fin l) R[X] :=
  Matrix.of fun μ ν ↦ hasseDeriv μ (φ ν)
theorem direct {R : Type*} [Semiring R] {l : ℕ} (φ : Fin l → R[X]) :
    matrix φ = Matrix.of (fun μ ν : Fin l ↦ hasseDeriv μ (φ ν)) := rfl
noncomputable def rectangular {R : Type*} [Semiring R] {ι κ : Type*}
    (order : ι → ℕ) (φ : κ → R[X]) : Matrix ι κ R[X] :=
  Matrix.of fun μ ν ↦ hasseDeriv (order μ) (φ ν)
theorem rectangular_bridge {R : Type*} [Semiring R] {l : ℕ} (φ : Fin l → R[X]) :
    matrix φ = rectangular (fun μ : Fin l ↦ (μ : ℕ)) φ := rfl

noncomputable def bundled {R : Type*} [Semiring R] {l : ℕ} :
    (Fin l → R[X]) →ₗ[R] Matrix (Fin l) (Fin l) R[X] :=
  LinearMap.pi fun i ↦ LinearMap.pi fun j ↦
    (hasseDeriv i).comp (LinearMap.proj j)
theorem bundled_bridge {R : Type*} [Semiring R] {l : ℕ} (φ : Fin l → R[X]) :
    bundled φ = matrix φ := rfl
theorem matrix_add {R : Type*} [Semiring R] {l : ℕ} (φ ψ : Fin l → R[X]) :
    matrix (φ + ψ) = matrix φ + matrix ψ :=
  (bundled : (Fin l → R[X]) →ₗ[R] Matrix (Fin l) (Fin l) R[X]).map_add φ ψ
theorem factorial_normalization_guard :
    hasseDeriv 2 (X ^ 2 : ℕ[X]) ≠ Polynomial.derivative (Polynomial.derivative (X ^ 2)) := by
  intro h
  have hc := congrArg (fun p : ℕ[X] ↦ p.coeff 0) h
  norm_num [hasseDeriv_coeff] at hc
