noncomputable def matrix {R : Type*} [Semiring R] {l : ℕ}
    (φ : Fin l → R[X]) : Matrix (Fin l) (Fin l) R[X] :=
  Matrix.of fun μ ν ↦ hasseDeriv μ (φ ν)
theorem direct {R : Type*} [Semiring R] {l : ℕ} (φ : Fin l → R[X]) :
    matrix φ = Matrix.of (fun μ ν : Fin l ↦ hasseDeriv μ (φ ν)) := rfl
noncomputable def rectangular {R : Type*} [Semiring R] {ι κ : Type*}
    (order : ι → ℕ) (φ : κ → R[X]) : Matrix ι κ R[X] :=
  Matrix.of fun μ ν ↦ hasseDeriv (order μ) (φ ν)
theorem rectangular_bridge {R : Type*} [Semiring R] {l : ℕ} (φ : Fin l → R[X]) :
    matrix φ = rectangular (fun μ : Fin l ↦ (μ : ℕ)) φ := rfl
example : Matrix (Fin 2) (Fin 2) ℕ[X] := matrix (fun _ ↦ X)
