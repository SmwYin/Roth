lemma inferred_irrational {α κ : ℝ} (hκ : 2 ≤ κ)
    (hinf : {x : ℚ | |α - x| < 1 / (x.den : ℝ) ^ κ}.Infinite) : Irrational α := by
  apply (Real.infinite_rat_abs_sub_lt_one_div_den_sq_iff_irrational α).mp
  refine hinf.mono fun x hx ↦ hx.trans_le ?_
  apply one_div_le_one_div_of_le (by positivity)
  rw [← Real.rpow_natCast]
  exact Real.rpow_le_rpow_of_exponent_le (by exact_mod_cast x.den_pos) hκ
lemma generalize
    (hR : ∀ {α : ℝ}, Irrational α → IsAlgebraic ℚ α → ∀ {κ : ℝ},
      {x : ℚ | |α - x| < 1 / (x.den : ℝ) ^ κ}.Infinite → κ ≤ 2)
    {α : ℝ} (halg : IsAlgebraic ℚ α) {κ : ℝ}
    (hinf : {x : ℚ | |α - x| < 1 / (x.den : ℝ) ^ κ}.Infinite) : κ ≤ 2 := by
  by_contra! hκ
  exact (hR (inferred_irrational hκ.le hinf) halg hinf).not_gt hκ
lemma recover
    (hR : ∀ {α : ℝ}, IsAlgebraic ℚ α → ∀ {κ : ℝ},
      {x : ℚ | |α - x| < 1 / (x.den : ℝ) ^ κ}.Infinite → κ ≤ 2)
    {α : ℝ} (_hα : Irrational α) (halg : IsAlgebraic ℚ α) {κ : ℝ}
    (hinf : {x : ℚ | |α - x| < 1 / (x.den : ℝ) ^ κ}.Infinite) : κ ≤ 2 :=
  hR halg hinf
lemma finite_iff_bound (α : ℝ) :
    (∀ {κ : ℝ}, {x : ℚ | |α - x| < 1 / (x.den : ℝ) ^ κ}.Infinite → κ ≤ 2) ↔
      ∀ κ : ℝ, 2 < κ → {x : ℚ | |α - x| < 1 / (x.den : ℝ) ^ κ}.Finite := by
  constructor
  · intro h κ hκ
    exact Set.not_infinite.mp fun hinf ↦ (h hinf).not_gt hκ
  · intro h κ hinf
    exact le_of_not_gt fun hκ ↦ hinf (h κ hκ)
example {α κ : ℝ} (_hα : Irrational α) (_halg : IsAlgebraic ℚ α)
    (_hinf : {x : ℚ | |α - x| < 1 / (x.den : ℝ) ^ κ}.Infinite) : True := by
  fail_if_success
    have : κ ≤ 2 := by exact?
  fail_if_success
    have : κ ≤ 2 := by aesop
  trivial
