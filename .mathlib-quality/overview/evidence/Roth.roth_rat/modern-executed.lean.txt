lemma cofinite_iff_finite (α κ : ℝ) :
    (∀ᶠ x : ℚ in cofinite, 1 / (x.den : ℝ) ^ κ ≤ |α - x|) ↔
      {x : ℚ | |α - x| < 1 / (x.den : ℝ) ^ κ}.Finite := by
  simp only [eventually_cofinite, not_le]
lemma exponent_iff_cofinite (α : ℝ) :
    (∀ {κ : ℝ}, {x : ℚ | |α - x| < 1 / (x.den : ℝ) ^ κ}.Infinite → κ ≤ 2) ↔
      ∀ κ : ℝ, 2 < κ → ∀ᶠ x : ℚ in cofinite, 1 / (x.den : ℝ) ^ κ ≤ |α - x| := by
  constructor
  · intro h κ hκ
    exact (cofinite_iff_finite α κ).mpr (Set.not_infinite.mp fun hinf ↦ (h hinf).not_gt hκ)
  · intro h κ hinf
    exact le_of_not_gt fun hκ ↦ hinf ((cofinite_iff_finite α κ).mp (h κ hκ))
lemma cofinite_consequence
    (hR : ∀ {α : ℝ}, IsAlgebraic ℚ α → ∀ {κ : ℝ},
      {x : ℚ | |α - x| < 1 / (x.den : ℝ) ^ κ}.Infinite → κ ≤ 2)
    {α κ₁ κ₂ : ℝ} (halg : IsAlgebraic ℚ α) (h₁ : 2 < κ₁) (h₂ : 2 < κ₂) :
    ∀ᶠ x : ℚ in cofinite,
      1 / (x.den : ℝ) ^ κ₁ ≤ |α - x| ∧ 1 / (x.den : ℝ) ^ κ₂ ≤ |α - x| :=
  ((exponent_iff_cofinite α).mp (hR halg) κ₁ h₁).and
    ((exponent_iff_cofinite α).mp (hR halg) κ₂ h₂)
lemma recover
    (hR : ∀ {α : ℝ}, IsAlgebraic ℚ α → ∀ κ : ℝ, 2 < κ →
      ∀ᶠ x : ℚ in cofinite, 1 / (x.den : ℝ) ^ κ ≤ |α - x|)
    {α κ : ℝ} (_hα : Irrational α) (halg : IsAlgebraic ℚ α)
    (hinf : {x : ℚ | |α - x| < 1 / (x.den : ℝ) ^ κ}.Infinite) : κ ≤ 2 :=
  (exponent_iff_cofinite α).mpr (hR halg) hinf
