#check @Real.sInf_nonneg

theorem abstractLaw {A : Type u} {R : Type v} {T : Type w} [Semiring R]
    [SemilatticeInf T] [OrderTop T] (D : A → T) (P : AddMonoidAlgebra R A) (c : T)
    (hD : ∀ d ∈ P.coeff.support, c ≤ D d) : c ≤ AddMonoidAlgebra.infDegree D P :=
  Finset.le_inf_iff.mpr hD

theorem projectedLaw {A : Type u} {R : Type v} [Semiring R]
    (D : A → ℝ) (P : AddMonoidAlgebra R A) (hD : ∀ d ∈ P.coeff.support, 0 ≤ D d) :
    0 ≤ (AddMonoidAlgebra.infDegree (fun d ↦ (D d : WithTop ℝ)) P).untopD 0 := by
  classical
  by_cases hs : P.coeff.support.Nonempty
  · change 0 ≤ (P.coeff.support.inf (WithTop.some ∘ D)).untopD 0
    rw [← Finset.coe_inf' hs D, WithTop.untopD_coe]
    exact (Finset.le_inf'_iff hs D).mpr hD
  · have he : P.coeff.support = ∅ := Finset.not_nonempty_iff_eq_empty.mp hs
    change 0 ≤ (P.coeff.support.inf (WithTop.some ∘ D)).untopD 0
    simp only [he, Finset.inf_empty, WithTop.untopD_top, le_refl]
#print axioms abstractLaw
#print axioms projectedLaw
end IndexNonneg138
