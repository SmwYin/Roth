noncomputable def indexViaInfDegree {σ : Type u} {R : Type v} [CommSemiring R]
    (P : MvPolynomial σ R) (α : σ → R) (w : (σ →₀ ℕ) → ℝ) : ℝ :=
  (AddMonoidAlgebra.infDegree (fun d ↦ (w d : WithTop ℝ))
    (aeval (fun j ↦ X j + C (α j)) P)).untopD 0
noncomputable def expandedIndex {σ : Type u} {R : Type v} [CommSemiring R]
    (P : MvPolynomial σ R) (α : σ → R) (w : (σ →₀ ℕ) → ℝ) : ℝ :=
  sInf (w '' ((aeval (fun j ↦ X j + C (α j)) P).support : Set (σ →₀ ℕ)))
theorem finsetBridge {ι : Type u} (s : Finset ι) (w : ι → ℝ) :
    (s.inf (fun d ↦ (w d : WithTop ℝ))).untopD 0 =
      sInf (w '' (s : Set ι)) := by
  classical
  rcases s.eq_empty_or_nonempty with h | h
  · subst s
    simp
  · change (s.inf (WithTop.some ∘ w)).untopD 0 = _
    rw [← Finset.coe_inf' h w, Finset.inf'_eq_csInf_image]
    rfl
theorem genericBridge {σ : Type u} {R : Type v} [CommSemiring R]
    (P : MvPolynomial σ R) (α : σ → R) (w : (σ →₀ ℕ) → ℝ) :
    indexViaInfDegree P α w = expandedIndex P α w :=
  finsetBridge _ _

noncomputable def denominatorIndex {σ : Type u} {R : Type v} [CommSemiring R]
    (P : MvPolynomial σ R) (α : σ → R) (r : σ → ℕ) : ℝ :=
  (AddMonoidAlgebra.infDegree
    (fun d : σ →₀ ℕ ↦ (Finsupp.weight (R := ℕ) (fun j ↦ (r j : ℝ)⁻¹) d : WithTop ℝ))
    (aeval (fun j ↦ X j + C (α j)) P)).untopD 0
theorem denominatorBridge {σ : Type u} {R : Type v} [CommSemiring R]
    (P : MvPolynomial σ R) (α : σ → R) (r : σ → ℕ) :
    denominatorIndex P α r = expandedIndex P α
      (fun d : σ →₀ ℕ ↦ d.sum fun j n ↦ (n : ℝ) * (r j : ℝ)⁻¹) := by
  simpa only [denominatorIndex, indexViaInfDegree, Finsupp.weight_apply, nsmul_eq_mul]
    using genericBridge P α (fun d : σ →₀ ℕ ↦ d.sum fun j n ↦ (n : ℝ) * (r j : ℝ)⁻¹)
#print axioms denominatorIndex
