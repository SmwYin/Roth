theorem law {s : Set ℝ} (hs : ∀ x ∈ s, 0 ≤ x) : 0 ≤ sSup s :=
  Real.sSup_nonneg hs
theorem family {A : Type u} {H : Type v} (f : A → H → ℝ) (s : Set A) (t : Set H)
    (hf : ∀ a ∈ s, ∀ h ∈ t, 0 ≤ f a h) : 0 ≤ sSup (Set.image2 f s t) :=
  Real.sSup_nonneg (Set.forall_mem_image2.mpr hf)
theorem witness {s : Set ℝ} (hs : ∃ x ∈ s, 0 ≤ x) : 0 ≤ sSup s :=
  Real.sSup_nonneg' hs
theorem empty : sSup (∅ : Set ℝ) = 0 := Real.sSup_empty
theorem unbounded {s : Set ℝ} (hs : ¬ BddAbove s) : sSup s = 0 :=
  Real.sSup_of_not_bddAbove hs
