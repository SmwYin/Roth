#check @lt_self_pow₀
theorem floorSquare {K : Type*} [Semiring K] [LinearOrder K] [IsStrictOrderedRing K]
    [FloorSemiring K] {X : K} (hX : 2 ≤ X) : X < (⌊X⌋₊ : K) ^ 2 := by
  have hn : 1 < ⌊X⌋₊ := (by decide : 1 < 2).trans_le (Nat.le_floor hX)
  have hp : ⌊X⌋₊ + 1 ≤ ⌊X⌋₊ ^ 2 := lt_self_pow₀ hn (by decide : 1 < 2)
  exact (Nat.lt_floor_add_one X).trans_le (by exact_mod_cast hp)
theorem fullBounds {X : ℝ} (hX : 2 ≤ X) :
    √X < (⌊X⌋₊ : ℝ) ∧ (⌊X⌋₊ : ℝ) ≤ X := by
  have hf : (0 : ℝ) < (⌊X⌋₊ : ℝ) := by
    exact_mod_cast (by decide : 0 < 2).trans_le (Nat.le_floor hX)
  exact ⟨(Real.sqrt_lt' hf).2 (floorSquare hX), Nat.floor_le (by positivity)⟩
