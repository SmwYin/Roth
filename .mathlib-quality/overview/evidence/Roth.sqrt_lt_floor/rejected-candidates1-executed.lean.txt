theorem halfRouteFails : ¬√(2 : ℝ) ≤ 2 / 2 := by
  intro h
  norm_num at h
  nlinarith only [h, Real.sqrt_nonneg 2, Real.sq_sqrt (by norm_num : (0 : ℝ) ≤ 2)]
theorem genericSearchProbe {X : ℝ} (hX : 2 ≤ X) :
    √X < (⌊X⌋₊ : ℝ) ∧ (⌊X⌋₊ : ℝ) ≤ X := by
  fail_if_success solve | exact?
  have hfloor : (2 : ℝ) ≤ ⌊X⌋₊ := by exact_mod_cast Nat.le_floor hX
  exact ⟨(Real.sqrt_lt' (by positivity)).2
    (by nlinarith only [Nat.lt_floor_add_one X, hfloor]), Nat.floor_le (by positivity)⟩
#print axioms halfRouteFails
