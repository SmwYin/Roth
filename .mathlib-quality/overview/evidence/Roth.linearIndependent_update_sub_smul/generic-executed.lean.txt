namespace IndependentUpdateProbe
theorem generic {R M ι : Type*} [Ring R] [AddCommGroup M] [Module R M] [DecidableEq ι]
    {v : ι → M} {i j : ι} (h : i ≠ j) (c : R) (hv : LinearIndependent R v) :
    LinearIndependent R (Function.update v j (v j - c • v i)) := by
  have hc : (Function.update (fun _ : ι ↦ (0 : R)) j (-c)) i = 0 := by
    simp [Function.update_of_ne h]
  have hx := (linearIndependent_add_smul_iff (v := v) hc).2 hv
  convert hx using 1
  funext k
  by_cases hk : k = j <;> simp [hk, Function.update_of_ne, sub_eq_add_neg]
#print axioms generic
end IndependentUpdateProbe
