variable {A α β : Type*} [CommSemiring A]
theorem extract (v : α → β → A) (s : Set α) (F : β → A)
    (hF : F ∈ Submodule.span A (v '' s)) :
    ∃ (t : Finset α) (c : α → A), (∀ a ∈ t, a ∈ s) ∧
      ∀ b, F b = ∑ a ∈ t, c a * v a b := by
  classical
  obtain ⟨c, hc, hsum⟩ := (Finsupp.mem_span_image_iff_linearCombination _).1 hF
  refine ⟨c.support, c, hc, fun b ↦ ?_⟩
  simpa [Finsupp.linearCombination_apply, Finsupp.sum] using congrFun hsum.symm b
#print axioms extract
end ExistsHasseExtraction
