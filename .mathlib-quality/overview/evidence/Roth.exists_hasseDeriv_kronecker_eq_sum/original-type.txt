Roth.exists_hasseDeriv_kronecker_eq_sum.{u_1, u_2} {K : Type u_1} [Field K] {σ : Type u_2} [Fintype σ] [CharZero K]
  (k μ : ℕ) :
  ∃ s c,
    (∀ d ∈ s, Finsupp.degree d ≤ μ) ∧
      ∀ (P : MvPolynomial σ K),
        (Polynomial.hasseDeriv μ) ((Roth.kronecker k) P) = ∑ d ∈ s, c d * (Roth.kronecker k) ((hasseDeriv d) P)
