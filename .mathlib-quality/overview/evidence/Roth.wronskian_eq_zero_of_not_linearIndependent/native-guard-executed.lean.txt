def matrix {R : Type*} [Semiring R] {l : ℕ} (φ : Fin l → R[X]) :
    Matrix (Fin l) (Fin l) R[X] := Matrix.of fun i j ↦ hasseDeriv i (φ j)
def W {R : Type*} [CommRing R] {l : ℕ} (φ : Fin l → R[X]) : R[X] := (matrix φ).det
theorem domain {R : Type*} [CommRing R] [IsDomain R] {l : ℕ} {φ : Fin l → R[X]}
    (h : ¬ LinearIndependent R φ) : W φ = 0 := by
  obtain ⟨g, hg, ν₀, hν₀⟩ := Fintype.not_linearIndependent_iff.1 h
  refine Matrix.exists_mulVec_eq_zero_iff.1 ⟨fun ν ↦ C (g ν), fun h0 ↦ hν₀ ?_, ?_⟩
  · simpa using congrFun h0 ν₀
  · ext1 μ
    simpa [matrix, Matrix.mulVec, dotProduct, ← smul_eq_C_mul, mul_comm]
      using congrArg (hasseDeriv μ) hg

theorem noZeroDivisors {R : Type*} [CommRing R] [NoZeroDivisors R]
    {l : ℕ} {φ : Fin l → R[X]} (h : ¬ LinearIndependent R φ) : W φ = 0 := by
  cases subsingleton_or_nontrivial R with
  | inl hR =>
    have := hR
    exact Subsingleton.elim _ _
  | inr hR =>
    have := hR
    have : IsDomain R := NoZeroDivisors.to_isDomain R
    exact domain h
theorem product_dependent :
    ¬ LinearIndependent (ℤ × ℤ)
      (fun _ : Fin 1 ↦ (C (1, 0) : (ℤ × ℤ)[X])) := by
  apply Fintype.not_linearIndependent_iff.mpr
  refine ⟨fun _ ↦ (0, 1), ?_, 0, by decide⟩
  simp [smul_eq_C_mul, ← C_mul]
theorem product_nonzero :
    W (fun _ : Fin 1 ↦ (C (1, 0) : (ℤ × ℤ)[X])) ≠ 0 := by
  simp [W, matrix, Matrix.det_unique]
