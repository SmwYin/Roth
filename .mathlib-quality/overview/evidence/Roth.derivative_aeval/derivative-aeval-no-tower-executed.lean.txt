theorem semiring {R σ : Type*} [CommSemiring R] [Fintype σ]
    (γ : σ → Polynomial R) (P : MvPolynomial σ R) :
    Polynomial.derivative (aeval γ P) =
      ∑ j, Polynomial.derivative (γ j) * aeval γ (pderiv j P) := by
  classical
  induction P using MvPolynomial.induction_on with
  | C a => simp
  | add p q hp hq => simp [hp, hq, Finset.sum_add_distrib, mul_add]
  | mul_X p j hp =>
    simp [hp, Finset.sum_add_distrib, ← Finset.mul_sum, add_mul, mul_assoc, mul_comm,
      Pi.single_apply, apply_ite (aeval γ)]
theorem abstract {R A M σ : Type*} [CommSemiring R] [CommSemiring A]
    [Algebra R A] [AddCommMonoid M] [Module R M] [Module A M]
    [Fintype σ] (D : Derivation R A M) (γ : σ → A) (P : MvPolynomial σ R) :
    D (aeval γ P) = ∑ j, aeval γ (pderiv j P) • D (γ j) := by
  classical
  induction P using MvPolynomial.induction_on with
  | C a => simp
  | add p q hp hq => simp [hp, hq, Finset.sum_add_distrib, add_smul]
  | mul_X p j hp =>
    simp [hp, Finset.sum_add_distrib, Finset.smul_sum, ← mul_smul, add_smul,
      Pi.single_apply, apply_ite (aeval γ)]

theorem finite_support {R A M σ : Type*} [CommSemiring R] [CommSemiring A]
    [Algebra R A] [AddCommMonoid M] [Module R M] [Module A M]
    (D : Derivation R A M) (γ : σ → A) (P : MvPolynomial σ R) :
    D (aeval γ P) = ∑ j ∈ P.vars, aeval γ (pderiv j P) • D (γ j) := by
  classical
  obtain ⟨Q, hQ⟩ := exists_rename_coe_vars_eq P
  have h := abstract D (γ ∘ Subtype.val) Q
  have hleft : aeval (γ ∘ Subtype.val) Q = aeval γ P := by
    rw [← aeval_rename, hQ]
  have hder (j : P.vars) :
      aeval (γ ∘ Subtype.val) (pderiv j Q) = aeval γ (pderiv (j : σ) P) := by
    rw [← aeval_rename, ← pderiv_rename Subtype.val_injective, hQ]
  calc
    D (aeval γ P) = ∑ j : P.vars, aeval γ (pderiv (j : σ) P) • D (γ j) := by
      simpa only [hleft, hder, Function.comp_apply] using h
    _ = ∑ j ∈ P.vars, aeval γ (pderiv j P) • D (γ j) :=
      Finset.sum_coe_sort P.vars (fun j ↦ aeval γ (pderiv j P) • D (γ j))
#print axioms finite_support
end DerivativeAEvalNoTower
