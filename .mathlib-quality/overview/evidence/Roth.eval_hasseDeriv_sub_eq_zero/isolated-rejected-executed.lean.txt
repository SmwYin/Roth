namespace Vanish251Isolated
open Polynomial
variable {R S A : Type*}
theorem eval_eq_of_mod_eq [Ring R] [CommRing S] (g : R →+* S) (x : S)
    {f p q : Polynomial R} (hf : eval₂ g x f = 0) (h : p %ₘ f = q %ₘ f) :
    eval₂ g x p = eval₂ g x q := by
  rw [← eval₂_modByMonic_eq_self_of_root hf,
    ← eval₂_modByMonic_eq_self_of_root (p := q) hf, h]
theorem observe [Ring R] [Ring S] [AddGroup A] (D : A →+ Polynomial R)
    (e : Polynomial R →+* S) {f : Polynomial R} (hf : e f = 0) {p q : A}
    (h : D p %ₘ f = D q %ₘ f) : e (D (p - q)) = 0 := by
  have hr (a : Polynomial R) : e (a %ₘ f) = e a := by
    rw [Polynomial.modByMonic_eq_sub_mul_div, map_sub, map_mul, hf, zero_mul, sub_zero]
  rw [map_sub, map_sub, sub_eq_zero, ← hr (D p), ← hr (D q), h]
theorem divisibility [Ring R] {f p q : Polynomial R} (h : p %ₘ f = q %ₘ f) :
    f ∣ p - q := by
  apply Polynomial.dvd_modByMonic_iff_dvd.mp
  simp only [Polynomial.sub_modByMonic, h, sub_self, dvd_zero]
theorem root_needed : (1 : Polynomial ℤ) %ₘ 1 = 0 %ₘ 1 ∧
    Polynomial.eval (0 : ℤ) ((1 : Polynomial ℤ) - 0) ≠ 0 := by simp
theorem equality_needed : Polynomial.eval (0 : ℤ) (Polynomial.X : Polynomial ℤ) = 0 ∧
    (1 : Polynomial ℤ) %ₘ Polynomial.X ≠ 0 %ₘ Polynomial.X ∧
    Polynomial.eval (0 : ℤ) ((1 : Polynomial ℤ) - 0) ≠ 0 := by simp
end Vanish251Isolated
