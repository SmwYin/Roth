namespace WeightDiv142
theorem mapWeight {σ R M N : Type*} [Semiring R] [AddCommMonoid M] [AddCommMonoid N]
    [Module R M] [Module R N] (f : M →ₗ[R] N) (w : σ → M) (d : σ →₀ R) :
    f (Finsupp.weight w d) = Finsupp.weight (f ∘ w) d :=
  Finsupp.apply_linearCombination R f w d
variable {σ K : Type*} [Semifield K]
theorem castWeight (w : σ → ℕ) (d : σ →₀ ℕ) :
    (Finsupp.weight w d : K) = Finsupp.weight (fun i ↦ (w i : K)) d := by
  exact Finsupp.apply_linearCombination ℕ (Nat.castAddMonoidHom K).toNatLinearMap w d
theorem scaleWeight (w : σ → K) (c : K) (d : σ →₀ ℕ) :
    c * Finsupp.weight w d = Finsupp.weight (fun i ↦ c * w i) d := by
  exact Finsupp.apply_linearCombination ℕ
    ((DistribSMul.toLinearMap ℕ K c) : K →ₗ[ℕ] K) w d
variable [CharZero K] [Fintype σ]
theorem clear (r : σ → ℕ) (hr : ∀ i, r i ≠ 0) (d : σ →₀ ℕ) :
    Finsupp.weight (fun i ↦ (r i : K)⁻¹) d =
      (Finsupp.weight (fun j ↦ (∏ i, r i) / r j) d : K) / ∏ i, (r i : K) := by
  have hD : (∏ i, (r i : K)) ≠ 0 := by
    exact_mod_cast (Finset.prod_ne_zero_iff.mpr (fun i _ ↦ hr i))
  rw [castWeight, div_eq_inv_mul, scaleWeight]
  congr 2
  funext j
  rw [Nat.cast_div_charZero (Finset.dvd_prod_of_mem r (Finset.mem_univ j)), Nat.cast_prod,
    div_eq_mul_inv, ← mul_assoc, inv_mul_cancel₀ hD, one_mul]
end WeightDiv142

namespace WeightDiv142Extra
variable {σ K : Type*} [Semifield K] [CharZero K]
theorem supportClear (r : σ → ℕ) (d : σ →₀ ℕ) (D : ℕ) (hD : D ≠ 0)
    (hdiv : ∀ j ∈ d.support, r j ∣ D) :
    Finsupp.weight (fun j ↦ (r j : K)⁻¹) d =
      (Finsupp.weight (fun j ↦ D / r j) d : K) / D := by
  rw [WeightDiv142.castWeight, div_eq_inv_mul, WeightDiv142.scaleWeight]
  simp only [Finsupp.weight_apply, Finsupp.sum]
  apply Finset.sum_congr rfl
  intro j hj
  rw [Nat.cast_div_charZero (hdiv j hj), div_eq_mul_inv, ← mul_assoc,
    inv_mul_cancel₀ (by exact_mod_cast hD), one_mul]
theorem mixedZero :
    Finsupp.weight (fun j : Bool ↦ ((if j then 1 else 0 : ℕ) : ℝ)⁻¹)
        (Finsupp.single true 1) ≠
      (Finsupp.weight (fun j : Bool ↦ (∏ i : Bool, (if i then 1 else 0 : ℕ)) /
        (if j then 1 else 0)) (Finsupp.single true 1) : ℝ) /
        (∏ i : Bool, ((if i then 1 else 0 : ℕ) : ℝ)) := by
  simp [Finsupp.weight_single]
omit [CharZero K] in
theorem zeroIndex (r : σ → ℕ) :
    Finsupp.weight (fun j ↦ (r j : K)⁻¹) (0 : σ →₀ ℕ) =
      (Finsupp.weight (fun j ↦ 0 / r j) (0 : σ →₀ ℕ) : K) / 0 := by simp
end WeightDiv142Extra
