#check_failure Roth.weight_eq_finsuppWeight_div
namespace WeightDiv142
variable {σ K : Type*} [Semifield K] [CharZero K]
theorem castWeight (w : σ → ℕ) (d : σ →₀ ℕ) :
    (Finsupp.weight w d : K) = Finsupp.weight (fun i ↦ (w i : K)) d := by
  exact Finsupp.apply_linearCombination ℕ (Nat.castAddMonoidHom K).toNatLinearMap w d
theorem scaleWeight (w : σ → K) (c : K) (d : σ →₀ ℕ) :
    c * Finsupp.weight w d = Finsupp.weight (fun i ↦ c * w i) d := by
  exact Finsupp.apply_linearCombination ℕ
    ((DistribSMul.toLinearMap ℕ K c) : K →ₗ[ℕ] K) w d
variable [Fintype σ]
theorem clear (r : σ → ℕ) (hr : ∀ i, r i ≠ 0) (d : σ →₀ ℕ) :
    Finsupp.weight (fun i ↦ (r i : K)⁻¹) d =
      (Finsupp.weight (fun j ↦ (∏ i, r i) / r j) d : K) / ∏ i, (r i : K) := by
  rw [castWeight, div_eq_inv_mul, scaleWeight]
  congr 1
  funext j
  rw [Nat.cast_div_charZero (Finset.dvd_prod_of_mem r (Finset.mem_univ j)), Nat.cast_prod,
    div_eq_mul_inv, ← mul_assoc, inv_mul_cancel₀ (by exact_mod_cast Finset.prod_ne_zero_iff.mpr
      (fun i _ ↦ hr i)), one_mul]
end WeightDiv142

#check @Finsupp.apply_linearCombination
#check @DistribSMul.toLinearMap
#print axioms WeightDiv142.castWeight
#print axioms WeightDiv142.scaleWeight
#print axioms WeightDiv142.clear
