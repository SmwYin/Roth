open MvPolynomial
 theorem abstract_monomial {R A σ : Type*} [CommSemiring R] [CommSemiring A]
    [Algebra R A] (g : σ → A) (d : σ →₀ ℕ) (a : R) :
    aeval g (monomial d a) = algebraMap R A a * d.prod (fun i n ↦ g i ^ n) :=
  aeval_monomial g d a
 theorem weighted_monomial {R σ : Type*} [CommSemiring R] (w : σ → ℕ)
    (d : σ →₀ ℕ) (a : R) :
    aeval (fun j ↦ (Polynomial.X : Polynomial R) ^ w j) (monomial d a) =
      Polynomial.C a * Polynomial.X ^ Finsupp.weight w d := by
  classical
  simp only [aeval_monomial, Polynomial.algebraMap_eq, Finsupp.prod, Finsupp.weight_apply,
    Finsupp.sum, smul_eq_mul, ← pow_mul, mul_comm, ← Finset.prod_pow_eq_pow_sum]
