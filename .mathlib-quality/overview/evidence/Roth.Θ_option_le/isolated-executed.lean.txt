theorem abstract {α ι : Type*} [ConditionallyCompleteLinearOrder α] {S : Set α}
    (hS : S.Nonempty) {s : Finset ι} (hs : s.Nonempty) (g : ι → α)
    (h : ∀ x ∈ S, ∃ i ∈ s, x ≤ g i) : sSup S ≤ s.sup' hs g :=
  csSup_le hS fun x hx ↦ by
    obtain ⟨i, hi, hxi⟩ := h x hx
    exact Finset.le_sup'_of_le g hi hxi
theorem real {ι : Type*} {S : Set ℝ} {s : Finset ι} (hs : s.Nonempty) (g : ι → ℝ)
    (h : ∀ x ∈ S, ∃ i ∈ s, x ≤ g i) (h0 : 0 ≤ s.sup' hs g) :
    sSup S ≤ s.sup' hs g :=
  Real.sSup_le (fun x hx ↦ by
    obtain ⟨i, hi, hxi⟩ := h x hx
    exact Finset.le_sup'_of_le g hi hxi) h0
theorem witness_iff {ι : Type*} {s : Finset ι} (hs : s.Nonempty)
    (g : ι → ℝ) (a : ℝ) : a ≤ s.sup' hs g ↔ ∃ i ∈ s, a ≤ g i :=
  Finset.le_sup'_iff hs
theorem empty_guard :
    (∀ x ∈ (∅ : Set ℝ), x ≤ -1) ∧ ¬sSup (∅ : Set ℝ) ≤ -1 := by
  simp [Real.sSup_empty]
theorem zero_order_guard (w : Unit → ℕ) :
    MvPowerSeries.weightedOrder w (0 : MvPowerSeries Unit ℝ) = ⊤ :=
  MvPowerSeries.weightedOrder_zero w
