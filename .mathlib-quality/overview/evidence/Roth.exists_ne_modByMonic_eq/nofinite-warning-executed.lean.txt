theorem noFinite {σ : Type*} (r : σ → ℕ) (B : ℕ)
    {f : Polynomial ℤ} (hf : f.Monic) (J : Finset (σ →₀ ℕ)) (C : ℕ)
    (hC : ∀ W ∈ boxFamily r B, ∀ j ∈ J, ∀ i < f.natDegree,
      |(diag (hasseDeriv j W) %ₘ f).coeff i| ≤ C)
    (hcount : ((2 * C + 1 : ℕ) : ℝ) ^ (f.natDegree * J.card) < ((B : ℝ) + 1) ^ ∏ᶠ j, (r j + 1)) :
    ∃ W' ∈ boxFamily r B, ∃ W'' ∈ boxFamily r B, W' ≠ W'' ∧
      ∀ j ∈ J, diag (hasseDeriv j W') %ₘ f = diag (hasseDeriv j W'') %ₘ f  := by
  classical
  by_cases hfin : Finite (Function.support r)
  · let : Finite (Function.support r) := hfin
    classical
    let : Fintype (boxFamily r B) := (boxFamily_finite r B).fintype
    let F (W : boxFamily r B) (p : J × Fin f.natDegree) : Set.Icc (-(C : ℤ)) C :=
      ⟨(diag (hasseDeriv p.1.1 W.1) %ₘ f).coeff p.2, abs_le.1 (hC W.1 W.2 p.1.1 p.1.2 p.2.1 p.2.2)⟩
    obtain ⟨W', W'', hne, h⟩ := Fintype.exists_ne_map_eq_of_card_lt F <| by
      simpa only [Fintype.card_fun, Fintype.card_ofFinset, Int.card_Icc, sub_neg_eq_add,
        Fintype.card_prod, Fintype.card_coe, Fintype.card_fin, mul_comm, Set.fintypeCard_eq_ncard,
        ncard_boxFamily, show ((C : ℤ) + 1 + C).toNat = 2 * C + 1 by lia] using
        (show (2 * C + 1) ^ (f.natDegree * J.card) < (B + 1) ^ ∏ᶠ j, (r j + 1) by
          exact_mod_cast hcount)
    refine ⟨W', W'.2, W'', W''.2, Subtype.coe_ne_coe.2 hne, fun j hj ↦ ?_⟩
    simpa [F, Polynomial.sum_modByMonic_coeff hf Polynomial.degree_le_natDegree] using
      congrArg (fun g ↦ ∑ i : Fin f.natDegree, Polynomial.monomial (i : ℕ)
        (g (⟨j, hj⟩, i) : ℤ)) h

  · have : Infinite (Function.support r) := not_finite_iff_infinite.mp hfin
    have hB : 0 < B := by
      by_contra hn
      have hzero : B = 0 := Nat.eq_zero_of_not_pos hn
      simp [hzero] at hcount
      exact (not_lt_of_ge (one_le_pow₀
        (le_add_of_nonneg_left (by positivity) : (1 : ℝ) ≤ 2 * C + 1))) hcount
    have hX (j : Function.support r) : (X j.1 : MvPolynomial σ ℤ) ∈ boxFamily r B := by
      refine ⟨fun k ↦ ?_, fun d ↦ ?_⟩
      · by_cases hk : k = j.1
        · subst k
          simpa only [degreeOf_X_self] using Nat.one_le_iff_ne_zero.2 j.2
        · simp only [degreeOf_X, hk, ite_false]
          exact Nat.zero_le _
      · rw [coeff_X]
        split_ifs <;> omega
    let F (j : Function.support r) (p : J × Fin f.natDegree) : Set.Icc (-(C : ℤ)) C :=
      ⟨(diag (hasseDeriv p.1.1 (X j.1)) %ₘ f).coeff p.2,
        abs_le.1 (hC _ (hX j) p.1.1 p.1.2 p.2.1 p.2.2)⟩
    obtain ⟨j, k, hne, he⟩ := Finite.exists_ne_map_eq_of_infinite F
    refine ⟨X j.1, hX j, X k.1, hX k, ?_, fun i hi ↦ ?_⟩
    · exact fun h ↦ hne (Subtype.ext (MvPolynomial.X_injective h))
    · simpa [F, Polynomial.sum_modByMonic_coeff hf Polynomial.degree_le_natDegree] using
        congrArg (fun g ↦ ∑ n : Fin f.natDegree, Polynomial.monomial (n : ℕ)
          (g (⟨i, hi⟩, n) : ℤ)) he
end Collision249NoFinite
