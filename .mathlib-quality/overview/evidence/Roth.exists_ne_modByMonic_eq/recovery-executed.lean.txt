theorem finiteObservations {α ι : Type*} [Fintype α] [Fintype ι]
    {β : ι → Type*} [∀ i, Fintype (β i)] (F : α → ∀ i, β i)
    (h : (∏ i, Fintype.card (β i)) < Fintype.card α) :
    ∃ x y, x ≠ y ∧ ∀ i, F x i = F y i := by
  classical
  obtain ⟨x, y, hne, he⟩ := Fintype.exists_ne_map_eq_of_card_lt F (by
    simpa only [Fintype.card_pi] using h)
  exact ⟨x, y, hne, fun i ↦ congrFun he i⟩

theorem polynomialObservations {α ι : Type*} [Fintype α] [Fintype ι]
    (Q : α → ι → Polynomial ℤ) {f : Polynomial ℤ} (hf : f.Monic) (C : ℕ)
    (hC : ∀ x j i, i < f.natDegree → |(Q x j %ₘ f).coeff i| ≤ C)
    (hcount : (2 * C + 1) ^ (f.natDegree * Fintype.card ι) < Fintype.card α) :
    ∃ x y, x ≠ y ∧ ∀ j, Q x j %ₘ f = Q y j %ₘ f := by
  classical
  let F (x : α) (p : ι × Fin f.natDegree) : Set.Icc (-(C : ℤ)) C :=
    ⟨(Q x p.1 %ₘ f).coeff p.2, abs_le.1 (hC x p.1 p.2 p.2.2)⟩
  obtain ⟨x, y, hne, he⟩ := Fintype.exists_ne_map_eq_of_card_lt F (by
    simpa only [Fintype.card_fun, Fintype.card_ofFinset, Int.card_Icc, sub_neg_eq_add,
      Fintype.card_prod, Fintype.card_fin, mul_comm,
      show ((C : ℤ) + 1 + C).toNat = 2 * C + 1 by omega] using hcount)
  refine ⟨x, y, hne, fun j ↦ ?_⟩
  simpa [F, Polynomial.sum_modByMonic_coeff hf Polynomial.degree_le_natDegree] using
    congrArg (fun g ↦ ∑ i : Fin f.natDegree, Polynomial.monomial (i : ℕ)
      (g (j, i) : ℤ)) he

theorem remainderRecover {R : Type*} [Ring R] {f P Q : Polynomial R} (hf : f.Monic)
    (h : ∀ i < f.natDegree, (P %ₘ f).coeff i = (Q %ₘ f).coeff i) : P %ₘ f = Q %ₘ f := by
  rw [← Polynomial.sum_modByMonic_coeff hf Polynomial.degree_le_natDegree (p := P),
    ← Polynomial.sum_modByMonic_coeff hf Polynomial.degree_le_natDegree (p := Q)]
  exact Finset.sum_congr rfl fun i _ ↦ congrArg (Polynomial.monomial (i : ℕ)) (h i i.2)

example (C N : ℕ) : (2 * C + 1) ^ (0 * N) = 1 := by simp
example (n C : ℕ) : (2 * C + 1) ^ (n * 0) = 1 := by simp
example (N : ℕ) : ¬ (1 : ℝ) < ((0 : ℝ) + 1) ^ N := by simp
example (P : Polynomial ℤ) : P %ₘ (1 : Polynomial ℤ) = 0 := by simp
example : (Polynomial.X : Polynomial ℤ) %ₘ 0 ≠ 0 %ₘ 0 := by simp
example : ∀ i < (0 : Polynomial ℤ).natDegree,
    ((Polynomial.X : Polynomial ℤ) %ₘ 0).coeff i = (0 %ₘ (0 : Polynomial ℤ)).coeff i := by
  simp
example : ¬ ∃ x y : Fin 1, x ≠ y := by simp

namespace Collision249Recovery
open Roth MvPolynomial
theorem recovered {σ : Type*} (r : σ → ℕ) [Finite (Function.support r)] (B : ℕ)
    {f : Polynomial ℤ} (hf : f.Monic) (J : Finset (σ →₀ ℕ)) (C : ℕ)
    (hC : ∀ W ∈ boxFamily r B, ∀ j ∈ J, ∀ i < f.natDegree,
      |(diag (hasseDeriv j W) %ₘ f).coeff i| ≤ C)
    (hcount : ((2 * C + 1 : ℕ) : ℝ) ^ (f.natDegree * J.card) < ((B : ℝ) + 1) ^ ∏ᶠ j, (r j + 1)) :
    ∃ W' ∈ boxFamily r B, ∃ W'' ∈ boxFamily r B, W' ≠ W'' ∧
      ∀ j ∈ J, diag (hasseDeriv j W') %ₘ f = diag (hasseDeriv j W'') %ₘ f  := by
  classical
  fail_if_success
    exact Fintype.exists_ne_map_eq_of_card_lt
      (fun W : boxFamily r B ↦ fun j : J ↦ diag (hasseDeriv j.1 W.1) %ₘ f) hcount
  let : Fintype (boxFamily r B) := (boxFamily_finite r B).fintype
  obtain ⟨x, y, hne, h⟩ := Collision249Generic.polynomialObservations
    (fun W : boxFamily r B ↦ fun j : J ↦ diag (hasseDeriv j.1 W.1)) hf C
    (fun W j i hi ↦ hC W.1 W.2 j.1 j.2 i hi) (by
      simpa only [Fintype.card_coe, Set.fintypeCard_eq_ncard, ncard_boxFamily] using
        (show (2 * C + 1) ^ (f.natDegree * J.card) < (B + 1) ^ ∏ᶠ j, (r j + 1) by
          exact_mod_cast hcount))
  exact ⟨x, x.2, y, y.2, Subtype.coe_ne_coe.2 hne, fun j hj ↦ h ⟨j, hj⟩⟩

example (B : ℕ) : (boxFamily (fun _ : Unit ↦ 0) B).ncard = B + 1 := by
  rw [ncard_boxFamily]
  simp
example (r : Unit → ℕ) : (boxFamily r 0).ncard = 1 := by
  rw [ncard_boxFamily]
  simp
end Collision249Recovery
