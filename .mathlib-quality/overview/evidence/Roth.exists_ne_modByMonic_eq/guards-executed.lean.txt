theorem zeroBoundConstant {σ : Type*} {W : MvPolynomial σ ℤ} {B : ℕ}
    (hW : W ∈ boxFamily (fun _ ↦ 0) B) : W = MvPolynomial.C (W.coeff 0) := by
  classical
  apply MvPolynomial.ext
  intro d
  by_cases hd : d = 0
  · simp only [hd, MvPolynomial.coeff_C, ite_true]
  have hz : W.coeff d = 0 := by
    by_contra hn
    have hs : d ∈ W.support := mem_support_iff.2 hn
    apply hd
    exact Finsupp.ext fun j ↦ Nat.eq_zero_of_le_zero
      ((monomial_le_degreeOf j hs).trans (hW.1 j))
  rw [hz, MvPolynomial.coeff_C_of_ne_zero hd]

theorem nonmonicCannotDrop :
    ¬ ∃ W' ∈ boxFamily (fun _ : Unit ↦ 0) 1,
      ∃ W'' ∈ boxFamily (fun _ : Unit ↦ 0) 1, W' ≠ W'' ∧
        ∀ j ∈ ({0} : Finset (Unit →₀ ℕ)),
          diag (hasseDeriv j W') %ₘ Polynomial.C (2 : ℤ) =
            diag (hasseDeriv j W'') %ₘ Polynomial.C (2 : ℤ) := by
  rintro ⟨W', hW', W'', hW'', hne, he⟩
  have hf : ¬ (Polynomial.C (2 : ℤ)).Monic := by
    simpa only [Polynomial.Monic, Polynomial.leadingCoeff_C] using
      (by norm_num : (2 : ℤ) ≠ 1)
  have eq := he 0 (Finset.mem_singleton_self _)
  rw [hasseDeriv_zero_apply, hasseDeriv_zero_apply,
    Polynomial.modByMonic_eq_of_not_monic _ hf,
    Polynomial.modByMonic_eq_of_not_monic _ hf] at eq
  rw [zeroBoundConstant hW', zeroBoundConstant hW''] at eq
  have hc : W'.coeff 0 = W''.coeff 0 := by
    simpa [diag] using congrArg (fun p : Polynomial ℤ ↦ p.coeff 0) eq
  exact hne ((zeroBoundConstant hW').trans (hc ▸ (zeroBoundConstant hW'').symm))

example : ((2 * 0 + 1 : ℕ) : ℝ) ^

      |(diag (hasseDeriv j W) %ₘ Polynomial.C (2 : ℤ)).coeff i| ≤ (0 : ℤ) := by simp
example : ∃ W' ∈ boxFamily (fun _ : Unit ↦ 0) 1,
    ∃ W'' ∈ boxFamily (fun _ : Unit ↦ 0) 1, W' ≠ W'' ∧
      ∀ j ∈ (∅ : Finset (Unit →₀ ℕ)),
        diag (hasseDeriv j W') %ₘ (1 : Polynomial ℤ) =
          diag (hasseDeriv j W'') %ₘ (1 : Polynomial ℤ) := by
  exact exists_ne_modByMonic_eq (fun _ ↦ 0) 1 Polynomial.monic_one ∅ 0 (by simp) (by norm_num)
