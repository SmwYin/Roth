theorem arbitraryFintype (i j : Fintype σ) (P : MvPolynomial σ ℝ) (α : σ → ℝ)
    (r : σ → ℕ) : @index σ i P α r = @index σ j P α r := by
  have : i = j := Subsingleton.elim _ _
  subst j
  rfl

theorem negativeScaleGuard :
    sInf ((fun n : ℕ ↦ -(n : ℝ)) '' ({1, 2} : Set ℕ)) ≠ -(1 : ℝ) := by
  have h : sInf ((fun n : ℕ ↦ -(n : ℝ)) '' ({1, 2} : Set ℕ)) ≤ -2 :=
    csInf_le ((Set.toFinite ({1, 2} : Set ℕ)).image _).bddBelow
      (Set.mem_image_of_mem _ (by simp))
  linarith
