namespace ClearDenoms171
open MvPolynomial
theorem originalProbe171 {σ : Type*} (G : MvPolynomial σ ℚ) :
    ∃ (N : ℕ) (U : MvPolynomial σ ℤ), 0 < N ∧
      map (Int.castRingHom ℚ) U = (N : ℚ) • G := by
  let : Algebra (MvPolynomial σ ℤ) (MvPolynomial σ ℚ) :=
    MvPolynomial.algebraMvPolynomial
  obtain ⟨⟨_, n, hn, rfl⟩, U, hU⟩ :=
    IsLocalization.exists_integer_multiple
      ((Submonoid.pos ℤ).map (C (σ := σ))) G
  have hn' : (0 : ℤ) < n := hn
  have hnat : (n.natAbs : ℤ) = n := Int.natAbs_of_nonneg hn'.le
  have hcast : (n.natAbs : ℚ) = (n : ℚ) := by
    simpa only [Int.cast_natCast] using congrArg (fun z : ℤ ↦ (z : ℚ)) hnat
  refine ⟨n.natAbs, U, Int.natAbs_pos.mpr (ne_of_gt hn'), ?_⟩
  have hU' : map (Int.castRingHom ℚ) U = C (n : ℚ) * G := by
    simpa only [MvPolynomial.algebraMap_def, Algebra.smul_def, map_C, algebraMap_int_eq,
      Int.coe_castRingHom] using hU
  simpa only [smul_eq_C_mul, hcast] using hU'

#print axioms ClearDenoms171.originalProbe171
end ClearDenoms171

namespace Clear171Generic
variable {σ R S : Type*} [CommRing R] [CommRing S] [Algebra R S]
    (M : Submonoid R) [IsLocalization M S]
attribute [local instance] MvPolynomial.algebraMvPolynomial
-- Existing generic theorem, exact one-call specialization, no project dependencies.
theorem clear (G : MvPolynomial σ S) :
    ∃ b : M.map (C (σ := σ)),
      IsLocalization.IsInteger (MvPolynomial σ R) ((b : MvPolynomial σ R) • G) :=
  IsLocalization.exists_integer_multiple (M.map (C (σ := σ))) G
#print axioms clear
-- The coefficient-family route works over commutative semirings already.
theorem finiteFamily {A B ι : Type*} [CommSemiring A] [CommSemiring B] [Algebra A B]
    (T : Submonoid A) [IsLocalization T B] (s : Finset ι) (f : ι → B) :
    ∃ b : T, ∀ i ∈ s, IsLocalization.IsInteger A ((b : A) • f i) :=
  IsLocalization.exist_integer_multiples T s f
#print axioms finiteFamily
end Clear171Generic


namespace Primitive172
open MvPolynomial
theorem primitive172 {σ : Type*} {G : MvPolynomial σ ℚ} (hG : G ≠ 0) :
    ∃ (g : ℚ) (U : MvPolynomial σ ℤ), g ≠ 0 ∧ map (Int.castRingHom ℚ) U = g • G ∧
      U.support.gcd (fun d ↦ U.coeff d) = 1 := by
  classical
  obtain ⟨N, U0, hN, hU0⟩ := ClearDenoms171.originalProbe171 G
  have hne : U0.support.Nonempty := support_nonempty.mpr fun h ↦ by
    simpa [h, smul_eq_zero, hN.ne', hG] using hU0.symm
  obtain ⟨g', hg', hg1⟩ := Finset.extract_gcd (fun d ↦ U0.coeff d) hne
  set g0 := U0.support.gcd (fun d ↦ U0.coeff d) with hg0def
  have hg0 : g0 ≠ 0 := by
    obtain ⟨d, hd⟩ := hne
    rw [hg0def, Ne, Finset.gcd_eq_zero_iff]
    exact fun h ↦ mem_support_iff.1 hd (h d hd)
  set U := ∑ d ∈ U0.support, monomial d (g' d) with hUdef
  have hUcoeff : ∀ d, U.coeff d = if d ∈ U0.support then g' d else 0 := by
    simp [hUdef, coeff_monomial, Finset.sum_ite_eq']
  have hU0coeff : ∀ d, U0.coeff d = g0 * U.coeff d := by
    intro d
    by_cases hd : d ∈ U0.support
    · simpa [hUcoeff, hd] using hg' d hd
    · simp [hUcoeff, hd, notMem_support_iff.1 hd]
  refine ⟨N / g0, U, div_ne_zero (by exact_mod_cast hN.ne') (by exact_mod_cast hg0), ?_, ?_⟩
  · ext d
    have h1 := congrArg (fun P ↦ P.coeff d) hU0
    simp only [coeff_map, coeff_smul, smul_eq_mul, eq_intCast] at h1 ⊢
    rw [hU0coeff] at h1
    push_cast at h1
    rwa [div_mul_eq_mul_div, eq_div_iff (by exact_mod_cast hg0), mul_comm]
  · have hsupp : U.support = U0.support := by
      ext d
      simp [mem_support_iff, hU0coeff d, hg0]
    rw [hsupp, ← hg1]
    exact Finset.gcd_congr rfl fun d hd ↦ by rw [hUcoeff, ite_eq_left hd]
#print axioms primitive172
end Primitive172

    {P : MvPolynomial σ R} (hP : P ≠ 0) :
    ∃ U : MvPolynomial σ R,
      P = C (P.support.gcd P.coeff) * U ∧ U.support.gcd U.coeff = 1 := by
  classical
  have hne : P.support.Nonempty := support_nonempty.mpr hP
  obtain ⟨g', hg', hg1⟩ := Finset.extract_gcd P.coeff hne
  set g0 := P.support.gcd P.coeff with hg0def
  have hg0 : g0 ≠ 0 := by
    obtain ⟨d, hd⟩ := hne
    rw [hg0def, Ne, Finset.gcd_eq_zero_iff]
    exact fun h ↦ mem_support_iff.1 hd (h d hd)
  set U := ∑ d ∈ P.support, monomial d (g' d) with hUdef
  have hUcoeff : ∀ d, U.coeff d = if d ∈ P.support then g' d else 0 := by
    simp [hUdef, coeff_monomial, Finset.sum_ite_eq']
  have hPcoeff : ∀ d, P.coeff d = g0 * U.coeff d := by

    · simpa [hUcoeff, hd] using hg' d hd
    · simp [hUcoeff, hd, notMem_support_iff.1 hd]
  refine ⟨U, ?_, ?_⟩
  · ext d
    simpa only [coeff_C_mul] using hPcoeff d
  · have hsupp : U.support = P.support := by
      ext d
      simp [mem_support_iff, hPcoeff d, hg0]

namespace Primitive172Guards
-- Nonzero is essential: mapping an integer polynomial to zero forces gcd zero.
theorem zeroImpossible {σ : Type*} :
    ¬ ∃ (g : ℚ) (U : MvPolynomial σ ℤ), g ≠ 0 ∧
      map (Int.castRingHom ℚ) U = g • (0 : MvPolynomial σ ℚ) ∧
      U.support.gcd U.coeff = 1 := by
  rintro ⟨g, U, hg, hU, hc⟩
  have hU0 : U = 0 := MvPolynomial.map_injective _ Int.cast_injective (by simpa using hU)
  simpa [hU0] using hc
-- Dropping both scalar nonvanishing and primitivity permits a vacuous zero choice.
theorem vacuous {σ : Type*} (G : MvPolynomial σ ℚ) :
    ∃ (g : ℚ) (U : MvPolynomial σ ℤ), map (Int.castRingHom ℚ) U = g • G := by
  exact ⟨0, 0, by simp⟩
#print axioms zeroImpossible
#print axioms vacuous
end Primitive172Guards

variable {σ R K : Type*} [CommRing R] [IsDomain R] [NormalizedGCDMonoid R]
    [Field K] [Algebra R K] [IsFractionRing R K]
theorem primitive {G : MvPolynomial σ K} (hG : G ≠ 0) :
    ∃ (g : K) (U : MvPolynomial σ R), g ≠ 0 ∧ map (algebraMap R K) U = g • G ∧
      U.support.gcd U.coeff = 1 := by
  classical
  let : Algebra (MvPolynomial σ R) (MvPolynomial σ K) :=
    MvPolynomial.algebraMvPolynomial
  obtain ⟨⟨_, N, hNmem, rfl⟩, U0, hU0raw⟩ :=
    IsLocalization.exists_integer_multiple ((nonZeroDivisors R).map (C (σ := σ))) G
  have hN : N ≠ 0 := mem_nonZeroDivisors_iff_ne_zero.mp hNmem
  have hinj : Function.Injective (algebraMap R K) := IsFractionRing.injective R K
  have hNcast : algebraMap R K N ≠ 0 := (map_ne_zero_iff _ hinj).mpr hN
  have hU0 : map (algebraMap R K) U0 = algebraMap R K N • G := by
    simpa only [MvPolynomial.algebraMap_def, Algebra.smul_def, map_C,
      smul_eq_C_mul] using hU0raw
  have hne : U0.support.Nonempty := support_nonempty.mpr fun h ↦ by
    simpa [h, smul_eq_zero, hNcast, hG] using hU0.symm
  obtain ⟨g', hg', hg1⟩ := Finset.extract_gcd (fun d ↦ U0.coeff d) hne
  set g0 := U0.support.gcd (fun d ↦ U0.coeff d) with hg0def
  have hg0 : g0 ≠ 0 := by
    obtain ⟨d, hd⟩ := hne
    rw [hg0def, Ne, Finset.gcd_eq_zero_iff]
    exact fun h ↦ mem_support_iff.1 hd (h d hd)
  set U := ∑ d ∈ U0.support, monomial d (g' d) with hUdef
  have hUcoeff : ∀ d, U.coeff d = if d ∈ U0.support then g' d else 0 := by
    simp [hUdef, coeff_monomial, Finset.sum_ite_eq']
  have hU0coeff : ∀ d, U0.coeff d = g0 * U.coeff d := by
    intro d
    by_cases hd : d ∈ U0.support
    · simpa [hUcoeff, hd] using hg' d hd
    · simp [hUcoeff, hd, notMem_support_iff.1 hd]
  refine ⟨algebraMap R K N / algebraMap R K g0, U,
    div_ne_zero hNcast ((map_ne_zero_iff _ hinj).mpr hg0), ?_, ?_⟩
  · ext d
    have h1 := congrArg (fun P ↦ P.coeff d) hU0
    simp only [coeff_map, coeff_smul, smul_eq_mul] at h1 ⊢
    rw [hU0coeff] at h1
    rw [map_mul] at h1
    rwa [div_mul_eq_mul_div, eq_div_iff ((map_ne_zero_iff _ hinj).mpr hg0), mul_comm]
  · have hsupp : U.support = U0.support := by
      ext d
      simp [mem_support_iff, hU0coeff d, hg0]
    rw [hsupp, ← hg1]
    exact Finset.gcd_congr rfl fun d hd ↦ by rw [hUcoeff, ite_eq_left hd]


#print axioms primitive
