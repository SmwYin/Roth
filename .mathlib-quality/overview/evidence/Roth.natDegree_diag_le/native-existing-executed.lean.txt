namespace MvPolynomial.DegreeDiagExisting
variable {σ R : Type*} [CommSemiring R]
theorem total (P : MvPolynomial σ R) :
    (aeval (fun _ ↦ (Polynomial.X : Polynomial R)) P).natDegree ≤ P.totalDegree := by
  simpa only [mul_one] using
    (aeval_natDegree_le P le_rfl (fun _ ↦ Polynomial.X) fun _ ↦
      (Polynomial.natDegree_X_le : (Polynomial.X : Polynomial R).natDegree ≤ 1))
theorem total_bound [Fintype σ] (P : MvPolynomial σ R) {r : σ → ℕ}
    (hP : ∀ j, degreeOf j P ≤ r j) : P.totalDegree ≤ ∑ j, r j := by
  classical
  refine (totalDegree_le_degrees_card P).trans ?_
  rw [← Multiset.sum_count_eq_card (s := Finset.univ) (fun _ _ ↦ Finset.mem_univ _)]
  exact Finset.sum_le_sum fun j _ ↦ by simpa only [← degreeOf_def] using hP j
theorem generic [Fintype σ] (P : MvPolynomial σ R) {r : σ → ℕ}
    (hP : ∀ j, degreeOf j P ≤ r j) :
    (aeval (fun _ ↦ (Polynomial.X : Polynomial R)) P).natDegree ≤ ∑ j, r j := by
  simpa only [mul_one] using
    (aeval_natDegree_le P (total_bound P hP) (fun _ ↦ Polynomial.X) fun _ ↦
      (Polynomial.natDegree_X_le : (Polynomial.X : Polynomial R).natDegree ≤ 1))
