namespace MvPolynomial.DiagSumRaw
open MvPolynomial
variable {σ R : Type*} [CommSemiring R]
theorem generic (P : MvPolynomial σ R) :
    aeval (fun _ ↦ (Polynomial.X : Polynomial R)) P =
      ∑ k ∈ P.support, Polynomial.monomial k.degree (P.coeff k) := by
  classical
  rw [aeval_def, eval₂_eq]
  simp_rw [Finset.prod_pow_eq_pow_sum, ← Finsupp.degree_apply, Polynomial.algebraMap_eq,
    Polynomial.C_mul_X_pow_eq_monomial]
theorem existingExpansion (P : MvPolynomial σ R) :
    aeval (fun _ ↦ (Polynomial.X : Polynomial R)) P =
      ∑ k ∈ P.support, Polynomial.monomial k.degree (P.coeff k) := by
  classical
  simpa only [MvPolynomial.aeval_def, Finset.prod_pow_eq_pow_sum,
    ← Finsupp.degree_apply, Polynomial.algebraMap_eq,
    Polynomial.C_mul_X_pow_eq_monomial] using
    (eval₂_eq (algebraMap R (Polynomial R)) (fun _ ↦ Polynomial.X) P)

namespace MvPolynomial.DegreeDiagRaw
variable {σ R : Type*} [CommSemiring R]
theorem total (P : MvPolynomial σ R) :
    (aeval (fun _ ↦ (Polynomial.X : Polynomial R)) P).natDegree ≤ P.totalDegree := by
  classical
  rw [DiagSumRaw.generic]
  refine Polynomial.natDegree_sum_le_of_forall_le _ _ fun k hk ↦
    (Polynomial.natDegree_monomial_le _).trans ?_
  change k.sum (fun _ e ↦ e) ≤ P.totalDegree
  exact le_totalDegree hk
theorem generic [Fintype σ] (P : MvPolynomial σ R) {r : σ → ℕ}
    (hP : ∀ j, degreeOf j P ≤ r j) :
    (aeval (fun _ ↦ (Polynomial.X : Polynomial R)) P).natDegree ≤ ∑ j, r j := by
  classical
  rw [DiagSumRaw.generic]
  refine Polynomial.natDegree_sum_le_of_forall_le _ _ fun k hk ↦
    (Polynomial.natDegree_monomial_le _).trans ?_
  rw [Finsupp.degree_eq_sum]
  exact Finset.sum_le_sum fun j _ ↦ (monomial_le_degreeOf j hk).trans (hP j)
