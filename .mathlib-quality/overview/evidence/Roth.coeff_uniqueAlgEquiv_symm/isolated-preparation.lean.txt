namespace UniqueCoeff161
open MvPolynomial
variable {R : Type*} [CommSemiring R] {σ : Type*} [Unique σ]
theorem oldForm (f : Polynomial R) (n : ℕ) :
    ((uniqueAlgEquiv R σ).symm f).coeff (Finsupp.single default n) = f.coeff n :=
  (MvPolynomial.coeff_uniqueAlgEquiv_symm R f _).trans
    (congrArg f.coeff Finsupp.single_eq_same)
theorem generalForm (f : Polynomial R) (d : σ →₀ ℕ) :
    ((uniqueAlgEquiv R σ).symm f).coeff d = f.coeff (d default) :=
  MvPolynomial.coeff_uniqueAlgEquiv_symm R f d
theorem arbitraryCoordinate (f : Polynomial R) (n : ℕ) (j : σ) :
    ((uniqueAlgEquiv R σ).symm f).coeff (Finsupp.single j n) = f.coeff n := by
  rw [Subsingleton.elim j default]
  exact oldForm f n
theorem representationBridge (f : Polynomial R) (d : σ →₀ ℕ) :
    d = Finsupp.single default (d default) := (Finsupp.unique_single d).symm
theorem falseGeneralization :
    (X true : MvPolynomial Bool ℕ).coeff (Finsupp.single false 1) ≠
      (Polynomial.X : Polynomial ℕ).coeff 1 := by
  simp [MvPolynomial.coeff_X, Finsupp.single_eq_single_iff]
#check_failure (MvPolynomial.uniqueAlgEquiv ℕ Bool)
