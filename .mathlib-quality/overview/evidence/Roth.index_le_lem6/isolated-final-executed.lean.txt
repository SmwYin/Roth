theorem numeric {θ Φ δ rp l : ℝ} (hΦ : 0 ≤ Φ) (hδ1 : δ ≤ 1) (hrp : 3 ≤ rp)
    (hl : 0 < l) (hl' : l ≤ rp + 1) (hmin : min (l * θ / 2) (rp * θ ^ 2 / 3) ≤ l * (Φ + δ)) :
    θ ≤ 2 * (Φ + √Φ + √δ) := by
  fail_if_success solve | simp
  fail_if_success solve | positivity
  -- The small-index comparison `δ ≤ √δ` also controls the linear bound.
  have hsΦ : 0 ≤ √Φ := Real.sqrt_nonneg _
  have hrp0 : 0 < rp := (by norm_num : (0 : ℝ) < 3).trans_le hrp
  rcases min_le_iff.1 hmin with hk | hk
  · have hθ : θ ≤ 2 * (Φ + δ) := by
      rwa [mul_div_assoc, mul_le_mul_iff_right₀ hl, div_le_iff₀ (by norm_num), mul_comm] at hk
    linarith only [hθ, Real.le_sqrt_self_iff.2 hδ1, hsΦ]
  · have hsq : θ ^ 2 ≤ 4 * (Φ + δ) := (mul_le_mul_iff_right₀ hrp0).1 (by
      have hsize : 3 * l ≤ 4 * rp := by linarith only [hl', hrp]
      linear_combination 3 * hk + mul_le_mul_of_nonneg_right hsize
        (nonneg_of_mul_nonneg_right
          ((div_nonneg (mul_nonneg hrp0.le (sq_nonneg θ)) (by norm_num)).trans hk) hl))
    have hθ : θ ≤ 2 * (√Φ + √δ) := le_of_sq_le_sq (by
      linear_combination hsq + 4 * ((le_max_left δ 0).trans_eq (Real.sq_sqrt').symm) +
        4 * (Real.sq_sqrt hΦ).symm + 8 * mul_nonneg hsΦ (Real.sqrt_nonneg δ)) (by positivity)
    linarith only [hθ, hΦ]


theorem improvedNumeric {θ Φ δ rp l : ℝ} (hΦ : 0 ≤ Φ) (hδ1 : δ ≤ 1) (hrp : 1 ≤ rp)
    (hl : 0 < l) (hl' : l ≤ rp + 1) (hmin : min (l * θ / 2) (rp * θ ^ 2 / 2) ≤ l * (Φ + δ)) :
    θ ≤ 2 * (Φ + √Φ + √δ) := by

theorem abstractWitness {θ δ rp : ℝ} {Φ : ℕ → ℝ} (hδ : δ ≤ 1) (hrp : 1 ≤ rp)
    (hΦ : ∀ l, 0 ≤ Φ l)
    (hw : ∃ l : ℕ, 1 ≤ l ∧ (l : ℝ) ≤ rp + 1 ∧
      min (l * θ / 2) (rp * θ ^ 2 / 2) ≤ l * (Φ l + δ)) :
    ∃ l : ℕ, 1 ≤ l ∧ (l : ℝ) ≤ rp + 1 ∧ θ ≤ 2 * (Φ l + √(Φ l) + √δ) := by
  obtain ⟨l, hl, hl', hm⟩ := hw
  refine ⟨l, hl, hl', ?_⟩
  exact improvedNumeric (hΦ l) hδ hrp (by exact_mod_cast hl) hl' hm

theorem zero_denominator_semantics : (1 : ℝ) / (0 : ℕ) = 0 := by norm_num

theorem integer_height_not_rational : |(100 : ℚ) * (1 / 100)| = 1 ∧ 1 < |(100 : ℚ)| := by
  norm_num

theorem missing_phi_nonneg_guard :
    min ((4 : ℝ) * (6 / 5) / 2) (3 * (6 / 5) ^ 2 / 3) ≤ 4 * (-1 / 2 + 1) ∧
      ¬(6 / 5 : ℝ) ≤ 2 * (-1 / 2 + √(-1 / 2) + √1) := by
  norm_num

theorem missing_positive_l_guard :
    min ((0 : ℝ) * 10 / 2) (3 * 10 ^ 2 / 3) ≤ 0 * (0 + 0) ∧
      ¬(10 : ℝ) ≤ 2 * (0 + √0 + √0) := by norm_num

theorem weightedOrder_hit {σ A : Type*} [CommSemiring A] [NoZeroDivisors A]
    (w : σ → ℕ) (f g : MvPowerSeries σ A) :
    MvPowerSeries.weightedOrder w (f * g) =
      MvPowerSeries.weightedOrder w f + MvPowerSeries.weightedOrder w g := by
  exact MvPowerSeries.weightedOrder_mul w f g
