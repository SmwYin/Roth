/-! Isolated iterativity assessment: raw coefficient and cocycle witnesses, no project import. -/
open MvPolynomial
namespace CoeffAssessment
variable {σ R : Type*} [CommSemiring R]
noncomputable def delta (d : σ →₀ ℕ) : MvPolynomial σ R →ₗ[R] MvPolynomial σ R :=
  Finsupp.lsum R (fun k : σ →₀ ℕ ↦
    (monomial (k - d)).comp (LinearMap.lsmul R R ((d.prod (fun j ↦ (k j).choose) : ℕ) : R))) ∘ₗ
      (AddMonoidAlgebra.coeffLinearEquiv R).toLinearMap

-- Coefficient formula is independently proved below.
theorem independent (d e : σ →₀ ℕ) (P : MvPolynomial σ R) :
    (delta d P).coeff e = ((d.prod (fun j ↦ ((e + d) j).choose) : ℕ) : R) * P.coeff (e + d) := by
  classical
  have hz (k : σ →₀ ℕ) : d.prod (fun j ↦ (k j).choose) = 0 ↔ ¬d ≤ k := by
    simp [Finsupp.prod, Finset.prod_eq_zero_iff, Nat.choose_eq_zero_iff, Finsupp.le_iff,
      not_forall]
  induction P using MvPolynomial.induction_on' with
  | monomial k a =>
    rw [show delta d (monomial k a) =
      monomial (k - d) (((d.prod (fun j ↦ (k j).choose) : ℕ) : R) * a) by simp [delta],
      coeff_monomial, coeff_monomial]
    by_cases hk : k = e + d
    · simp [hk]
    by_cases hdk : d ≤ k
    · simp [hk, tsub_eq_iff_eq_add_of_le hdk]
    · simp [hk, (hz k).2 hdk]
  | add p q hp hq => simp [hp, hq, mul_add]

def factor (d k : σ →₀ ℕ) : ℕ := d.prod fun j ↦ (k j).choose

theorem cocycle (f d e : σ →₀ ℕ) :
    factor d (f + d) * factor e (f + d + e) =
      factor d (d + e) * factor (d + e) (f + (d + e)) := by
  classical
  simp only [factor]
  have hd : d.support ⊆ d.support ∪ e.support := Finset.subset_union_left
  rw [Finsupp.prod_of_support_subset _ hd _ fun _ _ ↦ Nat.choose_zero_right _,
    Finsupp.prod_of_support_subset _ Finset.subset_union_right _
      fun _ _ ↦ Nat.choose_zero_right _,
    Finsupp.prod_of_support_subset _ hd _ fun _ _ ↦ Nat.choose_zero_right _,
    Finsupp.prod_of_support_subset _ Finsupp.support_add _ fun _ _ ↦ Nat.choose_zero_right _,
    ← Finset.prod_mul_distrib, ← Finset.prod_mul_distrib]
  exact Finset.prod_congr rfl fun i _ ↦ by
    simpa [← add_assoc, mul_comm, ← Nat.choose_symm_add (a := d i) (b := e i)] using
      (Nat.choose_mul (n := f i + d i + e i) (k := d i + e i) (s := e i)
        (Nat.le_add_left (e i) (d i))).symm

theorem iterativity (d e : σ →₀ ℕ) (P : MvPolynomial σ R) :
    delta d (delta e P) = (factor d (d + e) : R) • delta (d + e) P := by
  apply MvPolynomial.ext
  intro f
  change (delta d (delta e P)).coeff f = _
  rw [coeff_smul, independent, independent, independent, smul_eq_mul,
    ← mul_assoc, ← mul_assoc, ← Nat.cast_mul, ← Nat.cast_mul]
  change ((factor d (f + d) * factor e (f + d + e) : ℕ) : R) * _ = _
  rw [cocycle, add_assoc]
  rfl

example (d e : σ →₀ ℕ) :
    (delta (R := R) d).comp (delta e) = (factor d (d + e) : R) • delta (d + e) := by
  apply LinearMap.ext
  intro P
  exact iterativity d e P

theorem coeff_factor (d e : σ →₀ ℕ) (P : MvPolynomial σ R) :
    (delta d P).coeff e = (factor d (e + d) : R) * P.coeff (e + d) :=
  independent d e P
theorem succ_factor (j : σ) (d e : σ →₀ ℕ) :
    factor d (e + Finsupp.single j 1 + d) * (e j + 1) =
      (d j + 1) * factor (d + Finsupp.single j 1) (e + Finsupp.single j 1 + d) := by
  simpa [factor, add_comm, add_left_comm, add_assoc, mul_comm] using
    cocycle e (Finsupp.single j 1) d
theorem pderiv_delta (j : σ) (d : σ →₀ ℕ) (P : MvPolynomial σ R) :
    pderiv j (delta d P) = ((d j + 1 : ℕ) : R) • delta (d + Finsupp.single j 1) P := by
  apply MvPolynomial.ext
  intro e
  rw [coeff_pderiv, coeff_smul, coeff_factor, coeff_factor, smul_eq_mul,
    ← add_assoc e d, add_right_comm e d, mul_right_comm, ← Nat.cast_add_one, ← Nat.cast_mul,
    succ_factor, Nat.cast_mul, mul_assoc]

variable {S : Type*} [Semiring S]
theorem univariate (n : ℕ) (p : Polynomial S) :
    Polynomial.derivative (Polynomial.hasseDeriv n p) =
      (n + 1) • Polynomial.hasseDeriv (n + 1) p := by
  simpa only [LinearMap.comp_apply, Polynomial.hasseDeriv_one', LinearMap.smul_apply,
    Nat.choose_one_right, Nat.add_comm] using
    LinearMap.congr_fun (Polynomial.hasseDeriv_comp 1 n) p

namespace FactorialRecurrenceAssessment
variable {M : Type*} [AddCommMonoid M]
theorem recurrence (D : M →+ M) (H : ℕ → M → M)
    (hzero : ∀ x, H 0 x = x)
    (hsucc : ∀ n x, D (H n x) = (n + 1) • H (n + 1) x)
    (n : ℕ) (x : M) : n.factorial • H n x = (⇑D)^[n] x := by
  induction n with
  | zero => simp [hzero]
  | succ n ih =>
    rw [Function.iterate_succ_apply', ← ih, map_nsmul, hsucc, smul_smul,
      Nat.factorial_succ, Nat.mul_comm]

end FactorialRecurrenceAssessment

namespace RawFactorialAssessment
open MvPolynomial CoeffAssessment
variable {σ R : Type*} [CommSemiring R]
theorem delta_zero (P : MvPolynomial σ R) : delta 0 P = P := by
  apply MvPolynomial.ext
  intro e
  simp only [independent, Finsupp.prod_zero_index, Nat.cast_one, one_mul, add_zero]
theorem independent_factorial (j : σ) (n : ℕ) (P : MvPolynomial σ R) :
    (n.factorial : R) • delta (Finsupp.single j n) P = (⇑(pderiv (R := R) j))^[n] P := by
  induction n with
  | zero => simp [delta_zero]
  | succ n ih =>
    rw [Function.iterate_succ_apply', ← ih, Derivation.map_smul, pderiv_delta, smul_smul,
      Finsupp.single_eq_same, ← Finsupp.single_add, Nat.factorial_succ, Nat.cast_mul, mul_comm]

namespace GenericPolynomialConsumer
variable {S : Type*} [Semiring S]
theorem univariate_from_recurrence (n : ℕ) (p : Polynomial S) :
    n.factorial • Polynomial.hasseDeriv n p = (⇑(Polynomial.derivative (R := S)))^[n] p := by
  exact FactorialRecurrenceAssessment.recurrence
    (Polynomial.derivative (R := S)).toAddMonoidHom
    (fun m ↦ ⇑(Polynomial.hasseDeriv (R := S) m))
    (fun q ↦ Polynomial.hasseDeriv_zero' q)
    (fun m q ↦ by
      change Polynomial.derivative (Polynomial.hasseDeriv m q) = _
      simpa only [LinearMap.comp_apply, Polynomial.hasseDeriv_one', LinearMap.smul_apply,
        Nat.choose_one_right, Nat.add_comm] using
        LinearMap.congr_fun (Polynomial.hasseDeriv_comp 1 m) q) n p
#print axioms FactorialRecurrenceAssessment.recurrence
#print axioms univariate_from_recurrence
end GenericPolynomialConsumer
