namespace PrivateScalarAssessment
open MvPolynomial
variable {σ : Type*}
theorem replay (f d e : σ →₀ ℕ) :
    hasseCoeff d (f + d) * hasseCoeff e (f + d + e) =
      hasseCoeff d (d + e) * hasseCoeff (d + e) (f + (d + e)) := by
  classical
  simp only [hasseCoeff]
  have hd : d.support ⊆ d.support ∪ e.support := Finset.subset_union_left
  rw [Finsupp.prod_of_support_subset _ hd _ fun _ _ ↦ Nat.choose_zero_right _,
    Finsupp.prod_of_support_subset _ Finset.subset_union_right _
      fun _ _ ↦ Nat.choose_zero_right _,
    Finsupp.prod_of_support_subset _ hd _ fun _ _ ↦ Nat.choose_zero_right _,
    Finsupp.prod_of_support_subset _ Finsupp.support_add _ fun _ _ ↦ Nat.choose_zero_right _,
    ← Finset.prod_mul_distrib, ← Finset.prod_mul_distrib]
  exact Finset.prod_congr rfl fun i _ ↦ by
    simpa [← add_assoc, mul_comm, ← Nat.choose_symm_add (a := d i) (b := e i)] using
      (Nat.choose_mul (n := f i + d i + e i) (k := d i + e i) (s := e i)
        (Nat.le_add_left (e i) (d i))).symm
#check @replay
#print axioms replay
end PrivateScalarAssessment

-- Scoped first rewrite from the sole external caller, with arbitrary Option-indexed input.
theorem separated_caller {τ : Type*} (d : τ →₀ ℕ) (ν : ℕ)
    (P : MvPolynomial (Option τ) R) :
    hasseDeriv (d.mapDomain some + Finsupp.single none ν) P =
      hasseDeriv (d.mapDomain some) (hasseDeriv (Finsupp.single none ν) P) := by
  classical
  have hnone : ∀ {o : Option τ}, o ∈ (d.mapDomain some).support → o ≠ none := by simp
  have hdisj : Disjoint (d.mapDomain some).support (Finsupp.single none ν).support :=

      hnone ho (Finset.mem_singleton.1 (Finsupp.support_single_subset ho'))
