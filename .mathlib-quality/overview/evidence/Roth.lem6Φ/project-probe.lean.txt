theorem finset_bridge {ι : Type u} (s : Finset ι) (w : ι → ℝ) :
    (s.inf (fun d => (w d : WithTop ℝ))).untopD 0 = sInf (w '' (s : Set ι)) := by
  classical
  rcases s.eq_empty_or_nonempty with h | h
  · subst s
    simp
  · change (s.inf (WithTop.some ∘ w)).untopD 0 = _
    rw [← Finset.coe_inf' h w, Finset.inf'_eq_csInf_image]
    rfl

theorem index_bridge {σ : Type u} [Fintype σ] (P : MvPolynomial σ ℝ)
    (α : σ → ℝ) (r : σ → ℕ) : index P α r =
    (AddMonoidAlgebra.infDegree
      (fun d : σ →₀ ℕ => (Finsupp.weight (R := ℕ) (fun j => (r j : ℝ)⁻¹) d : WithTop ℝ))
      (aeval (fun j => X j + C (α j)) P)).untopD 0 := by
  have hw (d : σ →₀ ℕ) :
      Finsupp.weight (R := ℕ) (fun j => (r j : ℝ)⁻¹) d = weight r d := by
    simp only [weight, Finsupp.weight_eq_sum, nsmul_eq_mul, div_eq_mul_inv]
  simpa only [index, taylor, hw, AddMonoidAlgebra.infDegree,
    MvPolynomial.finsupp_support_eq_support] using
    (finset_bridge (taylor α P).support (weight r)).symm

theorem expanded_image2 {σ : Type u} [Fintype σ] (B : ℝ) (q r : σ → ℕ) :
    indexSet B q r = Set.image2 (fun R (h : σ → ℤ) =>
      (AddMonoidAlgebra.infDegree
        (fun d : σ →₀ ℕ =>
          (Finsupp.weight (R := ℕ) (fun j => (r j : ℝ)⁻¹) d : WithTop ℝ))
        (aeval (fun j => X j + C ((h j : ℝ) / q j))
          (map (Int.castRingHom ℝ) R))).untopD 0)
      (RSet B r) {h : σ → ℤ | ∀ j, IsCoprime (h j) (q j : ℤ)} := by
  change Set.image2 _ _ _ = _
  exact Set.image2_congr' fun R h => index_bridge _ _ _
variable {τ : Type u} [Fintype τ]
theorem literal (q r : Option τ → ℕ) (M : ℝ) (l : ℕ) : lem6Φ q r M l =
    Θ M (fun _ : Unit => q none) (fun _ => l * r none) +
      Θ M (q ∘ some) (fun j => l * r (some j)) := rfl
theorem constructor (q r : Option τ → ℕ) (M : ℝ) (l : ℕ) : lem6Φ q r M l =
    Add.add (sSup (indexSet M (fun _ : Unit => q none) (fun _ => l * r none)))
      (sSup (indexSet M (q ∘ some) (fun j => l * r (some j)))) := rfl
theorem image2_formula (q r : Option τ → ℕ) (M : ℝ) (l : ℕ) : lem6Φ q r M l =
    sSup (Set.image2 (fun R (h : Unit → ℤ) =>
      index (map (Int.castRingHom ℝ) R) (fun j => (h j : ℝ) / q none)
        (fun _ => l * r none))
      (RSet M (fun _ : Unit => l * r none))
      {h : Unit → ℤ | ∀ j, IsCoprime (h j) (q none : ℤ)}) +
    sSup (Set.image2 (fun R (h : τ → ℤ) =>
      index (map (Int.castRingHom ℝ) R) (fun j => (h j : ℝ) / q (some j))
        (fun j => l * r (some j)))
      (RSet M (fun j => l * r (some j)))
      {h : τ → ℤ | ∀ j, IsCoprime (h j) (q (some j) : ℤ)}) := rfl
noncomputable def expanded_family {σ : Type u} (M : ℝ) (q r : σ → ℕ) : Set ℝ :=
  Set.image2 (fun R (h : σ → ℤ) =>
    (AddMonoidAlgebra.infDegree
      (fun d : σ →₀ ℕ =>
        (Finsupp.weight (R := ℕ) (fun j => (r j : ℝ)⁻¹) d : WithTop ℝ))
      (aeval (fun j => X j + C ((h j : ℝ) / q j))
        (map (Int.castRingHom ℝ) R))).untopD 0)
    {R : MvPolynomial σ ℤ | R ≠ 0 ∧ (∀ j, degreeOf j R ≤ r j) ∧
      ∀ d, |((R.coeff d : ℤ) : ℝ)| ≤ M}
    {h : σ → ℤ | ∀ j, IsCoprime (h j) (q j : ℤ)}
theorem family_bridge {σ : Type u} [Fintype σ] (M : ℝ) (q r : σ → ℕ) :
    indexSet M q r = expanded_family M q r := by
  change Set.image2 _ _ _ = _
  exact Set.image2_congr' fun R h => index_bridge _ _ _
noncomputable def expanded {τ : Type u} (q r : Option τ → ℕ) (M : ℝ) (l : ℕ) : ℝ :=
  sSup (expanded_family M (fun _ : Unit => q none) (fun _ => l * r none)) +
    sSup (expanded_family M (q ∘ some) (fun j => l * r (some j)))
theorem expansion (q r : Option τ → ℕ) (M : ℝ) (l : ℕ) :
    lem6Φ q r M l = expanded q r M l := by
  change sSup (indexSet _ _ _) + sSup (indexSet _ _ _) = _
  rw [family_bridge, family_bridge]
  rfl
example (q r : Option Empty → ℕ) (M : ℝ) (l : ℕ) : ℝ := lem6Φ q r M l
example (q r : Option (ULift.{1} Unit) → ℕ) (M : ℝ) (l : ℕ) : ℝ := lem6Φ q r M l
example (q r : Option τ → ℕ) (M : ℝ) : lem6Φ q r M 0 = expanded q r M 0 :=
  expansion q r M 0
example (r : Option τ → ℕ) (M : ℝ) (l : ℕ) :
    lem6Φ (fun _ => 0) r M l = expanded (fun _ => 0) r M l := expansion _ r M l
