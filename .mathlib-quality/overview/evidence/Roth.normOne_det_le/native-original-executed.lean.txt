namespace MvPolynomial.NormDetAssessment
open Roth
#check @Roth.normOne_det_le
#print Roth.normOne_det_le
#print axioms Roth.normOne_det_le
variable {σ : Type*}
theorem replay {l : ℕ} (M : Matrix (Fin l) (Fin l) (MvPolynomial σ ℤ)) {C : ℝ}
    (hM : ∀ μ ν, normOne (M μ ν) ≤ C) :
    normOne M.det ≤ l.factorial * C ^ l := by
  classical
  rw [Matrix.det_apply']
  calc _ ≤ ∑ τ : Equiv.Perm (Fin l), normOne (_ * ∏ i, M (τ i) i) :=
          normOne_sum_le _ _
    _ ≤ ∑ _τ : Equiv.Perm (Fin l), C ^ l := Finset.sum_le_sum fun τ _ ↦ ?_
    _ = l.factorial * C ^ l := by simp [Fintype.card_perm]
  rcases Int.units_eq_one_or (Equiv.Perm.sign τ) with h | h <;>
    simpa [h, normOne_neg] using (normOne_prod_le Finset.univ fun i ↦ M (τ i) i).trans
      ((Finset.prod_le_prod₀ (fun _ _ ↦ normOne_nonneg _) fun i _ ↦ hM _ _).trans_eq (by simp))
#print axioms replay
end MvPolynomial.NormDetAssessment
