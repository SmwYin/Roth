namespace NormDetRaw
variable {R n : Type*} [CommRing R] [Fintype n] [DecidableEq n]
theorem genericDet (size : R → ℝ) (hzero : size 0 = 0)
    (hnonneg : ∀ a, 0 ≤ size a) (hone : size 1 ≤ 1)
    (hneg : ∀ a, size (-a) = size a)
    (hadd : ∀ a b, size (a + b) ≤ size a + size b)
    (hmul : ∀ a b, size (a * b) ≤ size a * size b)
    (M : Matrix n n R) {C : ℝ} (hM : ∀ i j, size (M i j) ≤ C) :
    size M.det ≤ (Fintype.card n).factorial * C ^ Fintype.card n := by
  classical
  rw [Matrix.det_apply']
  calc _ ≤ ∑ τ : Equiv.Perm n, size (_ * ∏ i, M (τ i) i) :=
      Finset.le_sum_of_subadditive size hzero.le hadd _ _
    _ ≤ ∑ _τ : Equiv.Perm n, C ^ Fintype.card n :=
      Finset.sum_le_sum fun τ _ ↦ ?_
    _ = _ := by simp [Fintype.card_perm]
  rcases Int.units_eq_one_or (Equiv.Perm.sign τ) with h | h <;>
    simpa [h, hneg] using
      (Finset.le_prod_of_submultiplicative_of_nonneg size hnonneg hone hmul
        Finset.univ fun i ↦ M (τ i) i).trans
        ((Finset.prod_le_prod₀ (fun _ _ ↦ hnonneg _) fun i _ ↦ hM _ _).trans_eq
          (by simp))
#print axioms genericDet
end NormDetRaw
