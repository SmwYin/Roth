section Bridge
variable {R : Type*} [CommSemiring R] {σ : Type*} [Unique σ]
theorem representationBridge (f : Polynomial R) :
    optionEquivLeft R {b : σ // b ≠ default}
      (rename (Equiv.optionSubtypeNe (default : σ)).symm ((uniqueAlgEquiv R σ).symm f)) =
        f.map MvPolynomial.C := by
  classical
  have h : (optionEquivLeft R {b : σ // b ≠ default}).toAlgHom.comp
      ((rename (Equiv.optionSubtypeNe (default : σ)).symm).comp
        (uniqueAlgEquiv R σ).symm.toAlgHom) =
      Polynomial.mapAlgHom (Algebra.ofId R _ : R →ₐ[R] MvPolynomial {b : σ // b ≠ default} R) := by
    apply Polynomial.algHom_ext
    simp [uniqueAlgEquiv_symm_apply]
  exact AlgHom.congr_fun h f
theorem canonicalRoute (f : Polynomial R) :
    degreeOf default ((uniqueAlgEquiv R σ).symm f) = f.natDegree := by
  classical
  rw [degreeOf_eq_natDegree, representationBridge]
  exact Polynomial.natDegree_map_eq_of_injective (MvPolynomial.C_injective _ _) f
end Bridge
section Domain
variable {R τ : Type*} [CommRing R] [IsDomain R]
theorem degreeOf_none_rename_mul_aeval {U : MvPolynomial τ R} (hU : U ≠ 0) (V : Polynomial R) :
    degreeOf none (rename some U * Polynomial.aeval (X none : MvPolynomial (Option τ) R) V) =
      V.natDegree := by
  classical
  by_cases hV : V = 0
  · simp [hV]
  rw [degreeOf_mul_eq ((rename_eq_zero_iff_of_injective U (Option.some_injective τ)).not.mpr hU)
      ((map_ne_zero_iff (Polynomial.aeval (X none))
        (Polynomial.toMvPolynomial_injective none)).2 hV),
    (show degreeOf none (rename some U) = 0 by
      simp [degreeOf, degrees_rename_of_injective (Option.some_injective τ)]),
    zero_add, aeval_X_eq_rename]
  exact (degreeOf_rename_of_injective (f := fun _ : Unit ↦ (none : Option τ))
    (fun _ _ _ ↦ rfl) ()).trans (canonicalRoute V)

end Domain
section Theta
variable {τ : Type*} [Fintype τ]
theorem thetaCaller {r : Option τ → ℕ} (hr0 : ∀ j, 0 < r j) {q : Option τ → ℕ}
    {h : Option τ → ℤ} (hh : ∀ j, IsCoprime (h j) (q j : ℤ)) {M : ℝ} {l : ℕ} (hl : 0 < l)
    {U : MvPolynomial τ ℤ} (hU : U ∈ RSet M (fun j ↦ l * r (some j))) {V : Polynomial ℤ}
    (hV : V ≠ 0) (hVd : V.natDegree ≤ l * r none) (hVc : ∀ b, |(V.coeff b : ℝ)| ≤ M) :
    index (map (Int.castRingHom ℝ)
        (rename some U * Polynomial.aeval (X none : MvPolynomial (Option τ) ℤ) V))
      (fun j ↦ (h j : ℝ) / q j) r ≤ l * lem6Φ q r M l := by
  classical
  set V' : MvPolynomial Unit ℤ := (uniqueAlgEquiv ℤ Unit).symm V with hV'
  have hV'0 : V' ≠ 0 := by
    rwa [hV', ne_eq, map_eq_zero_iff _ (uniqueAlgEquiv ℤ Unit).symm.injective]
  have hVmem : V' ∈ RSet M (fun _ : Unit ↦ l * r none) := by
    refine ⟨hV'0, fun j ↦ ?_, fun d ↦ ?_⟩
    · rw [Subsingleton.elim j default, hV', canonicalRoute]
      exact hVd
    · rw [hV', MvPolynomial.coeff_uniqueAlgEquiv_symm]
      exact hVc _
  have hinjV : Function.Injective (fun _ : Unit ↦ (none : Option τ)) :=
    fun a b _ ↦ Subsingleton.elim a b
  have hmU : map (Int.castRingHom ℝ) U ≠ 0 := fun h0 ↦
    hU.1 (map_injective _ Int.cast_injective (h0.trans (map_zero _).symm))
  have hmV : map (Int.castRingHom ℝ) V' ≠ 0 := fun h0 ↦
    hV'0 (map_injective _ Int.cast_injective (h0.trans (map_zero _).symm))
  rw [aeval_X_eq_rename, _root_.map_mul, map_rename, map_rename,
    index_mul ((map_ne_zero_iff (rename (R := ℝ) some)
      (rename_injective _ (Option.some_injective _))).2 hmU)
      ((map_ne_zero_iff (rename (R := ℝ) (fun _ : Unit ↦ (none : Option τ)))
        (rename_injective _ hinjV)).2 hmV) _ hr0,
    index_rename (Option.some_injective _), index_rename hinjV]
  have hUΘ := index_le_Θ hU (h := h ∘ some)
    fun j ↦ hh (some j)
  have hVΘ := index_le_Θ hVmem
    (q := fun _ : Unit ↦ q none) (h := fun _ ↦ h none) fun _ ↦ hh none
  rw [index_mul_left_weights] at hUΘ hVΘ
  have hlR : (0 : ℝ) < l := by exact_mod_cast hl
  rw [lem6Φ, mul_add, add_comm]
  exact add_le_add ((div_le_iff₀' hlR).1 hVΘ) ((div_le_iff₀' hlR).1 hUΘ)

end Theta
