namespace CoeffBoundSupportGenerality
theorem supportLocal {α M N : Type*} [Zero M] [AddCommMonoid N] [Preorder N]
    [AddLeftMono N] (f : α →₀ M) {g : M → N} (hg : g 0 = 0)
    (h : ∀ i ∈ f.support, 0 ≤ g (f i)) (a : α) : g (f a) ≤ f.sum (fun _ m ↦ g m) := by
  classical
  by_cases ha : a ∈ f.support
  · exact Finset.single_le_sum h ha
  · rw [Finsupp.notMem_support_iff.mp ha, hg]
    exact Finset.sum_nonneg h
#print axioms supportLocal
#check @supportLocal
end CoeffBoundSupportGenerality
