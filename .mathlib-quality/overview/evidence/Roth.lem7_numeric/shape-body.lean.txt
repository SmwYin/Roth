namespace Numeric219Shape
open Roth

theorem divided {p : ℕ} (hp : 2 ≤ p) {δ Φ : ℝ} (hδ0 : 0 < δ) (hδ1 : δ ≤ 1)
    (hΦ : Φ ≤ 3 * 10 ^ (p - 1) * δ ^ ((1 / 2 : ℝ) ^ (p - 1))) :
    Φ + √Φ + √δ < (10 ^ p * δ ^ ((1 / 2 : ℝ) ^ p)) / 2 := by
  exact (lt_div_iff₀' (by norm_num : (0 : ℝ) < 2)).2 (lem7_numeric hp hδ0 hδ1 hΦ)

theorem successor (k : ℕ) {δ Φ : ℝ} (hδ0 : 0 < δ) (hδ1 : δ ≤ 1)
    (hΦ : Φ ≤ 3 * 10 ^ (k + 1) * δ ^ ((1 / 2 : ℝ) ^ (k + 1))) :
    2 * (Φ + √Φ + √δ) < 10 ^ (k + 2) * δ ^ ((1 / 2 : ℝ) ^ (k + 2)) := by
  exact lem7_numeric (by omega) hδ0 hδ1 hΦ

theorem negativePhi {p : ℕ} (hp : 2 ≤ p) {δ Φ : ℝ} (hδ0 : 0 < δ) (hδ1 : δ ≤ 1)
    (hΦ : Φ ≤ 0) : 2 * (Φ + √Φ + √δ) < 10 ^ p * δ ^ ((1 / 2 : ℝ) ^ p) := by
  exact lem7_numeric hp hδ0 hδ1 (hΦ.trans (by positivity))

theorem equalityUpper (p : ℕ) (hp : 2 ≤ p) :
    2 * (3 * 10 ^ (p - 1) + √(3 * 10 ^ (p - 1)) + √(1 : ℝ)) < 10 ^ p := by
  simpa using lem7_numeric hp (by norm_num : (0 : ℝ) < 1) (le_refl (1 : ℝ))
    (Φ := 3 * 10 ^ (p - 1)) (by simp)

theorem noPhiCap :
    ¬ (2 * ((1000 : ℝ) + √1000 + √1) < 10 ^ 2 * 1 ^ ((1 / 2 : ℝ) ^ 2)) := by
  have h := Real.sqrt_nonneg (1000 : ℝ)
  norm_num
  linarith
