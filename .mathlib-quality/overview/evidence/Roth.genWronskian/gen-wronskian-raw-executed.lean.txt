namespace GenWronskianRaw
open MvPolynomial
noncomputable def coefficientOperator {R σ : Type*} [CommSemiring R]
    (d : σ →₀ ℕ) (P : MvPolynomial σ R) : MvPolynomial σ R := by
  classical
  exact ∑ k ∈ P.support, monomial (k - d)
    (((d.prod fun j n ↦ (k j).choose n : ℕ) : R) * P.coeff k)
noncomputable def raw {R σ ι : Type*} [CommRing R] [Fintype ι] [DecidableEq ι]
    (D : ι → σ →₀ ℕ) (φ : ι → MvPolynomial σ R) : MvPolynomial σ R :=
  (Matrix.of fun i j ↦ coefficientOperator (D i) (φ j)).det
