open MvPolynomial
namespace Primitive172FullGeneric
variable {σ R K : Type*} [CommRing R] [IsDomain R] [NormalizedGCDMonoid R]
    [Field K] [Algebra R K] [IsFractionRing R K]
theorem primitive {G : MvPolynomial σ K} (hG : G ≠ 0) :
    ∃ (g : K) (U : MvPolynomial σ R), g ≠ 0 ∧ map (algebraMap R K) U = g • G ∧
      U.support.gcd U.coeff = 1 := by
  classical
  let : Algebra (MvPolynomial σ R) (MvPolynomial σ K) :=
    MvPolynomial.algebraMvPolynomial
  obtain ⟨⟨_, N, hNmem, rfl⟩, U0, hU0raw⟩ :=
    IsLocalization.exists_integer_multiple ((nonZeroDivisors R).map (C (σ := σ))) G
  have hN : N ≠ 0 := mem_nonZeroDivisors_iff_ne_zero.mp hNmem
  have hinj : Function.Injective (algebraMap R K) := IsFractionRing.injective R K
  have hNcast : algebraMap R K N ≠ 0 := (map_ne_zero_iff _ hinj).mpr hN
  have hU0 : map (algebraMap R K) U0 = algebraMap R K N • G := by
    have h : map (algebraMap R K) U0 = C (algebraMap R K N) * G := by
      simpa only [MvPolynomial.algebraMap_def, Algebra.smul_def, map_C] using hU0raw
    simpa only [smul_eq_C_mul] using h
  have hne : U0.support.Nonempty := support_nonempty.mpr fun h ↦ by
    exact smul_ne_zero hNcast hG (by simpa only [h, map_zero] using hU0.symm)
  obtain ⟨g', hg', hg1⟩ := Finset.extract_gcd (fun d ↦ U0.coeff d) hne
  set g0 := U0.support.gcd (fun d ↦ U0.coeff d) with hg0def
  have hg0 : g0 ≠ 0 := by
    obtain ⟨d, hd⟩ := hne
    rw [hg0def, Ne, Finset.gcd_eq_zero_iff]
    exact fun h ↦ mem_support_iff.1 hd (h d hd)
  set U := ∑ d ∈ U0.support, monomial d (g' d) with hUdef
  have hUcoeff : ∀ d, U.coeff d = if d ∈ U0.support then g' d else 0 := by
    simp [hUdef, coeff_monomial, Finset.sum_ite_eq']
  have hU0coeff : ∀ d, U0.coeff d = g0 * U.coeff d := by
    intro d
    by_cases hd : d ∈ U0.support
    · simpa [hUcoeff, hd] using hg' d hd
    · simp [hUcoeff, hd, notMem_support_iff.1 hd]
  refine ⟨algebraMap R K N / algebraMap R K g0, U,
    div_ne_zero hNcast ((map_ne_zero_iff _ hinj).mpr hg0), ?_, ?_⟩
  · ext d
    have h1 := congrArg (fun P ↦ P.coeff d) hU0
    simp only [coeff_map, coeff_smul, smul_eq_mul] at h1 ⊢
    rw [hU0coeff] at h1
    rw [map_mul] at h1
    rwa [div_mul_eq_mul_div, eq_div_iff ((map_ne_zero_iff _ hinj).mpr hg0), mul_comm]
  · have hsupp : U.support = U0.support := by
      ext d
      simp [mem_support_iff, hU0coeff d, hg0]
    rw [hsupp, ← hg1]
    exact Finset.gcd_congr rfl fun d hd ↦ by rw [hUcoeff, ite_eq_left hd]


#print axioms primitive
end Primitive172FullGeneric


namespace Factors176GCDScalar
variable {R K : Type*} [CommRing R] [IsDomain R] [NormalizedGCDMonoid R]
    [Field K] [Algebra R K] [IsFractionRing R K]
omit [IsDomain R] in
theorem scalar_integer {ι : Type*} {s : Finset ι} {u : ι → R} (hu : s.gcd u = 1)
    {v : K} (hv : ∀ a ∈ s, ∃ z : R, algebraMap R K (u a) * v = algebraMap R K z) :
    ∃ z : R, v = algebraMap R K z := by
  obtain ⟨a, b, hab, he⟩ := IsLocalization.surj_of_gcd_domain (nonZeroDivisors R) v
  have hrel : IsRelPrime a b := gcd_isUnit_iff_isRelPrime.mp hab
  have hdiv : b ∣ s.gcd u := Finset.dvd_gcd fun i hi ↦ by
    obtain ⟨z, hz⟩ := hv i hi
    have hmul : u i * a = z * b := IsFractionRing.injective R K (by
      simp only [map_mul]
      rw [← he, ← mul_assoc, hz])
    apply hrel.symm.dvd_of_dvd_mul_right
    exact hmul ▸ dvd_mul_left b z
  have hb : IsUnit b := isUnit_iff_dvd_one.mpr (hu ▸ hdiv)
  refine ⟨a * (↑(hb.unit⁻¹) : R), ?_⟩
  rw [map_mul]
  have hmap := Units.coe_map_inv (algebraMap R K : R →* K) hb.unit
  simp only [MonoidHom.coe_coe] at hmap
  rw [← hmap, Units.eq_mul_inv_iff_mul_eq]
  exact he
#print axioms scalar_integer
end Factors176GCDScalar

namespace Factors176Coeff
open MvPolynomial
variable {τ R : Type*} [CommSemiring R]
noncomputable def coeffNone (P : MvPolynomial (Option τ) R) (ν : ℕ) : MvPolynomial τ R :=
  (optionEquivLeft R τ P).coeff ν

/-- The coefficient of `x_τ^d` in `coeffNone P ν` is the coefficient of `x_τ^d x_none^ν` in `P`. -/
theorem coeff_coeffNone (P : MvPolynomial (Option τ) R) (ν : ℕ) (d : τ →₀ ℕ) :
    (coeffNone P ν).coeff d = P.coeff (d.mapDomain some + Finsupp.single none ν) := by
  rw [coeffNone, optionEquivLeft_coeff_coeff]
  congr 1
  ext o
  cases o <;> simp [Finsupp.mapDomain_of_notMem_range,
    Finsupp.mapDomain_apply_of_injective (Option.some_injective τ)]

theorem coeff_rename_mul_aeval (U : MvPolynomial τ R) (V : Polynomial R) (a : τ →₀ ℕ) (b : ℕ) :
    (rename some U * Polynomial.aeval (X none : MvPolynomial (Option τ) R) V).coeff
        (a.mapDomain some + Finsupp.single none b) = U.coeff a * V.coeff b := by
  have h : (optionEquivLeft R τ).toAlgHom.comp (rename some) = Polynomial.CAlgHom := by
    ext i
    simp
  rw [← coeff_coeffNone, coeffNone, map_mul,
    (show optionEquivLeft R τ (rename some U) = Polynomial.C U from AlgHom.congr_fun h U),
    ← Polynomial.aeval_algHom_apply, optionEquivLeft_X_none, Polynomial.aeval_X_left_eq_map,
    Polynomial.coeff_C_mul, Polynomial.coeff_map, algebraMap_eq, mul_comm U, coeff_C_mul, mul_comm]

end Factors176Coeff

namespace Factors176GCD
open MvPolynomial Factors176Coeff
variable {τ R K : Type*} [CommRing R] [IsDomain R] [NormalizedGCDMonoid R]
    [Field K] [Algebra R K] [IsFractionRing R K]
theorem rescale_separated_factor_aux {U' U : MvPolynomial τ K}
    {V' : Polynomial K} {F : MvPolynomial (Option τ) K} {g : K} (hg : g ≠ 0) (hUmap : U = g • U')
    (hF : F = rename some U' * Polynomial.aeval (X none : MvPolynomial (Option τ) K) V') :
    F = rename some U * Polynomial.aeval (X none : MvPolynomial (Option τ) K)
      (Polynomial.C g⁻¹ * V') := by
  rw [hF, hUmap, map_mul, Polynomial.aeval_C, algebraMap_eq, smul_eq_C_mul, map_mul,
    rename_C, mul_mul_mul_comm, ← C_mul, mul_inv_cancel₀ hg, C_1, one_mul]

omit [IsDomain R] in
theorem exists_int_polynomial_of_primitive_factor_aux {U : MvPolynomial τ R}
    {V' : Polynomial K} {F : MvPolynomial (Option τ) R} (hgcd : U.support.gcd U.coeff = 1)
    (hF : map (algebraMap R K) F = rename some (map (algebraMap R K) U) *
      Polynomial.aeval (X none : MvPolynomial (Option τ) K) V') :
    ∃ V : Polynomial R, V.map (algebraMap R K) = V' := by
  apply (Polynomial.mem_map_range (algebraMap R K)).2
  intro b
  obtain ⟨z, hz⟩ := Factors176GCDScalar.scalar_integer hgcd fun a _ ↦
    ⟨F.coeff (a.mapDomain some + Finsupp.single none b), by
      have h := congrArg (fun P ↦ P.coeff (a.mapDomain some + Finsupp.single none b)) hF
      simp only [coeff_rename_mul_aeval, coeff_map] at h
      simpa using h.symm⟩
  exact ⟨z, hz.symm⟩

/-- If an integer polynomial `F` factors over `K` as `U'(x_τ) V'(x_none)` with nonzero factors,
then it factors over `R` as `U(x_τ) V(x_none)` with the same partial degrees.
Source: §2, proof of Lemma 2 ("Thus `U, V` are integral and nonzero, as required"), together
with the degree comparison at the end of the proof. -/
theorem exists_int_factors {U' : MvPolynomial τ K} {V' : Polynomial K} (hU : U' ≠ 0) (hV : V' ≠ 0)
    {F : MvPolynomial (Option τ) R} (hF : map (algebraMap R K) F =
      rename some U' * Polynomial.aeval (X none : MvPolynomial (Option τ) K) V') :
    ∃ (U : MvPolynomial τ R) (V : Polynomial R), U ≠ 0 ∧ V ≠ 0 ∧
      F = rename some U * Polynomial.aeval (X none : MvPolynomial (Option τ) R) V ∧
      (∀ j, degreeOf j U = degreeOf j U') ∧ V.natDegree = V'.natDegree := by
  fail_if_success exact hF
  fail_if_success simpa only using hF
  obtain ⟨g, U, hg, hUmap, hgcd⟩ := Primitive172FullGeneric.primitive (R := R) hU
  have hF' := rescale_separated_factor_aux hg hUmap hF
  obtain ⟨V, hVmap⟩ := exists_int_polynomial_of_primitive_factor_aux hgcd hF'
  refine ⟨U, V, ?_, ?_, ?_, fun j ↦ ?_, ?_⟩
  · exact fun h ↦ smul_ne_zero hg hU (by simpa [h] using hUmap.symm)
  · exact fun h ↦ mul_ne_zero (Polynomial.C_ne_zero.2 (inv_ne_zero hg)) hV
      (by simpa [h] using hVmap.symm)
  · apply MvPolynomial.map_injective (algebraMap R K) (IsFractionRing.injective R K)
    rw [hF', map_mul (map (algebraMap R K)), map_rename,
      Polynomial.map_aeval_eq_aeval_map (φ := algebraMap R K)
        (ψ := MvPolynomial.map (algebraMap R K)) (by ext r; simp), map_X,
      hVmap]
  · rw [degreeOf, ← degrees_map_of_injective U (f := algebraMap R K) (IsFractionRing.injective R K),
      hUmap, ← degreeOf, smul_eq_C_mul, degreeOf_C_mul j g (mem_nonZeroDivisors_of_ne_zero hg)]
  · rw [← Polynomial.natDegree_map_eq_of_injective (f := algebraMap R K)
      (IsFractionRing.injective R K), hVmap, Polynomial.natDegree_C_mul (inv_ne_zero hg)]

#print axioms exists_int_factors
end Factors176GCD

theorem original {τ : Type*} {U' : MvPolynomial τ ℚ} {V' : Polynomial ℚ}
    (hU : U' ≠ 0) (hV : V' ≠ 0) {F : MvPolynomial (Option τ) ℤ}
    (hF : map (Int.castRingHom ℚ) F =
      rename some U' * Polynomial.aeval (X none : MvPolynomial (Option τ) ℚ) V') :
    ∃ (U : MvPolynomial τ ℤ) (V : Polynomial ℤ), U ≠ 0 ∧ V ≠ 0 ∧
      F = rename some U * Polynomial.aeval (X none : MvPolynomial (Option τ) ℤ) V ∧
      (∀ j, degreeOf j U = degreeOf j U') ∧ V.natDegree = V'.natDegree :=
  Factors176GCD.exists_int_factors (R := ℤ) (K := ℚ) hU hV hF
