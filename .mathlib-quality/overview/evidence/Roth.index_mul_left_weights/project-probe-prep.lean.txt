namespace IndexScale148
open MvPolynomial Roth
variable {σ : Type*} [Fintype σ]
theorem replacement {l : ℕ} (P : MvPolynomial σ ℝ) (α : σ → ℝ) (r : σ → ℕ) :
    index P α (fun j ↦ l * r j) = index P α r / l := by
  have himg : weight (fun j ↦ l * r j) '' ((taylor α P).support : Set (σ →₀ ℕ)) =
      (l : ℝ)⁻¹ • (weight r '' ((taylor α P).support : Set (σ →₀ ℕ))) := by
    rw [← Set.image_smul, Set.image_image]
    refine Set.image_congr fun d _ ↦ ?_
    rw [weight_mul_left, smul_eq_mul, div_eq_inv_mul]
  rw [index, index, himg, Real.sInf_smul_of_nonneg (by positivity), smul_eq_mul, div_eq_inv_mul]
theorem zeroScale (P : MvPolynomial σ ℝ) (α : σ → ℝ) (r : σ → ℕ) :
    index P α (fun j ↦ 0 * r j) = 0 := by
  rw [replacement, Nat.cast_zero, div_zero]
theorem zeroPolynomial (α : σ → ℝ) (r : σ → ℕ) (l : ℕ) :
    index (0 : MvPolynomial σ ℝ) α (fun j ↦ l * r j) = 0 := by
  simp [index]
theorem noAutomaticUnfold {l : ℕ} (P : MvPolynomial σ ℝ) (α : σ → ℝ) (r : σ → ℕ) :
    index P α (fun j ↦ l * r j) = index P α r / l := by
  fail_if_success rfl
  exact replacement P α r
