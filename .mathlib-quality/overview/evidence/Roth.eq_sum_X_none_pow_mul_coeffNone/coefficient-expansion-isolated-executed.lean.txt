namespace Expansion156
open MvPolynomial
variable {τ R : Type*} [CommSemiring R]
theorem canonical (P : MvPolynomial (Option τ) R) {n : ℕ}
    (hn : degreeOf none P ≤ n) :
    optionEquivLeft R τ P = ∑ ν ∈ Finset.range (n + 1),
      Polynomial.C ((optionEquivLeft R τ P).coeff ν) * Polynomial.X ^ ν :=
  Polynomial.as_sum_range_C_mul_X_pow' _
    (Nat.lt_succ_of_le ((natDegree_optionEquivLeft R P).trans_le hn))
theorem inverse_C (Q : MvPolynomial τ R) :
    (optionEquivLeft R τ).symm (Polynomial.C Q) = rename some Q := by
  change Polynomial.aevalTower (rename some) (X none) (Polynomial.C Q) = _
  simp
theorem expanded (P : MvPolynomial (Option τ) R) {n : ℕ}
    (hn : degreeOf none P ≤ n) :
    P = ∑ ν ∈ Finset.range (n + 1),
      X none ^ ν * rename some ((optionEquivLeft R τ P).coeff ν) := by
  have h := congrArg (optionEquivLeft R τ).symm (canonical P hn)
  simpa only [AlgEquiv.symm_apply_apply, map_sum, map_mul, map_pow,
    optionEquivLeft_symm_X, inverse_C, mul_comm] using h

end Expansion156
