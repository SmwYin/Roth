-- Exact source proof with scalar mechanically generalized and only natural floor needed.
theorem genericFloor {K : Type*} [Field K] [LinearOrder K] [IsStrictOrderedRing K]
    [FloorSemiring K] {a : K} (ha : 0 ≤ a) :
    (⌊a⌋₊ : K) ^ 2 / 2 ≤ ∑ ν ∈ Finset.range (⌊a⌋₊ + 1), (a - ν) := by
  have hsum := congrArg (Nat.cast (R := K)) (Finset.sum_range_id_mul_two (⌊a⌋₊ + 1))
  simp only [Nat.add_sub_cancel] at hsum
  push_cast at hsum
  rw [Finset.sum_sub_distrib, Finset.sum_const, Finset.card_range, nsmul_eq_mul]
  push_cast
  nlinarith [Nat.floor_le ha]
-- The floor interface is stronger than the algebra actually needs: arbitrary natural cutoff.
theorem cutoff {K : Type*} [Field K] [LinearOrder K] [IsStrictOrderedRing K]
    {a : K} (n : ℕ) (hn : (n : K) ≤ a) :
    (n : K) ^ 2 / 2 ≤ ∑ ν ∈ Finset.range (n + 1), (a - ν) := by
  have hsum := congrArg (Nat.cast (R := K)) (Finset.sum_range_id_mul_two (n + 1))
  simp only [Nat.add_sub_cancel] at hsum
  push_cast at hsum
  rw [Finset.sum_sub_distrib, Finset.sum_const, Finset.card_range, nsmul_eq_mul]
  push_cast
  nlinarith
-- A stronger polynomial inequality removes division and all floor structure.
theorem ringCutoff {K : Type*} [CommRing K] [PartialOrder K] [IsOrderedRing K]
    {a : K} (n : ℕ) (hn : (n : K) ≤ a) :
    (n : K) ^ 2 + n ≤ 2 * ∑ ν ∈ Finset.range (n + 1), (a - ν) := by
  have hsum := congrArg (Nat.cast (R := K)) (Finset.sum_range_id_mul_two (n + 1))
  simp only [Nat.add_sub_cancel] at hsum
  push_cast at hsum
  rw [Finset.sum_sub_distrib, Finset.sum_const, Finset.card_range, nsmul_eq_mul]
  push_cast
  have hmul := mul_nonneg
    (add_nonneg (Nat.cast_nonneg n) zero_le_one) (sub_nonneg.mpr hn)
  calc
    (n : K) ^ 2 + n ≤ n ^ 2 + n + ((n + 1) * (a - n) + (n + 1) * (a - n)) :=
      le_add_of_nonneg_right (add_nonneg hmul hmul)
    _ = 2 * ((n + 1) * a - ∑ ν ∈ Finset.range (n + 1), (ν : K)) := by
      linear_combination -hsum
-- Closed arithmetic progression identity: no order or floor required.
theorem exactSum {K : Type*} [Field K] [CharZero K] (a : K) (n : ℕ) :
    (∑ ν ∈ Finset.range (n + 1), (a - ν)) = (n + 1) * (a - n / 2) := by
  have hsum := congrArg (Nat.cast (R := K)) (Finset.sum_range_id_mul_two (n + 1))
  simp only [Nat.add_sub_cancel] at hsum
  push_cast at hsum
  rw [Finset.sum_sub_distrib, Finset.sum_const, Finset.card_range, nsmul_eq_mul]
  push_cast
  linear_combination -hsum / 2
-- The actual ha is necessary for natural-floor convention at negative a.
theorem nonnegNecessary :
    ¬ (∀ a : ℝ, (⌊a⌋₊ : ℝ) ^ 2 / 2 ≤
      ∑ ν ∈ Finset.range (⌊a⌋₊ + 1), (a - ν)) := by
  intro h
  have h' := h (-1)
  have hf : ⌊(-1 : ℝ)⌋₊ = 0 := Nat.floor_eq_zero.mpr (by norm_num)
  norm_num [hf] at h'
-- Compiled bounded closure probes, never a global absence claim.
theorem closureGuards {a : ℝ} (ha : 0 ≤ a) :
    (⌊a⌋₊ : ℝ) ^ 2 / 2 ≤ ∑ ν ∈ Finset.range (⌊a⌋₊ + 1), (a - ν) := by
  fail_if_success rfl
  fail_if_success assumption
  fail_if_success
    solve | simp only [Finset.sum_sub_distrib, Finset.sum_const,
      Finset.card_range, nsmul_eq_mul]
  exact genericFloor ha
