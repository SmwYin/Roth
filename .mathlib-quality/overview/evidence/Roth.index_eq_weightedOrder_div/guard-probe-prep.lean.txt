namespace IndexWeightedGuards143
open MvPolynomial Roth
def mixedWeights (b : Bool) : ℕ := if b then 1 else 0
theorem mixedZeroIndex : index (X true) (fun _ ↦ 0) mixedWeights = 1 := by
  simp [index, taylor, mixedWeights, weight, Fintype.sum_bool]
theorem mixedZeroRhs :
    ((((taylor (fun _ ↦ 0) (X true : MvPolynomial Bool ℝ) : MvPowerSeries Bool ℝ).
      weightedOrder fun j ↦ (∏ i, mixedWeights i) / mixedWeights j).toNat : ℝ) /
      ∏ i, mixedWeights i) = 0 := by
  simp [mixedWeights, Fintype.prod_bool]
#print axioms mixedZeroIndex
#print axioms mixedZeroRhs
end IndexWeightedGuards143
