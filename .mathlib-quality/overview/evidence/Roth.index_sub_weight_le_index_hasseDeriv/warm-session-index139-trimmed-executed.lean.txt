noncomputable def genericIndex {σ : Type u} {R : Type v} [CommSemiring R]
    (P : MvPolynomial σ R) (α : σ → R) (w : σ → ℝ) : ℝ :=
  sInf ((fun d : σ →₀ ℕ ↦ d.sum fun j n ↦ (n : ℝ) * w j) ''
    ((aeval (fun j ↦ X j + C (α j)) P).support : Set (σ →₀ ℕ)))
theorem directUnfold {σ : Type u} [Fintype σ] (P : MvPolynomial σ ℝ) (α : σ → ℝ)
    (r : σ → ℕ) : Roth.index P α r =
      sInf (Roth.weight r '' ((taylor α P).support : Set (σ →₀ ℕ))) := rfl
theorem genericSpecialization {σ : Type u} [Fintype σ] (P : MvPolynomial σ ℝ) (α : σ → ℝ)
    (r : σ → ℕ) : genericIndex P α (fun j ↦ (r j : ℝ)⁻¹) = Roth.index P α r := by
  unfold genericIndex Roth.index Roth.weight taylor
  congr 2
  funext d
  rw [Finsupp.sum_fintype _ _ (by simp)]
  simp only [div_eq_mul_inv]
theorem zero {σ : Type u} [Fintype σ] (α : σ → ℝ) (r : σ → ℕ) :
    Roth.index 0 α r = 0 := by simp [Roth.index]
theorem zeroWeights {σ : Type u} [Fintype σ] (P : MvPolynomial σ ℝ) (α : σ → ℝ) :
    Roth.index P α (fun _ ↦ 0) = 0 := by
  by_cases hP : P = 0
  · subst P; simp [Roth.index]
  · have hne : ((taylor α P).support : Set (σ →₀ ℕ)).Nonempty := by simpa using hP
    have hw : Roth.weight (fun _ : σ ↦ 0) = fun _ ↦ (0 : ℝ) := by
      funext d
      simp only [Roth.weight, Nat.cast_zero, div_zero, Finset.sum_const_zero]
    rw [Roth.index, hw, hne.image_const]
    simp
theorem beta {σ : Type u} [Fintype σ] (P : MvPolynomial σ ℝ) (α : σ → ℝ)
    (r : σ → ℕ) : (fun Q ↦ Roth.index Q α r) P = Roth.index P α r := rfl

theorem universeProbe (α : ULift.{1} (Fin 2) → ℝ) (r : ULift.{1} (Fin 2) → ℕ) :
    Roth.index (0 : MvPolynomial (ULift.{1} (Fin 2)) ℝ) α r = 0 := by

theorem fintypeIndependence {σ : Type u} (i j : Fintype σ) (P : MvPolynomial σ ℝ)
    (α : σ → ℝ) (r : σ → ℕ) :
    @Roth.index σ i P α r = @Roth.index σ j P α r := by
  have : i = j := Subsingleton.elim _ _
  subst j
  rfl
theorem noAutomaticUnfold {σ : Type u} [Fintype σ] (P : MvPolynomial σ ℝ) (α : σ → ℝ)
    (r : σ → ℕ) : Roth.index P α r =
      sInf (Roth.weight r '' ((taylor α P).support : Set (σ →₀ ℕ))) := by
  fail_if_success simp only
  rfl

theorem positiveWeightedOrder {σ : Type u} [Fintype σ] (P : MvPolynomial σ ℝ) (α : σ → ℝ)
    (r : σ → ℕ) (hr : ∀ j, 0 < r j) : Roth.index P α r =
    (((taylor α P : MvPowerSeries σ ℝ).weightedOrder
      (fun j ↦ (∏ i, r i) / r j)).toNat : ℝ) / ∏ i, r i := by
  by_cases hP : P = 0
  · subst P; simp [Roth.index]
  · exact Roth.index_eq_weightedOrder_div hP hr α

theorem finsetBridge {ι : Type u} (s : Finset ι) (w : ι → ℝ) :
    (s.inf (fun d ↦ (w d : WithTop ℝ))).untopD 0 =
      sInf (w '' (s : Set ι)) := by
  classical
  rcases s.eq_empty_or_nonempty with h | h
  · subst s
    simp
  · change (s.inf (WithTop.some ∘ w)).untopD 0 = _
    rw [← Finset.coe_inf' h w, Finset.inf'_eq_csInf_image]
    rfl

theorem targetBridge {σ : Type u} [Fintype σ] (P : MvPolynomial σ ℝ)
    (α : σ → ℝ) (r : σ → ℕ) : Roth.index P α r =
    (AddMonoidAlgebra.infDegree (fun d ↦ (Roth.weight r d : WithTop ℝ))
      (taylor α P)).untopD 0 := by
  exact (finsetBridge (taylor α P).support (Roth.weight r)).symm

theorem indexSetBridge {σ : Type u} [Fintype σ] (B : ℝ) (q r : σ → ℕ) :
    Roth.indexSet B q r =
      {x | ∃ P ∈ Roth.RSet B r, ∃ h : σ → ℤ, (∀ j, IsCoprime (h j) (q j : ℤ)) ∧
        (AddMonoidAlgebra.infDegree (fun d ↦ (Roth.weight r d : WithTop ℝ))
          (taylor (fun j ↦ (h j : ℝ) / q j) (map (Int.castRingHom ℝ) P))).untopD 0 = x} := by
  unfold Roth.indexSet
  simp_rw [targetBridge]

theorem abstractLaw {A : Type u} {R : Type v} {T : Type w} [Semiring R]
    [LinearOrder T] [BoundedOrder T] [Nontrivial T] (D : A → T)
    (P : AddMonoidAlgebra R A) (a₀ : A) (hD : ∀ a, D a = ⊥ ↔ a = a₀) :
    AddMonoidAlgebra.infDegree D P = ⊥ ↔ P.coeff a₀ ≠ 0 := by
  simpa only [AddMonoidAlgebra.infDegree, hD, exists_eq_right, Finsupp.mem_support_iff]
    using (Finset.inf_eq_bot_iff (s := P.coeff.support) (f := D))
theorem projectedLaw {A : Type u} {R : Type v} [Semiring R]
    (D : A → ℝ) (P : AddMonoidAlgebra R A) (a₀ : A)
    (hs : P.coeff.support.Nonempty) (hD : ∀ a, 0 ≤ D a)
    (hz : ∀ a, D a = 0 ↔ a = a₀) :
    (AddMonoidAlgebra.infDegree (fun a ↦ (D a : WithTop ℝ)) P).untopD 0 = 0 ↔
      P.coeff a₀ ≠ 0 := by
  classical
  change (P.coeff.support.inf (WithTop.some ∘ D)).untopD 0 = 0 ↔ _
  rw [← Finset.coe_inf' hs D, WithTop.untopD_coe]
  have hn : 0 ≤ P.coeff.support.inf' hs D :=
    (Finset.le_inf'_iff hs D).mpr fun a _ ↦ hD a
  rw [show P.coeff.support.inf' hs D = 0 ↔ P.coeff.support.inf' hs D ≤ 0 from
    ⟨fun h ↦ h.le, fun h ↦ h.antisymm hn⟩]
  simpa only [hz, exists_eq_right, Finsupp.mem_support_iff] using
    (show P.coeff.support.inf' hs D ≤ 0 ↔ ∃ a ∈ P.coeff.support, D a = 0 from by
      rw [Finset.inf'_le_iff]
      exact exists_congr fun a ↦ and_congr_right fun _ ↦
        ⟨fun h ↦ h.antisymm (hD a), fun h ↦ h.le⟩)

theorem nonnegativeBridge {A : Type u} {R : Type v} [Semiring R]
    (D : A → ℝ) (P : AddMonoidAlgebra R A) (hD : ∀ a, 0 ≤ D a) :
    (AddMonoidAlgebra.infDegree (fun a ↦ (D a : WithTop ℝ)) P).untopD 0 =
      (((AddMonoidAlgebra.infDegree
        (fun a ↦ WithTop.some (NNReal.mk (D a) (hD a))) P).untopD 0 : NNReal) : ℝ) := by
  classical
  rcases P.coeff.support.eq_empty_or_nonempty with hs | hs
  · simp only [AddMonoidAlgebra.infDegree, hs, Finset.inf_empty, WithTop.untopD_top,
      NNReal.coe_zero]
  · change (P.coeff.support.inf (WithTop.some ∘ D)).untopD 0 =
      (((P.coeff.support.inf
        (WithTop.some ∘ (fun a ↦ NNReal.mk (D a) (hD a)))).untopD 0 : NNReal) : ℝ)
    rw [← Finset.coe_inf' hs D,
      ← Finset.coe_inf' hs (fun a ↦ NNReal.mk (D a) (hD a)), WithTop.untopD_coe,
      WithTop.untopD_coe]
    exact (Finset.apply_inf'_eq_inf'_comp (f := fun a ↦ NNReal.mk (D a) (hD a)) hs
      (fun x : NNReal ↦ (x : ℝ))
      (fun _ _ ↦ rfl)).symm

theorem nonnegativeZeroBridge {A : Type u} {R : Type v} [Semiring R]
    (D : A → ℝ) (P : AddMonoidAlgebra R A) (hD : ∀ a, 0 ≤ D a)
    (hs : P.coeff.support.Nonempty) :
    (AddMonoidAlgebra.infDegree (fun a ↦ (D a : WithTop ℝ)) P).untopD 0 = 0 ↔
      AddMonoidAlgebra.infDegree
        (fun a ↦ WithTop.some (NNReal.mk (D a) (hD a))) P = ⊥ := by
  rw [nonnegativeBridge D P hD]
  change (((P.coeff.support.inf
    (WithTop.some ∘ (fun a ↦ NNReal.mk (D a) (hD a)))).untopD 0 : NNReal) : ℝ) = 0 ↔ _
  rw [← Finset.coe_inf' hs (fun a ↦ NNReal.mk (D a) (hD a)), WithTop.untopD_coe]
  change _ ↔ P.coeff.support.inf (WithTop.some ∘ (fun a ↦ NNReal.mk (D a) (hD a))) = 0
  rw [← Finset.coe_inf' hs (fun a ↦ NNReal.mk (D a) (hD a))]
  simp only [WithTop.coe_eq_zero, NNReal.coe_eq_zero]
