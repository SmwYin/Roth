open MvPolynomial
noncomputable def minimum {σ R : Type*} [CommSemiring R]
    (D : (σ →₀ ℕ) → ℝ) (P : MvPolynomial σ R) : ℝ :=
  sInf (D '' (P.support : Set (σ →₀ ℕ)))
theorem supportShift {σ R : Type*} [CommSemiring R]
    (D : (σ →₀ ℕ) → ℝ) (P Q : MvPolynomial σ R) (k : σ →₀ ℕ)
    (hQ : Q ≠ 0) (hs : ∀ i, Q.coeff i ≠ 0 → P.coeff (i + k) ≠ 0)
    (hD : ∀ i, D (i + k) = D i + D k) :
    minimum D P - D k ≤ minimum D Q := by
  classical
  have hne : (D '' (Q.support : Set (σ →₀ ℕ))).Nonempty := by
    simpa using hQ
  obtain ⟨i, hi, himin⟩ := hne.csInf_mem (Q.support.finite_toSet.image D)
  have hp : minimum D P ≤ D (i + k) :=
    csInf_le (P.support.finite_toSet.image D).bddBelow
      (Set.mem_image_of_mem D (MvPolynomial.mem_support_iff.2 (hs i
        (MvPolynomial.mem_support_iff.1 hi))))
  exact sub_le_iff_le_add.2 (by simpa [hD, himin, minimum, add_comm] using hp)
