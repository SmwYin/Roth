-- This is ONLY generic logarithmic/order packaging with the exponential premise supplied.
-- It cannot establish that premise for integer polynomial root multiplicity.
theorem logarithmicOfPow {q B : ℝ} (hq : 1 < q) {t r : ℕ} (hpow : q ^ t ≤ B) :
    (t : ℝ) / r ≤ Real.log B / (r * Real.log q) := by
  have hlog := Real.log_le_log (pow_pos (zero_lt_one.trans hq) t) hpow
  rw [Real.log_pow] at hlog
  simpa [div_div, mul_comm] using
    div_le_div_of_nonneg_right
      ((le_div_iff₀ (Real.log_pos hq)).2 hlog) (Nat.cast_nonneg r)
theorem supremumOfLog {q B : ℝ} (hq : 1 < q) {r : ℕ} {s : Set ℝ}
    (hs : s.Nonempty) (hpoint : ∀ x ∈ s, ∃ t : ℕ, x = (t : ℝ) / r ∧ q ^ t ≤ B) :
    sSup s ≤ Real.log B / (r * Real.log q) := by
  apply csSup_le hs
  rintro x hx
  obtain ⟨t, rfl, ht⟩ := hpoint x hx
  exact logarithmicOfPow hq ht
