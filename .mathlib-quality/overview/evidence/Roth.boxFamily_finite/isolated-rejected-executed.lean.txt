def family {σ R : Type*} [CommSemiring R] (r : σ → ℕ)
    (t : (σ →₀ ℕ) → Set R) : Set (MvPolynomial σ R) :=
  {P | (∀ j, degreeOf j P ≤ r j) ∧ ∀ d, P.coeff d ∈ t d}

theorem finite_coeff_choices {σ R : Type*} [CommSemiring R] (s : Set (σ →₀ ℕ))
    (hs : s.Finite) (t : (σ →₀ ℕ) → Set R) (ht : ∀ d ∈ s, (t d).Finite) :
    {P : MvPolynomial σ R | (↑P.support : Set (σ →₀ ℕ)) ⊆ s ∧
      ∀ d, P.coeff d ∈ t d}.Finite := by
  classical
  let : Fintype s := hs.fintype
  let f : MvPolynomial σ R → s → R := fun P d ↦ P.coeff d
  have hfin : {g : s → R | ∀ d, g d ∈ t d}.Finite :=
    Set.Finite.pi' (fun d ↦ ht d d.property)
  refine hfin.of_injOn (f := f) (fun P hP d ↦ hP.2 d) ?_
  intro P hP Q hQ he
  apply MvPolynomial.ext
  intro d
  by_cases hd : d ∈ s
  · exact congrFun he ⟨d, hd⟩
  · have hp : P.coeff d = 0 :=
      Finsupp.notMem_support_iff.1 (fun hp ↦ hd (hP.1 hp))
    have hq : Q.coeff d = 0 :=
      Finsupp.notMem_support_iff.1 (fun hq ↦ hd (hQ.1 hq))
    exact hp.trans hq.symm

theorem finite_family_of_monomials {σ R : Type*} [CommSemiring R] (r : σ → ℕ)
    (hs : {d : σ →₀ ℕ | ∀ j, d j ≤ r j}.Finite)
    (t : (σ →₀ ℕ) → Set R) (ht : ∀ d, (t d).Finite) : (family r t).Finite := by
  refine (finite_coeff_choices _ hs t (fun d _ ↦ ht d)).subset ?_
  exact fun P hP ↦ ⟨fun d hd j ↦ (monomial_le_degreeOf j hd).trans (hP.1 j), hP.2⟩

theorem zero_bound {σ : Type*} (r : σ → ℕ) :
    family r (fun _ ↦ Set.Icc (0 : ℤ) 0) = {0} := by
  ext P
  constructor
  · intro h
    apply Set.mem_singleton_iff.2
    apply MvPolynomial.ext
    intro d
    exact le_antisymm (h.2 d).2 (h.2 d).1
  · rintro rfl
    simp [family]

theorem infinite_family : (family (fun _ : ℕ ↦ 1) (fun _ ↦ Set.Icc (0 : ℤ) 1)).Infinite := by
  have hi : Function.Injective (fun i : ℕ ↦ (X i : MvPolynomial ℕ ℤ)) :=
    fun _ _ h ↦ MvPolynomial.X_injective h
  refine (Set.infinite_range_of_injective hi).mono (Set.range_subset_iff.2 ?_)
  intro i
  constructor
  · intro j
    by_cases h : i = j
    · subst j
      simp
    · simp [degreeOf_X_of_ne (Ne.symm h)]
  · intro d
    simp only [MvPolynomial.coeff_X, Set.mem_Icc]
    split_ifs <;> norm_num

theorem abstract_image {α β : Type*} (s : Finset α) (f : α → β) : (f '' (s : Set α)).Finite :=
  s.finite_toSet.image f
