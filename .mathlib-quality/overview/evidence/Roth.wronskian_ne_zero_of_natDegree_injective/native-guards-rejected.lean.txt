namespace DistinctDegreeGuards
open Polynomial
abbrev family : Fin 2 → (ℤ × ℤ)[X] := ![monomial 0 (1, 0), monomial 1 (0, 1)]
theorem domain_needed :
    (∀ ν, family ν ≠ 0) ∧
    Function.Injective (fun ν ↦ ((family ν).natDegree : ℤ × ℤ)) ∧
    (Matrix.of fun μ ν : Fin 2 ↦ hasseDeriv (μ : ℕ) (family ν)).det = 0 := by
  constructor
  · intro ν
    fin_cases ν <;> simp [family]
  constructor
  · intro i j hij
    fin_cases i <;> fin_cases j <;> norm_num [family] at hij ⊢
  · norm_num [family, Matrix.det_fin_two, hasseDeriv_monomial]
#print axioms domain_needed
end DistinctDegreeGuards
