namespace PrivateCoeffProductAssessment
open Polynomial
theorem replay {R : Type*} [CommSemiring R] {ι : Type*}
    (s : Finset ι) (f : ι → R[X]) (n : ι → ℕ)
    (h : ∀ i ∈ s, (f i).natDegree ≤ n i) :
    (∏ i ∈ s, f i).coeff (∑ i ∈ s, n i) = ∏ i ∈ s, (f i).coeff (n i) := by
  classical
  induction s using Finset.induction_on with
  | empty => simp
  | insert a s has ih =>
    simp only [Finset.forall_mem_insert] at h
    rw [Finset.prod_insert has, Finset.sum_insert has, Finset.prod_insert has,
      coeff_mul_add_eq_of_natDegree_le h.1 ((natDegree_prod_le _ _).trans (Finset.sum_le_sum h.2)),
      ih h.2]
#check @replay
#print axioms replay
#check @Polynomial.coeff_prod_of_natDegree_le
end PrivateCoeffProductAssessment
namespace CoeffWronskianFiniteProbe
open Polynomial
noncomputable def W {K : Type*} [CommRing K] {ι : Type*} [Fintype ι] [DecidableEq ι]
    (r : ι → ℕ) (φ : ι → K[X]) : K[X] :=
  (Matrix.of fun μ ν : ι ↦ hasseDeriv (r μ) (φ ν)).det
theorem replay {K : Type*} [CommRing K] {ι : Type*} [Fintype ι] [DecidableEq ι]
    (r : ι → ℕ) {φ : ι → K[X]} :
    (W r φ).coeff (∑ ν, (φ ν).natDegree - ∑ ν : ι, r ν) =
      (∏ ν, (φ ν).leadingCoeff) *
        (Matrix.of fun μ ν : ι ↦ ((φ ν).natDegree.choose (r μ) : K)).det := by
  classical
  have key (σ : Equiv.Perm (ι)) : (∏ i, hasseDeriv (r (σ i)) (φ i)).coeff
      (∑ ν, (φ ν).natDegree - ∑ ν : ι, r ν) =
      ∏ i, ((φ i).natDegree.choose (r (σ i)) : K) * (φ i).leadingCoeff := by
    by_cases h : ∀ i, r (σ i) ≤ (φ i).natDegree
    · rw [← Equiv.sum_comp σ r,
        ← Finset.sum_tsub_distrib _ fun i _ ↦ h i,
        PrivateCoeffProductAssessment.replay _ _ _ fun i _ ↦ natDegree_hasseDeriv_le _ _]
      refine Finset.prod_congr rfl fun i _ ↦ ?_
      rw [hasseDeriv_coeff, Nat.sub_add_cancel (h i), leadingCoeff]
    · push Not at h
      obtain ⟨i, hi⟩ := h
      rw [Finset.prod_eq_zero (Finset.mem_univ i) (hasseDeriv_eq_zero_of_lt_natDegree _ _ hi),
        coeff_zero]
      exact (Finset.prod_eq_zero (Finset.mem_univ i)
        (by simp [Nat.choose_eq_zero_of_lt hi])).symm
  rw [W, Matrix.det_apply', Matrix.det_apply', finsetSum_coeff,
    Finset.mul_sum]
  refine Finset.sum_congr rfl fun σ _ ↦ ?_
  simp [coeff_intCast_mul, key, Finset.prod_mul_distrib, mul_comm, mul_assoc]
#check @replay
#print axioms replay

theorem expanded {K : Type*} [CommRing K] {ι : Type*} [Fintype ι] [DecidableEq ι]
    (r : ι → ℕ) {φ : ι → K[X]} :
    (Matrix.of fun μ ν : ι ↦ hasseDeriv (r μ) (φ ν)).det.coeff
        (∑ ν, (φ ν).natDegree - ∑ ν, r ν) =
      (∏ ν, (φ ν).leadingCoeff) *
        (Matrix.of fun μ ν : ι ↦ ((φ ν).natDegree.choose (r μ) : K)).det := by
  simpa only [W] using (replay (φ := φ) r)
#print axioms expanded
end CoeffWronskianFiniteProbe

namespace DetChooseRingProbe
theorem factorial_mul_det {R : Type*} [CommRing R] {l : ℕ} (d : Fin l → ℕ) :
    (∏ μ : Fin l, ((μ : ℕ).factorial : R)) *
      (Matrix.of fun μ ν : Fin l ↦ ((d ν).choose μ : R)).det =
        ∏ i : Fin l, ∏ j ∈ Finset.Ioi i, ((d j : R) - d i) := by
  nontriviality R
  have hm (n : ℕ) : (descPochhammer R n).Monic := by
    simpa only [descPochhammer_map] using
      Polynomial.Monic.map (Int.castRingHom R) (monic_descPochhammer ℤ n)
  have hd (n : ℕ) : (descPochhammer R n).natDegree = n := by
    simpa only [descPochhammer_map] using
      ((monic_descPochhammer ℤ n).natDegree_map (Int.castRingHom R)).trans
        (descPochhammer_natDegree ℤ n)
  have h1 : (Matrix.of fun i j : Fin l ↦ (descPochhammer R j).eval (d i : R)) =
      (Matrix.of fun μ ν : Fin l ↦ ((μ : ℕ).factorial : R) * ((d ν).choose μ : R)).transpose := by
    ext i j
    simp only [Matrix.of_apply, Matrix.transpose_apply,
      descPochhammer_eval_eq_descFactorial, Nat.descFactorial_eq_factorial_mul_choose,
      Nat.cast_mul]
  rw [← Matrix.det_mul_column, ← Matrix.det_transpose]
  simp only [Matrix.of_apply]
  rw [← h1, ← Matrix.det_eval_matrixOfPolynomials_eq_det_vandermonde (fun i ↦ (d i : R))
      (fun j ↦ descPochhammer R j) (fun i ↦ hd i)
      (fun i ↦ hm i), Matrix.det_vandermonde]
#print axioms factorial_mul_det
end DetChooseRingProbe

#check_failure Roth.wronskian_ne_zero_of_natDegree_injective
namespace DistinctDegreeProbe
open Polynomial
theorem generalized {K : Type*} [CommRing K] [IsDomain K] {l : ℕ} {φ : Fin l → K[X]}
    (hφ : ∀ ν, φ ν ≠ 0)
    (hd : Function.Injective fun ν ↦ ((φ ν).natDegree : K)) :
    (Matrix.of fun μ ν : Fin l ↦ hasseDeriv (μ : ℕ) (φ ν)).det ≠ 0 := by
  have hdet : (Matrix.of fun μ ν : Fin l ↦ ((φ ν).natDegree.choose μ : K)).det ≠ 0 := by
    intro hzero
    have h := DetChooseRingProbe.factorial_mul_det (R := K) (fun ν ↦ (φ ν).natDegree)
    have hv := Matrix.det_vandermonde_ne_zero_iff.mpr hd
    rw [Matrix.det_vandermonde] at hv
    exact hv (by simpa only [hzero, mul_zero] using h.symm)
  intro hzero
  have h := CoeffWronskianFiniteProbe.expanded (φ := φ) (fun μ : Fin l ↦ (μ : ℕ))
  have hp : (∏ ν, (φ ν).leadingCoeff) ≠ 0 :=
    Finset.prod_ne_zero_iff.mpr fun ν _ ↦ leadingCoeff_ne_zero.mpr (hφ ν)
  exact (mul_ne_zero hp hdet) (by simpa only [hzero, coeff_zero] using h.symm)

theorem original_corollary {K : Type*} [Field K] [CharZero K] {l : ℕ}
    {φ : Fin l → K[X]} (hφ : ∀ ν, φ ν ≠ 0)
    (hd : Function.Injective fun ν ↦ (φ ν).natDegree) :
    (Matrix.of fun μ ν : Fin l ↦ hasseDeriv (μ : ℕ) (φ ν)).det ≠ 0 :=
  generalized hφ (Nat.cast_injective.comp hd)

-- Concrete improvement: characteristic two with separated degree casts.
theorem positive_char :
    (Matrix.of fun μ ν : Fin 2 ↦ hasseDeriv (μ : ℕ)
      (![1, X] ν : (ZMod 2)[X])).det ≠ 0 := by
  apply generalized
  · intro ν
    fin_cases ν <;> simp
  · intro i j hij
    fin_cases i <;> fin_cases j <;> simp_all

-- Natural-degree separation alone does not suffice in characteristic two.
theorem char_two_guard :
    Function.Injective (fun ν : Fin 2 ↦ (![1, X^2] ν : (ZMod 2)[X]).natDegree) ∧
      (Matrix.of fun μ ν : Fin 2 ↦ hasseDeriv (μ : ℕ)
        (![1, X^2] ν : (ZMod 2)[X])).det = 0 := by
  constructor
  · intro i j hij
    fin_cases i <;> fin_cases j <;> simp_all
  · norm_num [Matrix.det_fin_two]
    exact (show (2 : ZMod 2) = 0 from by decide)
#print axioms generalized
#print axioms original_corollary
#print axioms positive_char
#print axioms char_two_guard
#check @Matrix.det_vandermonde_ne_zero_iff
end DistinctDegreeProbe
