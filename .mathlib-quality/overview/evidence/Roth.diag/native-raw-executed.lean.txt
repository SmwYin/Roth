namespace DiagonalRaw
open MvPolynomial
variable {σ R : Type*} [CommSemiring R]
noncomputable def replacement (P : MvPolynomial σ R) : Polynomial R :=
  aeval (fun _ ↦ Polynomial.X) P
theorem finiteExpansion (P : MvPolynomial σ R) :
    replacement P = ∑ k ∈ P.support, Polynomial.monomial k.degree (P.coeff k) := by
  rw [replacement, aeval_def, eval₂_eq]
  simp_rw [Finset.prod_pow_eq_pow_sum, ← Finsupp.degree_apply, Polynomial.algebraMap_eq,
    Polynomial.C_mul_X_pow_eq_monomial]
theorem variableCollapse :
    (uniqueAlgEquiv R PUnit).toAlgHom.comp (rename (fun _ : σ ↦ PUnit.unit)) =
      aeval (fun _ : σ ↦ (Polynomial.X : Polynomial R)) := by
  apply MvPolynomial.algHom_ext
  intro i
  change eval₂ Polynomial.C (fun _ ↦ Polynomial.X) (rename (fun _ : σ ↦ PUnit.unit) (X i)) =
    (aeval (fun _ : σ ↦ (Polynomial.X : Polynomial R))) (X i)
  simp only [rename_X, eval₂_X, aeval_X]
theorem evaluation {S : Type*} [CommSemiring S] [Algebra R S]
    (P : MvPolynomial σ R) (x : S) :
    Polynomial.aeval x (replacement P) = aeval (fun _ ↦ x) P := by
  simpa only [replacement, Polynomial.aeval_X] using
    (comp_aeval_apply (f := fun _ : σ ↦ (Polynomial.X : Polynomial R)) (Polynomial.aeval x) P)
