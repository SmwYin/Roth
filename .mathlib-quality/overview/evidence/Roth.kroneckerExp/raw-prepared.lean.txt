import Mathlib.Data.Finsupp.Weight
import Mathlib.Data.Fintype.EquivFin
#check_failure Roth.kroneckerExp
namespace KroneckerExpRawAssessment
example {σ : Type*} [Fintype σ] (k : ℕ) (d : σ →₀ ℕ) :
    (∑ j, d j * k ^ (Fintype.equivFin σ j : ℕ)) =
      Finsupp.weight (fun j ↦ k ^ (Fintype.equivFin σ j : ℕ)) d := by
  simpa only [smul_eq_mul] using
    (Finsupp.weight_eq_sum (fun j ↦ k ^ (Fintype.equivFin σ j : ℕ)) d).symm
example {σ M R : Type*} [Semiring R] [AddCommMonoid M] [Module R M]
    (w : σ → M) (d : σ →₀ R) : Finsupp.weight w d = Finsupp.linearCombination R w d := rfl
end KroneckerExpRawAssessment
