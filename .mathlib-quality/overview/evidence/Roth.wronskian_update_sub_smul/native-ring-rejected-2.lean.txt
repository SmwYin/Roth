open Polynomial
theorem roth48_ring {R : Type*} [CommRing R] {l : ℕ}
    {φ : Fin l → R[X]} {ν ν' : Fin l} (h : ν ≠ ν') (c : R) :
    (Matrix.of fun μ ρ : Fin l ↦ hasseDeriv μ
      ((Function.update φ ν' (φ ν' - C c * φ ν)) ρ)).det =
    (Matrix.of fun μ ρ : Fin l ↦ hasseDeriv μ (φ ρ)).det := by
  refine (congrArg Matrix.det ?_).trans
    (Matrix.det_updateCol_add_smul_self
      (Matrix.of fun μ ρ : Fin l ↦ hasseDeriv μ (φ ρ)) h.symm (-C c))
  ext μ ρ
  simp [Matrix.updateCol_apply, Function.update_apply, apply_ite,
    ← smul_eq_C_mul, sub_eq_add_neg]
#print axioms roth48_ring
