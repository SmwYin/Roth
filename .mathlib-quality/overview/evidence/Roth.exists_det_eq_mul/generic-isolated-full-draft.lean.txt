variable {σ R : Type*} [CommSemiring R]

/-- The binomial factor `∏ j, (k j).choose (d j)` (a product over the support of `d`). -/
def hasseCoeff (d k : σ →₀ ℕ) : ℕ := d.prod fun j ↦ (k j).choose

/-- The multivariate Hasse derivative `Δ_d`: the `R`-linear map sending `monomial k a` to
`monomial (k - d) (hasseCoeff d k * a)` (see `hasseDeriv_monomial`). The coefficient
`hasseCoeff d k` vanishes unless `d ≤ k`, so the truncated subtraction `k - d` is harmless.

Source: §1, definition of `Δ_i` on monomials. -/
noncomputable def hasseDeriv (d : σ →₀ ℕ) : MvPolynomial σ R →ₗ[R] MvPolynomial σ R :=
  Finsupp.lsum R (fun k : σ →₀ ℕ ↦
      (monomial (k - d)).comp (LinearMap.lsmul R R (hasseCoeff d k : R))) ∘ₗ
    (AddMonoidAlgebra.coeffLinearEquiv R).toLinearMap

/-- `Δ_d (a X^k) = hasseCoeff d k * a X^(k - d)`.

Source: §1, definition of `Δ_i` on monomials. -/
@[simp]
theorem hasseDeriv_monomial (d k : σ →₀ ℕ) (a : R) :
    hasseDeriv d (monomial k a) = monomial (k - d) ((hasseCoeff d k : R) * a) := by
  simp [hasseDeriv]

/-- Expansion of `Δ_d P` as a sum over the support of `P`.

Source: §1, definition of `Δ_i` on monomials. -/
theorem hasseDeriv_apply (d : σ →₀ ℕ) (P : MvPolynomial σ R) :
    hasseDeriv d P = ∑ k ∈ P.support, monomial (k - d) ((hasseCoeff d k : R) * P.coeff k) := by
  simp [hasseDeriv, sum_def]

/-- The binomial factor `hasseCoeff d k` vanishes exactly when `d ≰ k`.

Source: §1, "`0` otherwise" case of the definition of `Δ_i`. -/
theorem hasseCoeff_eq_zero_iff (d k : σ →₀ ℕ) : hasseCoeff d k = 0 ↔ ¬d ≤ k := by
  simp [hasseCoeff, Finsupp.prod, Finset.prod_eq_zero_iff, Nat.choose_eq_zero_iff, Finsupp.le_iff,
    not_forall]

/-- The coefficient identity `[X^e] Δ_d P = hasseCoeff d (e + d) * [X^(e + d)] P`. This is the
main computational interface to `hasseDeriv`; most identities below are proved by `ext` followed
by rewriting with it.

Source: §1, Lean implementation note. -/
theorem coeff_hasseDeriv (d e : σ →₀ ℕ) (P : MvPolynomial σ R) :
    (hasseDeriv d P).coeff e = (hasseCoeff d (e + d) : R) * P.coeff (e + d) := by
  classical
  induction P using MvPolynomial.induction_on' with
  | monomial k a =>
    rw [hasseDeriv_monomial, coeff_monomial, coeff_monomial]
    by_cases hk : k = e + d
    · simp [hk]
    by_cases hdk : d ≤ k
    · simp [hk, tsub_eq_iff_eq_add_of_le hdk]
    · simp [hk, (hasseCoeff_eq_zero_iff d k).2 hdk]
  | add p q hp hq => simp [hp, hq, mul_add]

/-- Over a finite set of variables, `hasseCoeff d k` is the full product
`∏ j, (k j).choose (d j)`.

Source: §1, Lean implementation note. -/
theorem hasseCoeff_eq_prod [Fintype σ] (d k : σ →₀ ℕ) : hasseCoeff d k = ∏ j, (k j).choose (d j) :=
  d.prod_fintype _ fun _ ↦ Nat.choose_zero_right _

/-- `Δ_0` is the identity, pointwise version of `hasseDeriv_zero`.

Source: §1. -/
theorem hasseDeriv_zero_apply (P : MvPolynomial σ R) : hasseDeriv 0 P = P := by
  ext e
  simp [coeff_hasseDeriv, hasseCoeff]

/-- `Δ_0` is the identity, since all binomial coefficients `(k j).choose 0` equal `1`.

Source: §1. -/
@[simp]
theorem hasseDeriv_zero : hasseDeriv (0 : σ →₀ ℕ) = (LinearMap.id : MvPolynomial σ R →ₗ[R] _) :=
  LinearMap.ext hasseDeriv_zero_apply

/-- A Hasse derivative of nonzero order kills constants.

Source: §1. -/
theorem hasseDeriv_C {d : σ →₀ ℕ} (hd : d ≠ 0) (a : R) : hasseDeriv d (C a) = 0 := by
  rw [C_apply, hasseDeriv_monomial, (hasseCoeff_eq_zero_iff d 0).2 (by simpa using hd)]
  simp

/-- `hasseDeriv` commutes with change of coefficients along a ring hom; used to pass between
`ℤ`, `ℚ` and `ℝ`.

Source: §1, Lean implementation note. -/
theorem hasseDeriv_map {S : Type*} [CommSemiring S] (f : R →+* S) (d : σ →₀ ℕ)
    (P : MvPolynomial σ R) : hasseDeriv d (map f P) = map f (hasseDeriv d P) := by
  ext e
  simp [coeff_hasseDeriv, coeff_map]

/-- `hasseDeriv` commutes with renaming variables along an injective map, provided the
multi-index is transported along the same map.

Source: §4, Lean implementation note after Lemma 4-2. -/
theorem hasseDeriv_rename {τ : Type*} {f : σ → τ} (hf : Function.Injective f) (d : σ →₀ ℕ)
    (P : MvPolynomial σ R) :
    hasseDeriv (d.mapDomain f) (rename f P) = rename f (hasseDeriv d P) := by
  classical
  induction P using MvPolynomial.induction_on' with
  | monomial k a =>
    simp [rename_monomial, hasseCoeff, Finsupp.prod_mapDomain_index_inj hf,
      Finsupp.mapDomain_apply_of_injective hf, ← Finsupp.mapDomain_tsub hf]
  | add p q hp hq => simp [hp, hq]

private lemma hasseCoeff_mul_hasseCoeff (f d e : σ →₀ ℕ) :
    hasseCoeff d (f + d) * hasseCoeff e (f + d + e) =
      hasseCoeff d (d + e) * hasseCoeff (d + e) (f + (d + e)) := by
  classical
  simp only [hasseCoeff]
  have hd : d.support ⊆ d.support ∪ e.support := Finset.subset_union_left
  rw [Finsupp.prod_of_support_subset _ hd _ fun _ _ ↦ Nat.choose_zero_right _,
    Finsupp.prod_of_support_subset _ Finset.subset_union_right _
      fun _ _ ↦ Nat.choose_zero_right _,
    Finsupp.prod_of_support_subset _ hd _ fun _ _ ↦ Nat.choose_zero_right _,
    Finsupp.prod_of_support_subset _ Finsupp.support_add _ fun _ _ ↦ Nat.choose_zero_right _,
    ← Finset.prod_mul_distrib, ← Finset.prod_mul_distrib]
  exact Finset.prod_congr rfl fun i _ ↦ by
    simpa [← add_assoc, mul_comm, ← Nat.choose_symm_add (a := d i) (b := e i)] using
      (Nat.choose_mul (n := f i + d i + e i) (k := d i + e i) (s := e i)
        (Nat.le_add_left (e i) (d i))).symm

/-- The composition rule `Δ_d Δ_e P = hasseCoeff d (d + e) • Δ_{d + e} P`, where
`hasseCoeff d (d + e) = ∏ j, (d j + e j).choose (d j)`. See
`hasseDeriv_hasseDeriv_of_disjoint` for the case where the factor is `1`.

Source: §4, proof of Lemma `index-of-derivative`. -/
theorem hasseDeriv_hasseDeriv (d e : σ →₀ ℕ) (P : MvPolynomial σ R) :
    hasseDeriv d (hasseDeriv e P) = (hasseCoeff d (d + e) : R) • hasseDeriv (d + e) P := by
  ext f
  rw [coeff_smul, coeff_hasseDeriv, coeff_hasseDeriv, coeff_hasseDeriv, smul_eq_mul, ← mul_assoc,
    ← mul_assoc, ← Nat.cast_mul, ← Nat.cast_mul, hasseCoeff_mul_hasseCoeff, add_assoc]

/-- The composition rule `Δ_d Δ_e = Δ_{d + e}` when `d` and `e` have disjoint supports, so that
the binomial factor in `hasseDeriv_hasseDeriv` is `1`.

Source: §4, proof of Lemma `index-of-derivative`. -/
theorem hasseDeriv_hasseDeriv_of_disjoint {d e : σ →₀ ℕ} (h : Disjoint d.support e.support)
    (P : MvPolynomial σ R) : hasseDeriv d (hasseDeriv e P) = hasseDeriv (d + e) P := by
  rw [hasseDeriv_hasseDeriv, show hasseCoeff d (d + e) = 1 from
    Finset.prod_eq_one fun i hi ↦ by
      simp [Finsupp.notMem_support_iff.1 (Finset.disjoint_left.1 h hi)], Nat.cast_one, one_smul]

private lemma hasseCoeff_mul_succ (j : σ) (d e : σ →₀ ℕ) :
    hasseCoeff d (e + Finsupp.single j 1 + d) * (e j + 1) =
      (d j + 1) * hasseCoeff (d + Finsupp.single j 1) (e + Finsupp.single j 1 + d) := by
  simpa [hasseCoeff, add_comm, add_left_comm, add_assoc, mul_comm] using
    hasseCoeff_mul_hasseCoeff e (Finsupp.single j 1) d

/-- A partial derivative of a Hasse derivative: `∂_j Δ_d = (d j + 1) • Δ_{d + e_j}`.

Source: §2, proof of Lemma 1. -/
theorem pderiv_hasseDeriv (j : σ) (d : σ →₀ ℕ) (P : MvPolynomial σ R) :
    pderiv j (hasseDeriv d P) = ((d j + 1 : ℕ) : R) • hasseDeriv (d + Finsupp.single j 1) P := by
  ext e
  rw [coeff_pderiv, coeff_smul, coeff_hasseDeriv, coeff_hasseDeriv, smul_eq_mul,
    ← add_assoc e d, add_right_comm e d, mul_right_comm, ← Nat.cast_add_one, ← Nat.cast_mul,
    hasseCoeff_mul_succ, Nat.cast_mul, mul_assoc]

/-- Hasse derivatives do not increase the degree in any variable.

Source: §2, proof of Lemma 2. -/
theorem degreeOf_hasseDeriv_le (j : σ) (d : σ →₀ ℕ) (P : MvPolynomial σ R) :
    degreeOf j (hasseDeriv d P) ≤ degreeOf j P := by
  refine degreeOf_le_iff.2 fun m hm ↦ ?_
  rw [mem_support_iff, coeff_hasseDeriv] at hm
  exact (monomial_le_degreeOf j (mem_support_iff.2 (right_ne_zero_of_mul hm))).trans' (by simp)

/-- `Δ_d P = 0` as soon as `d j` exceeds the degree of `P` in some variable `j`.

Source: §6, proof of Lemma 9. -/
theorem hasseDeriv_eq_zero_of_degreeOf_lt {j : σ} {d : σ →₀ ℕ} {P : MvPolynomial σ R}
    (h : degreeOf j P < d j) : hasseDeriv d P = 0 := by
  ext e
  simp [coeff_hasseDeriv, notMem_support_iff.1
    (notMem_support_of_degreeOf_lt j (s := e + d) (h.trans_le (by simp)))]

private lemma hasseDeriv_monomial_mul {d a : σ →₀ ℕ} (ha : ∀ j ∈ d.support, a j = 0) (c : R)
    (g : MvPolynomial σ R) : hasseDeriv d (monomial a c * g) = monomial a c * hasseDeriv d g := by
  classical
  induction g using MvPolynomial.induction_on' with
  | monomial b c' =>
    have hs : a + b - d = a + (b - d) := by
      ext j
      by_cases hj : j ∈ d.support <;> simp_all
    rw [monomial_mul_monomial, hasseDeriv_monomial, hasseDeriv_monomial,
      show hasseCoeff d (a + b) = hasseCoeff d b from
        Finset.prod_congr rfl fun j hj ↦ by simp [ha j hj]]
    simp [monomial_mul_monomial, hs, mul_left_comm]
  | add p q hp hq => simp [mul_add, hp, hq]

/-- A factor `f` not involving any variable in the support of `d` passes through `Δ_d`:
`Δ_d (f * g) = f * Δ_d g`.

Source: §2, proof of Lemma 2. -/
theorem hasseDeriv_mul_of_degreeOf_eq_zero {d : σ →₀ ℕ} {f g : MvPolynomial σ R}
    (hf : ∀ j ∈ d.support, degreeOf j f = 0) : hasseDeriv d (f * g) = f * hasseDeriv d g := by
  rw [f.as_sum, Finset.sum_mul, Finset.sum_mul, map_sum]
  exact Finset.sum_congr rfl fun a haf ↦ hasseDeriv_monomial_mul
    (fun j hj ↦ Nat.eq_zero_of_le_zero (hf j hj ▸ monomial_le_degreeOf j haf)) _ _

/-- The operator `Δ_{n e_j}` restricted to polynomials in the single variable `X j` is the
univariate Hasse derivative `Polynomial.hasseDeriv n`.

Source: §1. -/
theorem hasseDeriv_single_aeval_X (j : σ) (n : ℕ) (f : Polynomial R) :
    hasseDeriv (Finsupp.single j n) (Polynomial.aeval (X j : MvPolynomial σ R) f) =
      Polynomial.aeval (X j : MvPolynomial σ R) (Polynomial.hasseDeriv n f) := by
  induction f using Polynomial.induction_on' with
  | add p q hp hq => simp [hp, hq]
  | monomial m a =>
    simp [-map_natCast, X_pow_eq_monomial, C_mul_monomial, hasseCoeff, Finsupp.single_tsub,
      mul_assoc]

/-- For a single variable, `hasseDeriv` corresponds to `Polynomial.hasseDeriv` under
`MvPolynomial.uniqueAlgEquiv`.

Source: §1. -/
theorem uniqueAlgEquiv_hasseDeriv [Unique σ] (n : ℕ) (P : MvPolynomial σ R) :
    uniqueAlgEquiv R σ (hasseDeriv (R := R) (Finsupp.single (default : σ) n) P) =
      Polynomial.hasseDeriv n (uniqueAlgEquiv R σ P) := by
  ext m
  rw [coeff_uniqueAlgEquiv, coeff_hasseDeriv, Polynomial.hasseDeriv_coeff, coeff_uniqueAlgEquiv]
  simp [hasseCoeff, Finsupp.single_add]

/-- The binomial factor of a Hasse derivative is at most `2 ^ (∑ j, r j)` on the box `k ≤ r`.

Source: §3, proof of Lemma 3. -/
theorem hasseCoeff_le_two_pow [Fintype σ] {d k : σ →₀ ℕ} {r : σ → ℕ} (hk : ∀ j, k j ≤ r j) :
    hasseCoeff d k ≤ 2 ^ (∑ j, r j) := by
  rw [hasseCoeff_eq_prod, ← Finset.prod_pow_eq_pow_sum]
  exact Finset.prod_le_prod fun j _ ↦
    (Nat.choose_le_two_pow _ _).trans (Nat.pow_le_pow_right two_pos (hk j))

/-- `n! • Δ_{n e_j} = ∂_j ^ n`; as for `Polynomial.factorial_smul_hasseDeriv`, this holds over
any commutative semiring. Over a `ℚ`-algebra it identifies `Δ_{n e_j}` with `(1 / n!) ∂_j ^ n`.

Source: §1. -/
theorem factorial_smul_hasseDeriv_single (j : σ) (n : ℕ) (P : MvPolynomial σ R) :
    (n.factorial : R) • hasseDeriv (Finsupp.single j n) P = (⇑(pderiv (R := R) j))^[n] P := by
  induction n with
  | zero => simp
  | succ n ih =>
    rw [Function.iterate_succ_apply', ← ih, Derivation.map_smul, pderiv_hasseDeriv, smul_smul,
      Finsupp.single_eq_same, ← Finsupp.single_add, Nat.factorial_succ, Nat.cast_mul, mul_comm]

variable {K : Type*} [Field K] {l : ℕ}

/-- The Wronskian matrix `(hasseDeriv μ (φ ν))_{μ, ν}`. Source: §2, definition of `W`. -/
noncomputable def wronskianMatrix (φ : Fin l → K[X]) : Matrix (Fin l) (Fin l) K[X] :=
  Matrix.of fun μ ν ↦ hasseDeriv μ (φ ν)

/-- The Wronskian `det (hasseDeriv μ (φ ν))_{μ, ν}`. Source: §2, definition of `W`. -/
noncomputable def wronskian (φ : Fin l → K[X]) : K[X] := (wronskianMatrix φ).det

/-- A linearly dependent family has vanishing Wronskian; this holds over any field.
Source: §2, first lemma, easy direction ("Linear dependence gives a nontrivial constant linear
relation between the columns, so the determinant vanishes"). -/
theorem wronskian_eq_zero_of_not_linearIndependent {φ : Fin l → K[X]} (h : ¬ LinearIndependent K φ)
    : wronskian φ = 0 := by
  obtain ⟨g, hg, ν₀, hν₀⟩ := Fintype.not_linearIndependent_iff.1 h
  refine Matrix.exists_mulVec_eq_zero_iff.1 ⟨fun ν ↦ C (g ν), fun h0 ↦ hν₀ ?_, ?_⟩
  · simpa using congrFun h0 ν₀
  · ext1 μ
    simpa [wronskianMatrix, Matrix.mulVec, dotProduct, ← smul_eq_C_mul, mul_comm]
      using congrArg (hasseDeriv μ) hg

/-- The determinant of the binomial matrix `(d ν choose μ)_{μ, ν}` is the Vandermonde product
`∏_{i<j} (d j - d i)` times `∏ μ, (μ!)⁻¹`.
Source: §2, first lemma ("`det (d_ν choose μ) = ∏_{i<j} (d_j - d_i) / ∏ μ!`: `binom X μ` has
degree `μ` and leading coefficient `1/μ!`, so its evaluation determinant is the Vandermonde
determinant multiplied by `∏ 1/μ!`"). -/
theorem det_choose_eq [CharZero K] (d : Fin l → ℕ) :
    (Matrix.of fun μ ν : Fin l ↦ ((d ν).choose μ : K)).det =
      (∏ μ : Fin l, ((μ : ℕ).factorial : K)⁻¹) *
        ∏ i : Fin l, ∏ j ∈ Finset.Ioi i, ((d j : K) - d i) := by
  have h1 : (Matrix.of fun μ ν : Fin l ↦ ((d ν).choose μ : K)) =
      Matrix.of fun μ ν : Fin l ↦ ((μ : ℕ).factorial : K)⁻¹ *
        (Matrix.of fun i j : Fin l ↦ (descPochhammer K j).eval (d i : K)).transpose μ ν := by
    ext μ ν
    simp [Nat.cast_choose_eq_descPochhammer_div, div_eq_inv_mul]
  rw [h1, Matrix.det_mul_column, Matrix.det_transpose,
    ← Matrix.det_eval_matrixOfPolynomials_eq_det_vandermonde (fun i ↦ (d i : K))
      (fun j ↦ descPochhammer K j) (fun i ↦ descPochhammer_natDegree K _)
      (fun i ↦ monic_descPochhammer K _), Matrix.det_vandermonde]

private lemma coeff_prod_of_natDegree_le' {ι : Type*} (s : Finset ι) (f : ι → K[X]) (n : ι → ℕ)
    (h : ∀ i ∈ s, (f i).natDegree ≤ n i) :
    (∏ i ∈ s, f i).coeff (∑ i ∈ s, n i) = ∏ i ∈ s, (f i).coeff (n i) := by
  classical
  induction s using Finset.induction_on with
  | empty => simp
  | insert a s has ih =>
    simp only [Finset.forall_mem_insert] at h
    rw [Finset.prod_insert has, Finset.sum_insert has, Finset.prod_insert has,
      coeff_mul_add_eq_of_natDegree_le h.1 ((natDegree_prod_le _ _).trans (Finset.sum_le_sum h.2)),
      ih h.2]

/-- `l` distinct natural numbers sum to at least `0 + 1 + ⋯ + (l - 1)`.
Source: §2, first lemma (`D = ∑ d_ν - l(l-1)/2 ≥ 0` for distinct degrees). -/
theorem sum_range_le_sum_of_injective (d : Fin l → ℕ) (hd : Function.Injective d) :
    ∑ ν : Fin l, (ν : ℕ) ≤ ∑ ν, d ν := by
  exact_mod_cast (show ((∑ ν : Fin l, (ν : ℕ) : ℕ) : ℤ) ≤
      ((∑ ν, d ν : ℕ) : ℤ) by
    simpa [Finset.card_image_of_injective _ hd, Finset.sum_image hd.injOn,
      ← Fin.sum_univ_eq_sum_range, Nat.cast_sum] using
      Finset.sum_range_le_sum (s := (Finset.univ.image d).map Nat.castEmbedding)
        (c := 0) (by simp))

/-- For a family of pairwise distinct degrees `d ν`, the coefficient of the Wronskian in degree
`∑ d ν - ∑ ν` is `∏ (leading coefficients)` times the binomial determinant `det (d ν choose μ)`.
The hypotheses are not needed for the identity itself, which holds for any family (both sides
may vanish), but they are the setting in which it is used.
Source: §2, first lemma ("The coefficient of `t^D` in the new Wronskian is
`(∏ a_ν) det (d_ν choose μ)`... All other contributions have lower total degree"). -/
theorem coeff_wronskian_of_natDegree_injective {φ : Fin l → K[X]} :
    (wronskian φ).coeff (∑ ν, (φ ν).natDegree - ∑ ν : Fin l, (ν : ℕ)) = (∏ ν, (φ ν).leadingCoeff) *
      (Matrix.of fun μ ν : Fin l ↦ ((φ ν).natDegree.choose μ : K)).det := by
  classical
  have key (σ : Equiv.Perm (Fin l)) : (∏ i, hasseDeriv (σ i : ℕ) (φ i)).coeff
      (∑ ν, (φ ν).natDegree - ∑ ν : Fin l, (ν : ℕ)) =
      ∏ i, ((φ i).natDegree.choose (σ i) : K) * (φ i).leadingCoeff := by
    by_cases h : ∀ i, (σ i : ℕ) ≤ (φ i).natDegree
    · rw [← Equiv.sum_comp σ (fun i : Fin l ↦ (i : ℕ)), ← Finset.sum_tsub_distrib _ fun i _ ↦ h i,
        coeff_prod_of_natDegree_le' _ _ _ fun i _ ↦ natDegree_hasseDeriv_le _ _]
      refine Finset.prod_congr rfl fun i _ ↦ ?_
      rw [hasseDeriv_coeff, Nat.sub_add_cancel (h i), leadingCoeff]
    · push Not at h
      obtain ⟨i, hi⟩ := h
      rw [Finset.prod_eq_zero (Finset.mem_univ i) (hasseDeriv_eq_zero_of_lt_natDegree _ _ hi),
        coeff_zero]
      exact (Finset.prod_eq_zero (Finset.mem_univ i) (by simp [Nat.choose_eq_zero_of_lt hi])).symm
  rw [wronskian, wronskianMatrix, Matrix.det_apply', Matrix.det_apply', finsetSum_coeff,
    Finset.mul_sum]
  refine Finset.sum_congr rfl fun σ _ ↦ ?_
  simp [coeff_intCast_mul, key, Finset.prod_mul_distrib, mul_comm, mul_assoc]

/-- In characteristic zero, nonzero polynomials of pairwise distinct degrees have nonzero
Wronskian.
Source: §2, first lemma ("Thus the new Wronskian ... is nonzero" for a family with distinct
degrees). -/
theorem wronskian_ne_zero_of_natDegree_injective [CharZero K] {φ : Fin l → K[X]} (hφ : ∀ ν, φ ν ≠ 0)
    (hd : Function.Injective fun ν ↦ (φ ν).natDegree) : wronskian φ ≠ 0 := by
  intro h0
  simpa [h0, det_choose_eq, eq_comm (a := 0), Finset.prod_eq_zero_iff, sub_eq_zero, hφ,
    Nat.factorial_ne_zero, hd.eq_iff] using coeff_wronskian_of_natDegree_injective (φ := φ)

/-- Subtracting a constant multiple of one member of the family from another does not change
the Wronskian.
Source: §2, first lemma ("By invertible column operations over `ℚ`" — a single column
operation does not change the Wronskian). -/
theorem wronskian_update_sub_smul {φ : Fin l → K[X]} {ν ν' : Fin l} (h : ν ≠ ν') (c : K) :
    wronskian (Function.update φ ν' (φ ν' - C c * φ ν)) = wronskian φ := by
  refine (congrArg Matrix.det ?_).trans
    (Matrix.det_updateCol_add_smul_self (wronskianMatrix φ) h.symm (-C c))
  ext μ ρ
  simp [wronskianMatrix, Matrix.updateCol_apply, Function.update_apply, apply_ite,
    ← smul_eq_C_mul, sub_eq_add_neg]

/-- Subtracting a constant multiple of one member of a linearly independent family from another
gives a linearly independent family.
Source: §2, first lemma ("replace them by a basis `g_0, …, g_{l-1}` of their span": a column
operation preserves linear independence). -/
theorem linearIndependent_update_sub_smul {φ : Fin l → K[X]} {ν ν' : Fin l} (h : ν ≠ ν') (c : K)
    (hφ : LinearIndependent K φ) :
    LinearIndependent K (Function.update φ ν' (φ ν' - C c * φ ν)) := by
  rw [linearIndependent_iff_card_eq_finrank_span, Set.finrank, ← smul_eq_C_mul,
    Submodule.span_range_update_sub_smul h]
  exact linearIndependent_iff_card_eq_finrank_span.mp hφ

/-- If two members `φ ν`, `φ ν'` of a linearly independent family have the same degree, cancelling
the leading coefficient of `φ ν'` against `φ ν` strictly lowers the total degree.
Source: §2, first lemma ("Such a basis is obtained by eliminating equal leading degrees": a
column operation that cancels a leading coefficient strictly lowers the sum of the degrees; the
difference is nonzero because the family is linearly independent). -/
theorem sum_natDegree_update_lt {φ : Fin l → K[X]} (hφ : LinearIndependent K φ) {ν ν' : Fin l}
    (h : ν ≠ ν') (hdeg : (φ ν).natDegree = (φ ν').natDegree) : ∑ ρ, (Function.update φ ν'
      (φ ν' - C ((φ ν').leadingCoeff / (φ ν).leadingCoeff) * φ ν) ρ).natDegree
      < ∑ ρ, (φ ρ).natDegree := by
  set a := (φ ν').leadingCoeff / (φ ν).leadingCoeff
  have h0 : ∀ i, φ i ≠ 0 := hφ.ne_zero
  have hlc : (φ ν).leadingCoeff ≠ 0 := leadingCoeff_ne_zero.2 (h0 ν)
  have hne : φ ν' - C a * φ ν ≠ 0 := by
    simpa using (linearIndependent_update_sub_smul h a hφ).ne_zero ν'
  have hdeg' : degree (φ ν') = degree (C a * φ ν) := by
    rw [degree_C_mul (div_ne_zero (leadingCoeff_ne_zero.2 (h0 ν')) hlc),
      degree_eq_natDegree (h0 ν), degree_eq_natDegree (h0 ν'), hdeg]
  have hlt : (φ ν' - C a * φ ν).natDegree < (φ ν').natDegree :=
    natDegree_lt_natDegree hne (degree_sub_lt_left hdeg' (h0 ν') (by
      rw [leadingCoeff_mul, leadingCoeff_C, div_mul_cancel₀ _ hlc]))
  refine Finset.sum_lt_sum (fun i _ ↦ ?_) ⟨ν', Finset.mem_univ _, by simpa⟩
  by_cases hi : i = ν'
  · simpa [hi] using hlt.le
  · simp [Function.update_of_ne hi]

/-- In characteristic zero, a linearly independent family has nonzero Wronskian.
Source: §2, first lemma, hard direction ("Conversely, suppose the polynomials are linearly
independent ... Thus the new Wronskian, and hence the original one, is nonzero"). -/
theorem wronskian_ne_zero_of_linearIndependent [CharZero K] {φ : Fin l → K[X]}
    (h : LinearIndependent K φ) : wronskian φ ≠ 0 := by
  classical
  induction hN : ∑ ρ, (φ ρ).natDegree using Nat.strong_induction_on generalizing φ with | _ N ih =>
    by_cases hinj : Function.Injective fun ν ↦ (φ ν).natDegree
    · exact wronskian_ne_zero_of_natDegree_injective h.ne_zero hinj
    obtain ⟨ν, ν', hdeg, hne⟩ := Function.not_injective_iff.1 hinj
    rw [← wronskian_update_sub_smul hne ((φ ν').leadingCoeff / (φ ν).leadingCoeff)]
    exact ih _ (hN ▸ sum_natDegree_update_lt h hne hdeg)
      (linearIndependent_update_sub_smul hne _ h) rfl

/-- In characteristic zero, the Wronskian vanishes iff the family is linearly dependent.
Source: §2, first lemma ("the Wronskian vanishes if and only if they are linearly
dependent"). -/
theorem wronskian_eq_zero_iff [CharZero K] {φ : Fin l → K[X]} :
    wronskian φ = 0 ↔ ¬ LinearIndependent K φ :=
  ⟨imp_not_comm.mp wronskian_ne_zero_of_linearIndependent,
    wronskian_eq_zero_of_not_linearIndependent⟩

/-- The generalized Wronskian `det (Δ_{D μ} φ_ν)_{μ, ν}`. Source: §2, definition of
`W(φ_0, …, φ_{l-1}; Δ_0, …, Δ_{l-1})`. -/
noncomputable def genWronskian (D : Fin l → σ →₀ ℕ) (φ : Fin l → MvPolynomial σ K) :
    MvPolynomial σ K :=
  (Matrix.of fun μ ν ↦ hasseDeriv (D μ) (φ ν)).det

omit [Fintype σ] in
/-- Linearly dependent polynomials have identically vanishing generalized Wronskians.

Source: §2, second lemma ("If `φ_0, …, φ_{l-1}` ... are linearly dependent ..., then all of
their generalized Wronskians are identically zero"). -/
theorem genWronskian_eq_zero_of_not_linearIndependent (D : Fin l → (σ →₀ ℕ))
    {φ : Fin l → MvPolynomial σ K} (h : ¬ LinearIndependent K φ) : genWronskian D φ = 0 := by
  classical
  obtain ⟨g, hg, ν₀, hν₀⟩ := Fintype.not_linearIndependent_iff.1 h
  refine Matrix.exists_mulVec_eq_zero_iff.1 ⟨fun ν ↦ C (g ν), fun h0 ↦ hν₀ ?_, ?_⟩
  · simpa using congrFun h0 ν₀
  · ext1 μ
    simpa [Matrix.mulVec, dotProduct, map_smul, ← smul_eq_C_mul, mul_comm] using
      congrArg (hasseDeriv (D μ)) hg

/-- Kronecker substitution `x_j ↦ t^{k^{e j}}` where `e : σ ≃ Fin (card σ)` is the chosen
enumeration. Source: §2, proof of Lemma 1 (`h_ν(t) = φ_ν(t, t^k, …, t^{k^{p-1}})`). -/
noncomputable def kronecker (k : ℕ) : MvPolynomial σ K →ₐ[K] Polynomial K :=
  aeval fun j ↦ Polynomial.X ^ k ^ (Fintype.equivFin σ j : ℕ)

/-- The exponent `∑ j, d j * k^{e j}` of the image of the monomial `X^d`. -/
noncomputable def kroneckerExp (k : ℕ) (d : σ →₀ ℕ) : ℕ := ∑ j, d j * k ^ (Fintype.equivFin σ j : ℕ)

/-- The Kronecker substitution sends the monomial `a X^d` to `a t^{kroneckerExp k d}`. -/
theorem kronecker_monomial (k : ℕ) (d : σ →₀ ℕ) (a : K) :
    kronecker k (monomial d a) = Polynomial.C a * Polynomial.X ^ kroneckerExp k d := by
  simp [kronecker, kroneckerExp, aeval_monomial, Finsupp.prod_fintype, ← pow_mul, mul_comm,
    ← Finset.prod_pow_eq_pow_sum]

/-- The Kronecker exponent map is injective on exponent vectors whose entries are all `< k`.

Source: §2, proof of Lemma 1, FACT ("Since each `s_i` is strictly smaller than `k`, the
representation of any natural number in the form `s_1 + k s_2 + ⋯ + k^{p-1} s_p` is unique"). -/
theorem kroneckerExp_injOn {k : ℕ} (hk : 1 < k) :
    Set.InjOn (kroneckerExp (σ := σ) k) {d | ∀ j, d j < k} := by
  intro d hd d' hd' h
  set e := Fintype.equivFin σ
  have hsum (d : σ →₀ ℕ) : Nat.ofDigits k (List.ofFn fun i ↦ d (e.symm i)) = kroneckerExp k d := by
    simp only [Nat.ofDigits_eq_sum_mapIdx, List.mapIdx_eq_ofFn, Fin.sum_ofFn, kroneckerExp,
      ← e.symm.sum_comp, e, List.get_eq_getElem, List.getElem_ofFn, List.length_ofFn,
      Equiv.apply_symm_apply, Fin.val_cast, Fin.eta]
  apply DFunLike.coe_injective
  apply e.symm.surjective.injective_comp_right
  refine List.ofFn_injective (Nat.ofDigits_inj_of_len_eq hk (by simp)
    (List.forall_mem_ofFn_iff.2 fun i ↦ hd (e.symm i))
    (List.forall_mem_ofFn_iff.2 fun i ↦ hd' (e.symm i)) ?_)
  simpa [Function.comp_def, hsum] using h

/-- If all partial degrees of `P` are `< k`, the Kronecker substitution preserves the coefficient
of `X^d` for every `d` with entries `< k`.

Source: §2, proof of Lemma 1 ("Comparing the coefficient of `t^{s_1 + k s_2 + ⋯}` gives ..."). -/
theorem coeff_kronecker {k : ℕ} (hk : 1 < k) {P : MvPolynomial σ K} (hP : ∀ j, degreeOf j P < k)
    {d : σ →₀ ℕ} (hd : ∀ j, d j < k) : (kronecker k P).coeff (kroneckerExp k d) = P.coeff d := by
  classical
  conv_lhs => rw [P.as_sum]
  rw [map_sum, Polynomial.finsetSum_coeff]
  simp_rw [kronecker_monomial, Polynomial.coeff_C_mul_X_pow]
  rw [Finset.sum_eq_single d]
  · simp
  · intro e he hed
    exact ite_eq_right fun h ↦ hed (kroneckerExp_injOn hk
      (fun j ↦ (monomial_le_degreeOf j he).trans_lt (hP j)) hd h.symm)
  · intro hdP
    simp [notMem_support_iff.1 hdP]

/-- A polynomial with all partial degrees `< k` vanishes iff its Kronecker substitution does.

Source: §2, Lean implementation note ("prove injectivity only on the subspace where every
individual exponent is less than `k`"). -/
theorem kronecker_eq_zero_iff {k : ℕ} (hk : 1 < k) {P : MvPolynomial σ K}
    (hP : ∀ j, degreeOf j P < k) : kronecker k P = 0 ↔ P = 0 := by
  refine ⟨fun h ↦ support_eq_empty.1 <| Finset.eq_empty_iff_forall_notMem.2 fun d hd ↦ ?_,
    fun h ↦ by rw [h, map_zero]⟩
  apply mem_support_iff.1 hd
  simpa [h] using (coeff_kronecker hk hP fun j ↦ (monomial_le_degreeOf j hd).trans_lt (hP j)).symm

/-- The Kronecker substitution preserves linear independence of polynomials whose partial degrees
are all `< k`.

Source: §2, proof of Lemma 1 ("We claim that these polynomials are linearly independent"). -/
theorem linearIndependent_kronecker {k : ℕ} (hk : 1 < k) {φ : Fin l → MvPolynomial σ K}
    (hφ : ∀ ν j, degreeOf j (φ ν) < k) (h : LinearIndependent K φ) :
    LinearIndependent K fun ν ↦ kronecker k (φ ν) := by
  rw [Fintype.linearIndependent_iff] at h ⊢
  intro g hg
  apply h g
  refine (kronecker_eq_zero_iff hk fun j ↦ ?_).mp ?_
  · refine (degreeOf_sum_le _ _ _).trans_lt <|
      (Finset.sup_lt_iff (Nat.zero_lt_of_lt hk)).2 fun i _ ↦ ?_
    simpa [smul_eq_C_mul] using (degreeOf_C_mul_le (φ i) j (g i)).trans_lt (hφ i j)
  · simpa [map_sum, map_smul] using hg

/-- The algebraic chain rule for substituting univariate polynomials into a multivariate one.

Source: §2, proof of Lemma 1, FACT
(`d/dt h_ν(t) = ∑_j γ_j'(t) (∂φ_ν/∂x_j)(γ(t))`). -/
theorem derivative_aeval (γ : σ → Polynomial K) (P : MvPolynomial σ K) :
    Polynomial.derivative (aeval γ P) =
      ∑ j, Polynomial.derivative (γ j) * aeval γ (pderiv j P) := by
  classical
  induction P using MvPolynomial.induction_on with
  | C a => simp
  | add p q hp hq => simp [hp, hq, Finset.sum_add_distrib, mul_add]
  | mul_X p j hp =>
    simp [hp, Finset.sum_add_distrib, ← Finset.mul_sum, add_mul, mul_assoc, mul_comm,
      Pi.single_apply, apply_ite (aeval γ)]

/-- The operator `P ↦ kronecker k (Δ_d P)`, as an element of the `K[X]`-module of functions. -/
private noncomputable def kronHasse (k : ℕ) (d : σ →₀ ℕ) : MvPolynomial σ K → Polynomial K :=
  fun P ↦ kronecker k (hasseDeriv d P)

/-- The `K[X]`-span of the operators `kronHasse k d` with `|d| ≤ μ`. -/
private noncomputable def kronSpan (k μ : ℕ) : Submodule (Polynomial K)
    (MvPolynomial σ K → Polynomial K) :=
  Submodule.span (Polynomial K) (kronHasse (K := K) k '' {d | d.degree ≤ μ})

private lemma derivative_kronHasse_aux (k : ℕ) (d : σ →₀ ℕ) :
    (fun P ↦ Polynomial.derivative (kronHasse (K := K) k d P)) =
      ∑ j, (Polynomial.derivative (Polynomial.X ^ k ^ (Fintype.equivFin σ j : ℕ)) *
        Polynomial.C ((d j + 1 : ℕ) : K)) • kronHasse k (d + Finsupp.single j 1) := by
  classical
  funext P
  simp [kronHasse, kronecker, derivative_aeval, pderiv_hasseDeriv, map_smul,
    Polynomial.smul_eq_C_mul, mul_assoc, mul_comm, mul_left_comm]

private lemma derivative_mem_kronSpan (k μ : ℕ) {F : MvPolynomial σ K → Polynomial K}
    (hF : F ∈ kronSpan k μ) :
    (fun P ↦ Polynomial.derivative (F P)) ∈ kronSpan (K := K) (σ := σ) k (μ + 1) := by
  classical
  induction hF using Submodule.span_induction with
  | mem F hF =>
    obtain ⟨d, hd, rfl⟩ := hF
    rw [derivative_kronHasse_aux]
    refine Submodule.sum_mem _ fun j _ ↦ Submodule.smul_mem _ _ (Submodule.subset_span ?_)
    refine ⟨d + Finsupp.single j 1, ?_, rfl⟩
    simp only [Set.mem_ofPred_eq, map_add, Finsupp.degree_single] at hd ⊢
    exact Nat.add_le_add_right hd 1
  | zero =>
    convert (kronSpan (K := K) (σ := σ) k (μ + 1)).zero_mem using 1
    funext P
    simp
  | add F G _ _ hF hG =>
    convert Submodule.add_mem _ hF hG using 1
    funext P
    simp [Polynomial.derivative_add]
  | smul a F hF' hF =>
    convert Submodule.add_mem (kronSpan (K := K) (σ := σ) k (μ + 1))
      (Submodule.smul_mem _ (Polynomial.derivative a)
        (Submodule.span_mono (Set.image_mono fun _ hd ↦ le_trans hd (Nat.le_succ μ)) hF'))
      (Submodule.smul_mem _ a hF) using 1
    funext P
    simp [Polynomial.derivative_mul, add_comm]

private lemma hasseDeriv_kronecker_mem [CharZero K] (k μ : ℕ) :
    (fun P ↦ Polynomial.hasseDeriv μ (kronecker k P)) ∈ kronSpan (K := K) (σ := σ) k μ := by
  induction μ with
  | zero =>
    refine Submodule.subset_span ⟨0, by simp, ?_⟩
    funext P
    simp [kronHasse]
  | succ μ ih =>
    apply (((kronSpan (K := K) (σ := σ) k (μ + 1)).restrictScalars K).smul_mem_iff
      (s := ((μ + 1 : ℕ) : K)) (Nat.cast_ne_zero.2 μ.succ_ne_zero)).mp
    change (((μ + 1 : ℕ) : K) • fun P ↦ Polynomial.hasseDeriv (μ + 1) (kronecker k P)) ∈
      kronSpan (K := K) (σ := σ) k (μ + 1)
    convert derivative_mem_kronSpan k μ ih using 1
    funext P
    simpa [← Nat.cast_smul_eq_nsmul K, add_comm] using
      (congrArg (fun L ↦ L (kronecker k P)) (Polynomial.hasseDeriv_comp (R := K) 1 μ)).symm

/-- The `μ`-th Hasse derivative of a Kronecker substitution is a `K[t]`-linear combination of
Kronecker substitutions of Hasse derivatives `Δ_d` of order `|d| ≤ μ`, uniformly in `P`.

Source: §2, proof of Lemma 1 ("for each `μ`, there is a natural number `r(μ)` ..., polynomials
`f^μ_i(t)` with rational coefficients, and differential operators `Δ^{(i)}_μ` of order `≤ μ` such
that for all `ν`, `d^μ/dt^μ h_ν = ∑_i f^μ_i(t) (Δ^{(i)}_μ φ_ν)(γ(t))`"; here stated with Hasse
derivatives on both sides, which over `ℚ` differ from iterated derivatives by factorials). -/
theorem exists_hasseDeriv_kronecker_eq_sum [CharZero K] (k μ : ℕ) : ∃ (s : Finset (σ →₀ ℕ))
    (c : (σ →₀ ℕ) → Polynomial K), (∀ d ∈ s, d.degree ≤ μ) ∧ ∀ P : MvPolynomial σ K,
      Polynomial.hasseDeriv μ (kronecker k P) = ∑ d ∈ s, c d * kronecker k (hasseDeriv d P) := by
  classical
  obtain ⟨c, hc, hsum⟩ := (Finsupp.mem_span_image_iff_linearCombination _).1
    (hasseDeriv_kronecker_mem (K := K) (σ := σ) k μ)
  refine ⟨c.support, c, hc, fun P ↦ ?_⟩
  simpa [Finsupp.linearCombination_apply, Finsupp.sum, kronHasse] using congrFun hsum.symm P

/-- Expanding the determinant of a matrix whose rows are linear combinations, by multilinearity
in the rows.

Source: §2, proof of Lemma 1 ("multilinearity of the determinant in each row gives
`det (∑_i f^μ_i ⋯) = ∑_{i ∈ 𝒮} f_i(t) det (⋯)`"). -/
theorem det_sum_rows {A : Type*} [CommRing A] {ι : Type*} (s : Fin l → Finset ι) (c : Fin l → ι → A)
    (N : Fin l → ι → Fin l → A) :
    (Matrix.of fun μ ν ↦ ∑ i ∈ s μ, c μ i * N μ i ν).det =
      ∑ f ∈ Fintype.piFinset s, (∏ μ, c μ (f μ)) * (Matrix.of fun μ ν ↦ N μ (f μ) ν).det := by
  have h := (Matrix.detRowAlternating (R := A) (n := Fin l)).toMultilinearMap.map_sum_finset
    (fun μ i ↦ c μ i • N μ i) s
  simp_rw [MultilinearMap.map_smul_univ, smul_eq_mul] at h
  simp only [Matrix.det, Finset.sum_fn, Pi.smul_def, smul_eq_mul] at h ⊢
  exact h

private lemma wronskian_kronecker_eq_sum_aux (k : ℕ) (φ : Fin l → MvPolynomial σ K)
    (s : Fin l → Finset (σ →₀ ℕ)) (c : Fin l → (σ →₀ ℕ) → Polynomial K)
    (hc : ∀ μ ν : Fin l, Polynomial.hasseDeriv μ (kronecker k (φ ν)) =
      ∑ d ∈ s μ, c μ d * kronecker k (hasseDeriv d (φ ν))) : wronskian (fun ν ↦ kronecker k (φ ν)) =
    ∑ f ∈ Fintype.piFinset s, (∏ μ, c μ (f μ)) * kronecker k (genWronskian f φ) := by
  classical
  simpa [wronskian, wronskianMatrix, hc, genWronskian, AlgHom.map_det, Matrix.map] using
    det_sum_rows s c (fun μ d ν ↦ kronecker k (hasseDeriv d (φ ν)))

omit [Fintype σ] in
/-- **Lemma 1**: a linearly independent family of polynomials has a generalized Wronskian (with
Hasse operators of orders `|D μ| ≤ μ`) that is not identically zero.

Source: §2, Lemma `roth:lem1` ("If `φ_0, …, φ_{l-1}` are `l` linearly independent
polynomials in `p` variables with rational coefficients, then at least one of their generalized
Wronskians is not identically zero"). -/
theorem exists_genWronskian_ne_zero [Finite σ] [CharZero K] {φ : Fin l → MvPolynomial σ K}
    (h : LinearIndependent K φ) :
    ∃ D : Fin l → (σ →₀ ℕ), (∀ μ, (D μ).degree ≤ μ) ∧ genWronskian D φ ≠ 0 := by
  classical
  let := Fintype.ofFinite σ
  let k := 2 + ∑ ν, ∑ j, degreeOf j (φ ν)
  choose s c hs hc using fun μ : Fin l ↦ exists_hasseDeriv_kronecker_eq_sum (K := K) (σ := σ) k μ
  by_contra! hcon
  apply wronskian_ne_zero_of_linearIndependent
    (linearIndependent_kronecker (Nat.lt_add_right _ (by decide)) (fun ν j ↦
      lt_of_le_of_lt ((Finset.single_le_sum (f := fun j ↦ degreeOf j (φ ν))
        (by simp) (Finset.mem_univ j)).trans (Finset.single_le_sum
          (f := fun ν ↦ ∑ j, degreeOf j (φ ν)) (by simp) (Finset.mem_univ ν)))
        (Nat.lt_add_of_pos_left two_pos)) h)
  rw [wronskian_kronecker_eq_sum_aux k φ s c (fun μ ν ↦ hc μ (φ ν))]
  refine Finset.sum_eq_zero fun f hf ↦ ?_
  rw [hcon f fun μ ↦ hs μ _ (Fintype.mem_piFinset.1 hf μ), map_zero, mul_zero]

private lemma linearIndependent_coordinate_polynomials_aux {R M : Type*} [CommRing R]
    [AddCommMonoid M] [Module R M] {n l : ℕ} (v : Fin (n + 1) → M)
    (b : Module.Basis (Fin l) R (Submodule.span R (Set.range v))) :
    LinearIndependent R (fun ρ ↦ ∑ ν : Fin (n + 1),
      Polynomial.C (b.repr ⟨v ν, Submodule.subset_span ⟨ν, rfl⟩⟩ ρ) * Polynomial.X ^ (ν : ℕ)) := by
  classical
  let a := fun ν ρ ↦ b.repr ⟨v ν, Submodule.subset_span ⟨ν, rfl⟩⟩ ρ
  rw [Fintype.linearIndependent_iff]
  intro g hg ρ₀
  have hcoeff : ∀ ν : Fin (n + 1), ∑ ρ, g ρ * a ν ρ = 0 := by
    intro ν
    have h1 := congrArg (fun p : Polynomial R ↦ p.coeff ν) hg
    simp only [Polynomial.finsetSum_coeff, Polynomial.coeff_smul, smul_eq_mul,
      Polynomial.coeff_C_mul_X_pow, Polynomial.coeff_zero] at h1
    rw [← h1]
    refine Finset.sum_congr rfl fun ρ _ ↦ ?_
    rw [Finset.sum_eq_single ν (fun ν' _ hν' ↦ ite_eq_right fun h ↦ hν' (Fin.ext h).symm)
      (fun h ↦ absurd (Finset.mem_univ ν) h), ite_eq_left rfl]
  let f : Submodule.span R (Set.range v) →ₗ[R] R := ∑ ρ, g ρ • b.coord ρ
  have hf : f = 0 := LinearMap.ext_on Submodule.span_span_coe_preimage fun x hx ↦ by
    obtain ⟨ν, hν⟩ := hx
    obtain rfl : x = ⟨v ν, Submodule.subset_span ⟨ν, rfl⟩⟩ := Subtype.ext hν.symm
    simpa only [f, a, LinearMap.sum_apply, LinearMap.smul_apply, smul_eq_mul,
      Module.Basis.coord_apply, LinearMap.zero_apply] using hcoeff ν
  simpa only [f, LinearMap.sum_apply, LinearMap.smul_apply, Module.Basis.coord_apply,
    Module.Basis.repr_self, Finsupp.single_apply, smul_eq_mul, mul_ite, mul_one, mul_zero,
    Finset.sum_ite_eq, Finset.mem_univ, ite_true, LinearMap.zero_apply] using
    LinearMap.congr_fun hf (b ρ₀)

private lemma eq_sum_aeval_basis_coeffNone_aux {R : Type*} [CommSemiring R]
    {P : MvPolynomial (Option τ) R} {n l : ℕ}
    (hP : P = ∑ ν : Fin (n + 1), X none ^ (ν : ℕ) * rename some (coeffNone P ν))
    (b : Module.Basis (Fin l) R
      (Submodule.span R (Set.range (fun ν : Fin (n + 1) ↦ coeffNone P ν)))) :
    P = ∑ ρ, Polynomial.aeval (X none : MvPolynomial (Option τ) R) (∑ ν : Fin (n + 1),
        Polynomial.C (b.repr ⟨coeffNone P ν, Submodule.subset_span ⟨ν, rfl⟩⟩ ρ) *
          Polynomial.X ^ (ν : ℕ)) * rename some (b ρ : MvPolynomial τ R) := by
  let a := fun (ν : Fin (n + 1)) ρ ↦ b.repr ⟨coeffNone P ν, Submodule.subset_span ⟨ν, rfl⟩⟩ ρ
  have hexp : ∀ ν : Fin (n + 1), coeffNone P ν = ∑ ρ, a ν ρ • (b ρ : MvPolynomial τ R) := fun ν ↦ by
    have h := congrArg Subtype.val (b.sum_repr ⟨coeffNone P ν, Submodule.subset_span ⟨ν, rfl⟩⟩)
    simpa only [Submodule.coe_sum, Submodule.coe_smul] using h.symm
  change P = ∑ ρ, Polynomial.aeval (X none : MvPolynomial (Option τ) R)
    (∑ ν : Fin (n + 1), Polynomial.C (a ν ρ) * Polynomial.X ^ (ν : ℕ)) *
      rename some (b ρ : MvPolynomial τ R)
  conv_lhs => rw [hP]
  simp_rw [hexp, map_sum, Finset.mul_sum]
  rw [Finset.sum_comm]
  refine Finset.sum_congr rfl fun ρ _ ↦ ?_
  rw [Finset.sum_mul]
  refine Finset.sum_congr rfl fun ν _ ↦ ?_
  rw [map_smul, smul_eq_C_mul, map_mul, Polynomial.aeval_C, algebraMap_eq, map_pow,
    Polynomial.aeval_X]
  ac_rfl

private lemma exists_coeffNone_basis_aux {P : MvPolynomial (Option τ) K} (hP : P ≠ 0) {n : ℕ}
    (hP' : P = ∑ ν : Fin (n + 1), X none ^ (ν : ℕ) * rename some (coeffNone P ν)) :
    ∃ l, ∃ b : Module.Basis (Fin l) K
      (Submodule.span K (Set.range (fun ν : Fin (n + 1) ↦ coeffNone P ν))),

  let V := Submodule.span K (Set.range v)
  have : FiniteDimensional K V := FiniteDimensional.span_of_finite K (Set.finite_range v)
  let b := Module.finBasis K V
  have hl1 : 1 ≤ Module.finrank K V := by
    obtain ⟨ν, hν⟩ : ∃ ν, v ν ≠ 0 := by
      by_contra! h
      apply hP
      change P = ∑ ν : Fin (n + 1), X none ^ (ν : ℕ) * rename some (v ν) at hP'
      rw [hP']
      simp [h]
    exact Module.finrank_pos_iff_exists_ne_zero.2
      ⟨⟨v ν, Submodule.subset_span ⟨ν, rfl⟩⟩, fun h ↦ hν (congrArg Subtype.val h)⟩
  have hdegree : V ≤ restrictSupport K {d | ∀ j, d j ≤ degreeOf (some j) P} :=
    Submodule.span_le.2 (Set.range_subset_iff.2 fun (ν : Fin (n + 1)) d hd j ↦ by
      simpa using monomial_le_degreeOf (some j)
        ((mem_support_coeff_optionEquivLeft K).1 hd))
  exact ⟨Module.finrank K V, b, hl1,
    (finrank_range_le_card (R := K) v).trans_eq (Fintype.card_fin _),
    fun ρ j ↦ degreeOf_le_iff.2 fun d hd ↦ hdegree (b ρ).2 hd j⟩

/-- Over a field, a nonzero `P` of degree at most `n` in `x_none` can be written as
`∑_{ν < l} φ_ν(x_none) ψ_ν(x_τ)` with `1 ≤ l ≤ n + 1`, both families linearly independent,
`deg φ_ν ≤ n`, and each `ψ_ν` of degree in `x_j` at most that of `P`.
Source: §2, proof of Lemma 2 and its Lean implementation note ("Construct the separated
decomposition over `ℚ` from a basis of the span of the coefficient polynomials of `R` in `x_p`";
the `φ_ν` and `ψ_ν` are linearly independent). -/
theorem exists_separated_representation {P : MvPolynomial (Option τ) K} (hP : P ≠ 0) {n : ℕ}
    (hn : degreeOf none P ≤ n) :
    ∃ l, 1 ≤ l ∧ l ≤ n + 1 ∧ ∃ (φ : Fin l → Polynomial K) (ψ : Fin l → MvPolynomial τ K),
      LinearIndependent K φ ∧ LinearIndependent K ψ ∧ (∀ ν, (φ ν).natDegree ≤ n) ∧
      (∀ ν j, degreeOf j (ψ ν) ≤ degreeOf (some j) P) ∧
      P = ∑ ν, Polynomial.aeval (X none : MvPolynomial (Option τ) K) (φ ν) * rename some (ψ ν) := by
  classical
  let v : Fin (n + 1) → MvPolynomial τ K := fun ν ↦ coeffNone P ν
  have hP' : P = ∑ ν : Fin (n + 1), X none ^ (ν : ℕ) * rename some (v ν) := by
    rw [Fin.sum_univ_eq_sum_range (fun ν ↦ X none ^ ν * rename some (coeffNone P ν))]
    exact eq_sum_X_none_pow_mul_coeffNone P hn
  obtain ⟨l, b, hl1, hl2, hdegree⟩ := exists_coeffNone_basis_aux hP hP'
  refine ⟨l, hl1, hl2, fun ρ ↦ ∑ ν : Fin (n + 1),
      Polynomial.C (b.repr ⟨v ν, Submodule.subset_span ⟨ν, rfl⟩⟩ ρ) * Polynomial.X ^ (ν : ℕ),
    fun ρ ↦ (b ρ : MvPolynomial τ K), linearIndependent_coordinate_polynomials_aux v b,
    b.linearIndependent.map' (Submodule.span K (Set.range v)).subtype
      (Submodule.ker_subtype _), ?_, hdegree,
    eq_sum_aeval_basis_coeffNone_aux hP' b⟩
  exact fun ρ ↦ Polynomial.natDegree_sum_le_of_forall_le _ _ fun ν _ ↦
    (Polynomial.natDegree_C_mul_X_pow_le _ _).trans (Nat.lt_succ_iff.1 ν.2)

/-- The determinant of Hasse derivatives of a separated sum `∑ φ_ρ(x_none) ψ_ρ(x_τ)` is the
product of the generalised Wronskian of `ψ` and the Wronskian of `φ`.
Source: §2, proof of Lemma 2 ("The matrix defining `F` is `A Bᵀ`, so `F = det A det B` by
multiplicativity and invariance under transpose. Explicitly, we obtain `GW = det (...)`"). -/
theorem det_hasseDeriv_separated {l : ℕ} (D : Fin l → (τ →₀ ℕ)) (φ : Fin l → Polynomial K)
    (ψ : Fin l → MvPolynomial τ K) :
    (Matrix.of fun μ ν : Fin l ↦ hasseDeriv ((D μ).mapDomain some + Finsupp.single none (ν : ℕ))
        (∑ ρ, Polynomial.aeval (X none : MvPolynomial (Option τ) K) (φ ρ) *
          rename some (ψ ρ))).det = rename some (genWronskian D ψ) *
      Polynomial.aeval (X none : MvPolynomial (Option τ) K) (wronskian φ) := by
  rw [genWronskian, wronskian, AlgHom.map_det, AlgHom.map_det,
    ← Matrix.det_transpose ((Polynomial.aeval (X none)).mapMatrix (wronskianMatrix φ)),
    ← Matrix.det_mul]
  congr 1
  refine Matrix.ext fun μ ν ↦ ?_
  rw [Matrix.of_apply, map_sum, Matrix.mul_apply]
  refine Finset.sum_congr rfl fun ρ _ ↦ ?_
  rw [mul_comm (Polynomial.aeval (X none : MvPolynomial (Option τ) K) (φ ρ)),
    hasseDeriv_rename_mul_aeval]
  rfl

theorem primitive {G : MvPolynomial σ K} (hG : G ≠ 0) :
    ∃ (g : K) (U : MvPolynomial σ R), g ≠ 0 ∧ map (algebraMap R K) U = g • G ∧
      U.support.gcd U.coeff = 1 := by

  let : Algebra (MvPolynomial σ R) (MvPolynomial σ K) :=
    MvPolynomial.algebraMvPolynomial

  have hNcast : algebraMap R K N ≠ 0 := (map_ne_zero_iff _ hinj).mpr hN
  have hU0 : map (algebraMap R K) U0 = algebraMap R K N • G := by
    have h : map (algebraMap R K) U0 = C (algebraMap R K N) * G := by
      simpa only [MvPolynomial.algebraMap_def, Algebra.smul_def, map_C] using hU0raw
    simpa only [smul_eq_C_mul] using h
  have hne : U0.support.Nonempty := support_nonempty.mpr fun h ↦ by
    exact smul_ne_zero hNcast hG (by simpa only [h, map_zero] using hU0.symm)
  obtain ⟨g', hg', hg1⟩ := Finset.extract_gcd (fun d ↦ U0.coeff d) hne
  set g0 := U0.support.gcd (fun d ↦ U0.coeff d) with hg0def
  have hg0 : g0 ≠ 0 := by
    obtain ⟨d, hd⟩ := hne
    rw [hg0def, Ne, Finset.gcd_eq_zero_iff]
    exact fun h ↦ mem_support_iff.1 hd (h d hd)
  set U := ∑ d ∈ U0.support, monomial d (g' d) with hUdef
  have hUcoeff : ∀ d, U.coeff d = if d ∈ U0.support then g' d else 0 := by
    simp [hUdef, coeff_monomial, Finset.sum_ite_eq']
  have hU0coeff : ∀ d, U0.coeff d = g0 * U.coeff d := by
    intro d
    by_cases hd : d ∈ U0.support
    · simpa [hUcoeff, hd] using hg' d hd
    · simp [hUcoeff, hd, notMem_support_iff.1 hd]
  refine ⟨algebraMap R K N / algebraMap R K g0, U,
    div_ne_zero hNcast ((map_ne_zero_iff _ hinj).mpr hg0), ?_, ?_⟩
  · ext d
    have h1 := congrArg (fun P ↦ P.coeff d) hU0
    simp only [coeff_map, coeff_smul, smul_eq_mul] at h1 ⊢
    rw [hU0coeff] at h1
    rw [map_mul] at h1
    rwa [div_mul_eq_mul_div, eq_div_iff ((map_ne_zero_iff _ hinj).mpr hg0), mul_comm]
  · have hsupp : U.support = U0.support := by
      ext d
      simp [mem_support_iff, hU0coeff d, hg0]
    rw [hsupp, ← hg1]
    exact Finset.gcd_congr rfl fun d hd ↦ by rw [hUcoeff, ite_eq_left hd]

    apply hrel.symm.dvd_of_dvd_mul_right
    exact hmul ▸ dvd_mul_left b z
  have hb : IsUnit b := isUnit_iff_dvd_one.mpr (hu ▸ hdiv)
  refine ⟨a * (↑(hb.unit⁻¹) : R), ?_⟩
  rw [map_mul]
  have hmap := Units.coe_map_inv (algebraMap R K : R →* K) hb.unit
  simp only [MonoidHom.coe_coe] at hmap
  rw [← hmap, Units.eq_mul_inv_iff_mul_eq]
  exact he

theorem rescale_separated_factor_aux {U' U : MvPolynomial τ K}
    {V' : Polynomial K} {F : MvPolynomial (Option τ) K} {g : K} (hg : g ≠ 0) (hUmap : U = g • U')
    (hF : F = rename some U' * Polynomial.aeval (X none : MvPolynomial (Option τ) K) V') :
    F = rename some U * Polynomial.aeval (X none : MvPolynomial (Option τ) K)

theorem exists_int_polynomial_of_primitive_factor_aux {U : MvPolynomial τ R}
    {V' : Polynomial K} {F : MvPolynomial (Option τ) R} (hgcd : U.support.gcd U.coeff = 1)
    (hF : map (algebraMap R K) F = rename some (map (algebraMap R K) U) *
      Polynomial.aeval (X none : MvPolynomial (Option τ) K) V') :
    ∃ V : Polynomial R, V.map (algebraMap R K) = V' := by
  apply (Polynomial.mem_map_range (algebraMap R K)).2
  intro b
  obtain ⟨z, hz⟩ := Factors176GCDScalar.scalar_integer hgcd fun a _ ↦
    ⟨F.coeff (a.mapDomain some + Finsupp.single none b), by
      have h := congrArg (fun P ↦ P.coeff (a.mapDomain some + Finsupp.single none b)) hF
      simp only [coeff_rename_mul_aeval, coeff_map] at h
      simpa using h.symm⟩
  exact ⟨z, hz.symm⟩

/-- If an integer polynomial `F` factors over `K` as `U'(x_τ) V'(x_none)` with nonzero factors,
then it factors over `R` as `U(x_τ) V(x_none)` with the same partial degrees.
Source: §2, proof of Lemma 2 ("Thus `U, V` are integral and nonzero, as required"), together
with the degree comparison at the end of the proof. -/
theorem exists_int_factors {U' : MvPolynomial τ K} {V' : Polynomial K} (hU : U' ≠ 0) (hV : V' ≠ 0)
    {F : MvPolynomial (Option τ) R} (hF : map (algebraMap R K) F =
      rename some U' * Polynomial.aeval (X none : MvPolynomial (Option τ) K) V') :
    ∃ (U : MvPolynomial τ R) (V : Polynomial R), U ≠ 0 ∧ V ≠ 0 ∧
      F = rename some U * Polynomial.aeval (X none : MvPolynomial (Option τ) R) V ∧
      (∀ j, degreeOf j U = degreeOf j U') ∧ V.natDegree = V'.natDegree := by
  obtain ⟨g, U, hg, hUmap, hgcd⟩ := Primitive172FullGeneric.primitive (R := R) hU
  have hF' := rescale_separated_factor_aux hg hUmap hF
  obtain ⟨V, hVmap⟩ := exists_int_polynomial_of_primitive_factor_aux hgcd hF'
  refine ⟨U, V, ?_, ?_, ?_, fun j ↦ ?_, ?_⟩
  · exact fun h ↦ smul_ne_zero hg hU (by simpa [h] using hUmap.symm)
  · exact fun h ↦ mul_ne_zero (Polynomial.C_ne_zero.2 (inv_ne_zero hg)) hV
      (by simpa [h] using hVmap.symm)
  · apply MvPolynomial.map_injective (algebraMap R K) (IsFractionRing.injective R K)
    rw [hF', map_mul (map (algebraMap R K)), map_rename,
      Polynomial.map_aeval_eq_aeval_map (φ := algebraMap R K)
        (ψ := MvPolynomial.map (algebraMap R K)) (by ext r; simp), map_X,
      hVmap]
  · rw [degreeOf, ← degrees_map_of_injective U (f := algebraMap R K) (IsFractionRing.injective R K),
      hUmap, ← degreeOf, smul_eq_C_mul, degreeOf_C_mul j g (mem_nonZeroDivisors_of_ne_zero hg)]
  · rw [← Polynomial.natDegree_map_eq_of_injective (f := algebraMap R K)
      (IsFractionRing.injective R K), hVmap, Polynomial.natDegree_C_mul (inv_ne_zero hg)]

        hasseDeriv ((D μ).mapDomain Subtype.val) (φ ν)
      rw [← hasseDeriv_rename Subtype.val_injective, hψ]
    rw [← hrename]
    exact fun hz ↦ hne ((rename_injective (R := K) Subtype.val Subtype.val_injective)
