theorem oldForm (i : σ) (f : Polynomial R) :
    Polynomial.aeval (X i : MvPolynomial σ R) f =
      rename (fun _ : Unit ↦ i) ((uniqueAlgEquiv R Unit).symm f) :=
  AlgHom.congr_fun (Polynomial.toMvPolynomial_eq_rename_comp i) f
-- A bundled equality supports rewriting all inputs simultaneously.
theorem bundled (i : σ) : Polynomial.toMvPolynomial (R := R) i =
    (rename (fun _ : Unit ↦ i)).comp (uniqueAlgEquiv R Unit).symm :=
  Polynomial.toMvPolynomial_eq_rename_comp i
-- Arbitrary one-element representations are valid, without finiteness assumptions.
theorem uniqueForm {ι : Type*} [Unique ι] (i : σ) (f : Polynomial R) :
    Polynomial.aeval (X i : MvPolynomial σ R) f =
      rename (fun _ : ι ↦ i) ((uniqueAlgEquiv R ι).symm f) := by
  have h : Polynomial.aeval (X i : MvPolynomial σ R) =
      (rename (fun _ : ι ↦ i)).comp (uniqueAlgEquiv R ι).symm := by
    apply Polynomial.algHom_ext
    simp
  exact AlgHom.congr_fun h f
-- Replacing the named coordinate by an arbitrary element changes the meaning.
theorem falseGeneralization :
    ¬ (∀ f : Polynomial ℕ, Polynomial.aeval (0 : MvPolynomial Unit ℕ) f =
      rename (fun _ : Unit ↦ ()) ((uniqueAlgEquiv ℕ Unit).symm f)) := by
  intro h
  have hh := h Polynomial.X
  have hc := congrArg (fun P : MvPolynomial Unit ℕ ↦ P.coeff (Finsupp.single () 1)) hh
  norm_num at hc
