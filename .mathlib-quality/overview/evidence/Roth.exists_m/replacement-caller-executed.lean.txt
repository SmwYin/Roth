-- This is a NEW proof from existing mathlib building blocks, not an existing theorem.
theorem generic {a b κ : ℝ} (hκ : b < κ) :
    ∃ m : ℕ, a ^ 2 < m ∧ a * √(m : ℝ) < m ∧
      b * m / (m - a * √(m : ℝ)) < κ := by
  obtain ⟨s, hs⟩ := exists_nat_gt (max |a| (κ * a / (κ - b)))
  have hsabs : |a| < (s : ℝ) := (le_max_left _ _).trans_lt hs
  have hsa : a < (s : ℝ) := (le_abs_self a).trans_lt hsabs
  have hspos : (0 : ℝ) < s := (abs_nonneg a).trans_lt hsabs
  have hkey : κ * a < (s : ℝ) * (κ - b) :=
    (div_lt_iff₀ (sub_pos.mpr hκ)).mp ((le_max_right _ _).trans_lt hs)
  have hmul : a * (s : ℝ) < (s : ℝ) ^ 2 := by
    simpa only [pow_two] using mul_lt_mul_of_pos_right hsa hspos
  refine ⟨s ^ 2, ?_, ?_, ?_⟩
  · push_cast
    nlinarith only [mul_pos (sub_pos.mpr hsabs)
      (add_pos_of_pos_of_nonneg hspos (abs_nonneg a)), sq_abs a]
  · simpa only [Nat.cast_pow, Real.sqrt_sq (Nat.cast_nonneg s)] using hmul
  · push_cast
    rw [Real.sqrt_sq (Nat.cast_nonneg _), div_lt_iff₀ (sub_pos.mpr hmul)]
    nlinarith only [mul_lt_mul_of_pos_right hkey hspos]

theorem coefficient {t c b κ : ℝ} (hκ : b < κ) :
    ∃ m : ℕ, (c * t) ^ 2 < m ∧ c * t * √(m : ℝ) < m ∧
      b * m / (m - c * t * √(m : ℝ)) < κ :=
  generic hκ

theorem recovered {n : ℕ} {κ : ℝ} (hκ : 2 < κ) :
    ∃ m : ℕ, 16 * n ^ 2 < m ∧ 4 * n * √(m : ℝ) < m ∧
      2 * m / (m - 4 * n * √(m : ℝ)) < κ := by
  obtain ⟨m, hm, hs, hq⟩ := generic (a := 4 * n) hκ
  refine ⟨m, ?_, hs, hq⟩
  have hm' : (16 : ℝ) * (n : ℝ) ^ 2 < m := by nlinarith only [hm]
  exact_mod_cast hm'

end Dimension275Signed

namespace Dimension275ReplacementCaller
open Roth
theorem replay {α κ : ℝ} (hint : IsIntegral ℤ α) (hκ : 2 < κ) :
    ∃ (f : Polynomial ℤ) (A : ℝ) (m : ℕ) (δ : ℝ), f.Monic ∧ Polynomial.aeval α f = 0 ∧ 1 ≤ A ∧
      (∀ i, |(f.coeff i : ℝ)| ≤ A) ∧ 2 ≤ m ∧ 0 < δ ∧ δ < 1 / m ∧
      eta δ m + 2 * (1 + 3 * δ) * f.natDegree * √(m : ℝ) < m / 2 ∧
      (m : ℝ) * (1 + 4 * δ) / (gam δ f.natDegree m - eta δ m) < κ := by
  obtain ⟨f, hf, hfα, hdeg⟩ := exists_monic_aeval_eq_zero hint
  -- a bound for the coefficients of `f`
  let A : ℝ := 1 + ∑ i ∈ f.support, |(f.coeff i : ℝ)|
  -- choice of `m` and `δ`
  obtain ⟨m, hm, -, h47'⟩ := Dimension275Signed.recovered (n := f.natDegree) hκ
  obtain ⟨δ, hδ0, hδm, h30, h49⟩ := exists_delta hm h47'
  refine ⟨f, A, m, δ, hf, hfα, ?_, ?_, ?_, hδ0, hδm, h30, h49⟩
  · exact le_add_of_nonneg_right (Finset.sum_nonneg fun _ _ ↦ abs_nonneg _)
  · intro i
    exact (Finsupp.single_eval_le_sum f.toFinsupp.coeff (g := fun z : ℤ ↦ |(z : ℝ)|)
      (by simp) (fun _ ↦ abs_nonneg _) i).trans (le_add_of_nonneg_left zero_le_one)
  · nlinarith only [hm, hdeg]
end Dimension275ReplacementCaller
