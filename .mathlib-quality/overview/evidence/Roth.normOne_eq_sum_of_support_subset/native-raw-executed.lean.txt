namespace NormSupportRaw
variable {σ : Type*}
theorem current {P : MvPolynomial σ ℤ} {s : Finset (σ →₀ ℕ)} (hs : P.support ⊆ s) :
    P.coeff.sum (fun _ z ↦ |(z : ℝ)|) = ∑ d ∈ s, |((P.coeff d : ℤ) : ℝ)| :=
  Finsupp.sum_of_support_subset P.coeff hs (fun _ z ↦ |(z : ℝ)|)
    (fun _ _ ↦ (congrArg abs (Int.cast_zero : ((0 : ℤ) : ℝ) = 0)).trans abs_zero)
theorem supportLocal {ι R S : Type*} [Zero R] [AddCommMonoid S]
    (f : ι →₀ R) (s : Finset ι) (hs : f.support ⊆ s) (size : ι → R → S)
    (hsize : ∀ i ∈ s, size i 0 = 0) : f.sum size = ∑ i ∈ s, size i (f i) :=
  Finsupp.sum_of_support_subset f hs size hsize
theorem addedOnly {ι R S : Type*} [Zero R] [AddCommMonoid S]
    (f : ι →₀ R) (s : Finset ι) (hs : f.support ⊆ s) (size : ι → R → S)
    (hsize : ∀ i ∈ s, i ∉ f.support → size i 0 = 0) :
    f.sum size = ∑ i ∈ s, size i (f i) :=
  Finset.sum_subset hs fun i hi hn ↦ by
    rw [Finsupp.notMem_support_iff.mp hn]
    exact hsize i hi hn
theorem zeroSizeNecessary :
    (0 : Unit →₀ ℤ).sum (fun _ _ ↦ (1 : ℝ)) ≠ ∑ _i ∈ ({()} : Finset Unit), (1 : ℝ) := by
  simp
theorem containmentNecessary :
    (Finsupp.single () (1 : ℤ)).sum (fun _ z ↦ |(z : ℝ)|) ≠
      ∑ _i ∈ (∅ : Finset Unit), (0 : ℝ) := by
  simp
