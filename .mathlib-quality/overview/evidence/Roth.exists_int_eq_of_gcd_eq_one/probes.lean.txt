 theorem gcd_smul_mem {R M ι : Type*} [CommRing R] [IsBezout R]
    [NormalizedGCDMonoid R] [AddCommGroup M] [Module R M]
    (N : Submodule R M) (s : Finset ι) (u : ι → R) (hu : s.gcd u = 1)
    (v : M) (hv : ∀ i ∈ s, u i • v ∈ N) : v ∈ N := by
  classical
  obtain ⟨z, hz⟩ := Finset.gcd_eq_sum_mul s u
  have h : (∑ i ∈ s, u i * z i) • v ∈ N := by
    rw [Finset.sum_smul]
    exact N.sum_mem fun i hi ↦ by
      simpa only [mul_comm (u i), mul_smul] using N.smul_mem (z i) (hv i hi)
  simpa only [← hz, hu, one_smul] using h
 theorem original {ι : Type*} {s : Finset ι} {u : ι → ℤ} (hu : s.gcd u = 1)
    {v : ℚ} (hv : ∀ a ∈ s, ∃ z : ℤ, (u a : ℚ) * v = z) : ∃ z : ℤ, v = z := by
  fail_if_success exact?
  classical
  obtain ⟨z, hz⟩ := Finset.gcd_eq_sum_mul s u
  choose! w hw using hv
  refine ⟨∑ i ∈ s, z i * w i, ?_⟩
  calc v = ((∑ i ∈ s, u i * z i : ℤ) : ℚ) * v := by simp [← hz, hu]
       _ = ((∑ i ∈ s, z i * w i : ℤ) : ℚ) := by
         push_cast
         rw [Finset.sum_mul]
         exact Finset.sum_congr rfl fun i hi ↦ by rw [mul_comm (u i : ℚ), mul_assoc, hw i hi]
 theorem abstract_existing {R M : Type*} [Semiring R] [AddCommMonoid M] [Module R M]
    (N : Submodule R M) (s : Set R) (hs : Ideal.span s = ⊤) (v : M)
    (hv : ∀ r : s, (r : R) • v ∈ N) : v ∈ N :=
  N.mem_of_span_top_of_smul_mem s hs v hv
