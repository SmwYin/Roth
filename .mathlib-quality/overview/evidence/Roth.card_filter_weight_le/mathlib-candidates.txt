464: /-- **The Bhatia-Davis inequality on variance**
465: 
466: The variance of a random variable `X` satisfying `a ≤ X ≤ b` almost everywhere is at most
467: `(b - 𝔼 X) * (𝔼 X - a)`. -/
468: lemma variance_le_sub_mul_sub [IsProbabilityMeasure μ] {a b : ℝ} {X : Ω → ℝ}
469:     (h : ∀ᵐ ω ∂μ, X ω ∈ Set.Icc a b) (hX : AEMeasurable X μ) :
470:     variance X μ ≤ (b - μ[X]) * (μ[X] - a) := by
471:   have ha : ∀ᵐ ω ∂μ, a ≤ X ω := h.mono fun ω h => h.1
472:   have hb : ∀ᵐ ω ∂μ, X ω ≤ b := h.mono fun ω h => h.2
473:   have hX_int₂ : Integrable (fun ω ↦ -X ω ^ 2) μ :=
474:     (memLp_of_bounded h hX.aestronglyMeasurable 2).integrable_sq.neg
475:   have hX_int₁ : Integrable (fun ω ↦ (a + b) * X ω) μ :=
476:     ((integrable_const (max |a| |b|)).mono' hX.aestronglyMeasurable
477:       (by filter_upwards [ha, hb] with ω using abs_le_max_abs_abs)).const_mul (a + b)
478:   have h0 : 0 ≤ -μ[X ^ 2] + (a + b) * μ[X] - a * b :=
479:     calc
480:       _ ≤ ∫ ω, (b - X ω) * (X ω - a) ∂μ := by
481:         apply integral_nonneg_of_ae
482:         filter_upwards [ha, hb] with ω ha' hb'
483:         exact mul_nonneg (by linarith : 0 ≤ b - X ω) (by linarith : 0 ≤ X ω - a)
484:       _ = ∫ ω, -X ω ^ 2 + (a + b) * X ω - a * b ∂μ :=
485:         integral_congr_ae <| ae_of_all μ fun ω ↦ by ring
486:       _ = ∫ ω, - X ω ^ 2 + (a + b) * X ω ∂μ - ∫ _, a * b ∂μ :=
487:         integral_sub (by fun_prop) (integrable_const (a * b))
488:       _ = ∫ ω, - X ω ^ 2 + (a + b) * X ω ∂μ - a * b := by simp
489:       _ = - μ[X ^ 2] + (a + b) * μ[X] - a * b := by
490:         simp [← integral_neg, ← integral_const_mul, integral_add hX_int₂ hX_int₁]
491:   calc
492:     _ ≤ (a + b) * μ[X] - a * b - μ[X] ^ 2 := by
493:       rw [variance_eq_sub (memLp_of_bounded h hX.aestronglyMeasurable 2)]
494:       linarith
495:     _ = (b - μ[X]) * (μ[X] - a) := by ring
