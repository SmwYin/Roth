open scoped BigOperators
open Finset
namespace Audit132
theorem positive_bounds_necessary : ¬ (((((Roth.box (fun _ : Fin 9 ↦ 0)).filter
    fun d ↦ Roth.weight (fun _ : Fin 9 ↦ 0) d ≤ ((9 : ℝ) - 9) / 2).card : ℝ) ≤
      2 * Real.sqrt (9 : ℝ) / 9 * ∏ _j : Fin 9, ((0 : ℝ) + 1)) := by
  norm_num [Roth.box, Roth.weight, Real.sqrt_eq_iff_eq_sq]
theorem positive_lam_necessary : ¬ (((((Roth.box (fun _ : Fin 0 ↦ 1)).filter
    fun d ↦ Roth.weight (fun _ : Fin 0 ↦ 1) d ≤ ((0 : ℝ) - 0) / 2).card : ℝ) ≤
      2 * Real.sqrt (0 : ℝ) / 0 * ∏ _j : Fin 0, ((1 : ℝ) + 1)) := by
  norm_num [Roth.box, Roth.weight]
#print axioms positive_bounds_necessary
#print axioms positive_lam_necessary
end Audit132
