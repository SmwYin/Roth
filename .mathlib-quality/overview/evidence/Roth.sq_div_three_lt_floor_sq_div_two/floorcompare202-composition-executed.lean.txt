#check_failure Roth.sq_div_three_lt_floor_sq_div_two
namespace FloorCompare202Composition

theorem viaSubOne {a : ℝ} (ha : 10 < a) :
    a ^ 2 / 3 < (⌊a⌋₊ : ℝ) ^ 2 / 2 := by
  nlinarith [Nat.sub_one_lt_floor a]

theorem viaSquareMonotonicity {a : ℝ} (ha : 5 ≤ a) :
    a ^ 2 / 3 < (⌊a⌋₊ : ℝ) ^ 2 / 2 := by
  have hn : (5 : ℝ) ≤ ⌊a⌋₊ := by exact_mod_cast Nat.le_floor ha
  have hs : a ^ 2 < ((⌊a⌋₊ : ℝ) + 1) ^ 2 :=
    (sq_lt_sq₀ (by linarith : 0 ≤ a) (by positivity)).2 (Nat.lt_floor_add_one a)
  nlinarith [sq_nonneg ((⌊a⌋₊ : ℝ) - 5)]

theorem negativeInputFails : ¬ (2 * (-100 : ℝ) ^ 2 < 3 * (5 : ℝ) ^ 2) := by
  norm_num

theorem sqrtComparison {X : ℝ} (hX : 2 ≤ X) : √X < ⌊X⌋₊ := by
  have hfloor : (2 : ℝ) ≤ ⌊X⌋₊ := by exact_mod_cast Nat.le_floor hX
  exact (Real.sqrt_lt' (by positivity)).2
    (by nlinarith only [Nat.lt_floor_add_one X, hfloor])
