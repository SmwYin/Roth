theorem closureGuards {a : ℝ} (ha : 10 < a) :
    a ^ 2 / 3 < (⌊a⌋₊ : ℝ) ^ 2 / 2 := by
  fail_if_success solve | exact?
  fail_if_success solve | simp
  fail_if_success solve | linarith [Nat.lt_floor_add_one a]
  nlinarith [Nat.lt_floor_add_one a]

theorem genericOriginal {K : Type*} [Field K] [LinearOrder K] [IsStrictOrderedRing K]
    [FloorSemiring K] {a : K} (ha : 10 < a) :
    a ^ 2 / 3 < (⌊a⌋₊ : K) ^ 2 / 2 := by
  nlinarith [Nat.lt_floor_add_one a]

theorem floorFree {K : Type*} [CommRing K] [LinearOrder K] [IsStrictOrderedRing K]
    {a n : K} (ha : 0 ≤ a) (hn : 5 ≤ n) (hlt : a < n + 1) :
    2 * a ^ 2 < 3 * n ^ 2 := by
  nlinarith [mul_nonneg (sub_nonneg.mpr hn) (sub_nonneg.mpr hn)]

theorem genericFive {K : Type*} [Field K] [LinearOrder K] [IsStrictOrderedRing K]
    [FloorSemiring K] {a : K} (ha : 5 ≤ a) :
    a ^ 2 / 3 < (⌊a⌋₊ : K) ^ 2 / 2 := by
  have hn : (5 : K) ≤ ⌊a⌋₊ := by exact_mod_cast Nat.le_floor ha
  have h := floorFree (by linarith : 0 ≤ a) hn (Nat.lt_floor_add_one a)
  nlinarith only [h]

theorem realFive {a : ℝ} (ha : 5 ≤ a) :
    a ^ 2 / 3 < (⌊a⌋₊ : ℝ) ^ 2 / 2 := genericFive ha

theorem thresholdFourFails :
    ¬ ((99 / 20 : ℝ) ^ 2 / 3 < (⌊(99 / 20 : ℝ)⌋₊ : ℝ) ^ 2 / 2) := by
  have hf : ⌊(99 / 20 : ℝ)⌋₊ = 4 := (Nat.floor_eq_iff (by norm_num)).2 (by norm_num)
  rw [hf]
  norm_num

theorem noLowerBoundFails : ¬ ((0 : ℝ) ^ 2 / 3 < (⌊(0 : ℝ)⌋₊ : ℝ) ^ 2 / 2) := by
  norm_num

theorem realSpecialization {a : ℝ} (ha : 10 < a) :
    a ^ 2 / 3 < (⌊a⌋₊ : ℝ) ^ 2 / 2 := genericFive (by linarith)
