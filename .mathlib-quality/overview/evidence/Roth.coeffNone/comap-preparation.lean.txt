namespace CoeffNoneComap152
universe u v
variable {τ : Type v}
theorem insert_injective (ν : ℕ) :
    Function.Injective (fun d : τ →₀ ℕ => d.optionElim ν) := by
  intro d e h
  simpa only [Finsupp.some_optionElim] using congrArg Finsupp.some h
noncomputable def slice {A : Type u} [Zero A]
    (P : (Option τ →₀ ℕ) →₀ A) (ν : ℕ) : (τ →₀ ℕ) →₀ A :=
  Finsupp.comapDomain (fun d => d.optionElim ν) P (insert_injective ν).injOn
variable {R : Type u} [CommSemiring R]
theorem coefficient_bridge (P : MvPolynomial (Option τ) R) (ν : ℕ) :
    ((MvPolynomial.optionEquivLeft R τ P).coeff ν).coeff = slice P.coeff ν := by
  apply Finsupp.ext
  intro d
  exact MvPolynomial.optionEquivLeft_coeff_coeff R P ν d
