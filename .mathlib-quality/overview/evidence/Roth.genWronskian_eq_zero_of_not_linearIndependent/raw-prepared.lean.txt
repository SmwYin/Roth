 theorem abstract_zero {R A M ι : Type*} [CommSemiring R] [CommRing A] [IsDomain A]
    [Algebra R A] [FaithfulSMul R A] [AddCommMonoid M] [Module R M]
    [Fintype ι] [DecidableEq ι] (δ : ι → M →ₗ[R] A) {v : ι → M}
    (h : ¬ LinearIndependent R v) :
    (Matrix.of fun i j ↦ δ i (v j)).det = 0 :=
  Matrix.det_eq_zero_of_not_linearIndependent_cols fun hc ↦
    h (LinearIndependent.of_comp (LinearMap.pi δ) (hc.restrict_scalars' R))
theorem bundled_zero {R A M ι : Type*} [CommSemiring R] [CommRing A] [IsDomain A]
    [Algebra R A] [FaithfulSMul R A] [AddCommMonoid M] [Module R M]
    [Fintype ι] [DecidableEq ι] (δ : M →ₗ[R] (ι → A)) {v : ι → M}
    (h : ¬ LinearIndependent R v) :
    (Matrix.of fun i j ↦ δ (v j) i).det = 0 :=
  Matrix.det_eq_zero_of_not_linearIndependent_cols fun hc ↦
    h (LinearIndependent.of_comp δ (hc.restrict_scalars' R))
