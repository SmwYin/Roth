theorem universal {σ τ R : Type*} [CommSemiring R]
    (f : σ → MvPolynomial τ R) (P : MvPolynomial σ R) :
    (MvPolynomial.aeval f) P = MvPolynomial.aeval f P := rfl

theorem inline {σ R : Type*} [CommSemiring R] (α : σ → R) (P : MvPolynomial σ R) :
    (let T : MvPolynomial σ R →ₐ[R] MvPolynomial σ R :=
      MvPolynomial.aeval fun j ↦ MvPolynomial.X j + MvPolynomial.C (α j)
     T P) = MvPolynomial.aeval (fun j ↦ MvPolynomial.X j + MvPolynomial.C (α j)) P := rfl
