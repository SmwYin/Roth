theorem abstractLaw {A : Type u} {R : Type v} {T : Type w} [Semiring R]
    [SemilatticeInf T] [OrderTop T] (D : A → T) (P : AddMonoidAlgebra R A) (c : T) :
    c ≤ AddMonoidAlgebra.infDegree D P ↔ ∀ d ∈ P.coeff.support, c ≤ D d :=
  Finset.le_inf_iff

theorem projectedLaw {A : Type u} {R : Type v} [Semiring R]
    (D : A → ℝ) (P : AddMonoidAlgebra R A) (hP : P.coeff.support.Nonempty) (c : ℝ) :
    c ≤ (AddMonoidAlgebra.infDegree (fun d ↦ (D d : WithTop ℝ)) P).untopD 0 ↔
      ∀ d, P.coeff d ≠ 0 → c ≤ D d := by
  change c ≤ (P.coeff.support.inf (WithTop.some ∘ D)).untopD 0 ↔ _
  rw [← Finset.coe_inf' hP D]
  simp only [WithTop.untopD_coe]
  simpa only [Finsupp.mem_support_iff] using
    (show c ≤ P.coeff.support.inf' hP D ↔ ∀ d ∈ P.coeff.support, c ≤ D d from
      Finset.le_inf'_iff)
#print axioms abstractLaw
#print axioms projectedLaw
end LeIndexIff137
