namespace NormNonnegativeRaw
variable {σ : Type*}
theorem current (P : MvPolynomial σ ℤ) :
    0 ≤ ∑ d ∈ P.support, |((P.coeff d : ℤ) : ℝ)| :=
  Finset.sum_nonneg fun _ _ ↦ abs_nonneg _
theorem finsuppCurrent (P : MvPolynomial σ ℤ) :
    0 ≤ P.coeff.sum (fun _ z ↦ |(z : ℝ)|) :=
  Finsupp.sum_nonneg fun _ _ ↦ abs_nonneg _
theorem supportOnly {ι R S : Type*} [Zero R] [AddCommMonoid S] [Preorder S]
    [IsOrderedAddMonoid S] (f : ι →₀ R) (size : ι → R → S)
    (hsize : ∀ i ∈ f.support, 0 ≤ size i (f i)) : 0 ≤ f.sum size :=
  Finsupp.sum_nonneg hsize
