open Real
def UniformBound (t x : ℝ) : Prop :=
  ∀ ε : ℝ, 0 < ε → ∃ C : ℝ, 0 < C ∧
    ∀ p q : ℤ, 0 < q → C / (q : ℝ) ^ (t + ε) < |x - (p : ℝ) / q|
def pairSet (α κ : ℝ) : Set (ℤ × ℕ) :=
  {hq | 0 < hq.2 ∧ |α - (hq.1 : ℝ) / hq.2| < 1 / (hq.2 : ℝ) ^ κ}
-- These probes precede every authored equivalence/helper theorem.
example (x : ℝ) (hx : Irrational x) : True := by
  fail_if_success
    have : UniformBound 2 x ↔ ∀ {κ : ℝ}, (pairSet x κ).Infinite → κ ≤ 2 := by
      exact?
  fail_if_success
    have : UniformBound 2 x ↔ ∀ {κ : ℝ}, (pairSet x κ).Infinite → κ ≤ 2 := by
      simp only [UniformBound, pairSet]
  exact (show True from (fun (_ : Irrational x) ↦ True.intro) hx)
example (t x : ℝ) : True := by
  fail_if_success
    have : UniformBound t x ↔ ∀ {κ : ℝ}, (pairSet x κ).Infinite → κ ≤ t := by
      exact?
  fail_if_success
    have : UniformBound t x ↔
        Irrational x ∧ ∀ {p : ℝ}, LiouvilleWith p x → p ≤ t := by
      exact?
  trivial
