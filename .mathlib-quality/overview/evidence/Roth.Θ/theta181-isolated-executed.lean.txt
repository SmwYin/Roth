noncomputable def generic [SupSet V] (s : Set V) : V := sSup s
theorem exact_sup [SupSet V] (s : Set V) : generic s = SupSet.sSup s := rfl
theorem member (f : A → H → ℝ) (s : Set A) (t : Set H)
    (hb : BddAbove (Set.image2 f s t)) {a : A} (ha : a ∈ s) {h : H} (hh : h ∈ t) :
    f a h ≤ sSup (Set.image2 f s t) := le_csSup hb (Set.mem_image2_of_mem ha hh)
theorem upper (f : A → H → ℝ) (s : Set A) (t : Set H)
    (hne : (Set.image2 f s t).Nonempty) {c : ℝ}
    (hc : ∀ a ∈ s, ∀ h ∈ t, f a h ≤ c) : sSup (Set.image2 f s t) ≤ c :=
  csSup_le hne (Set.forall_mem_image2.mpr hc)
theorem empty : sSup (∅ : Set ℝ) = 0 := Real.sSup_empty
theorem unbounded {s : Set ℝ} (hs : ¬BddAbove s) : sSup s = 0 :=
  Real.sSup_of_not_bddAbove hs
