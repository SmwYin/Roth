def family {σ R : Type*} [CommSemiring R] (r : σ →₀ ℕ)
    (t : (σ →₀ ℕ) → Finset R) : Set (MvPolynomial σ R) :=
  {P | (∀ j, degreeOf j P ≤ r j) ∧ ∀ d, P.coeff d ∈ t d}

open scoped Classical in
theorem representation {σ R : Type*} [CommSemiring R] (r : σ →₀ ℕ)
    (t : (σ →₀ ℕ) → Finset R) (ht : ∀ d ∉ Finset.Iic r, 0 ∈ t d) :
    family r t = AddMonoidAlgebra.ofCoeff ''
      (↑((Finset.Iic r).finsupp t) : Set ((σ →₀ ℕ) →₀ R)) := by
  classical
  ext W
  constructor
  · intro hW
    refine ⟨W.coeff, ?_, rfl⟩
    rw [Finset.mem_coe, Finset.mem_finsupp_iff]
    refine ⟨fun d hd ↦ ?_, fun d _ ↦ hW.2 d⟩
    exact Finset.mem_Iic.2 fun j ↦ (monomial_le_degreeOf j hd).trans (hW.1 j)
  · rintro ⟨c, hc, rfl⟩
    obtain ⟨hs, hc⟩ := Finset.mem_finsupp_iff.1 hc
    refine ⟨fun j ↦ degreeOf_le_iff.2 fun d hd ↦ (Finset.mem_Iic.1 (hs hd)) j, fun d ↦ ?_⟩
    by_cases hd : d ∈ Finset.Iic r
    · exact hc d hd
    · simpa only [Finsupp.notMem_support_iff.1 (fun h ↦ hd (hs h))] using ht d hd

open scoped Classical in
theorem ncard_family {σ R : Type*} [CommSemiring R] (r : σ →₀ ℕ)
    (t : (σ →₀ ℕ) → Finset R) (ht : ∀ d ∉ Finset.Iic r, 0 ∈ t d) :
    (family r t).ncard = ∏ d ∈ Finset.Iic r, (t d).card := by
  classical
  rw [representation r t ht, Set.ncard_image_of_injective _
    AddMonoidAlgebra.ofCoeff_injective, Set.ncard_coe_finset, Finset.card_finsupp]

theorem constant_choices {σ R : Type*} [CommSemiring R] (r : σ →₀ ℕ)
    (t : Finset R) (ht : 0 ∈ t) :
    (family r (fun _ ↦ t)).ncard = t.card ^ ∏ j ∈ r.support, (r j + 1) := by
  classical
  rw [ncard_family r _ (fun _ _ ↦ ht), Finset.prod_const, Finsupp.card_Iic]
  simp only [Nat.card_Iic]

theorem abstract_image {α β : Type*} (s : Set α) (f : α → β)
    (hf : Function.Injective f) : (f '' s).ncard = s.ncard :=
  Set.ncard_image_of_injective s hf

theorem finite_choices {ι α : Type*} [Zero α] (s : Finset ι) (t : ι → Finset α) :
    (s.finsupp t).card = ∏ i ∈ s, (t i).card := Finset.card_finsupp s t

 theorem ncard_original {σ : Type*} (r : σ → ℕ) [Finite (Function.support r)] (B : ℕ) :
    (boxFamily r B).ncard = (B + 1) ^ ∏ᶠ j, (r j + 1) := by
  classical
  have h := Box246Generic.constant_choices (Finsupp.ofSupportFinite r (Set.toFinite _))
    (Finset.Icc (0 : ℤ) B) (by simp)
  have hc : (Finset.Icc (0 : ℤ) B).card = B + 1 := by simp
  rw [hc] at h
  have he : (Box246Generic.family (Finsupp.ofSupportFinite r (Set.toFinite _))
      (fun _ ↦ Finset.Icc (0 : ℤ) B)) = boxFamily r B := by
    ext P
    simp only [Box246Generic.family, boxFamily, Set.mem_setOf_eq, Finset.mem_Icc]
    rfl
  rw [he] at h
  rw [h]
  congr 1
  exact (finprod_eq_prod_of_mulSupport_subset _ fun j hj ↦ by
    simpa [Finsupp.ofSupportFinite_support, Function.support, Function.mulSupport] using hj).symm

theorem ncard_inline {σ : Type*} (r : σ → ℕ) [Finite (Function.support r)] (B : ℕ) :
    (boxFamily r B).ncard = (B + 1) ^ ∏ᶠ j, (r j + 1) := by
  rw [box246Representation, Set.ncard_image_of_injective _
    AddMonoidAlgebra.ofCoeff_injective, Set.ncard_coe_finset, Finset.card_finsupp]
  simp only [Int.card_Icc, sub_zero, Finset.prod_const, Finsupp.card_Iic, Nat.card_Iic]
  congr 1
  exact (finprod_eq_prod_of_mulSupport_subset _ fun j hj ↦ by
    simpa [Finsupp.ofSupportFinite_support, Function.support, Function.mulSupport] using hj).symm

theorem supplied_signature : @Roth.ncard_boxFamily = @ncard_original := rfl

theorem zero_coefficient_bound {σ : Type*} (r : σ → ℕ) :
    (boxFamily r 0).ncard = (0 + 1) ^ ∏ᶠ j, (r j + 1) := by
  have hz : boxFamily r 0 = {0} := by
    ext P
    constructor
    · intro h
      apply Set.mem_singleton_iff.2
      apply MvPolynomial.ext
      intro d
      exact le_antisymm (h.2 d).2 (h.2 d).1
    · rintro rfl
      simp [boxFamily]
  simp [hz]

theorem zero_degree {σ : Type*} (B : ℕ) :
    (boxFamily (fun _ : σ ↦ 0) B).ncard = B + 1 := by
  let : Fintype (Function.support (fun _ : σ ↦ 0)) :=
    (show (Function.support (fun _ : σ ↦ 0)).Finite by simp).fintype
  simpa using ncard_original (fun _ : σ ↦ 0) B

theorem empty_variables (r : Empty → ℕ) (B : ℕ) : (boxFamily r B).ncard = B + 1 := by
  simpa using ncard_original r B

theorem infinite_variables_finite_support (B : ℕ) :
    (boxFamily (Finsupp.single 0 2 : ℕ →₀ ℕ) B).ncard = (B + 1) ^ 3 := by
  classical
  have h := Box246Generic.constant_choices (Finsupp.single 0 2 : ℕ →₀ ℕ)
    (Finset.Icc (0 : ℤ) B) (by simp)
  simpa [Box246Generic.family, boxFamily] using h

theorem infinite_family : (boxFamily (fun _ : ℕ ↦ 1) 1).Infinite := by
  have hi : Function.Injective (fun i : ℕ ↦ (X i : MvPolynomial ℕ ℤ)) :=
    fun _ _ h ↦ MvPolynomial.X_injective h
  refine (Set.infinite_range_of_injective hi).mono (Set.range_subset_iff.2 ?_)
  intro i
  constructor
  · intro j
    by_cases h : i = j
    · subst j
      simp
    · simp [degreeOf_X_of_ne (Ne.symm h)]
  · intro d
    simp only [MvPolynomial.coeff_X]
    split_ifs <;> norm_num

theorem support_needed : (boxFamily (fun _ : ℕ ↦ 1) 1).ncard ≠
    (1 + 1) ^ ∏ᶠ _ : ℕ, (1 + 1) := by
  have hp : ∏ᶠ _ : ℕ, (1 + 1 : ℕ) = 1 := by
    apply finprod_of_infinite_mulSupport
    simpa [Function.mulSupport] using (Set.infinite_univ : (Set.univ : Set ℕ).Infinite)
  rw [hp]
  simpa using infinite_family.ncard

 theorem strict_no_direct {σ : Type*} (r : σ → ℕ) [Finite (Function.support r)] (B : ℕ) :
    (boxFamily r B).ncard = (B + 1) ^ ∏ᶠ j, (r j + 1) := by
  fail_if_success exact Finset.card_finsupp _ _
  fail_if_success exact Set.ncard_image_of_injective _ AddMonoidAlgebra.ofCoeff_injective
  fail_if_success solve | simp only [boxFamily]
  exact ncard_inline r B
theorem exists_ne_modByMonic_eq {σ : Type*} (r : σ → ℕ) [Finite (Function.support r)] (B : ℕ)
    {f : Polynomial ℤ} (hf : f.Monic) (J : Finset (σ →₀ ℕ)) (C : ℕ)
    (hC : ∀ W ∈ boxFamily r B, ∀ j ∈ J, ∀ i < f.natDegree,
      |(diag (hasseDeriv j W) %ₘ f).coeff i| ≤ C)
    (hcount : ((2 * C + 1 : ℕ) : ℝ) ^ (f.natDegree * J.card) < ((B : ℝ) + 1) ^ ∏ᶠ j, (r j + 1)) :
    ∃ W' ∈ boxFamily r B, ∃ W'' ∈ boxFamily r B, W' ≠ W'' ∧
      ∀ j ∈ J, diag (hasseDeriv j W') %ₘ f = diag (hasseDeriv j W'') %ₘ f := by
  classical
  let : Fintype (boxFamily r B) := (boxFamily_finite r B).fintype
  let F (W : boxFamily r B) (p : J × Fin f.natDegree) : Set.Icc (-(C : ℤ)) C :=
    ⟨(diag (hasseDeriv p.1.1 W.1) %ₘ f).coeff p.2, abs_le.1 (hC W.1 W.2 p.1.1 p.1.2 p.2.1 p.2.2)⟩
  obtain ⟨W', W'', hne, h⟩ := Fintype.exists_ne_map_eq_of_card_lt F <| by
    simpa only [Fintype.card_fun, Fintype.card_ofFinset, Int.card_Icc, sub_neg_eq_add,
      Fintype.card_prod, Fintype.card_coe, Fintype.card_fin, mul_comm, Set.fintypeCard_eq_ncard,
      show (boxFamily r B).ncard = (B + 1) ^ ∏ᶠ j, (r j + 1) from by
        rw [box246Representation, Set.ncard_image_of_injective _
          AddMonoidAlgebra.ofCoeff_injective, Set.ncard_coe_finset, Finset.card_finsupp]
        simp only [Int.card_Icc, sub_zero, Finset.prod_const, Finsupp.card_Iic, Nat.card_Iic]
        congr 1
        exact (finprod_eq_prod_of_mulSupport_subset _ fun j hj ↦ by
          simpa [Finsupp.ofSupportFinite_support, Function.support, Function.mulSupport]
            using hj).symm, show ((C : ℤ) + 1 + C).toNat = 2 * C + 1 by lia] using
      (show (2 * C + 1) ^ (f.natDegree * J.card) < (B + 1) ^ ∏ᶠ j, (r j + 1) by
        exact_mod_cast hcount)
  refine ⟨W', W'.2, W'', W''.2, Subtype.coe_ne_coe.2 hne, fun j hj ↦ ?_⟩
  simpa [F, Polynomial.sum_modByMonic_coeff hf Polynomial.degree_le_natDegree] using
    congrArg (fun g ↦ ∑ i : Fin f.natDegree, Polynomial.monomial (i : ℕ)
      (g (⟨j, hj⟩, i) : ℤ)) h
