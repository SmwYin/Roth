theorem proof1 {σ : Type*} [Fintype σ] {B : ℝ} (hB : 1 ≤ B) {q r : σ → ℕ} {c : ℝ}
    (hc : ∀ R ∈ Roth.RSet B r, ∀ h : σ → ℤ, (∀ j, IsCoprime (h j) (q j : ℤ)) →
      Roth.index (map (Int.castRingHom ℝ) R) (fun j ↦ (h j : ℝ) / q j) r ≤ c) :
    Roth.Θ B q r ≤ c := by
  exact csSup_le (Roth.indexSet_nonempty hB q r)
    (fun _ ⟨R, hR, h, hh, he⟩ ↦ he ▸ hc R hR h hh)
theorem proof2 {σ : Type*} [Fintype σ] {B : ℝ} {q r : σ → ℕ} {c : ℝ} (hc0 : 0 ≤ c)
    (hc : ∀ R ∈ Roth.RSet B r, ∀ h : σ → ℤ, (∀ j, IsCoprime (h j) (q j : ℤ)) →
      Roth.index (map (Int.castRingHom ℝ) R) (fun j ↦ (h j : ℝ) / q j) r ≤ c) :
    Roth.Θ B q r ≤ c := by
  exact Real.sSup_le (fun _ ⟨R, hR, h, hh, he⟩ ↦ he ▸ hc R hR h hh) hc0
-- Refute unconditional dropping of hB: empty family has default supremum zero.
theorem proof3 : ¬ (∀ (s : Set ℝ) (c : ℝ), (∀ x ∈ s, x ≤ c) → sSup s ≤ c) := by
  intro h
  have bad := h ∅ (-1) (by simp)
  norm_num at bad
theorem bridge {σ : Type*} [Fintype σ] (B : ℝ) (q r : σ → ℕ) :
    Roth.Θ B q r = sSup (Roth.indexSet B q r) := rfl
