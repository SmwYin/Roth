namespace MvPolynomial.TaylorZeroIffAssessment
variable {σ A : Type*} [CommRing A]
theorem replay (α : σ → A) (P : MvPolynomial σ A) : taylor α P = 0 ↔ P = 0 :=
  (taylor_injective α).eq_iff' (map_zero _)
theorem independent (α : σ → A) (P : MvPolynomial σ A) : taylor α P = 0 ↔ P = 0 :=
  (TaylorInjectiveAssessment.independent α).eq_iff' (map_zero _)
#print axioms replay
#print axioms independent
end MvPolynomial.TaylorZeroIffAssessment
